[3] (오류 상황) OUT 파라미터가 있는 함수를 SQL 에서 호출한다 — PLS-00230
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE FUNCTION f_with_out
  2     (p_sal IN NUMBER, p_grade OUT VARCHAR2)
  3     RETURN NUMBER
  4  IS
  5  BEGIN
  6     p_grade := CASE WHEN p_sal >= 10000 THEN 'A' ELSE 'B' END;
  7     RETURN ROUND(p_sal * 0.2);
  8  END f_with_out;
  9  /

Function created.

-- PL/SQL 안에서는 호출된다.
SQL> DECLARE
  2     v_grade   VARCHAR2(1);
  3     v_tax     NUMBER;
  4  BEGIN
  5     v_tax := f_with_out(24000, v_grade);
  6     DBMS_OUTPUT.PUT_LINE(v_tax || ' / ' || v_grade);
  7  END;
  8  /
4800 / A

PL/SQL procedure successfully completed.

-- 그러나 SQL 문에서는 호출할 수 없다.
SQL> SELECT f_with_out(salary) FROM employees WHERE rownum = 1;
SELECT f_with_out(salary) FROM employees WHERE rownum = 1
       *
ERROR at line 1:
ORA-00904: "F_WITH_OUT": invalid identifier

SQL> DECLARE
  2     v_x   NUMBER;
  3  BEGIN
  4     SELECT f_with_out(salary) INTO v_x FROM employees WHERE rownum = 1;
  5  END;
  6  /
   SELECT f_with_out(salary) INTO v_x FROM employees WHERE rownum = 1;
          *
ERROR at line 4:
ORA-06550: line 4, column 11:
PLS-00230: OUT and IN OUT modes cannot be used in this context
ORA-06550: line 4, column 4:
PL/SQL: SQL Statement ignored

-- [원인 파악]
--   SQL 엔진은 함수에 값을 넣어 주기만 할 뿐 돌려받을 변수를 제공하지 않는다.
--   따라서 SQL 문맥에서는 모든 파라미터가 IN 모드여야 한다.

-- [수정] 값을 두 개 돌려주어야 한다면 함수를 두 개로 나누거나
--        OUT 파라미터를 가진 프로시저로 분리한다.
SQL> CREATE OR REPLACE FUNCTION f_grade (p_sal IN NUMBER) RETURN VARCHAR2
  2  IS
  3  BEGIN
  4     RETURN CASE WHEN p_sal IS NULL THEN 'N/A'
  5                 WHEN p_sal >= 10000 THEN 'A' ELSE 'B' END;
  6  END f_grade;
  7  /

Function created.

  4     v_cnt   PLS_INTEGER;

  2   WHERE rownum <= 3;

-- [원인 파악]
--   BOOLEAN 은 PL/SQL 전용 타입으로 SQL 에는 존재하지 않는다.
--   RECORD 와 연관 배열도 마찬가지다.

-- [수정] SQL 에서 쓸 수 있는 자료형으로 바꾼다.
SQL> CREATE OR REPLACE FUNCTION is_manager_yn (p_emp_id IN NUMBER)
  2     RETURN VARCHAR2
  3  IS
  4     v_cnt   PLS_INTEGER;
  5  BEGIN
  6     SELECT COUNT(*) INTO v_cnt
  7       FROM employees WHERE manager_id = p_emp_id;
  8     RETURN CASE WHEN v_cnt > 0 THEN 'Y' ELSE 'N' END;
  9  END is_manager_yn;
 10  /

[5] (오류 상황) 질의 안에서 DML 을 수행한다 — ORA-14551
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE FUNCTION get_grade_log (p_sal IN NUMBER)
  2     RETURN VARCHAR2
  3  IS
  4  BEGIN
  5     INSERT INTO fn_call_log(called_at, arg_value)
  6          VALUES (SYSTIMESTAMP, p_sal);
  7     RETURN CASE WHEN p_sal >= 10000 THEN 'A' ELSE 'B' END;
  8  END get_grade_log;
  9  /

Function created.

-- 컴파일은 성공한다. PL/SQL 안에서 호출하면 잘 동작한다.
SQL> DECLARE
  2     v_g   VARCHAR2(1);
  3  BEGIN
  4     v_g := get_grade_log(24000);
  5     DBMS_OUTPUT.PUT_LINE('등급 : ' || v_g);
  6  END;
  7  /
등급 : A

PL/SQL procedure successfully completed.

SQL> SELECT COUNT(*) AS log_cnt FROM fn_call_log;

   LOG_CNT
----------
         1

SQL> ROLLBACK;

Rollback complete.

-- 그러나 SELECT 문에서 호출하면 실패한다.
SQL> SELECT last_name, get_grade_log(salary) AS g
  2    FROM employees
  3   WHERE rownum <= 3;
SELECT last_name, get_grade_log(salary) AS g
                  *
ERROR at line 1:
ORA-14551: cannot perform a DML operation inside a query
ORA-06512: at "HR.GET_GRADE_LOG", line 5
ORA-06512: at line 1

-- [원인 파악]
--   순수성 규칙 : SELECT 문에서 호출된 함수는 데이터베이스 상태를 바꿀 수 없다.
--   조회 도중 같은 트랜잭션의 데이터가 바뀌면 어떤 행은 변경 전 값을,
--   어떤 행은 변경 후 값을 보게 되어 읽기 일관성이 깨지기 때문이다.
--   COMMIT · ROLLBACK · DDL · SET ROLE 도 같은 이유로 금지된다.

-- [수정] 로그 기록을 자율 트랜잭션으로 분리한다.
SQL> CREATE OR REPLACE FUNCTION get_grade_log (p_sal IN NUMBER)
  2     RETURN VARCHAR2
  3  IS
  4     PRAGMA AUTONOMOUS_TRANSACTION;            -- 본체와 독립된 트랜잭션
  5  BEGIN
  6     INSERT INTO fn_call_log(called_at, arg_value)
  7          VALUES (SYSTIMESTAMP, p_sal);
  8     COMMIT;                                   -- 반드시 스스로 종료해야 한다
  9     RETURN CASE WHEN p_sal >= 10000 THEN 'A' ELSE 'B' END;
 10  END get_grade_log;
 11  /

Function created.

SQL> SELECT last_name, salary, get_grade_log(salary) AS g
  2    FROM employees
  3   WHERE department_id = 90
  4   ORDER BY employee_id;

LAST_NAME        SALARY G
------------- --------- ----------
King              24000 A
Kochhar           17000 A
De Haan           17000 A

SQL> SELECT COUNT(*) AS log_cnt FROM fn_call_log;

   LOG_CNT
----------
         3

-- 자율 트랜잭션 안에서 COMMIT 을 빠뜨리면 ORA-06519 가 발생한다는 점도
-- 3장에서 다룬 내용이다. 반드시 함께 기억한다.

[6] 호출 비용 줄이기 — DETERMINISTIC 과 RESULT_CACHE
--------------------------------------------------------------------------------

-- 함수는 결과 행마다 호출되므로 행이 많을수록 컨텍스트 스위치 비용이 커진다.
SQL> SET TIMING ON

----------

Elapsed: 00:00:00.09

SQL> SET TIMING OFF

-- 같은 입력에 항상 같은 값을 반환하면 DETERMINISTIC 을 선언한다.
SQL> CREATE OR REPLACE FUNCTION tax (p_salary IN NUMBER)
  2     RETURN NUMBER
  3     DETERMINISTIC
  4  IS
  5  BEGIN
  6     IF p_salary IS NULL THEN
  7        RETURN NULL;
  8     ELSIF p_salary >= 15000 THEN
  9        RETURN ROUND(p_salary * 0.35);
 10     ELSIF p_salary >=  8000 THEN
 11        RETURN ROUND(p_salary * 0.24);
 12     ELSE
 13        RETURN ROUND(p_salary * 0.15);
 14     END IF;
 15  END tax;
 16  /

-- 테이블을 조회하는 함수라면 RESULT_CACHE 가 효과적이다.
SQL> CREATE OR REPLACE FUNCTION get_dept_name (p_dept_id IN NUMBER)
  2     RETURN VARCHAR2
  3     RESULT_CACHE
  4  IS
  5     v_name   departments.department_name%TYPE;
  6  BEGIN
  7     SELECT department_name INTO v_name
  8       FROM departments WHERE department_id = p_dept_id;
  9     RETURN v_name;
 10  EXCEPTION
 11     WHEN NO_DATA_FOUND THEN
 12        RETURN '(미배정)';
 13  END get_dept_name;
 14  /

Function created.

SQL> SELECT last_name, get_dept_name(department_id) AS dept
  2    FROM employees
  3   WHERE employee_id IN (100, 145, 178)
  4   ORDER BY employee_id;

LAST_NAME     DEPT
------------- ------------------------------
King          Executive
Russell       Sales
Grant         (미배정)

-- RESULT_CACHE 는 참조 테이블이 변경되면 자동으로 무효화된다.

-- [주의] DETERMINISTIC 은 오라클이 검증해 주지 않는다.
--        SYSDATE 나 테이블 조회처럼 결과가 달라질 수 있는 함수에 붙이면
--        옵티마이저가 낡은 값을 재사용해 잘못된 결과가 나올 수 있다.

[7] 19c 의 WITH FUNCTION — SQL 문 안에 함수를 인라인 정의한다
--------------------------------------------------------------------------------

SQL> WITH
  2     FUNCTION quick_tax (p_sal IN NUMBER) RETURN NUMBER IS
  3     BEGIN
  4        RETURN ROUND(NVL(p_sal, 0) * 0.24);
  5     END;
  6  SELECT employee_id, last_name, salary, quick_tax(salary) AS t
  7    FROM employees
  8   WHERE department_id = 20
  9   ORDER BY employee_id;
 10  /

EMPLOYEE_ID LAST_NAME        SALARY          T
----------- ------------- --------- ----------
        201 Hartstein         13000       3120
        202 Fay                6000       1440

-- 인라인 함수는 저장되지 않으므로 딕셔너리에 객체가 생기지 않는다.
SQL> SELECT COUNT(*) AS cnt FROM user_objects WHERE object_name = 'QUICK_TAX';

       CNT
----------
         0

-- 한 SQL 문에서만 쓰는 로직이라면 컨텍스트 스위치 없이 처리할 수 있어 유리하다.

 실습 04 정리
--------------------------------------------------------------------------------
 · 사용자 정의 함수는 SELECT · WHERE · GROUP BY · ORDER BY · VALUES · SET
   절에서 호출할 수 있으나 CHECK 제약이나 컬럼 DEFAULT 에는 쓸 수 없다.
 · SQL 호출 조건 : 저장된 함수일 것, 모든 파라미터가 IN 일 것,
   자료형이 SQL 타입일 것, EXECUTE 권한이 있을 것.
 · OUT 파라미터가 있으면 PLS-00230, BOOLEAN 반환이면 ORA-00902 가 발생한다.
 · 순수성 규칙 위반(질의 안 DML)은 ORA-14551 이며
   PRAGMA AUTONOMOUS_TRANSACTION + COMMIT 으로 분리해 해결한다.
 · DETERMINISTIC · RESULT_CACHE 는 호출 비용을 줄이지만
   DETERMINISTIC 은 오라클이 검증하지 않으므로 신중히 사용한다.
 · 19c 의 WITH FUNCTION 절은 한 SQL 문에서만 쓰는 함수를 인라인 정의한다.
