[2] (오류 상황) RETURN 자료형에 크기를 지정한다 — PLS-00103
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE FUNCTION get_full_name (p_emp_id IN NUMBER)
  2     RETURN VARCHAR2(60)                       -- 크기를 지정했다
  3  IS
  4     v_name   VARCHAR2(60);
  5  BEGIN
  6     SELECT first_name || ' ' || last_name INTO v_name
  7       FROM employees WHERE employee_id = p_emp_id;
  8     RETURN v_name;
  9  END get_full_name;
 10  /

Warning: Function created with compilation errors.

SQL> SHOW ERRORS
Errors for FUNCTION GET_FULL_NAME:

LINE/COL ERROR
-------- -----------------------------------------------------------------
2/22     PLS-00103: Encountered the symbol "(" when expecting one of the
         following:  . @ % ; is authid as cluster order using external
         deterministic parallel_enable pipelined result_cache accessible

-- [원인 파악]
--   형식 파라미터와 마찬가지로 RETURN 자료형에도 길이·정밀도를 적을 수 없다.
--   컴파일러는 RETURN VARCHAR2 다음에 IS/AS 나 DETERMINISTIC 같은 절이
--   오기를 기대하는데 괄호가 나오니 PLS-00103 을 낸다.
--   실제 크기는 반환값을 받는 쪽 변수가 결정한다.

-- [수정] 크기를 제거한다.
SQL> CREATE OR REPLACE FUNCTION get_full_name (p_emp_id IN NUMBER)
  2     RETURN VARCHAR2
  3  IS
  4     v_name   VARCHAR2(60);
  5  BEGIN
  6     SELECT first_name || ' ' || last_name INTO v_name
  7       FROM employees WHERE employee_id = p_emp_id;
  8     RETURN v_name;
  9  EXCEPTION
 10     WHEN NO_DATA_FOUND THEN
 11        RETURN NULL;                           -- 모든 경로에서 RETURN 보장
 12  END get_full_name;
 13  /

Function created.

  2   WHERE object_name = 'GET_FULL_NAME';

----------------- ------------ -------

SQL> CREATE OR REPLACE FUNCTION get_annual_salary (p_emp_id IN NUMBER)
  2     RETURN NUMBER
  3  IS
  4     v_sal    employees.salary%TYPE;
  5     v_comm   employees.commission_pct%TYPE;
  6  BEGIN
  7     SELECT salary, NVL(commission_pct, 0)
  8       INTO v_sal, v_comm
  9       FROM employees
 10      WHERE employee_id = p_emp_id;
 11     RETURN ROUND(v_sal * 12 * (1 + v_comm));
 12  EXCEPTION
 13     WHEN NO_DATA_FOUND THEN
 14        RETURN NULL;
 15  END get_annual_salary;

[5] (오류 상황) 모든 경로에서 RETURN 하지 않는다 — ORA-06503
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE FUNCTION grade_of (p_sal IN NUMBER)
  2     RETURN VARCHAR2
  3  IS
  4  BEGIN
  5     IF p_sal >= 10000 THEN
  6        RETURN 'A';
  7     ELSIF p_sal >= 6000 THEN
  8        RETURN 'B';
  9     END IF;                                   -- ELSE 경로가 비어 있다
 10  END grade_of;
 11  /

Function created.

-- 주목 : 경고 하나 없이 정상 생성되었다. 상태도 VALID 다.
SQL> SELECT object_name, status FROM user_objects WHERE object_name='GRADE_OF';

OBJECT_NAME       STATUS
----------------- -------
GRADE_OF          VALID

-- 급여가 높은 사원은 문제없이 동작한다.
SQL> SELECT grade_of(24000) AS g FROM dual;

G
----------
A

SQL> SELECT grade_of(7000) AS g FROM dual;

G
----------
B

-- 그러나 6000 미만이 들어오는 순간 실패한다.
SQL> SELECT grade_of(3000) AS g FROM dual;
SELECT grade_of(3000) AS g FROM dual
       *
ERROR at line 1:
ORA-06503: PL/SQL: Function returned without value
ORA-06512: at "HR.GRADE_OF", line 10

-- 실제 데이터로 호출하면 일부 행에서만 오류가 난다.
SQL> SELECT last_name, salary, grade_of(salary) AS g
  2    FROM employees WHERE department_id = 50 AND rownum <= 5;
SELECT last_name, salary, grade_of(salary) AS g
                          *
ERROR at line 1:
ORA-06503: PL/SQL: Function returned without value
ORA-06512: at "HR.GRADE_OF", line 10

-- [원인 파악]
--   PL/SQL 컴파일러는 함수 본문에 RETURN 문이 최소 하나 있는지만 검사하고
--   모든 실행 경로가 RETURN 에 도달하는지는 검사하지 않는다.
--   따라서 ELSE 경로가 비어 있어도 컴파일에 성공하며,
--   그 경로로 실행이 흘러간 순간 ORA-06503 이 발생한다.
--   테스트 데이터에서 재현되지 않다가 운영에서 처음 터지는 대표적인 결함이다.

-- 컴파일 경고를 켜면 미리 알려 주는 경우도 있다.
SQL> ALTER SESSION SET PLSQL_WARNINGS = 'ENABLE:ALL';

Session altered.

SQL> ALTER FUNCTION grade_of COMPILE;
SP2-0805: Function altered with compilation warnings

SQL> SHOW ERRORS
Errors for FUNCTION GRADE_OF:

LINE/COL ERROR
-------- -----------------------------------------------------------------
5/4      PLW-06002: Unreachable code
10/1     PLW-05005: subprogram GRADE_OF returns without value at line 10

-- PLW-05005 가 정확히 이 문제를 지적하고 있다.

-- [수정] 모든 분기를 닫고 NULL 도 별도로 처리한다.
SQL> CREATE OR REPLACE FUNCTION grade_of (p_sal IN NUMBER)
  2     RETURN VARCHAR2
  3  IS
  4  BEGIN
  5     IF    p_sal IS NULL   THEN RETURN 'N/A';  -- NULL 을 최하 등급으로 두지 않는다
  6     ELSIF p_sal >= 10000  THEN RETURN 'A';
  7     ELSIF p_sal >=  6000  THEN RETURN 'B';
  8     ELSE                       RETURN 'C';
  9     END IF;
 10  END grade_of;
 11  /

Function created.

SQL> SELECT last_name, salary, grade_of(salary) AS g
  2    FROM employees
  3   WHERE employee_id IN (100, 145, 178, 132)
  4   ORDER BY salary DESC;

------------ --------- ----------
King             24000 A
Russell          14000 A
Grant             7000 B
Olson             2100 C

SQL> SELECT grade_of(NULL) AS g FROM dual;

----------

[6] 프로시저의 RETURN 과 함수의 RETURN
--------------------------------------------------------------------------------

-- 프로시저에서도 RETURN 을 쓸 수 있다. 단 표현식 없이 '즉시 종료'의 의미다.
SQL> CREATE OR REPLACE PROCEDURE check_emp (p_emp_id IN NUMBER)
  2  IS
  3     v_cnt   PLS_INTEGER;
  4  BEGIN
  5     SELECT COUNT(*) INTO v_cnt
  6       FROM employees WHERE employee_id = p_emp_id;
  7     IF v_cnt = 0 THEN
  8        DBMS_OUTPUT.PUT_LINE('사번 없음 — 처리 중단');
  9        RETURN;                                -- 표현식 없는 RETURN
 10     END IF;
 11     DBMS_OUTPUT.PUT_LINE('처리 계속 : ' || get_full_name(p_emp_id));
 12  END check_emp;
 13  /

Procedure created.

SQL> EXEC check_emp(9999);
사번 없음 — 처리 중단

PL/SQL procedure successfully completed.

SQL> EXEC check_emp(100);
처리 계속 : Steven King

PL/SQL procedure successfully completed.

-- (오류 상황) 프로시저의 RETURN 에 표현식을 붙이면 컴파일되지 않는다.
SQL> CREATE OR REPLACE PROCEDURE bad_return
  2  IS
  3  BEGIN
  4     RETURN 1;
  5  END bad_return;
  6  /

Warning: Procedure created with compilation errors.

SQL> SHOW ERRORS
Errors for PROCEDURE BAD_RETURN:

LINE/COL ERROR
-------- -----------------------------------------------------------------
4/4      PLS-00372: In a procedure, RETURN statement cannot contain an
         expression

-- [원인 파악] 프로시저는 값을 반환하지 않는 서브프로그램이다.
-- [수정] 값을 돌려주려면 OUT 파라미터를 쓰거나 함수로 만들어야 한다.

SQL> DROP PROCEDURE bad_return;

Procedure dropped.

 실습 02 정리
--------------------------------------------------------------------------------
 · 함수는 CREATE OR REPLACE FUNCTION ... RETURN 자료형 IS ... 로 만든다.
   RETURN 자료형에 크기를 지정하면 PLS-00103 이 발생한다.
 · 함수 호출 방법은 로컬 변수 대입, 호스트 변수 대입, 표현식의 일부,
   SQL 문 안에서의 호출 네 가지다. 문장처럼 단독 호출하면 PLS-00221 이다.
 · 컴파일러는 RETURN 문의 존재만 검사하므로, 일부 경로가 RETURN 없이
   끝나면 컴파일은 성공하고 실행 중에 ORA-06503 이 발생한다.
 · PLSQL_WARNINGS 를 켜면 PLW-05005 경고로 미리 확인할 수 있다.
 · 프로시저의 RETURN 은 표현식 없이 즉시 종료를 의미하며,
   표현식을 붙이면 PLS-00372 가 발생한다.
 · USER_ARGUMENTS 에서 POSITION 0 인 행이 함수의 반환값이다.
