SQL> CREATE OR REPLACE PROCEDURE print_dept_salary
  2     (p_dept_id IN NUMBER)
  3  IS
  4     v_total   NUMBER;
  5     v_cnt     NUMBER;

  7     SELECT COUNT(*), NVL(SUM(salary), 0)

 13  END print_dept_salary;

[4] (오류 상황) IN 파라미터에 값을 대입하려 한다 — PLS-00363
--------------------------------------------------------------------------------

-- NULL 이 들어오면 기본값으로 바꿔 쓰려는 자연스러운 발상이 오류가 된다.
SQL> CREATE OR REPLACE PROCEDURE calc_pay
  2     (p_emp_id IN NUMBER,
  3      p_comm   IN NUMBER)
  4  IS
  5     v_sal   employees.salary%TYPE;
  6  BEGIN
  7     p_comm := NVL(p_comm, 0);                 -- IN 파라미터에 대입 시도
  8     SELECT salary INTO v_sal
  9       FROM employees WHERE employee_id = p_emp_id;
 10     DBMS_OUTPUT.PUT_LINE(v_sal * (1 + p_comm));
 11  END calc_pay;
 12  /

Warning: Procedure created with compilation errors.

-- 'created' 라고 나왔지만 실제로는 INVALID 상태다. 원인을 확인한다.
SQL> SHOW ERRORS
Errors for PROCEDURE CALC_PAY:

LINE/COL ERROR
-------- -----------------------------------------------------------------
7/4      PL/SQL: Statement ignored
7/4      PLS-00363: expression 'P_COMM' cannot be used as an assignment
         target

-- [원인 파악]
--   IN 파라미터는 서브프로그램 안에서 읽기 전용 상수처럼 동작한다.
--   대입문의 왼쪽이나 SELECT INTO 의 대상이 될 수 없다.
--   이 제약이 있기 때문에 IN 을 참조로 전달해도 안전하고,
--   실인자로 리터럴이나 표현식을 넘길 수 있는 것이다.

-- [수정] 보정이 필요한 값은 로컬 변수에 옮겨 담는다.
SQL> CREATE OR REPLACE PROCEDURE calc_pay
  2     (p_emp_id IN NUMBER,
  3      p_comm   IN NUMBER)
  4  IS
  5     v_sal    employees.salary%TYPE;
  6     v_comm   NUMBER := NVL(p_comm, 0);        -- 로컬 변수에서 보정한다
  7  BEGIN
  8     SELECT salary INTO v_sal
  9       FROM employees WHERE employee_id = p_emp_id;
 10     DBMS_OUTPUT.PUT_LINE('총액 : ' || v_sal * (1 + v_comm));
 11  EXCEPTION
 12     WHEN NO_DATA_FOUND THEN
 13        RAISE_APPLICATION_ERROR(-20001, '사번 ' || p_emp_id || ' 없음');
 14  END calc_pay;
 15  /

Procedure created.

PL/SQL procedure successfully completed.

SQL> EXEC calc_pay(145, .4);

PL/SQL procedure successfully completed.

SQL> EXEC calc_pay(9999, .1);
BEGIN calc_pay(9999, .1); END;

*

ORA-06512: at line 1

-- 예외 처리부가 업무 메시지로 바꿔 주므로 호출자가 원인을 알 수 있다.

[5] OUT 파라미터 — 한 번의 호출로 여러 값을 돌려받는다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE get_dept_stat
  2     (p_dept_id  IN  NUMBER,
  3      p_cnt      OUT NUMBER,
  4      p_sum      OUT NUMBER,
  5      p_avg      OUT NUMBER)
  6  IS
  7  BEGIN
  8     SELECT COUNT(*), NVL(SUM(salary), 0), NVL(ROUND(AVG(salary)), 0)
  9       INTO p_cnt, p_sum, p_avg
 10       FROM employees
 11      WHERE department_id = p_dept_id;
 12  END get_dept_stat;
 13  /

Procedure created.

SQL> DECLARE
  2     v_cnt   NUMBER;
  3     v_sum   NUMBER;
  4     v_avg   NUMBER;
  5  BEGIN
  6     get_dept_stat(60, v_cnt, v_sum, v_avg);
  7     DBMS_OUTPUT.PUT_LINE(v_cnt || '명 / 합계 ' || v_sum ||
  8                          ' / 평균 ' || v_avg);
  9  END;
 10  /
5명 / 합계 28800 / 평균 5760

PL/SQL procedure successfully completed.

-- 함수는 값을 하나만 반환하므로 세 번 호출해야 한다.
-- OUT 파라미터를 쓰면 테이블을 한 번만 읽고 세 값을 모두 얻는다.

-- SQL*Plus 의 호스트 변수로도 결과를 받을 수 있다.
SQL> VARIABLE b_cnt NUMBER
SQL> VARIABLE b_sum NUMBER
SQL> VARIABLE b_avg NUMBER
SQL> EXEC get_dept_stat(90, :b_cnt, :b_sum, :b_avg);

[6] (오류 상황) OUT 실인자로 리터럴을 넘긴다 — PLS-00363
--------------------------------------------------------------------------------

SQL> BEGIN
  2     get_dept_stat(60, 0, 0, 0);
  3  END;
  4  /
   get_dept_stat(60, 0, 0, 0);
                     *
ERROR at line 2:
ORA-06550: line 2, column 22:
PLS-00363: expression '0' cannot be used as an assignment target
ORA-06550: line 2, column 4:
PL/SQL: Statement ignored

-- [원인 파악]
--   OUT 파라미터는 값을 돌려받는 통로이므로 실인자가 '값을 담을 수 있는 것'
--   이어야 한다. 리터럴 0 은 저장 공간이 아니므로 대입 대상이 될 수 없다.
--   IN 파라미터에 대입할 때와 같은 오류 번호(PLS-00363)가 나오지만
--   문제가 되는 위치는 형식 파라미터가 아니라 실인자 쪽이다.

-- [수정] 값을 받을 변수를 선언해 넘긴다.
SQL> DECLARE
  2     v_cnt NUMBER;  v_sum NUMBER;  v_avg NUMBER;
  3  BEGIN
  4     get_dept_stat(60, v_cnt, v_sum, v_avg);
  5     DBMS_OUTPUT.PUT_LINE('정상 호출 : ' || v_cnt || '명');
  6  END;
  7  /
정상 호출 : 5명

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[7] (오류 상황) OUT 파라미터의 초기값 NULL 함정
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE count_up (p_cnt OUT NUMBER)
  2  IS
  3  BEGIN
  4     p_cnt := p_cnt + 1;            -- 대입 전에 참조한다
  5  END count_up;
  6  /

Procedure created.

-- 컴파일은 성공한다. 실행 결과를 보자.
SQL> DECLARE
  2     v_n   NUMBER := 10;
  3  BEGIN
  4     count_up(v_n);
  5     DBMS_OUTPUT.PUT_LINE('결과 : ' || NVL(TO_CHAR(v_n), 'NULL'));
  6  END;
  7  /
결과 : NULL

PL/SQL procedure successfully completed.

-- [원인 파악]
--   10 이 든 변수를 넘겼는데 결과가 NULL 이다.
--   NOCOPY 가 없는 OUT 파라미터는 호출 시점에 실인자 값과 무관하게
--   NULL 로 초기화된다. 따라서 안쪽에서 p_cnt 는 NULL 이고
--   NULL + 1 은 NULL 이므로 결과가 NULL 이 된다.
--   오류가 발생하지 않기 때문에 잘못된 값이 조용히 흘러가는 위험한 결함이다.

-- [수정] 입력값을 기반으로 갱신하려면 IN OUT 모드를 사용한다.
SQL> CREATE OR REPLACE PROCEDURE count_up (p_cnt IN OUT NUMBER)
  2  IS
  3  BEGIN
  4     p_cnt := NVL(p_cnt, 0) + 1;
  5  END count_up;
  6  /

Procedure created.

SQL> DECLARE
  2     v_n   NUMBER := 10;
  3  BEGIN
  4     count_up(v_n);
  5     count_up(v_n);
  6     DBMS_OUTPUT.PUT_LINE('결과 : ' || v_n);
  7  END;
  8  /
결과 : 12

PL/SQL procedure successfully completed.

[8] IN OUT 파라미터 — 받아서 가공한 뒤 같은 통로로 돌려준다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE format_phone (p_phone IN OUT VARCHAR2)
  2  IS
  3     v_digit   VARCHAR2(20);
  4  BEGIN
  5     v_digit := REGEXP_REPLACE(p_phone, '[^0-9]', '');
  6     IF LENGTH(v_digit) != 10 THEN
  7        RAISE_APPLICATION_ERROR(-20002,
  8           '전화번호는 숫자 10자리여야 합니다 : ' || p_phone);
  9     END IF;
 10     p_phone := '(' || SUBSTR(v_digit, 1, 3) || ') ' ||
 11                SUBSTR(v_digit, 4, 3) || '-' || SUBSTR(v_digit, 7, 4);
 12  END format_phone;
 13  /

Procedure created.

SQL> DECLARE
  2     v_phone   VARCHAR2(30) := '800.633.0575';
  3  BEGIN
  4     format_phone(v_phone);
  5     DBMS_OUTPUT.PUT_LINE(v_phone);
  6  END;
  7  /
(800) 633-0575

PL/SQL procedure successfully completed.

-- IN OUT 은 읽기와 쓰기가 모두 가능하다는 점을 이용한 전형적인 패턴이다.
-- IN 만으로도, OUT 만으로도 구현할 수 없다.

-- 자릿수가 맞지 않으면 업무 오류가 발생한다.
SQL> DECLARE
  2     v_phone   VARCHAR2(30) := '633-0575';
  3  BEGIN
  4     format_phone(v_phone);
  5     DBMS_OUTPUT.PUT_LINE(v_phone);
  6  END;
  7  /
DECLARE
*
ERROR at line 1:
ORA-20002: 전화번호는 숫자 10자리여야 합니다 : 633-0575
ORA-06512: at "HR.FORMAT_PHONE", line 7
ORA-06512: at line 4

-- 예외로 종료했으므로 IN OUT 파라미터의 값은 실인자로 되복사되지 않는다.
-- v_phone 은 여전히 '633-0575' 그대로다.

 실습 01 정리
--------------------------------------------------------------------------------
 · 저장 프로시저는 CREATE OR REPLACE PROCEDURE ... IS ... BEGIN ... END 로 만든다.
   선언부에 DECLARE 를 쓰지 않으며 파라미터 자료형에 크기를 지정할 수 없다.
 · 호출 방법은 EXECUTE, 블록 안의 문장, CALL 문 세 가지다.
 · IN 파라미터는 읽기 전용이므로 대입하면 PLS-00363 이 발생한다.
   보정이 필요하면 로컬 변수에 옮겨 담는다.
 · OUT 파라미터의 실인자는 변수여야 하며 리터럴을 넘기면 PLS-00363 이 발생한다.
 · OUT 파라미터는 호출 시 NULL 로 초기화되므로 대입 전에 참조하면 NULL 이다.
   입력값을 기반으로 갱신하려면 IN OUT 을 사용한다.
 · 예외로 종료하면 OUT · IN OUT 파라미터의 값은 실인자로 되복사되지 않는다.
