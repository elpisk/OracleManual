[2] DEFAULT 절로 기본값 지정하기
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE hire_log
  2     (p_emp_id   IN NUMBER,
  3      p_action   IN VARCHAR2 DEFAULT 'HIRE',   -- DEFAULT 키워드
  4      p_when     IN DATE     := SYSDATE)       -- := 도 동일한 의미
  5  IS
  6  BEGIN
  7     DBMS_OUTPUT.PUT_LINE(RPAD(p_emp_id, 6) || RPAD(p_action, 10) ||
  8                          TO_CHAR(p_when, 'YYYY-MM-DD'));
  9  END hire_log;
 10  /

Procedure created.

SQL> EXEC hire_log(206);

PL/SQL procedure successfully completed.

SQL> EXEC hire_log(206, 'REHIRE');

PL/SQL procedure successfully completed.

-- 가운데를 건너뛰려면 이름 표기법을 쓴다.

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
[3] DEFAULT 와 명시적 NULL 전달의 차이 (자주 혼동하는 지점)
--------------------------------------------------------------------------------

SQL> EXEC hire_log(206);
206   HIRE      2025-08-19

PL/SQL procedure successfully completed.

SQL> EXEC hire_log(206, NULL);
206             2025-08-19

PL/SQL procedure successfully completed.

-- [확인] 인자를 생략하면 'HIRE' 가 들어가지만
--        명시적으로 NULL 을 넘기면 기본값이 적용되지 않고 NULL 이 들어간다.
--        따라서 NULL 보정이 목적이라면 DEFAULT 만으로는 부족하고
--        로컬 변수에서 NVL 로 처리해야 한다.

SQL> CREATE OR REPLACE PROCEDURE hire_log
  2     (p_emp_id   IN NUMBER,
  3      p_action   IN VARCHAR2 DEFAULT 'HIRE',
  4      p_when     IN DATE     DEFAULT SYSDATE)
  5  IS
  6     v_action   VARCHAR2(20) := NVL(p_action, 'HIRE');   -- 로컬 변수에서 보정
  7     v_when     DATE         := NVL(p_when, SYSDATE);
  8  BEGIN
  9     DBMS_OUTPUT.PUT_LINE(RPAD(p_emp_id, 6) || RPAD(v_action, 10) ||
 10                          TO_CHAR(v_when, 'YYYY-MM-DD'));
 11  END hire_log;
 12  /

Procedure created.

SQL> EXEC hire_log(206, NULL, NULL);
206   HIRE      2025-08-19

PL/SQL procedure successfully completed.

-- 이제 NULL 을 넘겨도 안전하다.

[4] (오류 상황) OUT 파라미터에 DEFAULT 를 주려 한다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE bad_default (p_cnt OUT NUMBER DEFAULT 0)
  2  IS
  3  BEGIN
  4     p_cnt := 1;
  5  END bad_default;
  6  /

Warning: Procedure created with compilation errors.

SQL> SHOW ERRORS
Errors for PROCEDURE BAD_DEFAULT:

LINE/COL ERROR
-------- -----------------------------------------------------------------
1/45     PLS-00230: OUT and IN OUT modes cannot be used in this context

-- [원인 파악]
--   OUT 과 IN OUT 은 값을 돌려받는 통로이므로 호출자가 반드시 변수를
--   제공해야 한다. 생략할 수 없으니 기본값 개념 자체가 성립하지 않는다.
--   DEFAULT 는 IN 파라미터에만 지정할 수 있다.

SQL> DROP PROCEDURE bad_default;

Procedure dropped.


--------------------------------------------------------------------------------
[5] (오류 상황) 기본값 파라미터를 앞에 선언한다 — PLS-00306
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE bad_order
  2     (p_action IN VARCHAR2 DEFAULT 'HIRE',     -- 기본값이 앞에
  3      p_emp_id IN NUMBER)                      -- 필수가 뒤에
  4  IS
  5  BEGIN
  6     DBMS_OUTPUT.PUT_LINE(p_action || ' / ' || p_emp_id);
  7  END bad_order;
  8  /

Procedure created.

-- 주목 : 선언 자체는 정상적으로 컴파일된다. Oracle 은 이 순서를 금지하지 않는다.
SQL> SELECT object_name, status FROM user_objects WHERE object_name='BAD_ORDER';

OBJECT_NAME       STATUS
----------------- -------
BAD_ORDER         VALID

-- 문제는 호출할 때 드러난다.
SQL> EXEC bad_order(206);
BEGIN bad_order(206); END;

*
ERROR at line 1:
ORA-06550: line 1, column 7:
PLS-00306: wrong number or types of arguments in call to 'BAD_ORDER'
ORA-06550: line 1, column 7:
PL/SQL: Statement ignored

-- [원인 파악]
--   위치 표기법에서 206 은 첫 번째 파라미터인 p_action 에 대응된다.
--   그러면 기본값이 없는 필수 파라미터 p_emp_id 가 비게 되므로
--   인자 개수·타입이 맞지 않는다는 PLS-00306 이 발생한다.
--   게다가 206 은 NUMBER 인데 p_action 은 VARCHAR2 라 타입도 어긋난다.

-- [해결 방법 1] 이름 표기법으로 호출한다.
SQL> EXEC bad_order(p_emp_id => 206);
HIRE / 206

PL/SQL procedure successfully completed.

-- [해결 방법 2 · 권장] 선언 순서를 바로잡는다.
SQL> CREATE OR REPLACE PROCEDURE bad_order
  2     (p_emp_id IN NUMBER,                      -- 필수를 앞으로
  3      p_action IN VARCHAR2 DEFAULT 'HIRE')     -- 기본값을 뒤로
  4  IS
  5  BEGIN
  6     DBMS_OUTPUT.PUT_LINE(p_action || ' / ' || p_emp_id);
  7  END bad_order;
  8  /

Procedure created.

SQL> EXEC bad_order(206);
HIRE / 206

PL/SQL procedure successfully completed.

-- 관례 : 기본값 없는 필수 파라미터를 앞에, 기본값 있는 파라미터를 뒤에 모은다.

[6] (오류 상황) 이름 표기법 뒤에 위치 표기법이 온다 — PLS-00312
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE log_action
  2     (p_emp_id IN NUMBER,
  3      p_action IN VARCHAR2,
  4      p_note   IN VARCHAR2)
  5  IS
  6  BEGIN
  7     DBMS_OUTPUT.PUT_LINE(p_emp_id || ' / ' || p_action || ' / ' || p_note);
  8  END log_action;
  9  /

Procedure created.

-- 위치 표기법
SQL> EXEC log_action(178, 'UPDATE', '급여 인상');
178 / UPDATE / 급여 인상

PL/SQL procedure successfully completed.

-- 이름 표기법 — 순서를 바꿔도 결과가 같다
SQL> EXEC log_action(p_note => '급여 인상', p_emp_id => 178, -
>                    p_action => 'UPDATE');
178 / UPDATE / 급여 인상

PL/SQL procedure successfully completed.

-- 혼합 표기법 — 위치가 먼저, 이름이 나중
SQL> EXEC log_action(178, p_note => '급여 인상', p_action => 'UPDATE');
178 / UPDATE / 급여 인상

PL/SQL procedure successfully completed.

-- (오류) 이름 뒤에 위치가 왔다
SQL> EXEC log_action(p_emp_id => 178, 'UPDATE', '급여 인상');
BEGIN log_action(p_emp_id => 178, 'UPDATE', '급여 인상'); END;

*
ERROR at line 1:
ORA-06550: line 1, column 35:
PLS-00312: a positional parameter association may not follow a named
association
ORA-06550: line 1, column 7:
PL/SQL: Statement ignored

-- [원인 파악]
--   이름 표기법이 등장한 순간 컴파일러는 파라미터 순서 정보를 더 이상
--   신뢰할 수 없다. 그 뒤에 오는 위치 인자를 어디에 대응시킬지 결정할 수
--   없으므로 문법 자체가 금지되어 있다.

-- [수정] 위치 인자를 앞으로 옮기거나 전부 이름 표기법으로 통일한다.
SQL> EXEC log_action(178, p_action => 'UPDATE', p_note => '급여 인상');
178 / UPDATE / 급여 인상

PL/SQL procedure successfully completed.

[7] 로컬 변수의 생명주기 — 호출 사이에 값이 유지되지 않는다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE call_counter
  2  IS
  3     v_cnt   PLS_INTEGER := 0;                 -- 호출할 때마다 새로 만들어진다
  4  BEGIN
  5     v_cnt := v_cnt + 1;
  6     DBMS_OUTPUT.PUT_LINE('호출 횟수 : ' || v_cnt);
  7  END call_counter;
  8  /

Procedure created.

SQL> BEGIN
  2     call_counter;
  3     call_counter;
  4     call_counter;
  5  END;
  6  /
호출 횟수 : 1
호출 횟수 : 1
호출 횟수 : 1

PL/SQL procedure successfully completed.

-- [확인]
--   저장 프로시저는 데이터베이스에 저장되지만 저장되는 것은 '코드'일 뿐이다.
--   로컬 변수는 호출될 때 생성되고 종료될 때 소멸하므로 값이 누적되지 않는다.
--   세션 단위로 상태를 유지하려면 5장에서 배울 패키지 변수를 사용해야 한다.

[8] (오류 상황) 파라미터 이름이 컬럼 이름과 같다 — 삭제 사고 재현
--------------------------------------------------------------------------------

-- 실제 EMPLOYEES 를 건드리지 않도록 복사 테이블에서 실습한다.
SQL> CREATE TABLE emp_copy AS SELECT * FROM employees;

Table created.

SQL> SELECT COUNT(*) AS total FROM emp_copy;

     TOTAL
----------
       107

SQL> CREATE OR REPLACE PROCEDURE del_emp (last_name IN VARCHAR2)
  2  IS
  3  BEGIN
  4     DELETE FROM emp_copy WHERE last_name = last_name;
  5     DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || ' 건 삭제');
  6  END del_emp;
  7  /

Procedure created.

SQL> EXEC del_emp('King');
107 건 삭제

PL/SQL procedure successfully completed.

-- [원인 파악]
--   'King' 인 사원 2명만 지워지길 기대했으나 107건 전부가 삭제되었다.
--   SQL 문 안에서는 컬럼 이름이 파라미터 이름보다 우선하므로
--   WHERE last_name = last_name 은 컬럼끼리 비교하는 조건이 된다.
--   이 조건은 last_name 이 NULL 이 아닌 모든 행에서 참이다.
--   컴파일 오류도 실행 오류도 나지 않아 커밋 전까지 발견하기 어렵다.

SQL> ROLLBACK;

Rollback complete.

SQL> SELECT COUNT(*) AS total FROM emp_copy;

     TOTAL
----------
       107

-- [수정] 파라미터에 p_ 접두사를 붙여 이름 충돌을 없앤다.
SQL> CREATE OR REPLACE PROCEDURE del_emp (p_last_name IN VARCHAR2)
  2  IS
  3  BEGIN
  4     DELETE FROM emp_copy WHERE last_name = p_last_name;
  5     DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || ' 건 삭제');
  6  END del_emp;
  7  /

Procedure created.

SQL> EXEC del_emp('King');
2 건 삭제

PL/SQL procedure successfully completed.

SQL> ROLLBACK;

Rollback complete.

-- 참고 : 프로시저 이름을 한정자로 써서 del_emp.last_name 형태로 참조할 수도
--        있으나, 명명 규칙을 지키는 편이 훨씬 안전하고 읽기 쉽다.

 실습 03 정리
--------------------------------------------------------------------------------
 · DEFAULT 는 IN 파라미터에만 지정할 수 있으며 OUT · IN OUT 에 지정하면
   PLS-00230 이 발생한다.
 · 인자를 생략하면 기본값이 적용되지만 명시적으로 NULL 을 넘기면 적용되지
   않는다. NULL 보정이 필요하면 로컬 변수에서 NVL 로 처리한다.
 · 기본값 파라미터를 앞에 선언해도 컴파일은 되지만 위치 표기법 호출이
   불가능해져 PLS-00306 이 발생한다. 필수를 앞, 기본값을 뒤에 모은다.
 · 혼합 표기법은 위치가 먼저, 이름이 나중이어야 하며 어기면 PLS-00312 다.
 · 로컬 변수는 호출 단위로 생성·소멸하므로 값이 누적되지 않는다.
 · SQL 문 안에서는 컬럼 이름이 파라미터 이름보다 우선하므로 p_ 접두사를
   반드시 붙인다. 이를 어기면 조건이 항상 참이 되는 삭제 사고가 발생한다.
