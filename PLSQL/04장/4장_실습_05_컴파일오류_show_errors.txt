[2] (오류 상황 1) 컬럼명·변수명 오타 — ORA-00904 와 PLS-00201
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE show_emp (p_emp_id IN NUMBER)
  2  IS
  3     v_name   VARCHAR2(60);
  4  BEGIN
  5     SELECT first_nam || ' ' || last_name           -- first_name 오타
  6       INTO v_name
  7       FROM employees
  8      WHERE employee_id = p_emp_id;
  9     DBMS_OUTPUT.PUT_LINE(v_nam);                   -- v_name 오타
 10  END show_emp;
 11  /

Warning: Procedure created with compilation errors.

-- 'created' 라고 나왔지만 실행할 수 없는 상태다. 반드시 확인해야 한다.
SQL> SELECT object_name, object_type, status
  2    FROM user_objects WHERE object_name = 'SHOW_EMP';

OBJECT_NAME       OBJECT_TYPE  STATUS
----------------- ------------ -------
SHOW_EMP          PROCEDURE    INVALID

SQL> SHOW ERRORS
Errors for PROCEDURE SHOW_EMP:

LINE/COL ERROR
-------- -----------------------------------------------------------------
5/4      PL/SQL: SQL Statement ignored
5/11     PL/SQL: ORA-00904: "FIRST_NAM": invalid identifier
9/4      PL/SQL: Statement ignored
9/26     PLS-00201: identifier 'V_NAM' must be declared

-- [읽는 방법]
--   네 줄이 나왔지만 실제 원인은 두 개다.
--   'Statement ignored' 로 시작하는 5/4 와 9/4 는 원인이 아니라 결과다.
--   즉 5행의 SQL 문장이 무시된 것은 5/11 의 ORA-00904 때문이고,
--   9행의 문장이 무시된 것은 9/26 의 PLS-00201 때문이다.
--   줄 번호는 CREATE 문의 첫 줄을 1로 세는 상대 번호다.

-- [원인 파악]
--   ORA-00904 : SQL 문 안의 컬럼명이 잘못되었다 → 테이블 정의를 확인한다.
--   PLS-00201 : PL/SQL 식별자가 선언되지 않았다 → 오타, 스코프 밖 참조,
--               또는 객체 권한 없음 세 가지 가능성을 모두 검토한다.

SQL> DESCRIBE employees
 Name                    Null?    Type
 ----------------------- -------- --------------------

 FIRST_NAME                       VARCHAR2(20)
 LAST_NAME               NOT NULL VARCHAR2(25)
 EMAIL                   NOT NULL VARCHAR2(25)
 PHONE_NUMBER                     VARCHAR2(20)
 HIRE_DATE               NOT NULL DATE
 JOB_ID                  NOT NULL VARCHAR2(10)
 SALARY                           NUMBER(8,2)
 COMMISSION_PCT                   NUMBER(2,2)
 MANAGER_ID                       NUMBER(6)
 DEPARTMENT_ID                    NUMBER(4)

-- 컬럼 이름은 FIRST_NAME 이 맞다.

-- [수정]
SQL> CREATE OR REPLACE PROCEDURE show_emp (p_emp_id IN NUMBER)
  2  IS
  3     v_name   VARCHAR2(60);

[3] INVALID 객체를 호출하면 어떻게 되는가 — PLS-00905
--------------------------------------------------------------------------------

-- 일부러 다시 오류가 있는 상태로 만들어 호출해 본다.
SQL> CREATE OR REPLACE PROCEDURE show_emp (p_emp_id IN NUMBER)
  2  IS
  3  BEGIN
  4     DBMS_OUTPUT.PUT_LINE(v_undefined);        -- 선언하지 않은 식별자
  5  END show_emp;
  6  /

Warning: Procedure created with compilation errors.

SQL> EXEC show_emp(100);
BEGIN show_emp(100); END;

*
ERROR at line 1:
ORA-06550: line 1, column 7:
PLS-00905: object HR.SHOW_EMP is invalid
ORA-06550: line 1, column 7:
PL/SQL: Statement ignored

-- [원인 파악]
--   INVALID 상태의 객체는 호출할 수 없다.
--   오라클이 자동 재컴파일을 시도하지만 소스 자체에 오류가 있으므로
--   재컴파일도 실패하고 PLS-00905 가 그대로 호출자에게 전달된다.

SQL> SHOW ERRORS PROCEDURE show_emp
Errors for PROCEDURE SHOW_EMP:

LINE/COL ERROR
-------- -----------------------------------------------------------------
4/4      PL/SQL: Statement ignored
4/26     PLS-00201: identifier 'V_UNDEFINED' must be declared

-- SHOW ERRORS 뒤에 객체 종류와 이름을 지정하면 방금 컴파일한 것이 아니어도
-- 오류를 조회할 수 있다.

-- [수정] 앞의 정상 버전으로 되돌린다.
SQL> CREATE OR REPLACE PROCEDURE show_emp (p_emp_id IN NUMBER)
  2  IS
  3     v_name   VARCHAR2(60);
  4  BEGIN
  5     SELECT first_name || ' ' || last_name INTO v_name
  6       FROM employees WHERE employee_id = p_emp_id;
  7     DBMS_OUTPUT.PUT_LINE(v_name);
  8  EXCEPTION
  9     WHEN NO_DATA_FOUND THEN
 10        DBMS_OUTPUT.PUT_LINE('사번 ' || p_emp_id || ' : 조회 결과 없음');
 11  END show_emp;
 12  /

Procedure created.

[4] USER_ERRORS 를 직접 조회하기 (스크립트 자동 점검용)
--------------------------------------------------------------------------------

-- 여러 개의 오류 객체를 한꺼번에 만들어 놓고 조회해 본다.
SQL> CREATE OR REPLACE FUNCTION bad_fn1 RETURN NUMBER
  2  IS
  3  BEGIN
  4     RETURN v_none;
  5  END bad_fn1;
  6  /

Warning: Function created with compilation errors.

SQL> CREATE OR REPLACE PROCEDURE bad_prc1
  2  IS
  3  BEGIN
  4     SELECT no_such_column INTO v_x FROM employees;
  5  END bad_prc1;
  6  /

Warning: Procedure created with compilation errors.

SQL> SELECT name, type, line, position, text
  2    FROM user_errors
  3   ORDER BY name, sequence;

NAME       TYPE       LINE POSITION TEXT
---------- --------- ----- -------- ------------------------------------------
BAD_FN1    FUNCTION      4        4 PL/SQL: Statement ignored
BAD_FN1    FUNCTION      4       11 PLS-00201: identifier 'V_NONE' must be
                                    declared
BAD_PRC1   PROCEDURE     4        4 PL/SQL: SQL Statement ignored
BAD_PRC1   PROCEDURE     4       11 PL/SQL: ORA-00904: "NO_SUCH_COLUMN":
                                    invalid identifier

-- 첫 번째 오류만 뽑아 요약하는 점검 질의 (배포 스크립트에서 유용하다)
SQL> SELECT o.object_type, o.object_name,
  2         MIN(e.text) KEEP (DENSE_RANK FIRST ORDER BY e.sequence) AS first_err
  3    FROM user_objects o
  4    JOIN user_errors  e ON e.name = o.object_name AND e.type = o.object_type
  5   WHERE o.status = 'INVALID'
  6   GROUP BY o.object_type, o.object_name
  7   ORDER BY o.object_type, o.object_name;

OBJECT_TYPE  OBJECT_NAME  FIRST_ERR
------------ ------------ ---------------------------------------------------
FUNCTION     BAD_FN1      PL/SQL: Statement ignored
PROCEDURE    BAD_PRC1     PL/SQL: SQL Statement ignored

-- 소스 확인은 USER_SOURCE 로 한다.
SQL> SELECT line, text FROM user_source
  2   WHERE name = 'BAD_FN1' AND type = 'FUNCTION'
  3   ORDER BY line;

      LINE TEXT
---------- ---------------------------------------------------------------
         1 FUNCTION bad_fn1 RETURN NUMBER
         2 IS
         3 BEGIN
         4    RETURN v_none;
         5 END bad_fn1;

-- USER_SOURCE 에는 CREATE OR REPLACE 를 제외한 부분이 저장된다.
-- 따라서 여기의 LINE 1 이 SHOW ERRORS 의 LINE 1 과 일치한다.

[5] (오류 상황 2) 문법 오류 — PLS-00103
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE bad_syntax (p_name IN VARCHAR2(30))
  2  IS
  3  BEGIN
  4     DBMS_OUTPUT.PUT_LINE(p_name);
  5  END bad_syntax;
  6  /

Warning: Procedure created with compilation errors.

SQL> SHOW ERRORS
Errors for PROCEDURE BAD_SYNTAX:

LINE/COL ERROR
-------- -----------------------------------------------------------------
1/51     PLS-00103: Encountered the symbol "(" when expecting one of the
         following:  := . ) , @ % default character
         The symbol ":=" was substituted for "(" to continue.

-- [원인 파악]
--   형식 파라미터의 자료형에는 길이·정밀도를 지정할 수 없다.
--   VARCHAR2(30) 의 괄호에서 컴파일러가 예상하지 못한 기호를 만난 것이다.

-- [수정] 크기를 제거하거나 %TYPE 으로 선언한다.
SQL> CREATE OR REPLACE PROCEDURE bad_syntax
  2     (p_name IN employees.last_name%TYPE)
  3  IS
  4  BEGIN
  5     DBMS_OUTPUT.PUT_LINE(p_name);
  6  END bad_syntax;
  7  /

Procedure created.

SQL> EXEC bad_syntax('King');
King

PL/SQL procedure successfully completed.

[6] (오류 상황 3) 중첩 서브프로그램 선언 위치 위반 — PLS-00103
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE dept_report (p_dept_id IN NUMBER)
  2  IS
  3     FUNCTION fmt_money (p_amt IN NUMBER) RETURN VARCHAR2 IS
  4     BEGIN
  5        RETURN TO_CHAR(NVL(p_amt, 0), 'FM999,999,999') || ' USD';
  6     END fmt_money;
  7
  8     v_title   VARCHAR2(60);                   -- 중첩 함수 뒤에 변수 선언
  9  BEGIN
 10     DBMS_OUTPUT.PUT_LINE(fmt_money(1000));
 11  END dept_report;
 12  /

Warning: Procedure created with compilation errors.

SQL> SHOW ERRORS
Errors for PROCEDURE DEPT_REPORT:

LINE/COL ERROR
-------- -----------------------------------------------------------------
8/4      PLS-00103: Encountered the symbol "V_TITLE" when expecting one of
         the following:  begin function pragma procedure subtype type
         current cursor delete exists prior

-- [원인 파악]
--   중첩 서브프로그램 선언은 선언부의 가장 마지막에 와야 한다.
--   그 뒤에는 BEGIN 이나 또 다른 서브프로그램만 올 수 있다.
--   오류 메시지가 'begin function procedure ...' 를 기대했다고 알려 준다.

-- [수정] 변수를 먼저 선언하고 중첩 서브프로그램을 뒤에 둔다.
SQL> CREATE OR REPLACE PROCEDURE dept_report (p_dept_id IN NUMBER)
  2  IS
  3     v_title   VARCHAR2(60);                   -- 변수가 먼저
  4
  5     FUNCTION fmt_money (p_amt IN NUMBER) RETURN VARCHAR2 IS
  6     BEGIN
  7        RETURN TO_CHAR(NVL(p_amt, 0), 'FM999,999,999') || ' USD';
  8     END fmt_money;                            -- 중첩 함수가 나중
  9  BEGIN
 10     SELECT '부서 ' || department_name INTO v_title
 11       FROM departments WHERE department_id = p_dept_id;
 12     DBMS_OUTPUT.PUT_LINE(v_title);
 13     FOR r IN (SELECT last_name, salary FROM employees
 14                WHERE department_id = p_dept_id ORDER BY salary DESC) LOOP
 15        DBMS_OUTPUT.PUT_LINE('  ' || RPAD(r.last_name, 15) ||
 16                             fmt_money(r.salary));
 17     END LOOP;
 18  END dept_report;
 19  /

Procedure created.

SQL> EXEC dept_report(60);
부서 IT
  Hunold         9,000 USD
  Ernst          6,000 USD
  Austin         4,800 USD
  Pataballa      4,800 USD
  Lorentz        4,200 USD

PL/SQL procedure successfully completed.

================================================================================
 실습 05 정리
--------------------------------------------------------------------------------
 · 'Warning: Procedure created with compilation errors.' 는 성공이 아니다.
   객체는 만들어지지만 상태가 INVALID 이므로 호출할 수 없다.
 · SHOW ERRORS 는 방금 컴파일한 객체를, SHOW ERRORS PROCEDURE 이름 은
   지정한 객체의 오류를 보여 준다. 내부적으로 USER_ERRORS 를 조회한다.
 · 오류 목록에서 'Statement ignored' 는 결과이고 그 아래·위의 구체적인
   오류가 원인이다. 줄 번호는 CREATE 문 첫 줄을 1로 세는 상대 번호다.
 · PLS-00201 은 오타·스코프 밖 참조·권한 없음을 모두 포함한다.
   ORA-00904 가 섞여 나오면 SQL 문 안의 컬럼명 문제다.
 · 파라미터 자료형에 크기를 쓰거나 중첩 서브프로그램 뒤에 변수를 선언하면
   PLS-00103 이 발생한다.
 · INVALID 객체를 호출하면 PLS-00905: object is invalid 가 발생한다.
 · 배포 스크립트에는 SHOW ERRORS 와 INVALID 객체 개수 점검을 포함한다.
