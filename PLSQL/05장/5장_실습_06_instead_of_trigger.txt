--------------------------------------------------------------------------------

ERROR at line 2:
ORA-25002: cannot create INSTEAD OF triggers on tables

-- 원인) INSTEAD OF 트리거는 갱신 불가능한 뷰에 대한 DML 을 대신 처리하기 위한
--       장치다. 테이블은 이미 DML 이 가능하므로 대상이 될 수 없다.
-- 조치) 대상이 테이블이라면 BEFORE 또는 AFTER 시점의 일반 DML 트리거를 쓴다.
--       뷰에 대한 DML 이 목적이라면 뷰에 INSTEAD OF 트리거를 만든다.

[5] 뷰에 INSTEAD OF UPDATE 트리거를 작성한다

SQL> CREATE OR REPLACE TRIGGER trg_emp_dept_v_iou
  2     INSTEAD OF UPDATE ON emp_dept_v
  3     FOR EACH ROW
  4  BEGIN
  5     -- 사원 정보는 EMPLOYEES 로
  6     UPDATE employees
  7        SET last_name     = :NEW.last_name,
  8            email         = UPPER(:NEW.email),
  9            salary        = :NEW.salary,
 10            department_id = :NEW.department_id
 11      WHERE employee_id = :OLD.employee_id;
 12
 13     IF SQL%ROWCOUNT = 0 THEN
 14        RAISE_APPLICATION_ERROR(-20601, '사원을 찾을 수 없습니다');
 15     END IF;
 16
 17     -- 부서 이름이 실제로 바뀐 경우에만 DEPARTMENTS 로
 18     IF NVL(:OLD.department_name, '~') <> NVL(:NEW.department_name, '~') THEN
 19        UPDATE departments
 20           SET department_name = :NEW.department_name
 21         WHERE department_id = :NEW.department_id;
 22     END IF;
 23  END trg_emp_dept_v_iou;

[8] (오류 상황) 뮤테이팅 테이블 오류 재현과 복합 트리거 회피
--------------------------------------------------------------------------------

-- 부서 평균 급여의 2배를 넘지 못하게 하려는 행 트리거를 만들어 본다.
SQL> CREATE OR REPLACE TRIGGER trg_emp_br_avg
  2     BEFORE INSERT OR UPDATE OF salary ON employees
  3     FOR EACH ROW
  4  DECLARE
  5     v_avg NUMBER;
  6  BEGIN
  7     SELECT AVG(salary) INTO v_avg          -- 자기 테이블을 읽는다
  8       FROM employees WHERE department_id = :NEW.department_id;
  9     IF :NEW.salary > v_avg * 2 THEN
 10        RAISE_APPLICATION_ERROR(-20701, '부서 평균의 2배를 초과할 수 없습니다');
 11     END IF;
 12  END trg_emp_br_avg;
 13  /

Trigger TRG_EMP_BR_AVG compiled

-- 컴파일은 통과했다. 실행해 보자.
SQL> UPDATE employees SET salary = salary * 1.1 WHERE department_id = 60;
UPDATE employees SET salary = salary * 1.1 WHERE department_id = 60
       *
ERROR at line 1:
ORA-04091: table HR.EMPLOYEES is mutating, trigger/function may not see it
ORA-06512: at "HR.TRG_EMP_BR_AVG", line 4
ORA-04088: error during execution of trigger 'HR.TRG_EMP_BR_AVG'

-- 원인) 행 트리거는 문장이 진행되는 도중에 실행된다. 그 시점의 EMPLOYEES 는
--       일부 행만 바뀐 중간 상태(mutating)이므로 일관된 읽기를 보장할 수 없다.
--       그래서 오라클은 자기 테이블 조회를 금지한다.
-- 조치) 복합 트리거로 옮긴다. 행 시점에는 대상만 수집하고, 문장이 끝난 뒤
--       테이블이 안정된 AFTER STATEMENT 시점에서 조회와 검증을 수행한다.

SQL> DROP TRIGGER trg_emp_br_avg;

Trigger dropped.

SQL> CREATE OR REPLACE TRIGGER trg_emp_comp
  2     FOR INSERT OR UPDATE OF salary ON employees
  3     COMPOUND TRIGGER
  4
  5     TYPE t_dept_set IS TABLE OF PLS_INTEGER INDEX BY PLS_INTEGER;
  6     g_depts t_dept_set;              -- 인덱스를 부서번호로 써서 중복 제거
  7
  8     BEFORE STATEMENT IS
  9     BEGIN
 10        g_depts.DELETE;
 11     END BEFORE STATEMENT;
 12
 13     AFTER EACH ROW IS
 14     BEGIN
 15        IF :NEW.department_id IS NOT NULL THEN
 16           g_depts(:NEW.department_id) := 1;   -- 조회하지 않고 수집만 한다
 17        END IF;
 18     END AFTER EACH ROW;
 19
 20     AFTER STATEMENT IS
 21        v_avg NUMBER;
 22        v_max NUMBER;
 23        v_idx PLS_INTEGER;
 24     BEGIN
 25        v_idx := g_depts.FIRST;
 26        WHILE v_idx IS NOT NULL LOOP
 27           SELECT AVG(salary), MAX(salary) INTO v_avg, v_max
 28             FROM employees WHERE department_id = v_idx;
 29           IF v_max > v_avg * 2 THEN
 30              RAISE_APPLICATION_ERROR(-20701,
 31                 v_idx || '번 부서에 평균의 2배를 넘는 급여가 있습니다');
 32           END IF;
 33           v_idx := g_depts.NEXT(v_idx);
 34        END LOOP;
 35     END AFTER STATEMENT;
 36
 37  END trg_emp_comp;
 38  /

 정리
--------------------------------------------------------------------------------
 1. 조인 뷰에서 키 보존되지 않은 테이블의 컬럼은 갱신할 수 없다(ORA-01779).
    삭제도 정확히 하나의 키 보존 테이블이 없으면 ORA-01752 로 거부된다.
 2. INSTEAD OF 트리거는 뷰 전용이다. 테이블에 만들면 ORA-25002 가 발생한다.
 3. INSTEAD OF 트리거는 항상 행 수준이며 BEFORE/AFTER 를 지정할 수 없다.
 4. INSTEAD OF 트리거가 생기면 USER_UPDATABLE_COLUMNS 가 모두 YES 로 바뀐다.
    실제 동작의 책임은 트리거를 작성한 개발자에게 있다.
 5. 행 트리거에서 자기 테이블을 조회하면 ORA-04091 이 발생한다.
    복합 트리거에서 행 시점에는 수집만 하고 AFTER STATEMENT 에서 검증하면
    같은 로직을 안전하게 구현할 수 있다.
