[5] 본문을 올바르게 작성한다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PACKAGE BODY emp_pkg IS
  2
  3     -- Private 변수 : 명세에 없으므로 패키지 밖에서 보이지 않는다
  4     g_call_count NUMBER := 0;
  5
  6     -- Private 함수 : 직무 코드가 유효한지 확인하는 내부 보조 로직
  7     FUNCTION valid_job(p_job_id IN jobs.job_id%TYPE) RETURN BOOLEAN IS
  8        v_cnt NUMBER;
  9     BEGIN
 10        SELECT COUNT(*) INTO v_cnt FROM jobs WHERE job_id = p_job_id;
 11        RETURN (v_cnt > 0);
 12     END valid_job;
 13
 14     -- 명세에 선언한 dept_exists 구현 : 헤더가 명세와 정확히 같다
 15     FUNCTION dept_exists(p_dept_id IN departments.department_id%TYPE)
 16        RETURN BOOLEAN IS
 17        v_cnt NUMBER;
 18     BEGIN
 19        SELECT COUNT(*) INTO v_cnt
 20          FROM departments WHERE department_id = p_dept_id;
 21        RETURN (v_cnt > 0);
 22     END dept_exists;
 23
 24     PROCEDURE add_employee(p_last_name IN employees.last_name%TYPE,
 25                            p_email     IN employees.email%TYPE,
 26                            p_job_id    IN employees.job_id%TYPE,
 27                            p_salary    IN employees.salary%TYPE DEFAULT 3000,
 28                            p_dept_id   IN employees.department_id%TYPE) IS
 29     BEGIN
 30        -- 같은 패키지 안이므로 패키지 이름 없이 이름만으로 호출한다
 31        IF NOT dept_exists(p_dept_id) THEN
 32           RAISE e_invalid_dept;              -- 명세에 선언한 공개 예외
 33        END IF;
 34        IF NOT valid_job(p_job_id) THEN       -- Private 함수 호출
 35           RAISE_APPLICATION_ERROR(-20101, '존재하지 않는 직무 : ' || p_job_id);
 36        END IF;
 37
 38        INSERT INTO employees(employee_id, last_name, email, hire_date,
 39                              job_id, salary, department_id)
 40        VALUES (employees_seq.NEXTVAL, p_last_name, UPPER(p_email), SYSDATE,
 41                p_job_id, GREATEST(p_salary, c_min_salary), p_dept_id);
 42
 43        g_call_count := g_call_count + 1;
 44     END add_employee;
 45
 46     PROCEDURE raise_salary(p_emp_id IN employees.employee_id%TYPE,
 47                            p_pct    IN NUMBER) IS
 48     BEGIN
 49        IF p_pct > c_max_raise THEN
 50           RAISE_APPLICATION_ERROR(-20102, '인상률 상한을 초과했습니다');
 51        END IF;
 52        UPDATE employees SET salary = salary * (1 + p_pct)
 53         WHERE employee_id = p_emp_id;
 54        g_call_count := g_call_count + 1;
 55     END raise_salary;
 56
 57     FUNCTION get_annual_comp(p_emp_id IN employees.employee_id%TYPE)
 58        RETURN NUMBER IS
 59        v_sal  employees.salary%TYPE;
 60        v_comm employees.commission_pct%TYPE;
 61     BEGIN
 62        SELECT salary, NVL(commission_pct, 0) INTO v_sal, v_comm
 63          FROM employees WHERE employee_id = p_emp_id;
 64        RETURN v_sal * 12 * (1 + v_comm);
 65     EXCEPTION
 66        WHEN NO_DATA_FOUND THEN
 67           RETURN NULL;
 68     END get_annual_comp;
 69
 70  END emp_pkg;
 71  /

================================================================================
 정리
--------------------------------------------------------------------------------
 1. 패키지는 명세(PACKAGE)와 본문(PACKAGE BODY) 두 개의 객체로 저장된다.
 2. 명세 없이 본문만 컴파일하면 PLS-00304 로 실패한다. 반대는 정상이다.
 3. 본문의 헤더는 파라미터 이름까지 명세와 같아야 한다. 다르면 PLS-00323.
 4. 명세만 있는 상태에서 호출하면 ORA-04063 / ORA-06508 이 발생한다.
 5. 본문만 재컴파일하면 호출 객체는 VALID 를 유지하지만,
    명세를 재컴파일하면 의존 객체가 전부 INVALID 로 바뀐다.
