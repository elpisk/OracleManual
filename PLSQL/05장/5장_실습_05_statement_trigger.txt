-- 원인) :OLD 와 :NEW 는 행 트리거 전용 상관 식별자다.
--       FOR EACH ROW 가 없으면 바인드 변수 NEW 자체가 정의되지 않으므로
--       컴파일 단계에서 잘못된 바인드 변수라는 오류가 발생한다.
--       문장 하나가 여러 행을 다룰 수 있어 어느 행의 값인지 특정할 수 없기
--       때문에 오라클이 애초에 제공하지 않는 것이다.

-- 이 상태에서 실제로 INSERT 를 시도하면 트리거가 INVALID 라 문장까지 실패한다.

INSERT INTO employees(employee_id, last_name, email, hire_date,

[5] 조건 술어로 이벤트를 구분해 문장 단위 로그를 남긴다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE TRIGGER trg_emp_as_log
  2     AFTER INSERT OR UPDATE OR DELETE ON employees
  3  DECLARE
  4     v_event VARCHAR2(10);
  5     v_cnt   NUMBER;
  6  BEGIN
  7     IF INSERTING THEN v_event := 'INSERT';
  8     ELSIF UPDATING THEN v_event := 'UPDATE';
  9     ELSE v_event := 'DELETE';
 10     END IF;
 11
 12     -- 문장 트리거에서는 자기 테이블을 조회할 수 있다(뮤테이팅 아님)
 13     SELECT COUNT(*) INTO v_cnt FROM employees;
 14
 15     INSERT INTO emp_stmt_log(event_type, row_count, done_by, done_at)
 16     VALUES (v_event, v_cnt, USER, SYSTIMESTAMP);
 17
 18     DBMS_OUTPUT.PUT_LINE('>> ' || v_event || ' 완료, 현재 사원 수 = ' || v_cnt);
 19  END trg_emp_as_log;
 20  /

================================================================================
 정리
--------------------------------------------------------------------------------
 1. FOR EACH ROW 가 없으면 문장 트리거이며 트리거링 문장당 정확히 1회 실행된다.
 2. 변경된 행이 0건이어도 문장 트리거는 발화한다. 행 트리거는 발화하지 않는다.
 3. 문장 트리거에서 :NEW / :OLD 를 참조하면 PLS-00049 가 발생한다.
    INVALID 상태의 트리거가 걸린 테이블에 DML 을 하면 ORA-04098 이 나온다.
 4. BEFORE 문장 트리거에서 예외를 던지면 트리거링 문장 전체가 취소된다.
 5. 문장 트리거에서는 자기 테이블을 조회해도 뮤테이팅 오류가 발생하지 않는다.
