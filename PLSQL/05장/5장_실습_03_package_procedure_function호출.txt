SQL> CREATE OR REPLACE PACKAGE BODY hr_pkg IS
  2
  3     -- 전방 선언 : 아래 show_dept 가 먼저 이 프로시저를 호출하기 때문
  4     PROCEDURE print_line(p_text IN VARCHAR2);
  5
  6     -- Private 변수 : 초기화 블록에서 채운다
  7     g_max_dept NUMBER;
  8
  9     FUNCTION dept_exists(p_dept_id IN NUMBER) RETURN BOOLEAN IS
 10        v_cnt NUMBER;
 11     BEGIN
 12        SELECT COUNT(*) INTO v_cnt
 13          FROM departments WHERE department_id = p_dept_id;
 14        RETURN (v_cnt > 0);
 15     END dept_exists;
 16
 17     FUNCTION annual_comp(p_emp_id IN NUMBER) RETURN NUMBER IS
 18        v_sal  employees.salary%TYPE;
 19        v_comm employees.commission_pct%TYPE;
 20     BEGIN
 21        SELECT salary, NVL(commission_pct, 0) INTO v_sal, v_comm
 22          FROM employees WHERE employee_id = p_emp_id;
 23        RETURN v_sal * 12 * (1 + v_comm);
 24     EXCEPTION
 25        WHEN NO_DATA_FOUND THEN RETURN NULL;
 26     END annual_comp;
 27
 28     FUNCTION net_pay(p_emp_id IN NUMBER) RETURN NUMBER IS
 29     BEGIN
 30        RETURN ROUND(annual_comp(p_emp_id) * (1 - c_tax_rate));
 31     END net_pay;
 32
 33     PROCEDURE find_emp(p_emp_id IN employees.employee_id%TYPE) IS
 34        v_name employees.last_name%TYPE;
 35     BEGIN
 36        SELECT last_name INTO v_name
 37          FROM employees WHERE employee_id = p_emp_id;
 38        print_line('사번 조회 : ' || p_emp_id || ' -> ' || v_name);
 39     EXCEPTION
 40        WHEN NO_DATA_FOUND THEN
 41           print_line('사번 조회 : 해당 사원이 없습니다');
 42     END find_emp;
 43
 44     PROCEDURE find_emp(p_last_name IN employees.last_name%TYPE) IS
 45     BEGIN
 46        FOR r IN (SELECT employee_id, last_name FROM employees
 47                   WHERE last_name = p_last_name ORDER BY employee_id) LOOP
 48           print_line('이름 조회 : ' || r.last_name || ' -> ' || r.employee_id);
 49        END LOOP;
 50     END find_emp;
 51
 52     PROCEDURE show_dept(p_dept_id IN NUMBER) IS
 53        v_name departments.department_name%TYPE;
 54        v_cnt  NUMBER;
 55     BEGIN
 56        IF NOT dept_exists(p_dept_id) THEN
 57           RAISE_APPLICATION_ERROR(-20301, '존재하지 않는 부서입니다');
 58        END IF;
 59        SELECT department_name INTO v_name
 60          FROM departments WHERE department_id = p_dept_id;
 61        SELECT COUNT(*) INTO v_cnt
 62          FROM employees WHERE department_id = p_dept_id;
 63        print_line(p_dept_id || ' : ' || v_name || ' / 사원 ' || v_cnt || '명');
 64     END show_dept;
 65
 66     -- 전방 선언의 실제 구현
 67     PROCEDURE print_line(p_text IN VARCHAR2) IS
 68     BEGIN
 69        DBMS_OUTPUT.PUT_LINE('  ' || p_text);
 70     END print_line;
 71
 72  BEGIN                                    -- 초기화 블록 : 세션당 1회
 73     SELECT MAX(department_id) INTO g_max_dept FROM departments;
 74     DBMS_OUTPUT.PUT_LINE('[hr_pkg 초기화] 최대 부서번호 = ' || g_max_dept);
 75  EXCEPTION
 76     WHEN OTHERS THEN
 77        g_max_dept := 0;
 78  END hr_pkg;

-- 원인) BOOLEAN 은 PL/SQL 전용 자료형이고 SQL 에는 대응하는 자료형이 없다.
--       그래서 SQL 문에서는 BOOLEAN 을 반환하는 함수를 부를 수 없다.
-- 조치 1) PL/SQL 블록 안에서 호출한다.
SQL> BEGIN
  2     IF hr_pkg.dept_exists(60) THEN
  3        DBMS_OUTPUT.PUT_LINE('60번 부서 존재');
  4     END IF;
  5  END;
  6  /
60번 부서 존재

PL/SQL procedure successfully completed.

-- 조치 2) SQL 에서도 써야 한다면 숫자나 문자로 반환하는 래퍼를 만든다.
SQL> CREATE OR REPLACE FUNCTION dept_exists_n(p_dept_id IN NUMBER)
  2     RETURN NUMBER IS
  3  BEGIN
  4     IF hr_pkg.dept_exists(p_dept_id) THEN RETURN 1; ELSE RETURN 0; END IF;

================================================================================
 정리
--------------------------------------------------------------------------------
 1. 외부 호출은 패키지이름.요소이름 점 표기법을 쓴다.
 2. 함수는 SELECT 문 안에서도 호출할 수 있지만 BOOLEAN 반환 함수는 불가능하다
    (ORA-06553 / PLS-382). 숫자나 문자를 반환하는 래퍼로 우회한다.
 3. 오버로딩은 실인자의 자료형으로 구분된다. USER_ARGUMENTS 의 OVERLOAD 열에서
    확인할 수 있다.
 4. 같은 계열의 자료형으로 오버로딩하면 컴파일은 되지만 호출 시 PLS-00307 이
    발생한다. 서로 다른 계열을 쓰거나 이름을 달리해야 한다.
 5. 초기화 블록은 세션이 패키지를 처음 참조할 때 한 번만 실행된다.
 6. 본문에서 정의보다 앞선 호출이 필요하면 전방 선언(헤더만 먼저 선언)을 쓴다.
