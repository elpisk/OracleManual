 제5장 실습 04 — BEFORE / AFTER Row Trigger 와 :OLD / :NEW 활용
--------------------------------------------------------------------------------
 대상 스키마 : HR (EMPLOYEES / DEPARTMENTS / JOBS / JOB_HISTORY /
                   LOCATIONS / COUNTRIES / REGIONS)
 실행 환경   : Oracle Database 19c (19.3) / SQL*Plus
 학습 목표   : 1) BEFORE 행 트리거로 입력값을 정규화하고 기본값을 보정한다
               2) AFTER 행 트리거로 변경 이력을 감사 테이블에 기록한다
               3) 이벤트별로 :OLD / :NEW 중 어느 쪽이 NULL 인지 확인한다
               4) AFTER 트리거에서 :NEW 를 수정하려다 나는 오류를 재현한다
               5) WHEN 절과 UPDATE OF 절로 발화 조건을 좁힌다

[2] BEFORE 행 트리거 — 입력값 정규화와 기본값 보정
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE TRIGGER trg_emp_biu_normalize
  2     BEFORE INSERT OR UPDATE ON employees
  3     FOR EACH ROW
  4  BEGIN
  5     -- 1) 이메일은 항상 대문자, 앞뒤 공백 제거
  6     :NEW.email := UPPER(TRIM(:NEW.email));
  7
  8     -- 2) 입사일이 비어 있으면 오늘 날짜로 채운다
  9     IF :NEW.hire_date IS NULL THEN
 10        :NEW.hire_date := TRUNC(SYSDATE);
 11     END IF;
 12
 13     -- 3) 급여 하한 보정
 14     IF :NEW.salary IS NULL OR :NEW.salary < 2000 THEN
 15        :NEW.salary := 2000;
 16     END IF;
 17  END trg_emp_biu_normalize;
 18  /

[4] AFTER 행 트리거 — 급여 변경 이력 기록
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE TRIGGER trg_emp_ar_salary
  2     AFTER UPDATE OF salary ON employees
  3     FOR EACH ROW
  4     WHEN (NVL(OLD.salary,-1) <> NVL(NEW.salary,-1))   -- 콜론을 붙이지 않는다
  5  BEGIN
  6     INSERT INTO emp_audit_detail(employee_id, event_type, column_name,
  7            old_value, new_value, changed_by, changed_at)
  8     VALUES (:OLD.employee_id, 'UPDATE', 'SALARY',
  9             TO_CHAR(:OLD.salary), TO_CHAR(:NEW.salary),
 10             USER, SYSTIMESTAMP);
 11  END trg_emp_ar_salary;

[6] 이벤트별로 :OLD / :NEW 값을 눈으로 확인한다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE TRIGGER trg_emp_ar_show
  2     AFTER INSERT OR UPDATE OR DELETE ON employees
  3     FOR EACH ROW
  4  DECLARE
  5     v_event VARCHAR2(10);
  6  BEGIN
  7     IF INSERTING THEN v_event := 'INSERT';
  8     ELSIF UPDATING THEN v_event := 'UPDATE';
  9     ELSE v_event := 'DELETE';
 10     END IF;
 11
 12     DBMS_OUTPUT.PUT_LINE(
 13        RPAD(v_event, 8) ||
 14        ' OLD.id=' || NVL(TO_CHAR(:OLD.employee_id), '(NULL)') ||
 15        ' OLD.sal=' || NVL(TO_CHAR(:OLD.salary), '(NULL)') ||
 16        ' | NEW.id=' || NVL(TO_CHAR(:NEW.employee_id), '(NULL)') ||
 17        ' NEW.sal=' || NVL(TO_CHAR(:NEW.salary), '(NULL)'));
 18  END trg_emp_ar_show;

[7] (오류 상황) WHEN 절 안에 콜론을 붙여 본다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE TRIGGER trg_emp_when_bad
  2     AFTER UPDATE OF salary ON employees
  3     FOR EACH ROW
  4     WHEN (:NEW.salary > 10000)
  5  BEGIN
  6     NULL;
  7  END;
  8  /
    WHEN (:NEW.salary > 10000)
          *
ERROR at line 4:
ORA-04079: invalid trigger specification

-- 원인) WHEN 절은 트리거 본문이 아니라 트리거 헤더의 일부다.
--       바인드 변수 표기인 콜론을 여기서는 쓸 수 없다.
-- 조치) 콜론을 빼고 NEW.salary 로 쓴다.

SQL> CREATE OR REPLACE TRIGGER trg_emp_when_ok
  2     AFTER UPDATE OF salary ON employees
  3     FOR EACH ROW
  4     WHEN (NEW.salary > 10000)
  5  BEGIN
  6     DBMS_OUTPUT.PUT_LINE('고액 급여 변경 : ' || :NEW.employee_id
  7                          || ' -> ' || :NEW.salary);
  8  END;
  9  /

================================================================================
 정리
--------------------------------------------------------------------------------
 1. FOR EACH ROW 를 붙이면 영향받는 행마다 트리거 본문이 실행된다.
 2. :NEW 에 값을 대입할 수 있는 곳은 BEFORE 행 트리거뿐이다.
    AFTER 에서 대입하면 PLS-00363, 실행 시점에는 ORA-04084 가 발생한다.
 3. INSERT 는 :OLD 가 전부 NULL, DELETE 는 :NEW 가 전부 NULL 이다.
 4. WHEN 절은 본문 실행 여부를, UPDATE OF 절은 트리거 발화 여부를 결정한다.
 5. WHEN 절에서는 콜론을 붙이지 않는다. 붙이면 ORA-04079 가 발생한다.
