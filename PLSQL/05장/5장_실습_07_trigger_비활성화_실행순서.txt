SQL> CREATE OR REPLACE TRIGGER trg_emp_br_validate
  2     BEFORE INSERT ON employees
  3     FOR EACH ROW
  4  BEGIN
  5     IF :NEW.salary IS NULL THEN
  6        RAISE_APPLICATION_ERROR(-20801, '급여는 필수입니다');
  7     END IF;
  8     DBMS_OUTPUT.PUT_LINE('   [검증] 급여 확인 완료');
  9  END trg_emp_br_validate;
 10  /

Trigger TRG_EMP_BR_VALIDATE compiled

SQL> CREATE OR REPLACE TRIGGER trg_emp_br_default
  2     BEFORE INSERT ON employees
  3     FOR EACH ROW
  4     FOLLOWS trg_emp_br_validate        -- 검증 트리거 이후에 실행
  5  BEGIN
  6     :NEW.email := UPPER(:NEW.email);
  7     DBMS_OUTPUT.PUT_LINE('   [보정] 이메일 대문자 변환');
  8  END trg_emp_br_default;
  9  /

[6] 안전한 작업 패턴 — 실패해도 반드시 다시 켜지도록 감싼다

-- 대량 적재 작업을 흉내 내되, 실패해도 트리거가 켜지도록 익명 블록으로 감싼다.
SQL> DECLARE
  2     PROCEDURE set_trg(p_name IN VARCHAR2, p_state IN VARCHAR2) IS
  3     BEGIN
  4        EXECUTE IMMEDIATE 'ALTER TRIGGER ' || p_name || ' ' || p_state;
  5     END set_trg;
  6  BEGIN
  7     set_trg('trg_emp_ar_salary', 'DISABLE');
  8     DBMS_OUTPUT.PUT_LINE('트리거 비활성화, 적재 시작');
  9
 10     BEGIN
 11        -- 여기서 일부러 오류를 낸다(적재 중 실패 상황 재현)
 12        UPDATE employees SET salary = 'ABC' WHERE department_id = 60;
 13     EXCEPTION
 14        WHEN OTHERS THEN
 15           DBMS_OUTPUT.PUT_LINE('적재 실패 : ' || SQLERRM);
 16     END;
 17
 18     set_trg('trg_emp_ar_salary', 'ENABLE');
 19     DBMS_OUTPUT.PUT_LINE('트리거 재활성화 완료');
 20  EXCEPTION
 21     WHEN OTHERS THEN
 22        set_trg('trg_emp_ar_salary', 'ENABLE');   -- 어떤 경우에도 다시 켠다
 23        RAISE;
 24  END;
 25  /
트리거 비활성화, 적재 시작
적재 실패 : ORA-01722: invalid number
트리거 재활성화 완료

 정리
--------------------------------------------------------------------------------
 1. 실행 순서는 BEFORE 문장 1회 → (BEFORE 행 → 변경 → AFTER 행) 반복 →
    AFTER 문장 1회다. 변경된 행이 없으면 문장 트리거만 발화한다.
 2. 같은 시점·같은 이벤트의 트리거는 순서가 정의되지 않으므로 FOLLOWS 또는
    PRECEDES 로 명시해야 한다.
 3. ALTER TRIGGER ... DISABLE / ENABLE 로 개별 트리거를,
    ALTER TABLE ... DISABLE ALL TRIGGERS 로 테이블 단위 일괄 처리를 한다.
 4. 이름을 잘못 적으면 ORA-04080 이 발생한다. USER_TRIGGERS 로 확인한다.
 5. 트리거를 끈 채 작업하고 다시 켜지 않으면 감사 이력이 누락되고 되돌릴 수
    없다. 작업을 익명 블록으로 감싸 EXCEPTION 절에서도 반드시 ENABLE 하고,
    STATUS = 'DISABLED' 인 트리거를 매일 점검해야 한다.
 6. CREATE TRIGGER ... DISABLE 로 처음부터 꺼진 트리거를 만들 수 있다.
