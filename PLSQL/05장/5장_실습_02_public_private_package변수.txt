                   LOCATIONS / COUNTRIES / REGIONS)

 학습 목표   : 1) 명세에 없는 Private 요소를 외부에서 부르면 PLS-00302 가 난다
               2) Private 요소를 공개 인터페이스로 감싸 사용하는 방법을 익힌다
               3) 패키지 변수가 세션 동안 값을 유지하는 것을 확인한다
               4) 패키지 변수가 세션 간에 공유되지 않는 것을 두 세션으로 확인한다
               5) 재컴파일 시 ORA-04068 이 발생하는 상황을 재현한다

[2] Public 과 Private 를 나누어 패키지를 만든다
--------------------------------------------------------------------------------

-- 명세에는 외부가 알아야 하는 것만 선언한다.
SQL> CREATE OR REPLACE PACKAGE sal_pkg IS
  2
  3     -- Public 상수
  4     c_max_raise CONSTANT NUMBER := 0.30;
  5
  6     -- Public 프로시저 / 함수
  7     PROCEDURE raise_salary(p_emp_id IN employees.employee_id%TYPE,
  8                            p_pct    IN NUMBER);
  9     FUNCTION  raise_count  RETURN NUMBER;
 10     PROCEDURE reset_count;
 11
 12  END sal_pkg;
 13  /

Package SAL_PKG compiled

-- 본문에는 명세의 구현과 함께 Private 요소를 둔다.
SQL> CREATE OR REPLACE PACKAGE BODY sal_pkg IS
  2
  3     -- Private 패키지 변수 : 세션 단위로 값이 유지된다
  4     g_raise_count NUMBER := 0;
  5     g_last_emp_id NUMBER;
  6
  7     -- Private 함수 : 명세에 없으므로 외부에서 호출할 수 없다
  8     FUNCTION valid_pct(p_pct IN NUMBER) RETURN BOOLEAN IS
  9     BEGIN
 10        RETURN (p_pct IS NOT NULL AND p_pct > 0 AND p_pct <= c_max_raise);
 11     END valid_pct;
 12
 13     -- Private 프로시저 : 내부 기록용
 14     PROCEDURE mark(p_emp_id IN NUMBER) IS
 15     BEGIN
 16        g_raise_count := g_raise_count + 1;
 17        g_last_emp_id := p_emp_id;
 18     END mark;
 19
 20     PROCEDURE raise_salary(p_emp_id IN employees.employee_id%TYPE,
 21                            p_pct    IN NUMBER) IS
 22     BEGIN
 23        IF NOT valid_pct(p_pct) THEN          -- Private 함수 호출
 24           RAISE_APPLICATION_ERROR(-20201,
 25              '인상률은 0 초과 ' || c_max_raise || ' 이하여야 합니다');
 26        END IF;
 27        UPDATE employees SET salary = salary * (1 + p_pct)
 28         WHERE employee_id = p_emp_id;
 29        IF SQL%ROWCOUNT = 0 THEN
 30           RAISE_APPLICATION_ERROR(-20202, '없는 사번입니다 : ' || p_emp_id);
 31        END IF;
 32        mark(p_emp_id);                       -- Private 프로시저 호출
 33     END raise_salary;
 34
 35     FUNCTION raise_count RETURN NUMBER IS
 36     BEGIN
 37        RETURN g_raise_count;
 38     END raise_count;
 39
 40     PROCEDURE reset_count IS
 41     BEGIN
 42        g_raise_count := 0;
 43        g_last_emp_id := NULL;
 44     END reset_count;
 45
 46  END sal_pkg;
 47  /

Package Body SAL_PKG compiled

[3] (오류 상황) Private 요소를 외부에서 호출해 본다 — PLS-00302
--------------------------------------------------------------------------------

SQL> BEGIN
  2     IF sal_pkg.valid_pct(0.10) THEN
  3        DBMS_OUTPUT.PUT_LINE('유효한 인상률');
  4     END IF;
  5  END;
  6  /
BEGIN
*
ERROR at line 1:
ORA-06550: line 2, column 16:
PLS-00302: component 'VALID_PCT' must be declared
ORA-06550: line 2, column 7:
PL/SQL: Statement ignored

-- 원인) valid_pct 는 본문에만 있는 Private 함수다. 명세에 없으므로 외부에서
--       그 이름을 볼 수 없다. 오라클은 "그런 구성 요소가 선언되어 있지 않다"
--       라고 알려 준다.

-- Private 변수도 마찬가지다.
SQL> EXEC DBMS_OUTPUT.PUT_LINE(sal_pkg.g_raise_count)

BEGIN DBMS_OUTPUT.PUT_LINE(sal_pkg.g_raise_count); END;

*
ERROR at line 1:
ORA-06550: line 1, column 34:
PLS-00302: component 'G_RAISE_COUNT' must be declared

-- 조치 1) 캡슐화를 유지한다 — 공개 함수 raise_count 를 통해 값을 읽는다.
SQL> EXEC DBMS_OUTPUT.PUT_LINE('현재 누적 : ' || sal_pkg.raise_count)
현재 누적 : 0

PL/SQL procedure successfully completed.

-- 조치 2) 정말 공개가 필요하면 명세에 헤더를 추가한다.
--         단, 명세가 바뀌면 그 패키지를 참조하는 모든 객체가 INVALID 가 된다.
--         이번 실습에서는 캡슐화를 유지하는 조치 1을 택한다.

[4] 패키지 변수가 세션 동안 값을 유지하는 것을 확인한다
--------------------------------------------------------------------------------

SQL> SELECT employee_id, last_name, salary

누적 : 1

누적 : 3

-- 서로 다른 익명 블록에서 호출했는데도 값이 계속 누적되었다.
-- 로컬 변수였다면 호출이 끝날 때마다 사라졌을 것이다.

--------------------------------------------------------------------------------
[5] Private 함수의 검증이 동작하는지 확인한다
--------------------------------------------------------------------------------

SQL> EXEC sal_pkg.raise_salary(103, 0.50)

BEGIN sal_pkg.raise_salary(103, 0.50); END;

*
ERROR at line 1:
ORA-20201: 인상률은 0 초과 .3 이하여야 합니다
ORA-06512: at "HR.SAL_PKG", line 24
ORA-06512: at line 1

-- 상한을 넘는 인상률은 Private 함수 valid_pct 가 막았다.
-- 호출자는 valid_pct 의 존재를 알 필요가 없다. 이것이 캡슐화의 이점이다.

SQL> EXEC DBMS_OUTPUT.PUT_LINE('누적 : ' || sal_pkg.raise_count)
누적 : 3

-- 실패한 호출은 mark 까지 도달하지 못했으므로 카운터가 올라가지 않았다.


--------------------------------------------------------------------------------
[6] 두 번째 세션에서 같은 패키지 변수를 조회한다 — 세션 간 비공유
--------------------------------------------------------------------------------

-- 새 터미널을 열어 두 번째 세션으로 접속한다.

$ sqlplus hr/hr@localhost:1521/orclpdb

SQL> SET SERVEROUTPUT ON

SQL> SELECT SYS_CONTEXT('USERENV','SID') AS my_sid FROM dual;

MY_SID
------
   304

-- 첫 번째 세션(SID 271)에서는 누적이 3 이었다.
SQL> EXEC DBMS_OUTPUT.PUT_LINE('누적 : ' || sal_pkg.raise_count)
누적 : 0

PL/SQL procedure successfully completed.

-- 확인) 같은 패키지, 같은 변수인데 값이 0 이다.
--       패키지 변수는 UGA(User Global Area)에 세션마다 별도의 사본으로
--       저장되기 때문이다. 세션을 넘는 공유는 일어나지 않는다.
--       세션 간 공유가 필요하면 테이블이나 RESULT_CACHE 를 써야 한다.

SQL> EXEC sal_pkg.raise_salary(206, 0.03)

PL/SQL procedure successfully completed.

SQL> EXEC DBMS_OUTPUT.PUT_LINE('두 번째 세션 누적 : ' || sal_pkg.raise_count)
두 번째 세션 누적 : 1

SQL> COMMIT;

Commit complete.

-- 다시 첫 번째 세션(SID 271)으로 돌아온다.
SQL> EXEC DBMS_OUTPUT.PUT_LINE('첫 번째 세션 누적 : ' || sal_pkg.raise_count)
첫 번째 세션 누적 : 3

-- 두 세션의 값이 서로 영향을 주지 않는다.


--------------------------------------------------------------------------------
[7] (오류 상황) 재컴파일로 상태가 무효화된다 — ORA-04068
--------------------------------------------------------------------------------

-- 첫 번째 세션은 sal_pkg 를 이미 사용 중이고 g_raise_count 는 3 이다.
-- 이 상태에서 다른 세션(또는 배포 스크립트)이 패키지를 재컴파일했다고 하자.

SQL> ALTER PACKAGE sal_pkg COMPILE BODY;

Package body altered.

-- 재컴파일 직후 첫 번째 세션에서 다시 호출해 본다.
SQL> EXEC DBMS_OUTPUT.PUT_LINE('누적 : ' || sal_pkg.raise_count)

BEGIN DBMS_OUTPUT.PUT_LINE('누적 : ' || sal_pkg.raise_count); END;

*
ERROR at line 1:
ORA-04068: existing state of packages has been discarded
ORA-04061: existing state of package body "HR.SAL_PKG" has been invalidated
ORA-06512: at line 1

-- 원인) 재컴파일로 이 세션이 들고 있던 패키지 상태가 무효화되었다.
--       오라클은 낡은 상태를 그대로 쓰지 못하게 하려고 한 번 오류를 던진다.
-- 조치) 그 시점에 상태가 초기화되므로 같은 문장을 다시 실행하면 성공한다.

SQL> EXEC DBMS_OUTPUT.PUT_LINE('누적 : ' || sal_pkg.raise_count)
누적 : 0

PL/SQL procedure successfully completed.

-- 확인) 재시도는 성공했지만 누적값 3 은 사라지고 0 으로 초기화되었다.
--       운영 애플리케이션은 이 오류를 만나면 한 번 재시도하도록 설계하고,
--       중요한 상태는 패키지 변수가 아니라 테이블에 보관해야 한다.

================================================================================
 정리
--------------------------------------------------------------------------------
 1. 명세에 없는 Private 요소를 외부에서 부르면 PLS-00302 가 발생한다.
 2. 캡슐화를 유지하려면 공개 인터페이스를 통해 간접적으로 사용한다.
    명세에 헤더를 추가하는 방법은 의존 객체 전면 INVALID 라는 대가가 있다.
 3. 패키지 변수는 세션이 끝날 때까지 값을 유지한다(Persistent State).
 4. 패키지 상태는 UGA 에 세션별로 저장되므로 세션 간에 공유되지 않는다.
 5. 재컴파일로 상태가 무효화되면 다음 호출에서 ORA-04068 이 한 번 발생하고
    그 시점에 상태가 초기화된다. 재시도하면 성공하지만 값은 사라진다.
