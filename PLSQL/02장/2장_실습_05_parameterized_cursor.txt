[2] 파라미터 없이 외부 변수를 참조하는 커서 — 동작은 하지만 함정이 있다
--------------------------------------------------------------------------------

-- 커서 질의는 블록의 지역 변수를 그대로 참조할 수 있다.
-- 다만 그 값이 "언제" 반영되는지가 문제다.

SQL> DECLARE
  2     v_dept   employees.department_id%TYPE := 60;
  3
  4     CURSOR c_emp IS
  5        SELECT last_name
  6          FROM employees
  7         WHERE department_id = v_dept       -- 외부 변수 참조
  8         ORDER BY employee_id;
  9
 10     v_name   employees.last_name%TYPE;
 11  BEGIN
 12     OPEN c_emp;                            -- 이 시점에 v_dept = 60 으로 확정
 13     v_dept := 90;                          -- OPEN 이후에 값을 바꿔 본다
 14
 15     LOOP
 16        FETCH c_emp INTO v_name;
 17        EXIT WHEN c_emp%NOTFOUND;
 18        DBMS_OUTPUT.PUT_LINE(v_name);
 19     END LOOP;
 20     CLOSE c_emp;
 21
 22     DBMS_OUTPUT.PUT_LINE('블록 종료 시 v_dept = ' || v_dept);
 23  END;

-- ▶ 확인 사항

[3] 파라미터 커서 — 값을 OPEN에서 명시적으로 넘긴다
--------------------------------------------------------------------------------

-- CURSOR 커서명(파라미터명 데이터타입) IS SELECT …
-- 파라미터는 커서 질의 안에서만 쓰이는 지역 이름이며, 모드는 항상 IN이다.

SQL> DECLARE
  2     CURSOR c_emp_by_dept(p_dept_id NUMBER) IS
  3        SELECT employee_id, last_name, salary
  4          FROM employees
  5         WHERE department_id = p_dept_id
  6         ORDER BY employee_id;
  7
  8     v_row   c_emp_by_dept%ROWTYPE;
  9  BEGIN
 10     -- 첫 번째 사용 : 부서 60
 11     OPEN c_emp_by_dept(60);
 12     LOOP
 13        FETCH c_emp_by_dept INTO v_row;
 14        EXIT WHEN c_emp_by_dept%NOTFOUND;
 15        DBMS_OUTPUT.PUT_LINE('[60] ' || v_row.employee_id || ' ' || v_row.last_name);
 16     END LOOP;
 17     CLOSE c_emp_by_dept;

 23        EXIT WHEN c_emp_by_dept%NOTFOUND;
 24        DBMS_OUTPUT.PUT_LINE('[90] ' || v_row.employee_id || ' ' || v_row.last_name);
 25     END LOOP;

[4] Cursor FOR LOOP에 파라미터 넘기기

-- FOR 레코드 IN 커서명(인자) LOOP 형태로 그대로 쓴다.
-- OPEN/CLOSE가 없으니 인자 전달 위치는 FOR 절 한 곳뿐이다.

SQL> DECLARE
  2     CURSOR c_emp_by_dept(p_dept_id NUMBER) IS
  3        SELECT last_name, salary
  4          FROM employees
  5         WHERE department_id = p_dept_id
  6         ORDER BY salary DESC;
  7
  8     v_tot   NUMBER;
  9  BEGIN
 10     FOR d IN (SELECT department_id, department_name
 11                 FROM departments
 12                WHERE department_id IN (30, 60, 90)
 13                ORDER BY department_id) LOOP
 14
 15        v_tot := 0;

[5] 파라미터 기본값(DEFAULT)
--------------------------------------------------------------------------------

-- DEFAULT 또는 := 로 기본값을 준다. 기본값이 있는 파라미터는 호출 시 생략 가능하다.
-- 기본값 없는 파라미터를 기본값 있는 파라미터 뒤에 두면 위치 표기 호출이
-- 곤란해지므로, 기본값 있는 것을 뒤쪽에 모아 선언한다.

SQL> DECLARE
  2     CURSOR c_emp(p_dept_id NUMBER,
  3                  p_min_sal NUMBER DEFAULT 0) IS
  4        SELECT last_name, salary
  5          FROM employees
  6         WHERE department_id = p_dept_id
  7           AND salary >= p_min_sal
  8         ORDER BY salary DESC;
  9  BEGIN
 10     DBMS_OUTPUT.PUT_LINE('--- 부서 30, 하한 생략(기본 0) ---');
 11     FOR r IN c_emp(30) LOOP
 12        DBMS_OUTPUT.PUT_LINE(RPAD(r.last_name, 12) || LPAD(r.salary, 6));
 13     END LOOP;
 14
 15     DBMS_OUTPUT.PUT_LINE('--- 부서 30, 하한 2800 ---');
 16     FOR r IN c_emp(30, 2800) LOOP
 17        DBMS_OUTPUT.PUT_LINE(RPAD(r.last_name, 12) || LPAD(r.salary, 6));
 18     END LOOP;
 19
 20     DBMS_OUTPUT.PUT_LINE('--- 이름 지정 표기 ---');
 21     FOR r IN c_emp(p_min_sal => 9000, p_dept_id => 60) LOOP
 22        DBMS_OUTPUT.PUT_LINE(RPAD(r.last_name, 12) || LPAD(r.salary, 6));
 23     END LOOP;
 24  END;

[8] (오류 재현 3) 인자 타입 불일치 — ORA-06502
--------------------------------------------------------------------------------

-- 부서를 이름으로 넘기려다 숫자 파라미터에 문자열을 전달한 경우다.
-- 컴파일은 통과한다. 문자열 → 숫자 변환은 런타임에 시도되기 때문이다.

SQL> DECLARE
  2     CURSOR c_emp(p_dept_id NUMBER) IS
  3        SELECT last_name FROM employees WHERE department_id = p_dept_id;
  4
  5     v_name   employees.last_name%TYPE;
  6     v_input  VARCHAR2(30) := 'IT';    -- 사용자가 부서명을 입력했다고 가정
  7  BEGIN
  8     OPEN c_emp(v_input);
  9     FETCH c_emp INTO v_name;
 10     DBMS_OUTPUT.PUT_LINE(v_name);
 11     CLOSE c_emp;
 12  END;
 13  /
DECLARE
*
ERROR at line 1:
ORA-06502: PL/SQL: numeric or value error: character to number conversion error
ORA-06512: at line 8

-- ▶ 원인 파악
--    p_dept_id는 NUMBER다. 전달된 'IT'를 암묵적으로 숫자로 바꾸려다 실패했다.
--    'IT' 대신 '60'을 넘겼다면 암묵 변환이 성공해 오류 없이 동작했을 것이다.
--    즉 이 오류는 "입력값에 따라 났다 안 났다" 하는 유형이라 더 위험하다.

-- ▶ 수정 : 부서명을 받고 싶다면 파라미터 타입 자체를 문자열로 두고
--          커서 질의에서 조인으로 해결한다.

SQL> DECLARE
  2     CURSOR c_emp(p_dept_name departments.department_name%TYPE) IS
  3        SELECT e.last_name, e.salary
  4          FROM employees e
  5          JOIN departments d ON e.department_id = d.department_id
  6         WHERE d.department_name = p_dept_name
  7         ORDER BY e.employee_id;
  8
  9     v_input  departments.department_name%TYPE := 'IT';
 10  BEGIN
 11     FOR r IN c_emp(v_input) LOOP
 12        DBMS_OUTPUT.PUT_LINE(RPAD(r.last_name, 12) || LPAD(r.salary, 6));
 13     END LOOP;
 14  END;

[9] 실무 적용 — 하나의 파라미터 커서로 부서별 급여 등급 집계
--------------------------------------------------------------------------------

SQL> DECLARE
  2     CURSOR c_emp(p_dept_id NUMBER) IS
  3        SELECT last_name, salary
  4          FROM employees
  5         WHERE department_id = p_dept_id;
  6
  7     v_high   NUMBER;
  8     v_mid    NUMBER;
  9     v_low    NUMBER;
 10  BEGIN
 11     FOR d IN (SELECT department_id, department_name
 12                 FROM departments
 13                WHERE department_id IN (30, 60, 90)
 14                ORDER BY department_id) LOOP
 15
 16        v_high := 0;  v_mid := 0;  v_low := 0;
 17
 18        FOR e IN c_emp(d.department_id) LOOP
 19           IF    e.salary >= 10000 THEN v_high := v_high + 1;
 20           ELSIF e.salary >=  5000 THEN v_mid  := v_mid  + 1;
 21           ELSE                         v_low  := v_low  + 1;
 22           END IF;
 23        END LOOP;
 24
 25        DBMS_OUTPUT.PUT_LINE(RPAD(d.department_name, 14) ||
 26                             ' 상위 ' || v_high ||

[10] 정리
--------------------------------------------------------------------------------

-- 1) 커서 질의의 변수 값은 OPEN 시점에 확정된다. OPEN 이후 변수를 바꿔도
--    결과 집합은 변하지 않는다.
-- 2) 파라미터를 쓰면 "어떤 값으로 도는지"가 호출부에 드러나고, 하나의 커서를
--    여러 값으로 재사용할 수 있다.
-- 3) 커서 파라미터의 모드는 항상 IN이며 OUT/IN OUT은 없다.
-- 4) 파라미터 타입에 길이·정밀도를 붙일 수 없다(PLS-00103). %TYPE을 쓴다.
-- 5) 기본값 없는 파라미터를 생략하면 PLS-00306, 타입이 맞지 않으면
--    런타임에 ORA-06502다. 후자는 입력값에 따라 간헐적으로 나므로 더 위험하다.
-- 6) DEFAULT는 뒤쪽 파라미터에 몰아 두고, 인자가 3개를 넘으면 이름 지정
--    표기(p_name => 값)를 쓰는 편이 안전하다.

SQL> EXIT
