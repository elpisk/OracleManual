[2] 왜 명시적 커서가 필요한가 — SELECT INTO의 한계 확인
--------------------------------------------------------------------------------

-- SELECT INTO는 정확히 1행을 반환할 때만 성공한다.
-- 여러 행이 나오면 TOO_MANY_ROWS 예외가 발생한다.

SQL> DECLARE
  2     v_name   employees.last_name%TYPE;
  3  BEGIN
  4     SELECT last_name INTO v_name
  5       FROM employees
  6      WHERE department_id = 60;
  7
  8     DBMS_OUTPUT.PUT_LINE(v_name);
  9  END;
 10  /
DECLARE
*
ERROR at line 1:
ORA-01422: exact fetch returns more than requested number of rows
ORA-06512: at line 4

-- ▶ 여러 행을 한 건씩 처리하려면 결과 집합에 이름을 붙이고
--    한 행씩 꺼내 쓰는 장치가 필요하다. 그것이 명시적 커서다.


--------------------------------------------------------------------------------
[3] 명시적 커서 4단계 — DECLARE / OPEN / FETCH / CLOSE
--------------------------------------------------------------------------------

-- DECLARE : 질의에 이름을 붙인다. 이 시점에는 아직 실행되지 않는다.
-- OPEN    : 질의를 실행하고 결과 집합을 확정한다. 커서 포인터는 첫 행 앞에 선다.
-- FETCH   : 현재 행을 변수에 담고 포인터를 한 칸 내린다.
-- CLOSE   : 결과 집합을 해제한다. 이후 FETCH는 불가능하다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name, salary
  4          FROM employees
  5         WHERE department_id = 60
  6         ORDER BY employee_id;
  7
  8     v_empno   employees.employee_id%TYPE;
  9     v_name    employees.last_name%TYPE;
 10     v_sal     employees.salary%TYPE;
 11  BEGIN
 12     OPEN c_emp;
 13
 14     LOOP
 15        FETCH c_emp INTO v_empno, v_name, v_sal;
 16        EXIT WHEN c_emp%NOTFOUND;
 17
 18        DBMS_OUTPUT.PUT_LINE(v_empno || ' ' || RPAD(v_name, 12) || v_sal);
 19     END LOOP;
 20
 21     CLOSE c_emp;
 22  END;
 23  /

104 Ernst       6000
105 Austin      4800
106 Pataballa   4800
107 Lorentz     4200

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    FETCH INTO의 변수 개수·순서·타입은 SELECT 목록과 정확히 일치해야 한다.
--    개수가 다르면 PLS-00394(wrong number of values in the INTO list)가 난다.


--------------------------------------------------------------------------------
[4] %ROWTYPE 레코드로 받기 — 변수 나열을 없앤다
--------------------------------------------------------------------------------

-- 커서 이름%ROWTYPE은 그 커서의 SELECT 목록 구조를 그대로 가진 레코드 타입이다.
-- 컬럼이 늘거나 순서가 바뀌어도 FETCH 문을 고칠 필요가 없다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name, salary, hire_date
  4          FROM employees
  5         WHERE department_id = 60
  6         ORDER BY salary DESC;
  7
  8     v_row   c_emp%ROWTYPE;           -- 커서 구조를 그대로 복사
  9  BEGIN
 10     OPEN c_emp;
 11
 12     LOOP
 13        FETCH c_emp INTO v_row;
 14        EXIT WHEN c_emp%NOTFOUND;
 15
 16        DBMS_OUTPUT.PUT_LINE(RPAD(v_row.last_name, 12) ||
 17                             LPAD(v_row.salary, 6)     || '  ' ||
 18                             TO_CHAR(v_row.hire_date, 'YYYY-MM-DD'));
 19     END LOOP;
 20
 21     DBMS_OUTPUT.PUT_LINE('처리 건수 = ' || c_emp%ROWCOUNT);
 22     CLOSE c_emp;
 23  END;
 24  /
Hunold        9000  2006-01-03
Ernst         6000  2007-05-21
Austin        4800  2005-06-25
Pataballa     4800  2006-02-05
Lorentz       4200  2007-02-07

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    %ROWCOUNT는 "지금까지 FETCH에 성공한 행 수"다. 마지막에 실패한 FETCH는
--    세지 않으므로 5가 나온다. CLOSE 전에 읽어야 한다는 점을 기억한다.


--------------------------------------------------------------------------------
[5] 커서 하나를 여러 번 열어 쓰기 — 반드시 CLOSE 후 재OPEN
--------------------------------------------------------------------------------

-- 같은 커서를 조건만 바꿔 반복 사용할 수 있다. 단, OPEN → 처리 → CLOSE의
-- 한 사이클이 끝난 뒤에 다시 OPEN해야 한다.

SQL> DECLARE
  2     CURSOR c_dept IS
  3        SELECT department_id, department_name
  4          FROM departments
  5         WHERE department_id IN (10, 20)
  6         ORDER BY department_id;
  7
  8     v_row   c_dept%ROWTYPE;
  9  BEGIN
 10     FOR pass IN 1 .. 2 LOOP
 11        DBMS_OUTPUT.PUT_LINE('--- ' || pass || '회차 ---');
 12
 13        OPEN c_dept;
 14        LOOP
 15           FETCH c_dept INTO v_row;
 16           EXIT WHEN c_dept%NOTFOUND;
 17           DBMS_OUTPUT.PUT_LINE(v_row.department_id || ' ' || v_row.department_name);
 18        END LOOP;
 19        CLOSE c_dept;                 -- 다음 OPEN을 위해 반드시 닫는다
 20     END LOOP;
 21  END;
 22  /
--- 1회차 ---
10 Administration
20 Marketing
--- 2회차 ---
10 Administration
20 Marketing

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[6] (오류 재현 1) CLOSE 없이 다시 OPEN — ORA-06511
--------------------------------------------------------------------------------

-- 위 블록에서 CLOSE 한 줄만 빼 보면 두 번째 OPEN에서 바로 실패한다.
-- 문법에는 문제가 없으므로 컴파일은 통과하고 런타임에 터진다.

SQL> DECLARE
  2     CURSOR c_dept IS
  3        SELECT department_id, department_name
  4          FROM departments
  5         WHERE department_id IN (10, 20)
  6         ORDER BY department_id;
  7
  8     v_row   c_dept%ROWTYPE;
  9  BEGIN
 10     FOR pass IN 1 .. 2 LOOP
 11        DBMS_OUTPUT.PUT_LINE('--- ' || pass || '회차 ---');
 12
 13        OPEN c_dept;
 14        LOOP
 15           FETCH c_dept INTO v_row;
 16           EXIT WHEN c_dept%NOTFOUND;
 17           DBMS_OUTPUT.PUT_LINE(v_row.department_id || ' ' || v_row.department_name);
 18        END LOOP;
 19        -- CLOSE c_dept;              ← 실수로 빠뜨렸다
 20     END LOOP;
 21  END;
 22  /
--- 1회차 ---
10 Administration
20 Marketing
--- 2회차 ---
DECLARE
*
ERROR at line 1:
ORA-06511: PL/SQL: cursor already open
ORA-06512: at line 13

-- ▶ 원인 파악
--    1회차 종료 시점에 c_dept는 여전히 열린 상태다. 결과 집합을 다 읽어
--    %NOTFOUND가 TRUE가 되었다고 해서 커서가 자동으로 닫히지는 않는다.
--    그 상태에서 13행의 OPEN이 다시 실행되어 ORA-06511이 발생했다.
--    "다 읽으면 닫힌다"는 오해가 이 오류의 사실상 유일한 원인이다.

-- ▶ 수정 (A) : 사이클마다 CLOSE 한다(= [5]의 코드).

-- ▶ 수정 (B) : 열려 있는지 %ISOPEN으로 확인한 뒤 방어적으로 닫고 연다.

SQL> DECLARE
  2     CURSOR c_dept IS
  3        SELECT department_id, department_name
  4          FROM departments
  5         WHERE department_id IN (10, 20)
  6         ORDER BY department_id;
  7
  8     v_row   c_dept%ROWTYPE;
  9  BEGIN
 10     FOR pass IN 1 .. 2 LOOP
 11        IF c_dept%ISOPEN THEN         -- 방어 코드
 12           CLOSE c_dept;
 13        END IF;
 14
 15        OPEN c_dept;
 16        LOOP
 17           FETCH c_dept INTO v_row;
 18           EXIT WHEN c_dept%NOTFOUND;
 19           DBMS_OUTPUT.PUT_LINE(pass || '회차 : ' || v_row.department_name);
 20        END LOOP;
 21        CLOSE c_dept;
 22     END LOOP;
 23  END;
 24  /
1회차 : Administration
1회차 : Marketing
2회차 : Administration
2회차 : Marketing

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[7] (오류 재현 2) 닫힌 커서에 FETCH — ORA-01001
--------------------------------------------------------------------------------

-- CLOSE를 루프 안에 잘못 넣으면 첫 행을 처리한 직후 커서가 닫히고,
-- 두 번째 FETCH에서 유효하지 않은 커서 오류가 난다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name
  4          FROM employees
  5         WHERE department_id = 60
  6         ORDER BY employee_id;
  7
  8     v_row   c_emp%ROWTYPE;
  9  BEGIN
 10     OPEN c_emp;
 11
 12     LOOP
 13        FETCH c_emp INTO v_row;
 14        EXIT WHEN c_emp%NOTFOUND;
 15
 16        DBMS_OUTPUT.PUT_LINE(v_row.employee_id || ' ' || v_row.last_name);
 17        CLOSE c_emp;                  -- 들여쓰기에 속아 루프 안에 들어갔다
 18     END LOOP;
 19  END;
 20  /
103 Hunold
DECLARE
*
ERROR at line 1:
ORA-01001: invalid cursor
ORA-06512: at line 13

-- ▶ 원인 파악
--    17행의 CLOSE가 END LOOP 안쪽에 있다. 첫 행 출력 직후 커서가 닫히고
--    루프는 계속 돌아 13행의 FETCH가 다시 실행되었다. 닫힌 커서에 대한
--    FETCH·CLOSE·속성 참조는 모두 ORA-01001로 거부된다.

-- ▶ 수정 : CLOSE를 END LOOP 밖으로 옮기고, 예외가 나더라도 반드시 닫히도록
--          EXCEPTION 절에서도 정리한다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name
  4          FROM employees
  5         WHERE department_id = 60
  6         ORDER BY employee_id;
  7
  8     v_row   c_emp%ROWTYPE;
  9  BEGIN
 10     OPEN c_emp;
 11
 12     LOOP
 13        FETCH c_emp INTO v_row;
 14        EXIT WHEN c_emp%NOTFOUND;
 15        DBMS_OUTPUT.PUT_LINE(v_row.employee_id || ' ' || v_row.last_name);
 16     END LOOP;
 17
 18     CLOSE c_emp;                     -- 루프 밖
 19  EXCEPTION
 20     WHEN OTHERS THEN
 21        IF c_emp%ISOPEN THEN
 22           CLOSE c_emp;               -- 예외 경로에서도 자원 회수
 23        END IF;
 24        RAISE;
 25  END;
 26  /
103 Hunold
104 Ernst
105 Austin
106 Pataballa
107 Lorentz

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[8] 커서를 닫지 않으면 무엇이 남는가 — 세션 열린 커서 확인
--------------------------------------------------------------------------------

-- 열린 커서는 세션 자원이다. 상한은 OPEN_CURSORS 파라미터로 정해지고,
-- 넘어서면 ORA-01000(maximum open cursors exceeded)이 발생한다.

SQL> SELECT value FROM v$parameter WHERE name = 'open_cursors';

VALUE
--------------------
300

-- ▶ PL/SQL 블록이 끝나면 그 블록에서 선언한 커서는 정리되지만,
--    패키지 전역 커서나 REF CURSOR는 세션이 끝날 때까지 남을 수 있다.
--    "연 것은 반드시 닫는다"를 습관으로 만든다.


--------------------------------------------------------------------------------
[9] 정리
--------------------------------------------------------------------------------

-- 1) 여러 행을 한 건씩 처리해야 하면 SELECT INTO가 아니라 명시적 커서다.
--    SELECT INTO는 0행이면 ORA-01403, 2행 이상이면 ORA-01422다.
-- 2) DECLARE는 이름만 붙이고, 질의는 OPEN 시점에 실행된다.
-- 3) FETCH 직후 %NOTFOUND를 검사해 EXIT한다. 순서를 바꾸면 마지막 행이
--    한 번 더 처리된다(실습 06에서 상세히 확인).
-- 4) 결과를 다 읽어도 커서는 자동으로 닫히지 않는다.
--    닫지 않고 다시 열면 ORA-06511, 닫은 뒤 만지면 ORA-01001이다.
-- 5) 변수 나열 대신 커서명%ROWTYPE을 쓰면 SELECT 목록 변경에 강해진다.
-- 6) 예외 처리 절에서도 %ISOPEN을 확인해 CLOSE하는 패턴을 기본으로 삼는다.

SQL> EXIT
