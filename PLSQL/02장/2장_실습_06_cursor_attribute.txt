-- ▶ 네 속성 요약
--    %ISOPEN   : 커서가 열려 있으면 TRUE. BOOLEAN.
--    %FOUND    : 직전 FETCH가 행을 가져왔으면 TRUE.

[2] %ISOPEN — OPEN 전 / OPEN 후 / CLOSE 후

[3] %FOUND / %NOTFOUND / %ROWCOUNT를 한 걸음씩 추적
--------------------------------------------------------------------------------

-- FETCH를 한 번씩 하면서 매번 세 속성을 찍어 본다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT last_name FROM employees WHERE department_id = 20
  4         ORDER BY employee_id;
  5
  6     v_name   employees.last_name%TYPE;
  7
  8     FUNCTION b2c(p_flag BOOLEAN) RETURN VARCHAR2 IS
  9     BEGIN
 10        RETURN CASE WHEN p_flag IS NULL THEN 'NULL'
 11                    WHEN p_flag         THEN 'TRUE'
 12                    ELSE                     'FALSE'
 13               END;
 14     END;
 15
 16     PROCEDURE trace(p_step VARCHAR2) IS
 17     BEGIN
 18        DBMS_OUTPUT.PUT_LINE(RPAD(p_step, 14) ||
 19           ' %FOUND='    || RPAD(b2c(c_emp%FOUND), 6) ||
 20           ' %NOTFOUND=' || RPAD(b2c(c_emp%NOTFOUND), 6) ||
 21           ' %ROWCOUNT=' || c_emp%ROWCOUNT);
 22     END;
 23  BEGIN
 24     OPEN c_emp;
 25     trace('OPEN 직후');
 26
 27     FETCH c_emp INTO v_name;
 28     trace('FETCH 1회');
 29
 30     FETCH c_emp INTO v_name;
 31     trace('FETCH 2회');
 32
 33     FETCH c_emp INTO v_name;
 34     trace('FETCH 3회');
 35
 36     CLOSE c_emp;
 37  END;
 38  /
OPEN 직후       %FOUND=NULL   %NOTFOUND=NULL   %ROWCOUNT=0
FETCH 1회       %FOUND=TRUE   %NOTFOUND=FALSE  %ROWCOUNT=1
FETCH 2회       %FOUND=TRUE   %NOTFOUND=FALSE  %ROWCOUNT=2
FETCH 3회       %FOUND=FALSE  %NOTFOUND=TRUE   %ROWCOUNT=2

PL/SQL procedure successfully completed.

-- ▶ 확인 사항 세 가지
--    1) OPEN 직후 %FOUND/%NOTFOUND는 FALSE가 아니라 NULL이다.
--       아직 FETCH를 한 적이 없으므로 "가져왔다/못 가져왔다"를 말할 수 없다.
--    2) %ROWCOUNT는 OPEN 직후 0이고, 실패한 FETCH에서는 증가하지 않는다.
--       3회 FETCH 후에도 2로 남는다.
--    3) 실패한 FETCH는 변수 값을 바꾸지 않는다. v_name에는 여전히 2번째 행의
--       값(Fay)이 남아 있다. 다음 절의 논리 오류가 여기서 비롯된다.

[4] (오류 재현 1) %NOTFOUND 검사 위치를 FETCH 앞에 두어 마지막 행이 중복 처리
--------------------------------------------------------------------------------

-- 컴파일도 되고 실행도 되며 예외도 나지 않는다.
-- 다만 결과가 한 줄 더 나온다. 실무에서 가장 잦은 커서 버그다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name, salary
  4          FROM employees
  5         WHERE department_id = 20
  6         ORDER BY employee_id;
  7
  8     v_row   c_emp%ROWTYPE;
  9     v_tot   NUMBER := 0;
 10  BEGIN
 11     OPEN c_emp;
 12
 13     LOOP
 14        EXIT WHEN c_emp%NOTFOUND;      -- FETCH 하기 전에 검사했다
 15        FETCH c_emp INTO v_row;
 16
 17        v_tot := v_tot + v_row.salary;
 18        DBMS_OUTPUT.PUT_LINE(v_row.employee_id || ' ' ||
 19                             RPAD(v_row.last_name, 12) || LPAD(v_row.salary, 6));
 20     END LOOP;
 21
 22     DBMS_OUTPUT.PUT_LINE('출력 합계 = ' || v_tot ||
 23                          ' / 실제 FETCH 건수 = ' || c_emp%ROWCOUNT);
 24     CLOSE c_emp;
 25  END;
 26  /
201 Hartstein     13000
202 Fay            6000
202 Fay            6000
출력 합계 = 25000 / 실제 FETCH 건수 = 2

PL/SQL procedure successfully completed.

-- ▶ 원인 파악
--    루프의 흐름을 그대로 따라가면 이렇다.
--      1회전 : %NOTFOUND = NULL → EXIT 안 함 → FETCH 성공(201) → 출력
--      2회전 : %NOTFOUND = FALSE → FETCH 성공(202) → 출력
--      3회전 : %NOTFOUND = FALSE(아직 2회전 FETCH 기준) → FETCH 실패
--              → v_row는 갱신되지 않고 202 값이 그대로 남아 있음 → 재출력
--      4회전 : %NOTFOUND = TRUE → EXIT
--    검사 시점의 %NOTFOUND가 "직전 FETCH"의 결과라는 점이 핵심이다.
--    이번 FETCH의 성공 여부는 FETCH를 해 봐야 알 수 있다.
--    합계 25000은 실제 부서 20 급여 합계 19000과 다르다. 집계 로직이었다면
--    아무도 모른 채 잘못된 숫자가 보고되었을 것이다.

-- ▶ 수정 : FETCH 직후에 %NOTFOUND를 검사한다. 이 순서가 표준 관용구다.

[6] %ROWCOUNT를 종료 조건으로 쓰기 — 상위 N건만 처리
--------------------------------------------------------------------------------

-- 커서 전체를 다 읽을 필요가 없을 때 %ROWCOUNT로 중간에 빠져나온다.
-- Cursor FOR LOOP로는 이런 조기 종료를 자연스럽게 쓰기 어려워
-- 명시적 커서를 택하는 대표적 사례다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT last_name, salary
  4          FROM employees
  5         ORDER BY salary DESC, last_name;
  6
  7     v_row   c_emp%ROWTYPE;
  8  BEGIN
  9     OPEN c_emp;
 10     LOOP
 11        FETCH c_emp INTO v_row;
 12        EXIT WHEN c_emp%NOTFOUND OR c_emp%ROWCOUNT > 5;
 13
 14        DBMS_OUTPUT.PUT_LINE(LPAD(c_emp%ROWCOUNT, 2) || '위 ' ||
 15                             RPAD(v_row.last_name, 12) || LPAD(v_row.salary, 6));
 16     END LOOP;
 17     CLOSE c_emp;
 18  END;
 19  /

 2위 De Haan       17000
 3위 Kochhar       17000
 4위 Russell       14000
 5위 Partners      13500

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    EXIT 조건을 %ROWCOUNT > 5 로 썼다. > 5 대신 >= 5 로 쓰면 4건만 출력된다.
--    "이번 행을 처리하고 나갈 것인가, 처리하지 않고 나갈 것인가"를
--    조건에 정확히 반영해야 한다.

[7] 암시적 커서 속성 — SQL%ROWCOUNT / SQL%FOUND / SQL%NOTFOUND
--------------------------------------------------------------------------------

-- PL/SQL 안의 모든 DML과 SELECT INTO는 SQL이라는 이름의 암시적 커서를 쓴다.
-- 직전에 실행된 SQL 문장의 결과를 이 속성들로 확인한다.

SQL> DECLARE
  2     FUNCTION b2c(p_flag BOOLEAN) RETURN VARCHAR2 IS
  3     BEGIN
  4        RETURN CASE WHEN p_flag IS NULL THEN 'NULL'
  5                    WHEN p_flag         THEN 'TRUE'
  6                    ELSE                     'FALSE'
  7               END;
  8     END;
  9  BEGIN
 10     UPDATE employees
 11        SET salary = salary * 1.05
 12      WHERE department_id = 20;
 13
 14     DBMS_OUTPUT.PUT_LINE('UPDATE  SQL%ROWCOUNT = ' || SQL%ROWCOUNT ||
 15                          ' , SQL%FOUND = ' || b2c(SQL%FOUND));
 16
 17     UPDATE employees
 18        SET salary = salary * 1.05
 19      WHERE department_id = 999;       -- 존재하지 않는 부서
 20
 21     DBMS_OUTPUT.PUT_LINE('UPDATE  SQL%ROWCOUNT = ' || SQL%ROWCOUNT ||
 22                          ' , SQL%NOTFOUND = ' || b2c(SQL%NOTFOUND));
 23
 24     DBMS_OUTPUT.PUT_LINE('SQL%ISOPEN = ' || b2c(SQL%ISOPEN));
 25  END;
 26  /
UPDATE  SQL%ROWCOUNT = 2 , SQL%FOUND = TRUE
UPDATE  SQL%ROWCOUNT = 0 , SQL%NOTFOUND = TRUE
SQL%ISOPEN = FALSE

PL/SQL procedure successfully completed.

SQL> ROLLBACK;

Rollback complete.

-- ▶ 확인 사항
--    1) 0건을 갱신한 UPDATE는 예외가 아니다. SQL%ROWCOUNT = 0으로만 알 수 있다.
--       "분명 UPDATE 했는데 왜 안 바뀌었지"의 원인 대부분이 여기다.
--    2) SQL%ISOPEN은 항상 FALSE다. 암시적 커서는 문장이 끝나는 즉시 닫히기
--       때문에 사용자가 열려 있는 순간을 볼 수 없다.
--    3) SQL 속성은 직전 문장 기준이다. 두 번째 UPDATE 이후 첫 UPDATE의
--       건수는 이미 사라졌으므로, 필요하면 그때그때 변수에 보관한다.


--------------------------------------------------------------------------------
[8] 정리
--------------------------------------------------------------------------------

-- 1) %FOUND / %NOTFOUND는 "직전 FETCH"의 결과다. FETCH 전에 검사하면
--    마지막 행이 한 번 더 처리된다. 반드시 FETCH 바로 다음 줄에서 검사한다.
-- 2) 실패한 FETCH는 INTO 변수 값을 바꾸지 않는다. 이전 행 값이 그대로 남는
--    것이 중복 처리 버그의 실체다.
-- 3) OPEN 직후 %FOUND/%NOTFOUND는 NULL, %ROWCOUNT는 0이다.
-- 4) %ROWCOUNT는 성공한 FETCH만 센다. 조기 종료 조건으로 쓸 때는
--    경계값(> N 인가 >= N 인가)을 반드시 확인한다.
-- 5) CLOSE 후에 참조할 수 있는 속성은 %ISOPEN뿐이다. 나머지는 ORA-01001이다.
--    건수는 CLOSE 전에 변수로 옮겨 둔다.
-- 6) 암시적 커서 SQL의 속성으로 DML 영향 행 수를 확인한다.
--    0건 갱신은 예외가 아니므로 SQL%ROWCOUNT 확인이 곧 검증이다.

SQL> EXIT
