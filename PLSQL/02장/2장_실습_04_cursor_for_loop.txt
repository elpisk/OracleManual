[2] Cursor FOR LOOP 기본형 — 네 단계가 한 줄로 줄어든다
--------------------------------------------------------------------------------

-- FOR 레코드명 IN 커서명 LOOP … END LOOP;
-- 이 한 줄이 아래 네 가지를 모두 대신한다.
--   · OPEN 커서명
--   · FETCH 커서명 INTO 레코드
--   · EXIT WHEN 커서명%NOTFOUND
--   · CLOSE 커서명
-- 레코드 변수도 선언하지 않는다. 커서명%ROWTYPE으로 자동 생성된다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name, salary
  4          FROM employees
  5         WHERE department_id = 100
  6         ORDER BY salary DESC;
  7  BEGIN
  8     FOR emp_rec IN c_emp LOOP
  9        DBMS_OUTPUT.PUT_LINE(emp_rec.employee_id || ' ' ||
 10                             RPAD(emp_rec.last_name, 12) ||
 11                             LPAD(emp_rec.salary, 6));
 12     END LOOP;
 13  END;
 14  /

109 Faviet        9000
110 Chen          8200
112 Urman         7800

-- ▶ 확인 사항
--    emp_rec는 DECLARE에 없다. FOR가 시작될 때 c_emp%ROWTYPE으로 암묵
--    선언되고, END LOOP에서 사라진다. FOR 정수 루프의 인덱스와 같은 규칙이다.

--------------------------------------------------------------------------------
[4] 명시적 4단계와 Cursor FOR LOOP를 나란히 비교
--------------------------------------------------------------------------------

-- (A) 명시적 4단계 — 관리해야 할 요소가 레코드 변수 1개 + 문장 4개다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT last_name, salary FROM employees WHERE department_id = 100;
  4     v_row   c_emp%ROWTYPE;
  5     v_tot   NUMBER := 0;
  6  BEGIN
  7     OPEN c_emp;
  8     LOOP
  9        FETCH c_emp INTO v_row;
 10        EXIT WHEN c_emp%NOTFOUND;
 11        v_tot := v_tot + v_row.salary;
 12     END LOOP;
 13     CLOSE c_emp;
 14
 15     DBMS_OUTPUT.PUT_LINE('(A) 급여 합계 = ' || v_tot);
 16  END;
 17  /
(A) 급여 합계 = 51600

PL/SQL procedure successfully completed.

-- (B) Cursor FOR LOOP — 같은 결과, 관리 요소는 0개다.

SQL> DECLARE
  2     v_tot   NUMBER := 0;
  3  BEGIN
  4     FOR r IN (SELECT salary FROM employees WHERE department_id = 100) LOOP
  5        v_tot := v_tot + r.salary;
  6     END LOOP;
  7
  8     DBMS_OUTPUT.PUT_LINE('(B) 급여 합계 = ' || v_tot);
  9  END;
 10  /
(B) 급여 합계 = 51600

PL/SQL procedure successfully completed.

-- ▶ Cursor FOR LOOP는 예외가 발생해도 커서를 닫아 준다.
--    ORA-06511이나 ORA-01001을 구조적으로 만들 수 없는 형태다.
--    "닫는 것을 잊었다"는 사고를 문법으로 차단한다는 점이 가장 큰 장점이다.

[5] (오류 재현 1) 표현식 컬럼에 별칭을 빠뜨림 — PLS-00302
--------------------------------------------------------------------------------

-- 레코드의 필드 이름은 SELECT 목록의 "컬럼 이름"이다.
-- salary * 12 처럼 계산식을 넣으면 컬럼 이름이 식별자 형태가 아니어서
-- 레코드 필드로 참조할 수 없다.

  4          FROM employees
  5         WHERE department_id = 100
  6         ORDER BY last_name;
  7  BEGIN
  8     FOR emp_rec IN c_emp LOOP
  9        DBMS_OUTPUT.PUT_LINE(emp_rec.last_name || ' : ' || emp_rec.annual_sal);
 10     END LOOP;
 11  END;

--    커서 레코드에는 ANNUAL_SAL이라는 필드가 없다. 두 번째 컬럼의 이름은

[6] (오류 재현 2) Cursor FOR LOOP 안에서 커서를 직접 OPEN — ORA-06511
--------------------------------------------------------------------------------

-- 명시적 커서 습관이 남아 "루프에 들어가려면 열어야지" 하고 OPEN을 넣으면,
-- FOR가 이미 열어 둔 커서를 한 번 더 여는 셈이 된다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name
  4          FROM employees
  5         WHERE department_id = 100
  6         ORDER BY employee_id;
  7  BEGIN
  8     FOR emp_rec IN c_emp LOOP
  9        OPEN c_emp;                   -- 습관적으로 넣은 OPEN
 10        DBMS_OUTPUT.PUT_LINE(emp_rec.employee_id || ' ' || emp_rec.last_name);
 11        CLOSE c_emp;
 12     END LOOP;
 13  END;
 14  /
DECLARE
*
ERROR at line 1:
ORA-06511: PL/SQL: cursor already open
ORA-06512: at line 9

-- ▶ 원인 파악
--    FOR emp_rec IN c_emp 는 루프에 진입하기 전에 이미 c_emp를 OPEN한 상태다.
--    Cursor FOR LOOP는 OPEN·FETCH·CLOSE 전부를 엔진이 관리하므로,
--    사용자가 같은 커서에 이 세 문장을 직접 쓰면 안 된다.
--    (CLOSE를 먼저 썼다면 이번에는 ORA-01001: invalid cursor가 났을 것이다.)

-- ▶ 수정 : OPEN / CLOSE 문장을 제거한다. 그것만으로 끝이다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT employee_id, last_name
  4          FROM employees
  5         WHERE department_id = 100
  6         ORDER BY employee_id;
  7  BEGIN
  8     FOR emp_rec IN c_emp LOOP
  9        DBMS_OUTPUT.PUT_LINE(emp_rec.employee_id || ' ' || emp_rec.last_name);

[7] (오류 재현 3) 루프 레코드의 필드에 값을 대입 — PLS-00363
--------------------------------------------------------------------------------

-- 루프 레코드는 FOR 인덱스와 마찬가지로 읽기 전용이다.
-- 레코드 값을 바꿔도 테이블에 반영되지 않으므로 애초에 막아 둔 것이다.

SQL> DECLARE
  2     CURSOR c_emp IS
  3        SELECT last_name, salary FROM employees WHERE department_id = 100;
  4  BEGIN
  5     FOR emp_rec IN c_emp LOOP
  6        emp_rec.salary := emp_rec.salary * 1.1;    -- 인상분을 넣으려는 시도
  7        DBMS_OUTPUT.PUT_LINE(emp_rec.last_name || ' ' || emp_rec.salary);
  8     END LOOP;
  9  END;

      emp_rec.salary := emp_rec.salary * 1.1;    -- 인상분을 넣으려는 시도

-- ▶ 원인 파악
--    커서 레코드에 대입해 봐야 메모리 사본만 바뀔 뿐 DB에는 영향이 없다.
--    PL/SQL은 이런 착각을 막기 위해 대입 자체를 금지한다.

-- ▶ 수정 (A) : 계산 결과만 필요하면 별도 지역 변수에 담는다.

SQL> DECLARE
  2     v_new_sal   employees.salary%TYPE;
  3  BEGIN
  4     FOR emp_rec IN (SELECT last_name, salary
  5                       FROM employees
  6                      WHERE department_id = 100

  9        DBMS_OUTPUT.PUT_LINE(RPAD(emp_rec.last_name, 12) ||
 10                             LPAD(emp_rec.salary, 7) || ' -> ' ||
 11                             LPAD(v_new_sal, 7));
 12     END LOOP;
 13  END;

[8] 결과가 0건일 때의 동작 — 예외가 아니라 0회 반복
--------------------------------------------------------------------------------

-- 부서 190(Contracting)에는 소속 사원이 없다.

SQL> SELECT COUNT(*) FROM employees WHERE department_id = 190;

  COUNT(*)
----------
         0

SQL> DECLARE
  2     v_cnt   NUMBER := 0;
  3  BEGIN
  4     FOR r IN (SELECT last_name FROM employees WHERE department_id = 190) LOOP
  5        v_cnt := v_cnt + 1;
  6     END LOOP;
  7
  8     IF v_cnt = 0 THEN
  9        DBMS_OUTPUT.PUT_LINE('해당 부서에 사원이 없습니다.');
 10     END IF;
 11  END;
 12  /
해당 부서에 사원이 없습니다.

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    커서는 결과가 0건이어도 NO_DATA_FOUND를 던지지 않는다.
--    NO_DATA_FOUND는 SELECT INTO의 예외다. 커서에서 "없음"을 다루려면
--    카운터를 두거나 %ROWCOUNT를 확인해야 한다.


--------------------------------------------------------------------------------
[9] 정리
--------------------------------------------------------------------------------

-- 1) Cursor FOR LOOP는 OPEN/FETCH/EXIT WHEN %NOTFOUND/CLOSE를 한 줄로 대체한다.
-- 2) 루프 레코드는 커서명%ROWTYPE으로 암묵 선언되며 루프 밖에서는 사라진다.
-- 3) 표현식 컬럼에는 반드시 별칭을 준다. 없으면 PLS-00302다.
-- 4) 같은 커서에 OPEN/FETCH/CLOSE를 직접 쓰면 ORA-06511 또는 ORA-01001이다.
--    Cursor FOR LOOP를 쓰기로 했으면 커서 관리는 엔진에 완전히 맡긴다.
-- 5) 루프 레코드는 읽기 전용이다(PLS-00363). 값 변경은 지역 변수나 UPDATE로.
-- 6) 결과 0건은 예외가 아니라 0회 반복이다.
-- 7) 커서 전체를 처음부터 끝까지 한 방향으로 읽는 경우라면 Cursor FOR LOOP가
--    기본 선택이다. 중간에 멈추거나 여러 커서를 교차 제어해야 할 때만
--    명시적 4단계를 쓴다.

SQL> EXIT
