[6] 중첩 커서를 조인 커서 하나로 대체 — N+1 질의 제거
--------------------------------------------------------------------------------

-- [2]의 중첩 구조는 부서 3개에 대해 질의를 1 + 3 = 4회 실행한다.
-- 부서가 27개면 28회다. 이것이 이른바 N+1 문제다.
-- 조인 커서 하나로 정렬해 받아 오면 질의는 언제나 1회다.
-- 그룹이 바뀌는 지점은 "직전 부서 ID"를 기억해 직접 판단한다.

SQL> DECLARE
  2     CURSOR c_join IS
  3        SELECT d.department_id,
  4               d.department_name,
  5               e.employee_id,
  6               e.last_name,
  7               e.salary
  8          FROM departments d
  9          JOIN employees   e ON e.department_id = d.department_id
 10         WHERE d.department_id IN (20, 60, 90)
 11         ORDER BY d.department_id, e.salary DESC, e.last_name;
 12
 13     v_prev_dept   departments.department_id%TYPE := NULL;
 14     v_tot         NUMBER := 0;
 15     v_grand       NUMBER := 0;
 16  BEGIN
 17     FOR r IN c_join LOOP
 18        -- 부서가 바뀌는 지점에서 이전 부서의 소계를 출력한다
 19        IF v_prev_dept IS NOT NULL AND r.department_id <> v_prev_dept THEN
 20           DBMS_OUTPUT.PUT_LINE('    소계 : ' || v_tot);
 21           v_grand := v_grand + v_tot;
 22           v_tot   := 0;
 23        END IF;
 24
 25        IF v_prev_dept IS NULL OR r.department_id <> v_prev_dept THEN
 26           DBMS_OUTPUT.PUT_LINE('[' || r.department_id || '] ' ||
 27                                r.department_name);
 28        END IF;
 29
 30        DBMS_OUTPUT.PUT_LINE('    ' || r.employee_id || ' ' ||
 31                             RPAD(r.last_name, 12) || LPAD(r.salary, 6));
 32
 33        v_tot       := v_tot + r.salary;
 34        v_prev_dept := r.department_id;
 35     END LOOP;
 36
 37     -- 마지막 그룹의 소계는 루프 종료 후에 처리한다
 38     IF v_prev_dept IS NOT NULL THEN
 39        DBMS_OUTPUT.PUT_LINE('    소계 : ' || v_tot);
 40        v_grand := v_grand + v_tot;
 41     END IF;
 42
 43     DBMS_OUTPUT.PUT_LINE('총계 : ' || v_grand);
 44  END;

--------------------------------------------------------------------------------
[7] 정리
--------------------------------------------------------------------------------

-- 1) 중첩 커서는 "외부 = 그룹, 내부 = 상세" 구조다. 내부 커서에는 외부
--    커서의 값을 파라미터로 넘긴다.
-- 2) 커서 파라미터 이름이 컬럼명과 겹치면 SQL에서는 컬럼이 우선한다.
--    WHERE department_id = department_id 는 항상 참이 되어 조건이 사라진다.
--    p_ 접두사, 테이블 별칭 한정, 블록 레이블 한정으로 막는다.
-- 3) 명시적 커서를 중첩할 때 OPEN과 CLOSE는 반드시 같은 중첩 레벨에 둔다.
--    CLOSE가 한 단계 밖으로 나가면 두 번째 그룹에서 ORA-06511이다.
-- 4) 중첩 구조에서는 Cursor FOR LOOP가 기본 선택이다. OPEN/CLOSE 실수 자체가
--    불가능해진다.
-- 5) 중첩 커서는 그룹 수만큼 질의를 반복한다(N+1). 조인 커서 하나로 정렬해
