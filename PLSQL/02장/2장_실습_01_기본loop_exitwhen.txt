[2] 가장 단순한 기본 LOOP — 카운터를 직접 선언하고 직접 증가시킨다
--------------------------------------------------------------------------------

-- 기본 LOOP에는 조건 자리가 없다. 종료는 오직 EXIT 문장으로만 이루어진다.
-- 카운터 변수는 반드시 초기값을 지정한다. 초기값이 없으면 NULL이 되고
-- NULL과의 비교 결과도 NULL이라 EXIT 조건이 결코 참이 되지 않는다.

SQL> DECLARE
  2     v_counter   NUMBER := 1;
  3  BEGIN
  4     LOOP
  5        DBMS_OUTPUT.PUT_LINE('반복 ' || v_counter || '회차');
  6        v_counter := v_counter + 1;
  7        EXIT WHEN v_counter > 3;
  8     END LOOP;
  9
 10     DBMS_OUTPUT.PUT_LINE('루프 종료. v_counter = ' || v_counter);
 11  END;
 12  /
반복 1회차
반복 2회차
반복 3회차
루프 종료. v_counter = 4

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    루프를 빠져나온 시점의 v_counter는 3이 아니라 4다.
--    카운터를 처리 건수로 그대로 사용하면 한 건이 더 많이 세어진다.


--------------------------------------------------------------------------------
[3] EXIT WHEN의 위치가 실행 횟수를 결정한다
--------------------------------------------------------------------------------

-- (A) EXIT WHEN을 본문 끝에 둔 경우 — 본문은 최소 1회 실행된다

SQL> DECLARE
  2     v_counter   NUMBER := 100;      -- 이미 종료 조건을 넘긴 값
  3  BEGIN
  4     LOOP
  5        DBMS_OUTPUT.PUT_LINE('(A) 본문 실행됨');
  6        EXIT WHEN v_counter > 3;
  7     END LOOP;
  8  END;
  9  /
(A) 본문 실행됨

PL/SQL procedure successfully completed.

-- (B) EXIT WHEN을 본문 맨 앞에 둔 경우 — 본문이 한 번도 실행되지 않는다

SQL> DECLARE
  2     v_counter   NUMBER := 100;
  3  BEGIN
  4     LOOP
  5        EXIT WHEN v_counter > 3;
  6        DBMS_OUTPUT.PUT_LINE('(B) 본문 실행됨');
  7     END LOOP;
  8
  9     DBMS_OUTPUT.PUT_LINE('(B) 본문은 한 번도 실행되지 않았다');
 10  END;
 11  /
(B) 본문은 한 번도 실행되지 않았다

PL/SQL procedure successfully completed.

-- ▶ 정리
--    "기본 LOOP는 최소 1회 실행된다"는 설명은 EXIT WHEN을 본문 끝에 두었을
--    때만 성립한다. 위치가 곧 의미다.


--------------------------------------------------------------------------------
[4] IF 안의 단독 EXIT와 EXIT WHEN은 결과가 같다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_i   NUMBER := 1;
  3     v_j   NUMBER := 1;
  4  BEGIN
  5     -- (A) IF 안의 단독 EXIT
  6     LOOP
  7        v_i := v_i + 1;
  8        IF v_i > 3 THEN
  9           EXIT;
 10        END IF;
 11     END LOOP;
 12
 13     -- (B) WHEN 절을 붙인 형태
 14     LOOP
 15        v_j := v_j + 1;
 16        EXIT WHEN v_j > 3;
 17     END LOOP;
 18
 19     DBMS_OUTPUT.PUT_LINE('v_i = ' || v_i || ' , v_j = ' || v_j);
 20  END;
 21  /
v_i = 4 , v_j = 4

PL/SQL procedure successfully completed.

-- ▶ 두 방식의 결과는 같다. 다만 (B)가 조건과 종료를 한 줄에 드러내므로
--    코드 리뷰에서 흐름을 따라가기 훨씬 쉽다.

  4   ORDER BY location_id;

  2     v_countryid   locations.country_id%TYPE := 'CA';
  3     v_loc_id      locations.location_id%TYPE;

  5     v_new_city    locations.city%TYPE := 'Montreal';

  8       FROM locations
  9      WHERE country_id = v_countryid;

 16        v_counter := v_counter + 1;

 18     END LOOP;

--------------------------------------------------------------------------------
[6] (오류 재현 1) 종료 조건 변수를 갱신하지 않은 무한 루프
--------------------------------------------------------------------------------

-- 아래 블록은 문법 오류가 전혀 없다. 컴파일도 정상적으로 통과한다.
-- 그러나 v_counter를 증가시키는 문장이 IF 분기 안에 들어가 있어,
-- 조건을 만족하지 않는 경로에서는 카운터가 영원히 변하지 않는다.

SQL> DECLARE
  2     v_counter   NUMBER := 1;
  3     v_flag      BOOLEAN := FALSE;
  4  BEGIN
  5     LOOP
  6        IF v_flag THEN
  7           v_counter := v_counter + 1;    -- 이 경로로는 절대 들어오지 않는다
  8        END IF;
  9        EXIT WHEN v_counter > 3;
 10     END LOOP;
 11
 12     DBMS_OUTPUT.PUT_LINE('여기까지 오지 못한다');
 13  END;
 14  /
^C
ERROR at line 1:
ORA-01013: user requested cancel of current operation

-- ▶ 원인 파악
--    v_flag가 FALSE로 고정되어 있으므로 v_counter는 1에서 변하지 않는다.
--    EXIT WHEN v_counter > 3 이 결코 참이 되지 않아 루프가 끝나지 않는다.
--    무한 루프는 컴파일 오류가 아니다. 실행 중에 멈추지 않을 뿐이다.
--    Ctrl+C로 중단했으므로 ORA-01013이 표시되었다.

-- ▶ 수정 : 조건 변수를 갱신하는 문장을 루프 본문의 최상위 레벨로 옮긴다.
--          아울러 최대 반복 횟수 안전장치를 함께 둔다.

SQL> DECLARE
  2     v_counter   NUMBER := 1;
  3     v_guard     NUMBER := 0;
  4  BEGIN
  5     LOOP
  6        v_counter := v_counter + 1;       -- 분기 밖, 최상위 레벨
  7
  8        v_guard := v_guard + 1;           -- 안전장치
  9        IF v_guard > 10000 THEN
 10           RAISE_APPLICATION_ERROR(-20001, '반복 횟수 상한 초과');
 11        END IF;
 12
 13        EXIT WHEN v_counter > 3;
 14     END LOOP;
 15
 16     DBMS_OUTPUT.PUT_LINE('정상 종료. v_counter = ' || v_counter);
 17  END;
 18  /
정상 종료. v_counter = 4

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[7] (오류 재현 2) EXIT를 루프 밖에서 사용 — PLS-00376
--------------------------------------------------------------------------------

SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('시작');
  3     EXIT;                          -- 루프 밖에서 EXIT를 사용했다
  4     DBMS_OUTPUT.PUT_LINE('끝');
  5  END;
  6  /
   EXIT;
   *
ERROR at line 3:
ORA-06550: line 3, column 4:
PLS-00376: illegal EXIT statement; it must appear inside a loop
ORA-06550: line 3, column 4:
PL/SQL: Statement ignored

-- ▶ 원인 파악
--    EXIT는 반복문을 종료하는 문장이므로 반드시 LOOP … END LOOP 안에
--    있어야 한다. 블록을 빠져나가려는 목적이라면 EXIT가 아니라 RETURN을
--    사용하거나(프로시저·함수) GOTO 레이블을 사용해야 한다.
--    CONTINUE도 같은 제약을 가지며 루프 밖에 두면 PLS-00375 계열 오류가 난다.

-- ▶ 수정 : 익명 블록에서 조기 종료가 필요하면 RETURN을 사용한다.

SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('시작');
  3     RETURN;                        -- 블록 실행을 여기서 종료
  4     DBMS_OUTPUT.PUT_LINE('끝');
  5  END;
  6  /
시작

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[8] 정리
--------------------------------------------------------------------------------

-- 1) 기본 LOOP는 조건 자리가 없고 EXIT 또는 EXIT WHEN으로만 종료된다.
-- 2) EXIT WHEN을 본문 끝에 두어야 "최소 1회 실행"이 보장된다.
-- 3) EXIT WHEN의 조건은 TRUE일 때만 종료한다. FALSE와 NULL은 계속이다.
-- 4) 종료 조건에 쓰이는 변수는 초기값을 지정하고, 갱신 문장을 분기 안에
--    넣지 않는다. 무한 루프의 90퍼센트는 이 두 가지에서 나온다.
-- 5) EXIT와 CONTINUE는 반드시 루프 안에 있어야 한다(PLS-00376 / PLS-00375).

SQL> EXIT
