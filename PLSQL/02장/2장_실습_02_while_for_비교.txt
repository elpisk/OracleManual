[2] WHILE LOOP — 조건을 먼저 평가하고 참일 때만 본문에 들어간다
--------------------------------------------------------------------------------

-- 기본 LOOP와 달리 WHILE은 반복 여부를 결정하는 조건이 헤더에 있다.
-- 본문에 들어가기 전에 조건을 평가하므로, 처음부터 조건이 거짓이면
-- 본문은 단 한 번도 실행되지 않는다.

SQL> DECLARE
  2     v_counter   NUMBER := 1;
  3  BEGIN
  4     WHILE v_counter <= 3 LOOP
  5        DBMS_OUTPUT.PUT_LINE('반복 ' || v_counter || '회차');
  6        v_counter := v_counter + 1;
  7     END LOOP;
  8
  9     DBMS_OUTPUT.PUT_LINE('루프 종료. v_counter = ' || v_counter);
 10  END;
 11  /
반복 1회차
반복 2회차
반복 3회차
루프 종료. v_counter = 4

PL/SQL procedure successfully completed.

-- 처음부터 조건이 거짓인 경우

SQL> DECLARE
  2     v_counter   NUMBER := 100;
  3  BEGIN
  4     WHILE v_counter <= 3 LOOP
  5        DBMS_OUTPUT.PUT_LINE('본문 실행됨');
  6     END LOOP;
  7
  8     DBMS_OUTPUT.PUT_LINE('본문은 한 번도 실행되지 않았다');
  9  END;
 10  /
본문은 한 번도 실행되지 않았다

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    기본 LOOP는 "최소 1회", WHILE LOOP는 "0회부터"가 기본 성질이다.
--    반복 횟수를 미리 알 수 없고 0번일 수도 있으면 WHILE이 맞다.


--------------------------------------------------------------------------------
[3] (논리 함정) WHILE 조건이 NULL이면 루프는 시작조차 하지 않는다
--------------------------------------------------------------------------------

-- WHILE은 조건이 TRUE일 때만 반복한다. FALSE와 NULL은 모두 "반복하지 않음"이다.
-- 초기값을 주지 않은 변수를 조건에 쓰면 첫 평가가 NULL이 되어 루프가 통째로
-- 건너뛰어진다. 오류 메시지가 나지 않으므로 발견하기 가장 어려운 유형이다.

SQL> DECLARE
  2     v_sal   NUMBER;                 -- 초기값 없음 → NULL
  3     v_cnt   NUMBER := 0;
  4  BEGIN
  5     WHILE v_sal < 5000 LOOP
  6        v_cnt := v_cnt + 1;
  7        v_sal := NVL(v_sal, 0) + 1000;
  8     END LOOP;
  9
 10     DBMS_OUTPUT.PUT_LINE('반복 횟수 = ' || v_cnt);
 11     DBMS_OUTPUT.PUT_LINE('v_sal    = ' || NVL(TO_CHAR(v_sal), 'NULL'));
 12  END;
 13  /
반복 횟수 = 0
v_sal    = NULL

PL/SQL procedure successfully completed.

-- ▶ 수정 : 선언 시점에 초기값을 주거나, 조건에서 NVL로 방어한다.

SQL> DECLARE
  2     v_sal   NUMBER := 0;            -- 초기값 지정
  3     v_cnt   NUMBER := 0;
  4  BEGIN
  5     WHILE v_sal < 5000 LOOP
  6        v_cnt := v_cnt + 1;
  7        v_sal := v_sal + 1000;
  8     END LOOP;
  9
 10     DBMS_OUTPUT.PUT_LINE('반복 횟수 = ' || v_cnt || ' , v_sal = ' || v_sal);
 11  END;
 12  /
반복 횟수 = 5 , v_sal = 5000

PL/SQL procedure successfully completed.

[4] FOR LOOP — 인덱스는 선언하지 않고, 증가도 직접 하지 않는다
--------------------------------------------------------------------------------

-- FOR 루프의 인덱스는 DECLARE에 쓰지 않는다. LOOP가 시작될 때
-- PLS_INTEGER 형태로 암묵 선언되고, END LOOP에서 사라진다.

SQL> BEGIN
  2     FOR i IN 1 .. 3 LOOP
  3        DBMS_OUTPUT.PUT_LINE('반복 ' || i || '회차');
  4     END LOOP;
  5  END;
  6  /
반복 1회차
반복 2회차
반복 3회차

PL/SQL procedure successfully completed.

  3        DBMS_OUTPUT.PUT_LINE('여기는 실행되지 않는다');

--------------------------------------------------------------------------------
[5] REVERSE — 범위 표기는 그대로 두고 진행 방향만 뒤집는다
--------------------------------------------------------------------------------

-- 흔한 착각: REVERSE를 쓰면 범위를 3 .. 1 로 적어야 한다고 생각한다.
-- 실제로는 범위를 항상 "작은값 .. 큰값"으로 쓰고 REVERSE만 붙인다.

SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('--- REVERSE 1 .. 3 ---');
  3     FOR i IN REVERSE 1 .. 3 LOOP
  4        DBMS_OUTPUT.PUT_LINE('i = ' || i);
  5     END LOOP;
  6
  7     DBMS_OUTPUT.PUT_LINE('--- REVERSE 3 .. 1 ---');
  8     FOR i IN REVERSE 3 .. 1 LOOP
  9        DBMS_OUTPUT.PUT_LINE('i = ' || i);
 10     END LOOP;
 11     DBMS_OUTPUT.PUT_LINE('(위 루프는 0회 실행되었다)');
 12  END;
 13  /
--- REVERSE 1 .. 3 ---
i = 3
i = 2
i = 1
--- REVERSE 3 .. 1 ---
(위 루프는 0회 실행되었다)

PL/SQL procedure successfully completed.

[6] 같은 결과를 세 가지 반복문으로 작성해 본다
--------------------------------------------------------------------------------

-- 1부터 5까지의 합을 세 방식으로 구한다.

SQL> DECLARE
  2     v_sum1   NUMBER := 0;
  3     v_sum2   NUMBER := 0;
  4     v_sum3   NUMBER := 0;
  5     v_i      NUMBER := 1;
  6  BEGIN
  7     -- (A) 기본 LOOP
  8     LOOP
  9        v_sum1 := v_sum1 + v_i;
 10        v_i := v_i + 1;
 11        EXIT WHEN v_i > 5;
 12     END LOOP;
 13
 14     -- (B) WHILE LOOP
 15     v_i := 1;
 16     WHILE v_i <= 5 LOOP
 17        v_sum2 := v_sum2 + v_i;
 18        v_i := v_i + 1;
 19     END LOOP;
 20
 21     -- (C) FOR LOOP
 22     FOR i IN 1 .. 5 LOOP
 23        v_sum3 := v_sum3 + i;
 24     END LOOP;
 25
 26     DBMS_OUTPUT.PUT_LINE('LOOP=' || v_sum1 ||
 27                          ' WHILE=' || v_sum2 ||
 28                          ' FOR=' || v_sum3);
 29  END;
 30  /
LOOP=15 WHILE=15 FOR=15

PL/SQL procedure successfully completed.

-- ▶ 선택 기준
--    반복 횟수가 상수/변수로 이미 정해져 있다  → FOR   (카운터 관리 코드가 사라진다)
--    조건이 만족될 때까지 돌려야 한다          → WHILE (0회 실행 허용)
--    무조건 한 번은 실행한 뒤 판단해야 한다    → LOOP + EXIT WHEN


--------------------------------------------------------------------------------
[7] (오류 재현 1) FOR 루프 인덱스에 값을 대입 — PLS-00363
--------------------------------------------------------------------------------

-- FOR 루프의 인덱스는 읽기 전용이다. 루프 제어권이 PL/SQL 엔진에 있기 때문에
-- 사용자가 값을 바꿀 수 없다. C나 Java의 for(i=0; i<n; i++)와 다른 점이다.

SQL> BEGIN
  2     FOR i IN 1 .. 5 LOOP
  3        DBMS_OUTPUT.PUT_LINE('i = ' || i);
  4        i := i + 1;                   -- 두 칸씩 건너뛰게 하려는 시도
  5     END LOOP;
  6  END;
  7  /
      i := i + 1;                   -- 두 칸씩 건너뛰게 하려는 시도
      *
ERROR at line 4:
ORA-06550: line 4, column 7:
PLS-00363: expression 'I' cannot be used as an assignment target
ORA-06550: line 4, column 7:
PL/SQL: Statement ignored

-- ▶ 원인 파악
--    FOR 인덱스는 상수처럼 취급된다. 증가폭(STEP) 지정 문법도 PL/SQL에는
--    없으므로, 인덱스를 조작해 간격을 바꾸는 방식은 아예 불가능하다.

-- ▶ 수정 (A) : 인덱스를 그대로 두고 MOD로 걸러낸다.

SQL> BEGIN
  2     FOR i IN 1 .. 5 LOOP
  3        IF MOD(i, 2) = 1 THEN
  4           DBMS_OUTPUT.PUT_LINE('i = ' || i);
  5        END IF;
  6     END LOOP;
  7  END;

i = 1
i = 3
i = 5

--------------------------------------------------------------------------------
[8] (오류 재현 2) FOR 루프 인덱스를 루프 밖에서 참조 — PLS-00201
--------------------------------------------------------------------------------

-- 인덱스의 유효 범위(scope)는 LOOP … END LOOP 안이다.
-- 루프가 끝나면 식별자 자체가 사라진다.

SQL> BEGIN
  2     FOR i IN 1 .. 3 LOOP
  3        NULL;
  4     END LOOP;
  5
  6     DBMS_OUTPUT.PUT_LINE('마지막 i = ' || i);
  7  END;
  8  /
      DBMS_OUTPUT.PUT_LINE('마지막 i = ' || i);
                                            *
ERROR at line 6:
ORA-06550: line 6, column 45:
PLS-00201: identifier 'I' must be declared
ORA-06550: line 6, column 4:
PL/SQL: Statement ignored

-- ▶ 원인 파악
--    FOR 인덱스는 루프에 종속된 지역 식별자다. WHILE이나 기본 LOOP에서
--    쓰던 카운터 변수를 루프 종료 후에도 읽던 습관 때문에 자주 나온다.

-- ▶ 수정 : 루프 밖에서도 값이 필요하면 별도 변수에 담아 둔다.

SQL> DECLARE
  2     v_last   NUMBER;
  3  BEGIN
  4     FOR i IN 1 .. 3 LOOP
  5        v_last := i;
  6     END LOOP;
  7
  8     DBMS_OUTPUT.PUT_LINE('마지막 i = ' || v_last);
  9  END;
 10  /
마지막 i = 3

PL/SQL procedure successfully completed.

-- ▶ 참고
--    바깥 블록에 i라는 변수가 이미 있어도 FOR 인덱스가 그 이름을 가리므로,
--    루프 안의 i는 여전히 인덱스다. 혼동을 피하려면 인덱스 이름을 겹치지
--    않게 짓는다.

-- (A) WHILE : 사원 107(Diana Lorentz)의 급여가 매년 5퍼센트 인상될 때
--             6000을 넘기까지 몇 년이 걸리는지 계산한다.
--             몇 번 반복할지 미리 알 수 없으므로 WHILE이 맞다.

SQL> SELECT employee_id, first_name || ' ' || last_name AS emp_name, salary
  2    FROM employees
  3   WHERE employee_id = 107;

EMPLOYEE_ID EMP_NAME                           SALARY
----------- ------------------------------ ----------
        107 Diana Lorentz                        4200

SQL> DECLARE
  2     v_sal    employees.salary%TYPE;
  3     v_year   NUMBER := 0;
  4     v_goal   NUMBER := 6000;
  5  BEGIN
  6     SELECT salary INTO v_sal
  7       FROM employees
  8      WHERE employee_id = 107;
  9
 10     WHILE v_sal <= v_goal LOOP
 11        v_year := v_year + 1;
 12        v_sal  := ROUND(v_sal * 1.05, 2);
 13        DBMS_OUTPUT.PUT_LINE(v_year || '년차 급여 = ' || v_sal);
 14     END LOOP;
 15
 16     DBMS_OUTPUT.PUT_LINE('목표 ' || v_goal || ' 초과까지 ' || v_year || '년 소요');
 17  END;
 18  /

-- (B) FOR : 부서 10 / 20 / 30 의 인원수를 차례로 확인한다.
--           반복 횟수가 3회로 고정이므로 FOR가 맞다.

SQL> DECLARE
  2     v_dept_id   departments.department_id%TYPE;
  3     v_dname     departments.department_name%TYPE;
  4     v_cnt       NUMBER;
  5  BEGIN
  6     FOR i IN 1 .. 3 LOOP
  7        v_dept_id := i * 10;
  8
  9        SELECT department_name INTO v_dname
 10          FROM departments
 11         WHERE department_id = v_dept_id;
 12
 13        SELECT COUNT(*) INTO v_cnt
 14          FROM employees
 15         WHERE department_id = v_dept_id;
 16
 17        DBMS_OUTPUT.PUT_LINE(RPAD(v_dname, 20) || ' 인원 ' || v_cnt || '명');
 18     END LOOP;
 19  END;
 20  /
Administration       인원 1명
Marketing            인원 2명
Purchasing           인원 6명

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    (B)처럼 행마다 SELECT를 반복하는 방식은 이해에는 좋지만 실무 성능에는
--    나쁘다. 같은 결과를 GROUP BY 한 번으로 얻을 수 있다면 그쪽이 맞다.
--    다음 실습부터 다루는 커서는 "여러 행을 한 번의 질의로 받아 반복 처리"
--    하는 정석 방법이다.


--------------------------------------------------------------------------------
[10] 정리
--------------------------------------------------------------------------------

-- 1) WHILE은 조건을 먼저 평가한다. 0회 실행이 정상 동작이다.
-- 2) WHILE 조건이 NULL이면 FALSE와 똑같이 "반복하지 않음"으로 처리된다.
--    조건에 쓰는 변수는 반드시 초기값을 준다.
-- 3) FOR 인덱스는 선언하지 않고, 대입할 수 없으며(PLS-00363),
--    루프 밖에서는 존재하지 않는다(PLS-00201).
-- 4) FOR의 범위는 항상 "작은값 .. 큰값"으로 쓰고, 역순은 REVERSE로 지정한다.
--    REVERSE 3 .. 1 은 오류가 아니라 0회 반복이다.
-- 5) 횟수가 정해졌으면 FOR, 조건 기반이면 WHILE, 최소 1회가 필요하면
--    LOOP + EXIT WHEN. 이 세 줄이 선택 기준의 전부다.

SQL> EXIT
