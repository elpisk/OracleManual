[2] (오해하기 쉬운 상황) 없는 객체를 조회하면 오류가 아니다
--------------------------------------------------------------------------------

SQL> SELECT object_name, object_type, status FROM user_objects
  2   WHERE object_name = 'NO_SUCH_OBJECT';

no rows selected

-- 이것은 오류가 아니다. 딕셔너리 뷰 USER_OBJECTS 자체는 존재하므로
-- 질의는 정상적으로 수행되었고 조건에 맞는 행이 없을 뿐이다.

-- 반면 존재하지 않는 테이블을 직접 조회하면 오류가 난다.

SQL> SELECT COUNT(*) FROM no_such_object;
SELECT COUNT(*) FROM no_such_object
                     *
ERROR at line 1:
ORA-00942: table or view does not exist

-- 두 상황의 차이를 자동화 코드에 적용하면 다음과 같다.

SQL> CREATE OR REPLACE FUNCTION object_exists(p_name IN VARCHAR2)
  2     RETURN VARCHAR2 IS
  3     v_cnt   NUMBER;
  4  BEGIN
  5     SELECT COUNT(*) INTO v_cnt FROM user_objects
  6      WHERE object_name = UPPER(TRIM(p_name));
  7     RETURN CASE WHEN v_cnt > 0 THEN 'YES' ELSE 'NO' END;
  8  END object_exists;
  9  /

Function created.

SQL> SELECT object_exists('EMP_COPY')       AS a,
  2         object_exists('NO_SUCH_OBJECT') AS b FROM dual;

A   B
--- ---

-- 존재 여부는 예외가 아니라 COUNT(*) = 0 으로 판정해야 한다.

[5] 동적 SQL 로 INVALID 객체를 일괄 재컴파일한다
--------------------------------------------------------------------------------

SQL> CREATE OR REPLACE PROCEDURE recompile_invalid IS
  2     v_stmt   VARCHAR2(400);
  3     v_ok     NUMBER := 0;
  4     v_fail   NUMBER := 0;
  5  BEGIN
  6     FOR r IN (SELECT object_name, object_type
  7                 FROM user_objects
  8                WHERE status = 'INVALID'
  9                  AND object_type IN ('PROCEDURE','FUNCTION','VIEW',
 10                                      'PACKAGE','PACKAGE BODY','TRIGGER')) LOOP
 11        BEGIN
 12           IF r.object_type = 'PACKAGE BODY' THEN
 13              v_stmt := 'ALTER PACKAGE "' || r.object_name || '" COMPILE BODY';
 14           ELSE
 15              v_stmt := 'ALTER ' || r.object_type || ' "'
 16                        || r.object_name || '" COMPILE';
 17           END IF;
 18           EXECUTE IMMEDIATE v_stmt;
 19           v_ok := v_ok + 1;
 20           DBMS_OUTPUT.PUT_LINE('OK   : ' || r.object_name);
 21        EXCEPTION
 22           WHEN OTHERS THEN
 23              v_fail := v_fail + 1;
 24              DBMS_OUTPUT.PUT_LINE('FAIL : ' || r.object_name
 25                                   || ' - ' || SQLERRM);
 26        END;
 27     END LOOP;
 28     DBMS_OUTPUT.PUT_LINE('성공 ' || v_ok || ' / 실패 ' || v_fail);
 29  END recompile_invalid;
 30  /

Procedure created.

SQL> EXEC recompile_invalid
OK   : BROKEN_PROC
OK   : USE_EMP_COPY
성공 2 / 실패 0

PL/SQL procedure successfully completed.

-- ALTER ... COMPILE 자체는 성공했지만 소스에 오류가 있는 BROKEN_PROC 은
-- 여전히 INVALID 로 남는다. 재컴파일 성공과 유효성은 다른 문제다.

SQL> SELECT object_name, status FROM user_objects
  2   WHERE object_name IN ('BROKEN_PROC', 'USE_EMP_COPY') ORDER BY object_name;

OBJECT_NAME          STATUS
-------------------- -------
BROKEN_PROC          INVALID
USE_EMP_COPY         VALID

-- 주의 : PACKAGE BODY 를 ALTER PACKAGE BODY x COMPILE 로 쓰면 ORA-00922 가 난다.
--        위 코드처럼 ALTER PACKAGE x COMPILE BODY 로 변환해야 한다.

[9] 점검 결과를 plsql_job_log 에 기록한다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_start    TIMESTAMP := SYSTIMESTAMP;
  3     v_invalid  NUMBER;
  4     v_stale    NUMBER;
  5  BEGIN
  6     SELECT COUNT(*) INTO v_invalid FROM user_objects WHERE status = 'INVALID';
  7     SELECT COUNT(*) INTO v_stale   FROM user_tables
  8      WHERE last_analyzed IS NULL OR last_analyzed < SYSDATE - 30;
  9
 10     INSERT INTO plsql_job_log(job_name, start_time, end_time, status,
 11                               processed_cnt, error_message, created_by)
 12     VALUES ('SCHEMA_CHECK', v_start, SYSTIMESTAMP,
 13             CASE WHEN v_invalid = 0 THEN 'SUCCESS' ELSE 'WARN' END,
 14             v_invalid + v_stale,
 15             'INVALID=' || v_invalid || ', STALE_STATS=' || v_stale, USER);
 16     COMMIT;
 17     DBMS_OUTPUT.PUT_LINE('점검 완료 : INVALID ' || v_invalid
 18                          || ' / 통계 미수집 ' || v_stale);
 19  END;

================================================================================
 정리
--------------------------------------------------------------------------------
 1. USER_OBJECTS 는 소유 객체의 이름·종류·상태·생성일·최종 DDL 시각을 보여 준다.
 2. 없는 객체명으로 조회하면 오류가 아니라 빈 결과다. 존재 여부는 COUNT(*)=0 으로
    판정하며, 조건 값은 반드시 UPPER 로 변환해야 한다.
 3. INVALID 객체는 USER_OBJECTS 로 찾고 동적 SQL 로 일괄 재컴파일한다.
    PACKAGE BODY 는 ALTER PACKAGE x COMPILE BODY 형태여야 한다.
 4. 재컴파일에 실패해 남는 INVALID 는 USER_ERRORS 로 원인을 확인한다.
 5. USER_TABLES 의 NUM_ROWS 는 통계 수집 시점의 값이며 실시간 값이 아니다.
    USER_INDEXES 의 STATUS 가 UNUSABLE 이면 인덱스가 사용되지 않는다.
 6. USER_PROCEDURES 의 AUTHID 로 정의자·호출자 권한을 확인할 수 있고,
    USER_SOURCE 로 코드 표준 점검을 자동화할 수 있다.
 7. 점검 결과는 화면 출력이 아니라 plsql_job_log 같은 로그 테이블에 남긴다.
