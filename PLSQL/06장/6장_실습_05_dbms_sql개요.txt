--   커서를 닫지 않아 세션의 열린 커서 수가 OPEN_CURSORS(300)를 넘었다.

SQL> CREATE OR REPLACE FUNCTION safe_count RETURN NUMBER IS
  2     v_cur   INTEGER;
  3     v_dummy NUMBER;
  4     v_cnt   NUMBER;
  5  BEGIN
  6     v_cur := DBMS_SQL.OPEN_CURSOR;
  7     DBMS_SQL.PARSE(v_cur, 'SELECT COUNT(*) FROM employees', DBMS_SQL.NATIVE);
  8     DBMS_SQL.DEFINE_COLUMN(v_cur, 1, v_cnt);
  9     v_dummy := DBMS_SQL.EXECUTE(v_cur);
 10     IF DBMS_SQL.FETCH_ROWS(v_cur) > 0 THEN
 11        DBMS_SQL.COLUMN_VALUE(v_cur, 1, v_cnt);
 12     END IF;
 13     DBMS_SQL.CLOSE_CURSOR(v_cur);
 14     RETURN v_cnt;
 15  EXCEPTION
 16     WHEN OTHERS THEN
 17        IF DBMS_SQL.IS_OPEN(v_cur) THEN
 18           DBMS_SQL.CLOSE_CURSOR(v_cur);
 19        END IF;
 20        RAISE;
 21  END safe_count;
 22  /

[6] TO_REFCURSOR 로 NDS 방식으로 넘긴다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_cur     INTEGER;
  3     v_rc      SYS_REFCURSOR;
  4     v_dummy   INTEGER;
  5     v_id      employees.employee_id%TYPE;
  6     v_name    employees.last_name%TYPE;
  7  BEGIN
  8     v_cur := DBMS_SQL.OPEN_CURSOR;
  9     DBMS_SQL.PARSE(v_cur,
 10        'SELECT employee_id, last_name FROM employees
 11          WHERE department_id = :d ORDER BY employee_id', DBMS_SQL.NATIVE);
 12     DBMS_SQL.BIND_VARIABLE(v_cur, ':d', 20);
 13     v_dummy := DBMS_SQL.EXECUTE(v_cur);
 14
 15     -- 여기서부터는 NDS 방식으로 다룬다
 16     v_rc := DBMS_SQL.TO_REFCURSOR(v_cur);
 17     LOOP
 18        FETCH v_rc INTO v_id, v_name;
 19        EXIT WHEN v_rc%NOTFOUND;
 20        DBMS_OUTPUT.PUT_LINE(v_id || ' / ' || v_name);
 21     END LOOP;
 22     CLOSE v_rc;
 23  END;
 24  /
201 / Hartstein
202 / Fay

PL/SQL procedure successfully completed.

-- TO_REFCURSOR 를 호출한 뒤에는 원래의 커서 번호로 DBMS_SQL 호출을 할 수 없다.
-- 소유권이 REF CURSOR 로 넘어간 것으로 보고 CLOSE 문으로 닫아야 한다.

SQL> DECLARE
  2     v_cur     INTEGER;
  3     v_rc      SYS_REFCURSOR;
  4     v_dummy   INTEGER;
  5  BEGIN
  6     v_cur := DBMS_SQL.OPEN_CURSOR;
  7     DBMS_SQL.PARSE(v_cur, 'SELECT 1 FROM dual', DBMS_SQL.NATIVE);
  8     v_dummy := DBMS_SQL.EXECUTE(v_cur);
  9     v_rc := DBMS_SQL.TO_REFCURSOR(v_cur);
 10     DBMS_SQL.CLOSE_CURSOR(v_cur);       -- 이미 넘어간 커서를 닫으려 한다
 11  END;
 12  /
DECLARE

================================================================================
 정리
--------------------------------------------------------------------------------
 1. DBMS_SQL 의 순서는 OPEN_CURSOR → PARSE → BIND_VARIABLE → EXECUTE →
    CLOSE_CURSOR 이며, 질의라면 DEFINE_COLUMN 과 FETCH_ROWS/COLUMN_VALUE 가 추가된다.
 2. PARSE 전에 BIND_VARIABLE 을 호출하면 ORA-01001 이 발생한다.
 3. CLOSE_CURSOR 를 빠뜨리면 커서가 누수되어 ORA-01000 이 발생한다.
    예외 절에 IS_OPEN 검사와 CLOSE_CURSOR 를 반드시 둔다.
 4. DESCRIBE_COLUMNS 는 결과 구조를 실행 시점에 조사한다. NDS 로는 불가능한
    Method 4 의 영역이며 이때만 DBMS_SQL 이 필요하다.
 5. TO_REFCURSOR 로 변환하면 이후에는 NDS 방식으로 다루며, 원래 커서 번호로
    DBMS_SQL 을 호출하면 ORA-29471 이 발생한다.
 6. DBMS_SQL 은 DDL 을 PARSE 시점에 곧바로 실행한다. EXECUTE 호출이 불필요하다.
