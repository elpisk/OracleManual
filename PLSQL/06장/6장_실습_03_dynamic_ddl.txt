 제6장 실습 03 — Dynamic DDL 실행
--------------------------------------------------------------------------------
 대상 스키마 : HR (EMPLOYEES / DEPARTMENTS / JOBS / JOB_HISTORY /
                   LOCATIONS / COUNTRIES / REGIONS)
 실행 환경   : Oracle Database 19c (19.3) / SQL*Plus
 학습 목표   : 1) PL/SQL 안에 DDL 을 직접 쓸 수 없음을 확인한다(PLS-00103)
               2) EXECUTE IMMEDIATE 로 CREATE / ALTER / DROP 을 실행한다
               3) 이미 존재하는 이름으로 CREATE 할 때의 ORA-00955 를 재현한다
               4) 없는 객체를 DROP 할 때의 ORA-00942 를 예외로 처리한다
               5) DDL 이 암묵적 COMMIT 을 유발함을 직접 확인한다
               6) plsql_job_log 테이블을 동적 DDL 로 생성한다

[2] (오류 상황) PL/SQL 블록 안에 DDL 을 직접 썼다
--------------------------------------------------------------------------------

SQL> BEGIN
  2     CREATE TABLE emp_backup AS SELECT * FROM employees;
  3  END;
  4  /
   CREATE TABLE emp_backup AS SELECT * FROM employees;
   *
ERROR at line 2:
ORA-06550: line 2, column 4:
PLS-00103: Encountered the symbol "CREATE" when expecting one of the following:
   begin case declare exit for goto if loop mod null pragma
   raise return select update while with <an identifier>

-- 원인 파악
--   PL/SQL 컴파일러는 DML 과 트랜잭션 제어문만 직접 받아들인다.
--   DDL 은 실행 시점에 객체를 만들거나 없애므로, 컴파일 시점에 아직
--   존재하지 않는 객체를 참조하는 코드는 컴파일될 수 없기 때문이다.
--   파서는 CREATE 라는 단어가 그 자리에 올 수 없다고 판단해 PLS-00103 을 낸다.

-- 수정 : EXECUTE IMMEDIATE 로 감싼다.

SQL> BEGIN
  2     EXECUTE IMMEDIATE 'CREATE TABLE emp_backup AS SELECT * FROM employees';
  3  END;
  4  /

PL/SQL procedure successfully completed.

SQL> SELECT table_name, num_rows FROM user_tables
  2   WHERE table_name = 'EMP_BACKUP';

TABLE_NAME           NUM_ROWS
-------------------- --------
EMP_BACKUP


SQL> SELECT COUNT(*) FROM emp_backup;

  COUNT(*)
----------
       107


--------------------------------------------------------------------------------
[3] (오류 상황) 이미 존재하는 이름으로 다시 CREATE 했다
--------------------------------------------------------------------------------

SQL> BEGIN
  2     EXECUTE IMMEDIATE 'CREATE TABLE emp_backup AS SELECT * FROM employees';
  3  END;
  4  /
BEGIN

*
ERROR at line 1:
ORA-00955: name is already used by an existing object
ORA-06512: at line 2

-- 원인 파악
--   EMP_BACKUP 이라는 이름이 이미 사용 중이다.
--   동적 DDL 은 컴파일러가 검사해 주지 않으므로 이런 충돌이 실행 시점에 드러난다.

-- 수정 방법 1 : 딕셔너리로 존재 여부를 먼저 확인한다.

SQL> DECLARE
  2     v_cnt   NUMBER;
  3  BEGIN
  4     SELECT COUNT(*) INTO v_cnt FROM user_tables
  5      WHERE table_name = 'EMP_BACKUP';
  6     IF v_cnt = 0 THEN
  7        EXECUTE IMMEDIATE 'CREATE TABLE emp_backup AS SELECT * FROM employees';
  8        DBMS_OUTPUT.PUT_LINE('EMP_BACKUP 생성 완료');
  9     ELSE
 10        DBMS_OUTPUT.PUT_LINE('EMP_BACKUP 이 이미 존재합니다.');
 11     END IF;
 12  END;
 13  /
EMP_BACKUP 이 이미 존재합니다.

PL/SQL procedure successfully completed.

-- 수정 방법 2 : 예외로 잡아 무시한다(더 견고하다).
--   확인과 실행 사이에 다른 세션이 객체를 만들 수도 있기 때문이다.

SQL> BEGIN
  2     EXECUTE IMMEDIATE 'CREATE TABLE emp_backup AS SELECT * FROM employees';
  3  EXCEPTION
  4     WHEN OTHERS THEN
  5        IF SQLCODE = -955 THEN
  6           DBMS_OUTPUT.PUT_LINE('이미 존재 - 건너뜀');
  7        ELSE
  8           RAISE;
  9        END IF;
 10  END;
 11  /
이미 존재 - 건너뜀

PL/SQL procedure successfully completed.

[4] (오류 상황) 없는 객체를 DROP 했다
--------------------------------------------------------------------------------

SQL> BEGIN
  2     EXECUTE IMMEDIATE 'DROP TABLE no_such_table PURGE';
  3  END;
  4  /
BEGIN

*
ERROR at line 1:
ORA-00942: table or view does not exist
ORA-06512: at line 2

-- 원인 파악
--   존재하지 않는 객체를 지우려 했다. 자동화 스크립트에서 "있으면 지운다"
--   패턴을 쓸 때 이 오류를 예외로 흡수해야 한다.

-- 수정 : ORA-00942 만 무시하는 프로시저를 만든다.

SQL> CREATE OR REPLACE PROCEDURE drop_if_exists(p_table IN VARCHAR2) IS
  2  BEGIN
  3     EXECUTE IMMEDIATE 'DROP TABLE '
  4                       || DBMS_ASSERT.SIMPLE_SQL_NAME(p_table) || ' PURGE';
  5     DBMS_OUTPUT.PUT_LINE(p_table || ' 삭제 완료');
  6  EXCEPTION
  7     WHEN OTHERS THEN
  8        IF SQLCODE = -942 THEN
  9           DBMS_OUTPUT.PUT_LINE(p_table || ' 없음 - 건너뜀');
 10        ELSE
 11           RAISE;
 12        END IF;
 13  END drop_if_exists;
 14  /

Procedure created.

SQL> EXEC drop_if_exists('NO_SUCH_TABLE')
NO_SUCH_TABLE 없음 - 건너뜀

PL/SQL procedure successfully completed.

SQL> EXEC drop_if_exists('EMP_BACKUP')
EMP_BACKUP 삭제 완료

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[5] DDL 은 암묵적 COMMIT 을 유발한다
--------------------------------------------------------------------------------

-- 먼저 현재 건수를 확인한다.

SQL> SELECT COUNT(*) FROM emp_copy WHERE department_id = 10;

  COUNT(*)
----------
         1

SQL> DELETE FROM emp_copy WHERE department_id = 10;

1 row deleted.

-- 아직 커밋하지 않았다. 여기서 동적 DDL 을 한 줄 실행한다.

SQL> BEGIN
  2     EXECUTE IMMEDIATE 'CREATE TABLE t_tmp (id NUMBER)';
  3  END;
  4  /

PL/SQL procedure successfully completed.

-- 이제 롤백을 시도한다.

SQL> ROLLBACK;

Rollback complete.

SQL> SELECT COUNT(*) FROM emp_copy WHERE department_id = 10;

  COUNT(*)
----------
         0

-- 원인 파악
--   삭제된 행이 돌아오지 않았다. 오라클은 DDL 문을 실행하기 직전에 한 번,
--   실행한 직후에 한 번 커밋을 수행하기 때문이다(암묵적 COMMIT).
--   따라서 동적 DDL 을 트랜잭션 한가운데 넣으면 앞선 DML 이 확정되어 버린다.

-- 복구 : 삭제한 행을 원본에서 다시 넣는다.

SQL> INSERT INTO emp_copy
  2  SELECT * FROM employees WHERE department_id = 10;

1 row created.

SQL> COMMIT;

Commit complete.

SQL> SELECT COUNT(*) FROM emp_copy;

  COUNT(*)
----------
       107

SQL> EXEC drop_if_exists('T_TMP')
T_TMP 삭제 완료

PL/SQL procedure successfully completed.

================================================================================
 정리
--------------------------------------------------------------------------------
 1. PL/SQL 안에 DDL 을 직접 쓰면 PLS-00103 이 발생한다.
    DDL 은 반드시 EXECUTE IMMEDIATE 로 감싸 실행한다.
 2. 이미 있는 이름으로 CREATE 하면 ORA-00955, 없는 객체를 DROP 하면 ORA-00942.
    자동화 코드에서는 SQLCODE 로 해당 오류만 골라 흡수한다.
 3. DDL 은 실행 직전과 직후에 각각 커밋을 수행한다(암묵적 COMMIT).
    앞선 DML 이 확정되므로 ROLLBACK 으로 되돌릴 수 없다.
 4. 동적 DDL 에는 바인드 변수를 쓸 수 없으므로 객체 이름은 문자열로 결합한다.
    DBMS_ASSERT.SIMPLE_SQL_NAME 으로 반드시 검증한다.
 5. 조립한 문장을 출력하거나 로그에 남겨 두면 ORA-00906 · ORA-00922 같은
    조립 오류의 원인을 즉시 확인할 수 있다.
