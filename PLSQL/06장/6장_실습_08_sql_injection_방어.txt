 제6장 실습 08 — Dynamic SQL Injection 위험성 확인 및 Bind Variable 방어
--------------------------------------------------------------------------------
 대상 스키마 : HR (EMPLOYEES / DEPARTMENTS / JOBS / JOB_HISTORY /
                   LOCATIONS / COUNTRIES / REGIONS)
 실행 환경   : Oracle Database 19c (19.3) / SQL*Plus
 주의        : 이 실습은 데이터를 변경하는 시나리오를 포함한다.
               반드시 복사본 테이블(EMP_COPY)에서만 수행하고 확인 즉시 ROLLBACK 한다.
 학습 목표   : 1) 문자열 결합 동적 SQL 의 SQL Injection 을 안전하게 재현한다
               2) 조회 유출과 데이터 변경 두 가지 피해 유형을 확인한다
               3) Bind Variable 로 값 자리를 방어한다
               4) DBMS_ASSERT 로 식별자 자리를 검증한다
               5) 딕셔너리 대조 화이트리스트로 최종 방어선을 만든다
               6) 방어 결과를 plsql_job_log 에 기록한다

SQL> EXEC find_emp_bad('x'' OR ''1''=''1')

-- 원인 파악
--   입력값의 작은따옴표가 WHERE 절의 문자열을 닫아 버렸다.
--   그 뒤의 OR '1'='1' 이 조건절의 일부로 해석되어 WHERE 절이 항상 참이 되었다.
--   그 결과 특정 사원 한 명을 조회하려던 문장이 테이블 전체를 반환했다.
--   출력된 "실행 문장"을 보면 조립 결과가 어떻게 바뀌었는지 한눈에 확인할 수 있다.

-- 더 진행된 공격 : UNION 으로 다른 테이블의 데이터를 끌어온다.

department_id FROM departments WHERE '1'='1'

   Benefits / 160

-- SELECT 목록의 개수와 자료형만 맞추면 전혀 다른 테이블을 조회할 수 있다.

[5] (방어 1) 값은 예외 없이 바인딩한다

SQL> CREATE OR REPLACE PROCEDURE find_emp_safe(p_name IN VARCHAR2) IS
  2     TYPE t_tab IS TABLE OF VARCHAR2(120);
  3     v_rows   t_tab;
  4  BEGIN
  5     EXECUTE IMMEDIATE
  6        'SELECT last_name || '' / '' || salary FROM emp_copy
  7          WHERE last_name = :name'
  8        BULK COLLECT INTO v_rows USING p_name;
  9     DBMS_OUTPUT.PUT_LINE('조회 건수 : ' || v_rows.COUNT);
 10     FOR i IN 1 .. LEAST(v_rows.COUNT, 3) LOOP
 11        DBMS_OUTPUT.PUT_LINE('   ' || v_rows(i));
 12     END LOOP;
 13  END find_emp_safe;
 14  /

Procedure created.

SQL> EXEC find_emp_safe('King')
조회 건수 : 2
   King / 24000
   King / 10000

PL/SQL procedure successfully completed.

-- 같은 공격 문자열을 넣어 본다.

SQL> EXEC find_emp_safe('x'' OR ''1''=''1')

PL/SQL procedure successfully completed.

SQL> EXEC find_emp_safe('X'' OR 1=1 --')
조회 건수 : 0

-- 원리
--   바인드 변수로 넘긴 값은 문장이 파싱된 뒤에 전달된다.
--   따라서 그 안에 어떤 문자가 들어 있어도 문장의 구조를 바꿀 수 없고
--   입력 전체가 "이름이 이상한 검색어" 하나로 취급된다.
--   문장 텍스트가 항상 동일해지므로 하드 파스도 사라진다. 보안과 성능을 함께 얻는다.

-- 삭제 프로시저도 같은 방식으로 고친다.

SQL> CREATE OR REPLACE PROCEDURE del_emp_safe(p_job IN VARCHAR2) IS
  2  BEGIN
  3     EXECUTE IMMEDIATE 'DELETE FROM emp_copy WHERE job_id = :j'
  4        USING p_job;
  5     DBMS_OUTPUT.PUT_LINE('삭제 건수 : ' || SQL%ROWCOUNT);
  6  END del_emp_safe;
  7  /

Procedure created.

  5  END;
  6  /

-- 원인 : 바인드 변수는 리터럴이 올 수 있는 자리에만 놓을 수 있다.
--        테이블 이름 자리에는 플레이스홀더를 쓸 수 없다.

-- 따라서 문자열 결합이 불가피하다. 대신 DBMS_ASSERT 로 검증한다.

SQL> CREATE OR REPLACE FUNCTION count_rows_assert(p_table IN VARCHAR2)
  2     RETURN NUMBER IS
  3     v_cnt   NUMBER;
  4  BEGIN
  5     EXECUTE IMMEDIATE 'SELECT COUNT(*) FROM '
  6                       || DBMS_ASSERT.SIMPLE_SQL_NAME(p_table)
  7        INTO v_cnt;
  8     RETURN v_cnt;
  9  END count_rows_assert;
 10  /

Function created.

[7] (방어 3) DBMS_ASSERT 만으로는 부족하다 — 화이트리스트를 더한다
--------------------------------------------------------------------------------

-- DBMS_ASSERT 는 "문법적으로 올바른 이름인가"만 검사한다.
-- 아래 이름은 문법적으로 올바르므로 그대로 통과한다.

SQL> SELECT DBMS_ASSERT.SIMPLE_SQL_NAME('PLSQL_JOB_LOG') AS ok FROM dual;

OK
--------------------
PLSQL_JOB_LOG

-- 즉 사용자가 의도하지 않은 테이블을 지정해도 막지 못한다.

SQL> SELECT count_rows_assert('PLSQL_JOB_LOG') AS cnt FROM dual;

       CNT
----------
         2

-- 최종 방어선 : 딕셔너리를 조회해 허용된 대상인지 대조한다.

SQL> CREATE OR REPLACE FUNCTION count_rows_safe(p_table IN VARCHAR2)
  2     RETURN NUMBER IS
  3     v_name   VARCHAR2(128);
  4     v_cnt    NUMBER;
  5  BEGIN
  6     -- 1) 딕셔너리 대조(화이트리스트) : 존재하고 내가 소유한 테이블인가
  7     SELECT table_name INTO v_name
  8       FROM user_tables
  9      WHERE table_name = UPPER(TRIM(p_table))
 10        AND table_name IN ('EMP_COPY', 'EMPLOYEES', 'DEPARTMENTS');
 11
 12     -- 2) 확인된 이름만 문장에 결합한다
 13     EXECUTE IMMEDIATE 'SELECT COUNT(*) FROM "' || v_name || '"' INTO v_cnt;
 14     RETURN v_cnt;
 15  EXCEPTION
 16     WHEN NO_DATA_FOUND THEN
 17        RAISE_APPLICATION_ERROR(-20010, '허용되지 않은 테이블: ' || p_table);
 18  END count_rows_safe;
 19  /

Function created.

SQL> SELECT count_rows_safe('emp_copy') AS cnt FROM dual;

       CNT
----------
       107

SQL> SELECT count_rows_safe('PLSQL_JOB_LOG') AS cnt FROM dual;
SELECT count_rows_safe('PLSQL_JOB_LOG') AS cnt FROM dual
       *
ERROR at line 1:
ORA-20010: 허용되지 않은 테이블: PLSQL_JOB_LOG
ORA-06512: at "HR.COUNT_ROWS_SAFE", line 17

-- 이제 문장에 들어가는 것은 사용자 입력이 아니라 딕셔너리에서 읽어 온 값이다.

[9] 방어 점검 결과를 plsql_job_log 에 기록한다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_start   TIMESTAMP := SYSTIMESTAMP;
  3     v_bad     NUMBER;
  4  BEGIN
  5     -- 소스에서 문자열 결합 패턴이 남아 있는 프로그램을 찾는다
  6     SELECT COUNT(DISTINCT name) INTO v_bad
  7       FROM user_source
  8      WHERE UPPER(text) LIKE '%EXECUTE IMMEDIATE%'
  9         OR UPPER(text) LIKE '%|| P_%';
 10
 11     INSERT INTO plsql_job_log(job_name, start_time, end_time, status,
 12                               processed_cnt, error_message, created_by)
 13     VALUES ('INJECTION_SCAN', v_start, SYSTIMESTAMP,
 14             CASE WHEN v_bad = 0 THEN 'SUCCESS' ELSE 'WARN' END,
 15             v_bad,
 16             '문자열 결합 의심 프로그램 ' || v_bad || '건 - 코드 리뷰 필요',
 17             USER);
 18     COMMIT;
 19     DBMS_OUTPUT.PUT_LINE('점검 완료 : 의심 프로그램 ' || v_bad || '건');
 20  END;
 21  /

 정리
--------------------------------------------------------------------------------
 1. 문자열 결합으로 만든 동적 SQL 은 입력값이 코드로 해석될 수 있다.
    ' OR '1'='1 같은 입력 하나로 WHERE 절이 항상 참이 되어 전체가 노출된다.
 2. 같은 취약점이 DELETE·UPDATE 에 있으면 데이터가 통째로 사라진다.
    실습은 반드시 복사본에서 수행하고 확인 즉시 ROLLBACK 한다.
 3. 값은 예외 없이 바인딩한다. 바인드 값은 파싱 이후에 전달되므로
    문장의 구조를 바꿀 수 없고, 덤으로 하드 파스도 사라진다.
 4. 테이블·컬럼·정렬 컬럼은 식별자라 바인딩할 수 없다(ORA-00903).
    DBMS_ASSERT.SIMPLE_SQL_NAME 으로 문법을 검증한다(위반 시 ORA-44003).
 5. DBMS_ASSERT 는 문법만 검사하므로 허용 대상까지 막지는 못한다.
    딕셔너리 대조 화이트리스트를 더해 확인된 이름만 문장에 결합한다.
 6. 조립한 문장을 로그로 남기면 공격 시도와 조립 오류를 사후에 추적할 수 있다.
