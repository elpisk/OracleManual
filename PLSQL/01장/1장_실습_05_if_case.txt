SQL> DECLARE
  2     v_last_name  employees.last_name%TYPE;
  3     v_salary     employees.salary%TYPE;
  4     v_job_id     employees.job_id%TYPE;
  5     v_grade      VARCHAR2(20);
  6     v_bonus      NUMBER(10,2);
  7  BEGIN
  8     SELECT last_name, salary, job_id
  9       INTO v_last_name, v_salary, v_job_id
 10       FROM employees
 11      WHERE employee_id = 103;
 12
 13     -- 범위 조건은 반드시 '큰 값부터' 검사한다
 14     IF    v_salary >= 15000 THEN  v_grade := 'EXECUTIVE';
 15     ELSIF v_salary >= 10000 THEN  v_grade := 'SENIOR';
 16     ELSIF v_salary >=  6000 THEN  v_grade := 'MIDDLE';
 17     ELSE                          v_grade := 'JUNIOR';
 18     END IF;
 19
 20     -- 논리 연산자를 사용한 복합 조건

 25     END IF;
 26
 27     DBMS_OUTPUT.PUT_LINE(v_last_name || ' / ' || v_job_id
 28                          || ' / ' || v_salary);

[4] NULL 조건식과 3값 논리
--------------------------------------------------------------------------------

-- HR의 commission_pct는 영업직이 아닌 사원에게 NULL이다.
SQL> SELECT employee_id, last_name, commission_pct
  2    FROM employees WHERE employee_id IN (103, 145);

EMPLOYEE_ID LAST_NAME                 COMMISSION_PCT
----------- ------------------------- --------------
        103 Hunold
        145 Russell                              .4

SQL> DECLARE
  2     v_comm   employees.commission_pct%TYPE;
  3  BEGIN
  4     SELECT commission_pct INTO v_comm
  5       FROM employees WHERE employee_id = 103;    -- NULL
  6
  7     DBMS_OUTPUT.PUT_LINE('v_comm = ' || NVL(TO_CHAR(v_comm), 'NULL'));
  8
  9     IF v_comm > 0 THEN
 10        DBMS_OUTPUT.PUT_LINE('[A] v_comm > 0 이 참');
 11     ELSE
 12        DBMS_OUTPUT.PUT_LINE('[B] v_comm > 0 이 참이 아님 (ELSE 실행)');
 13     END IF;
 14
 15     IF NOT (v_comm > 0) THEN
 16        DBMS_OUTPUT.PUT_LINE('[C] NOT (v_comm > 0) 이 참');
 17     ELSE
 18        DBMS_OUTPUT.PUT_LINE('[D] NOT (v_comm > 0) 도 참이 아님 (ELSE 실행)');
 19     END IF;
 20
 21     IF v_comm IS NULL THEN
 22        DBMS_OUTPUT.PUT_LINE('[E] IS NULL 로 검사하면 정확히 판정된다');
 23     END IF;
 24  END;
 25  /
v_comm = NULL
[B] v_comm > 0 이 참이 아님 (ELSE 실행)
[D] NOT (v_comm > 0) 도 참이 아님 (ELSE 실행)
[E] IS NULL 로 검사하면 정확히 판정된다

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    v_comm이 NULL이면 v_comm > 0 의 결과는 TRUE도 FALSE도 아닌 NULL이다.
--    IF는 NULL을 '만족하지 않음'으로 보고 ELSE로 넘어가며,
--    NULL을 부정해도 NULL이므로 NOT (v_comm > 0) 역시 실행되지 않는다.
--    NULL 가능성이 있는 값은 반드시 IS NULL 또는 NVL로 처리해야 한다.

-- ▶ 실무 방식: NVL로 기본값을 주어 NULL 자체를 제거한다
SQL> DECLARE
  2     v_comm    employees.commission_pct%TYPE;
  3     v_salary  employees.salary%TYPE;
  4     v_total   NUMBER(10,2);
  5  BEGIN
  6     SELECT salary, commission_pct INTO v_salary, v_comm
  7       FROM employees WHERE employee_id = 103;
  8
  9     v_total := v_salary * (1 + NVL(v_comm, 0));   -- NULL이면 0으로 취급
 10     DBMS_OUTPUT.PUT_LINE('총 보상액: ' || v_total);
 11  END;
 12  /

[5] CASE 표현식 — 값을 만들어 변수에 대입한다 (END로 종료)
--------------------------------------------------------------------------------

SQL> DECLARE
  2     rec_emp      employees%ROWTYPE;
  3     v_appraisal  VARCHAR2(20);
  4     v_job_kor    VARCHAR2(30);
  5  BEGIN
  6     SELECT * INTO rec_emp FROM employees WHERE employee_id = 178;
  7
  8     -- 검색 CASE 표현식: 범위 조건 판정에 적합
  9     v_appraisal := CASE
 10                       WHEN rec_emp.salary >= 20000 THEN 'TOP'
 11                       WHEN rec_emp.salary >= 10000 THEN 'HIGH'
 12                       WHEN rec_emp.salary >=  6000 THEN 'MIDDLE'
 13                       ELSE 'ENTRY'
 14                    END;                    -- 표현식이므로 END
 15
 16     -- 단순 CASE 표현식: 코드 값 매핑에 적합
 17     v_job_kor := CASE rec_emp.job_id
 18                     WHEN 'IT_PROG' THEN '프로그래머'
 19                     WHEN 'SA_REP'  THEN '영업담당'
 20                     WHEN 'AD_PRES' THEN '대표이사'
 21                     ELSE '기타직무'
 22                  END;
 23
 24     DBMS_OUTPUT.PUT_LINE(rec_emp.last_name || ' / ' || v_job_kor

 26  END;

[9] 단순 CASE의 함정 — 셀렉터가 NULL이면 어디에도 걸리지 않는다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_dept   employees.department_id%TYPE;
  3     v_nm     VARCHAR2(30);
  4  BEGIN
  5     SELECT department_id INTO v_dept
  6       FROM employees WHERE employee_id = 178;    -- 부서 미배정(NULL)
  7
  8     -- NVL 없이 작성하면 WHEN NULL 가지에도 걸리지 않는다
  9     v_nm := CASE v_dept
 10                WHEN 20   THEN 'Marketing'
 11                WHEN NULL THEN '(부서 미배정)'   -- 절대 선택되지 않음
 12                ELSE 'Others'
 13             END;
 14     DBMS_OUTPUT.PUT_LINE('NVL 없음 : ' || v_nm);
 15
 16     -- NVL로 셀렉터를 치환하면 의도한 분기가 동작한다
 17     v_nm := CASE NVL(v_dept, -1)
 18                WHEN 20 THEN 'Marketing'
 19                WHEN -1 THEN '(부서 미배정)'
 20                ELSE 'Others'
 21             END;
 22     DBMS_OUTPUT.PUT_LINE('NVL 사용 : ' || v_nm);
 23  END;
 24  /
NVL 없음 : Others
NVL 사용 : (부서 미배정)

PL/SQL procedure successfully completed.

-- ▶ 단순 CASE는 셀렉터와 WHEN 값을 '등호'로 비교하는데
--    NULL = NULL은 참이 아니므로 WHEN NULL 가지는 결코 선택되지 않는다.
--    NULL 분기가 필요하면 NVL로 치환하거나 검색 CASE에 IS NULL을 쓴다.

[10] 정리
--------------------------------------------------------------------------------
--  · 다중 분기 키워드는 ELSIF (ELSEIF/ELIF 아님), 종료는 END IF;
--  · 범위 조건은 반드시 큰 값부터 검사한다 (도달 불가 코드 방지)
--  · 조건식이 NULL이면 IF는 ELSE로 간다. NOT NULL도 여전히 NULL이다
--  · CASE 표현식 → 값 반환, END로 종료, 미일치 시 NULL
--  · CASE 문     → 문장 실행, END CASE;로 종료, 미일치 시 ORA-06592
--  · 재현한 오류 : PLS-00103(ELSEIF) → ELSIF로 수정
--                  논리 오류(ELSIF 순서) → 큰 값부터 검사로 수정
--                  PLS-00103(END CASE 오용) → END로 수정
--                  ORA-06592(CASE_NOT_FOUND) → ELSE 절 추가로 수정
