[1] 기본 변수 선언과 초기화 — 초기값이 없으면 NULL이다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_emp_id      NUMBER(6);                  -- 초기값 없음 → NULL
  3     v_last_name   VARCHAR2(25);               -- 초기값 없음 → NULL
  4     v_salary      NUMBER(8,2) := 0;           -- := 로 초기화
  5     v_hire_date   DATE DEFAULT SYSDATE;       -- DEFAULT 로 초기화
  6     v_dept_id     NUMBER(4) NOT NULL := 50;   -- NOT NULL은 초기값 필수
  7     v_is_manager  BOOLEAN := FALSE;           -- PL/SQL 전용 타입
  8  BEGIN
  9     DBMS_OUTPUT.PUT_LINE('v_emp_id    = ' || NVL(TO_CHAR(v_emp_id), 'NULL'));
 10     DBMS_OUTPUT.PUT_LINE('v_last_name = ' || NVL(v_last_name, 'NULL'));
 11     DBMS_OUTPUT.PUT_LINE('v_salary    = ' || v_salary);
 12     DBMS_OUTPUT.PUT_LINE('v_hire_date = ' || TO_CHAR(v_hire_date, 'YYYY-MM-DD'));
 13     DBMS_OUTPUT.PUT_LINE('v_dept_id   = ' || v_dept_id);
 14     -- BOOLEAN은 직접 출력할 수 없어 IF로 분기해 문자열로 바꾼다
 15     IF v_is_manager THEN
 16        DBMS_OUTPUT.PUT_LINE('v_is_manager= TRUE');
 17     ELSE
 18        DBMS_OUTPUT.PUT_LINE('v_is_manager= FALSE');
 19     END IF;
 20  END;
 21  /

[2] (오류 재현) 초기화하지 않은 숫자 변수로 누적 계산하기
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_total   NUMBER;                      -- 초기값 없음 → NULL
  3  BEGIN
  4     FOR i IN 1..3 LOOP
  5        v_total := v_total + i;             -- NULL + 숫자 = NULL
  6     END LOOP;
  7     DBMS_OUTPUT.PUT_LINE('합계: ' || NVL(TO_CHAR(v_total), 'NULL'));
  8  END;
  9  /

-- ▶ 원인 파악
--    오류 메시지는 나오지 않지만 결과가 틀렸다(논리 오류).
--    PL/SQL 변수는 타입과 무관하게 초기값이 NULL이다.
--    숫자 변수라고 해서 0에서 시작하지 않으며, NULL에 무엇을 더해도 NULL이다.

-- ▶ 수정: 선언 시점에 0으로 초기화한다(근본 해결)
SQL> DECLARE
  2     v_total   NUMBER := 0;
  3  BEGIN
  4     FOR i IN 1..3 LOOP
  5        v_total := v_total + i;
  6     END LOOP;
  7     DBMS_OUTPUT.PUT_LINE('합계: ' || v_total);
  8  END;

[5] %TYPE이 상속하지 않는 것 — NOT NULL 제약은 따라오지 않는다
--------------------------------------------------------------------------------

-- LAST_NAME 컬럼은 NOT NULL이지만, %TYPE으로 선언한 변수는 NULL을 가질 수 있다.
SQL> DECLARE
  2     v_last_name   employees.last_name%TYPE;   -- 컬럼은 NOT NULL이지만
  3  BEGIN
  4     v_last_name := NULL;                      -- 변수에는 NULL 할당이 가능하다
  5     DBMS_OUTPUT.PUT_LINE('v_last_name = ' || NVL(v_last_name, '(NULL)'));
  6  END;
  7  /
v_last_name = (NULL)

-- ▶ %TYPE이 상속하는 것    : 데이터 타입, 길이/정밀도/스케일
-- ▶ %TYPE이 상속하지 않는 것: 컬럼의 NOT NULL 제약, DEFAULT 값

-- ▶ 원인 파악
--    EMPLOYEES 테이블에는 EMP_NAME 컬럼이 없다(정확한 이름은 LAST_NAME).

[7] %ROWTYPE — 행 전체를 하나의 레코드 변수에 담는다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     rec_emp    employees%ROWTYPE;      -- EMPLOYEES 한 행 전체
  3     rec_dept   departments%ROWTYPE;    -- DEPARTMENTS 한 행 전체
  4  BEGIN
  5     -- %ROWTYPE 레코드는 SELECT * INTO 로 채우는 것이 원칙이다
  6     SELECT * INTO rec_emp
  7       FROM employees
  8      WHERE employee_id = 100;
  9
 10     -- 레코드의 필드를 다시 다른 SQL의 조건으로 사용할 수 있다
 11     SELECT * INTO rec_dept
 12       FROM departments
 13      WHERE department_id = rec_emp.department_id;
 14
 15     DBMS_OUTPUT.PUT_LINE('사원   : ' || rec_emp.first_name
 16                          || ' ' || rec_emp.last_name);
 17     DBMS_OUTPUT.PUT_LINE('직무   : ' || rec_emp.job_id
 18                          || ' / 급여: ' || rec_emp.salary);
 19     DBMS_OUTPUT.PUT_LINE('입사일 : '
 20                          || TO_CHAR(rec_emp.hire_date, 'YYYY-MM-DD'));
 21     DBMS_OUTPUT.PUT_LINE('부서   : ' || rec_dept.department_name
 22                          || ' (' || rec_dept.department_id || ')');
 23  END;

--------------------------------------------------------------------------------
[10] 정리
--------------------------------------------------------------------------------
--  · 변수 초기값은 항상 NULL — 누적 변수는 반드시 := 0 으로 초기화
--  · 컬럼 값을 담는 변수는 예외 없이 %TYPE으로 선언한다
--  · %TYPE은 타입·길이만 상속하고 NOT NULL/DEFAULT는 상속하지 않는다
--  · %ROWTYPE은 행 전체를 담으며 SELECT * INTO 와 짝지어 사용한다
--  · 재현한 오류 : ORA-06502(길이 부족) → %TYPE으로 해결
--                  PLS-00302(잘못된 컬럼명) → 올바른 컬럼명으로 해결
--                  PLS-00394(INTO 개수 불일치) → SELECT * 또는 스칼라 변수로 해결
