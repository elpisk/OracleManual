[2] 가장 단순한 블록 — BEGIN과 END;만 있는 형태
--------------------------------------------------------------------------------

-- 실행부에는 최소 한 개의 문장이 있어야 한다.
-- 아무 동작도 하지 않는 블록은 NULL; 문장을 넣어 만든다.
SQL> BEGIN
  2     NULL;
  3  END;
  4  /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
[3] (오류 재현) 실행부를 비워 두면 어떻게 되는가
--------------------------------------------------------------------------------

SQL> BEGIN
  2  END;
  3  /
END;
*
ERROR at line 2:
ORA-06550: line 2, column 1:
PLS-00103: Encountered the symbol "END" when expecting one of the following:
   ( begin case declare exit for goto if loop mod null pragma
   raise return select update while with <an identifier> ...

-- ▶ 원인 파악
--    PL/SQL의 실행부(BEGIN ~ END)는 '최소 한 개의 문장'을 요구한다.
--    BEGIN 다음에 곧바로 END가 오면 컴파일러는 문장을 기대하다가
--    END를 만나 PLS-00103(구문 오류)을 발생시킨다.
--    앞줄의 ORA-06550은 '컴파일에 실패했다'는 껍데기 메시지이고,
--    실제 원인은 그 다음 줄의 PLS- 메시지에 있다.
--    컴파일 오류이므로 블록은 한 줄도 실행되지 않았다.

-- ▶ 수정
SQL> BEGIN
  2     NULL;                      -- 아무 동작도 하지 않는 문장
  3  END;
  4  /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
[4] 네 개의 절을 모두 사용하는 완전한 블록
--------------------------------------------------------------------------------

SQL> DECLARE                                   -- 선언부(선택)
  2     v_emp_id      employees.employee_id%TYPE := 100;
  3     v_last_name   employees.last_name%TYPE;
  4     v_salary      employees.salary%TYPE;
  5  BEGIN                                     -- 실행부(필수)
  6     SELECT last_name, salary
  7       INTO v_last_name, v_salary
  8       FROM employees
  9      WHERE employee_id = v_emp_id;
 10
 11     DBMS_OUTPUT.PUT_LINE('사번   : ' || v_emp_id);
 12     DBMS_OUTPUT.PUT_LINE('사원명 : ' || v_last_name);
 13     DBMS_OUTPUT.PUT_LINE('급여   : ' || v_salary);
 14  EXCEPTION                                 -- 예외 처리부(선택)
 15     WHEN NO_DATA_FOUND THEN
 16        DBMS_OUTPUT.PUT_LINE('해당 사번의 사원이 없습니다.');
 17     WHEN TOO_MANY_ROWS THEN
 18        DBMS_OUTPUT.PUT_LINE('조건에 맞는 사원이 2명 이상입니다.');
 19  END;                                      -- 종료(필수)
 20  /
사번   : 100
사원명 : King
급여   : 24000

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    DECLARE, BEGIN, EXCEPTION 키워드 뒤에는 세미콜론을 붙이지 않는다.
--    반면 END 뒤에는 반드시 세미콜론을 붙이고,
--    모든 문장(변수 선언 포함)도 각각 세미콜론으로 끝난다.

--------------------------------------------------------------------------------
[5] 예외 처리부가 실제로 동작하는지 확인한다
--------------------------------------------------------------------------------

-- 존재하지 않는 사번 9999로 바꾸어 NO_DATA_FOUND를 유도한다.
SQL> DECLARE
  2     v_emp_id      employees.employee_id%TYPE := 9999;
  3     v_last_name   employees.last_name%TYPE;
  4  BEGIN
  5     SELECT last_name INTO v_last_name
  6       FROM employees
  7      WHERE employee_id = v_emp_id;
  8     DBMS_OUTPUT.PUT_LINE('사원명 : ' || v_last_name);
  9  EXCEPTION
 10     WHEN NO_DATA_FOUND THEN
 11        DBMS_OUTPUT.PUT_LINE('사번 ' || v_emp_id || ' 사원이 존재하지 않습니다.');
 12  END;
 13  /
사번 9999 사원이 존재하지 않습니다.

PL/SQL procedure successfully completed.

-- ▶ 예외가 처리되었으므로 블록은 '실패'가 아니라 '정상 종료'된다.
--    예외 처리부가 없었다면 ORA-01403으로 블록 전체가 실패했을 것이다.

--------------------------------------------------------------------------------
[6] 슬래시(/)의 역할 — 세미콜론만으로는 블록이 실행되지 않는다
--------------------------------------------------------------------------------

-- END; 를 입력해도 슬래시를 넣지 않으면 SQL*Plus는 계속 입력을 기다린다.
SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('슬래시 테스트');
  3  END;
  4
  5
  6

-- ▶ 원인 파악
--    세미콜론은 'PL/SQL 문장의 끝'을 뜻할 뿐이고,
--    슬래시는 'SQL*Plus 버퍼의 내용을 서버로 보내 실행하라'는 명령이다.
--    따라서 익명 블록은 반드시 마지막 줄에 슬래시를 단독으로 입력해야 한다.

-- ▶ 수정: 다음 줄에 슬래시를 입력하면 그 시점에 블록이 실행된다.
  7  /
슬래시 테스트

PL/SQL procedure successfully completed.

-- 참고: 단일 SQL 문장은 세미콜론만으로 실행되므로 슬래시가 필요 없다.
SQL> SELECT COUNT(*) AS emp_cnt FROM employees;

[7] 중첩 블록 — 실행부 안에 또 다른 블록을 넣을 수 있다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_outer_msg   VARCHAR2(50) := '바깥 블록';
  3  BEGIN
  4     DBMS_OUTPUT.PUT_LINE('[1] ' || v_outer_msg || ' 시작');
  5
  6     BEGIN                                  -- 중첩 블록 시작
  7        DBMS_OUTPUT.PUT_LINE('[2] 안쪽 블록 실행 (바깥 변수 참조: '
  8                             || v_outer_msg || ')');
  9     END;                                   -- 중첩 블록 종료
 10
 11     DBMS_OUTPUT.PUT_LINE('[3] ' || v_outer_msg || ' 종료');
 12  END;
 13  /
[1] 바깥 블록 시작
[2] 안쪽 블록 실행 (바깥 변수 참조: 바깥 블록)
[3] 바깥 블록 종료

PL/SQL procedure successfully completed.

-- ▶ 주의: 블록은 실행부와 예외 처리부 안에만 중첩할 수 있고,
--         선언부(DECLARE 절) 안에는 중첩할 수 없다.

--------------------------------------------------------------------------------
[8] 정리
--------------------------------------------------------------------------------
--  · 필수 절  : BEGIN, END;      (실행부에는 최소 한 문장 필요 → NULL;)
--  · 선택 절  : DECLARE, EXCEPTION
--  · 세미콜론 : 문장의 끝        슬래시(/) : 블록 전체의 실행 지시
--  · 컴파일 오류(PLS-)는 실행 자체를 막고, 런타임 오류(ORA-)는 실행 중 발생한다
--  · 재현한 오류 : PLS-00103 (실행부에 문장이 없음) → NULL; 추가로 해결

SQL> EXIT
