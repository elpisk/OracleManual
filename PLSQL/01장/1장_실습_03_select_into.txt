[2] (오류 재현) 조회 결과가 0건일 때 — NO_DATA_FOUND
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_last_name   employees.last_name%TYPE;
  3  BEGIN
  4     SELECT last_name INTO v_last_name
  5       FROM employees
  6      WHERE employee_id = 9999;             -- 존재하지 않는 사번
  7     DBMS_OUTPUT.PUT_LINE('사원명 : ' || v_last_name);
  8  END;
  9  /
DECLARE
*
ERROR at line 1:
ORA-01403: no data found
ORA-06512: at line 4

-- ▶ 원인 파악
--    SQL의 SELECT는 0건이어도 그냥 'no rows selected'로 끝나지만,
--    PL/SQL의 SELECT INTO는 값을 변수에 담아야 하므로
--    한 행도 없으면 NO_DATA_FOUND(ORA-01403) 예외를 발생시킨다.
--    예외 처리부가 없으면 오류가 호출자에게 전파되어 블록 전체가 실패한다.

-- ▶ 참고: 같은 조건을 일반 SQL로 실행하면 예외가 아니다
SQL> SELECT last_name FROM employees WHERE employee_id = 9999;

no rows selected

-- ▶ 수정: 예외 처리부에서 NO_DATA_FOUND를 처리한다
SQL> DECLARE
  2     v_emp_id      employees.employee_id%TYPE := 9999;
  3     v_last_name   employees.last_name%TYPE;
  4  BEGIN
  5     SELECT last_name INTO v_last_name
  6       FROM employees
  7      WHERE employee_id = v_emp_id;
  8     DBMS_OUTPUT.PUT_LINE('사원명 : ' || v_last_name);
  9  EXCEPTION
 10     WHEN NO_DATA_FOUND THEN
 11        DBMS_OUTPUT.PUT_LINE('사번 ' || v_emp_id
 12                             || ' 에 해당하는 사원이 없습니다.');
 13  END;
 14  /
사번 9999 에 해당하는 사원이 없습니다.

PL/SQL procedure successfully completed.

--    SELECT INTO는 변수 하나에 값 하나를 담으므로

-- ▶ 수정 1: 조건을 기본키로 바꾸어 단일 행을 보장한다

-- ▶ 수정 2: 집계 결과가 필요한 것이라면 집계 함수로 한 행을 만든다
SQL> DECLARE
  2     v_cnt        NUMBER;
  3     v_max_sal    employees.salary%TYPE;
  4     v_avg_sal    NUMBER;
  5  BEGIN
  6     SELECT COUNT(*), MAX(salary), ROUND(AVG(salary), 2)

 11     DBMS_OUTPUT.PUT_LINE('최고 : ' || v_max_sal);

-- ▶ 수정 3: 여러 행을 하나씩 처리해야 한다면 커서를 사용한다(2장에서 학습)

--------------------------------------------------------------------------------
[4] 집계 함수는 대상 행이 없어도 예외가 발생하지 않는다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_max   employees.salary%TYPE;
  3     v_cnt   NUMBER;
  4  BEGIN
  5     SELECT MAX(salary) INTO v_max
  6       FROM employees WHERE department_id = 9999;   -- 존재하지 않는 부서
  7     SELECT COUNT(*)    INTO v_cnt
  8       FROM employees WHERE department_id = 9999;
  9
 10     DBMS_OUTPUT.PUT_LINE('MAX   = ' || NVL(TO_CHAR(v_max), 'NULL'));
 11     DBMS_OUTPUT.PUT_LINE('COUNT = ' || v_cnt);
 12  END;
 13  /
MAX   = NULL
COUNT = 0

PL/SQL procedure successfully completed.

-- ▶ 집계 함수를 쓰는 SELECT는 조건에 맞는 행이 없어도 항상 한 행을 반환한다.
--    MAX는 NULL을, COUNT는 0을 돌려주므로 NO_DATA_FOUND가 발생하지 않는다.
--    이 때문에 '값이 없는 것'과 '행이 없는 것'을 구분하지 못하는 함정이 생긴다.

--------------------------------------------------------------------------------
[5] (오류 재현) 이름 모호성 — 변수 이름이 컬럼 이름과 같을 때
--------------------------------------------------------------------------------

SQL> DECLARE
  2     last_name    employees.last_name%TYPE := 'King';   -- 컬럼명과 동일!
  3     v_salary     employees.salary%TYPE;
  4  BEGIN
  5     SELECT salary INTO v_salary
  6       FROM employees
  7      WHERE last_name = last_name;          -- 항상 참인 조건이 된다
  8     DBMS_OUTPUT.PUT_LINE('급여: ' || v_salary);
  9  END;
 10  /
DECLARE
*
ERROR at line 1:
ORA-01422: exact fetch returns more than requested number of rows
ORA-06512: at line 5

-- ▶ 원인 파악
--    SQL 문장 안에서 PL/SQL 변수 이름과 테이블 컬럼 이름이 같으면
--    Oracle은 그 식별자를 '컬럼'으로 해석한다.
--    그 결과 WHERE 절은 last_name = last_name, 즉 '컬럼 = 컬럼'이 되어
--    NULL이 아닌 모든 행에서 참이 되고 107건이 반환된다.
--    문법 오류가 아니라 논리 오류이므로 컴파일 단계에서 잡히지 않는다.

-- ▶ 수정 1: v_ 접두어 명명 규칙을 지킨다 (가장 확실한 방법)
SQL> DECLARE
  2     v_last_name  employees.last_name%TYPE := 'King';
  3     v_salary     employees.salary%TYPE;
  4  BEGIN
  5     SELECT salary INTO v_salary
  6       FROM employees
  7      WHERE last_name = v_last_name;
  8     DBMS_OUTPUT.PUT_LINE('급여: ' || v_salary);
  9  EXCEPTION
 10     WHEN NO_DATA_FOUND THEN
 11        DBMS_OUTPUT.PUT_LINE('해당 사원이 없습니다.');
 12     WHEN TOO_MANY_ROWS THEN
 13        DBMS_OUTPUT.PUT_LINE('동명이인이 있습니다. 사번으로 조회하십시오.');
 14  END;
 15  /
급여: 24000

PL/SQL procedure successfully completed.

[6] 실무 주의 — UPDATE에서의 이름 모호성은 예외조차 발생하지 않는다
--------------------------------------------------------------------------------

-- 아래 블록은 절대로 운영 환경에서 실행하지 말 것 (실습용 확인만).
SQL> DECLARE
  2     last_name    employees.last_name%TYPE := 'King';
  3  BEGIN
  4     UPDATE employees
  5        SET salary = salary * 1.1
  6      WHERE last_name = last_name;          -- 전 사원이 대상이 된다!
  7     DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || '건이 수정되었습니다.');
  8     ROLLBACK;                              -- 실습이므로 즉시 되돌린다
  9     DBMS_OUTPUT.PUT_LINE('ROLLBACK 완료');
 10  END;
 11  /
107건이 수정되었습니다.
ROLLBACK 완료

PL/SQL procedure successfully completed.

-- ▶ 원인 파악
--    의도는 King 한 명(1건)이었지만 실제로는 107건 전부가 수정되었다.
--    UPDATE는 대상이 많다고 예외를 발생시키지 않으므로
--    오류 없이 조용히 잘못된 결과만 남는다. 가장 위험한 형태의 버그다.

-- ▶ 수정: 접두어 규칙을 적용하면 의도한 1건만 수정된다
SQL> DECLARE
  2     v_last_name  employees.last_name%TYPE := 'King';
  3  BEGIN
  4     UPDATE employees
  5        SET salary = salary * 1.1
  6      WHERE last_name = v_last_name;
  7     DBMS_OUTPUT.PUT_LINE(SQL%ROWCOUNT || '건이 수정되었습니다.');
  8     ROLLBACK;
  9  END;
 10  /
1건이 수정되었습니다.

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
[7] 정리
--------------------------------------------------------------------------------
--  · PL/SQL의 SELECT에는 반드시 INTO가 있어야 하고 정확히 한 행이어야 한다
--  · 0건 → NO_DATA_FOUND(ORA-01403), 2건 이상 → TOO_MANY_ROWS(ORA-01422)
--  · SELECT INTO를 쓰는 블록에는 두 예외 처리를 두는 것을 원칙으로 한다
--  · 집계 함수 SELECT는 대상이 없어도 항상 한 행(MAX=NULL, COUNT=0)을 반환한다
--  · 변수 이름은 절대 컬럼 이름과 같게 짓지 않는다 (v_ 접두어 규칙)
--  · 재현한 오류 : ORA-01403 → 예외 처리부 추가로 해결
--                  ORA-01422 → 기본키 조건/집계 함수/커서로 해결
--                  이름 모호성 → v_ 접두어 또는 블록 레이블 한정자로 해결
