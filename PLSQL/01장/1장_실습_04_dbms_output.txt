[1] (오류 재현) SET SERVEROUTPUT을 켜지 않으면 출력이 보이지 않는다
--------------------------------------------------------------------------------

SQL> SHOW SERVEROUTPUT
serveroutput OFF

SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('첫 번째 출력 테스트');
  3  END;
  4  /

PL/SQL procedure successfully completed.

-- ▶ 원인 파악
--    블록은 오류 없이 정상 종료됐는데 출력만 보이지 않는다.
--    DBMS_OUTPUT.PUT_LINE은 화면에 직접 쓰는 함수가 아니라
--    '서버 측 버퍼'에 문자열을 쌓아 두는 함수다.
--    실제 출력은 블록이 끝난 뒤 클라이언트가 그 버퍼를 읽어 표시하는데,
--    SET SERVEROUTPUT이 OFF이면 클라이언트가 버퍼를 읽지 않는다.
--    즉 '오류'가 아니라 '설정 누락'이다.

-- ▶ 수정: SET SERVEROUTPUT ON 실행 후 다시 시도한다
SQL> SET SERVEROUTPUT ON SIZE UNLIMITED
SQL> SHOW SERVEROUTPUT
serveroutput ON SIZE UNLIMITED FORMAT WORD_WRAPPED

SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('첫 번째 출력 테스트');
  3  END;
  4  /
첫 번째 출력 테스트

PL/SQL procedure successfully completed.

-- ▶ 주의: 이 설정은 세션 단위다. 재접속하거나 다른 창을 열면 다시 꺼진다.

--------------------------------------------------------------------------------
[2] PUT_LINE / PUT / NEW_LINE의 차이
--------------------------------------------------------------------------------

SQL> DECLARE
  2     rec_emp   employees%ROWTYPE;
  3     c_line    CONSTANT VARCHAR2(44) := RPAD('-', 44, '-');
  4  BEGIN
  5     SELECT * INTO rec_emp FROM employees WHERE employee_id = 103;
  6
  7     DBMS_OUTPUT.PUT_LINE(c_line);
  8
  9     -- PUT은 버퍼에 넣기만 하고 줄을 바꾸지 않는다
 10     DBMS_OUTPUT.PUT('사원명 : ');
 11     DBMS_OUTPUT.PUT(rec_emp.first_name || ' ' || rec_emp.last_name);
 12     DBMS_OUTPUT.NEW_LINE;                  -- 여기서 비로소 줄이 마감된다
 13
 14     DBMS_OUTPUT.PUT_LINE('직무   : ' || rec_emp.job_id);
 15     DBMS_OUTPUT.PUT_LINE('급여   : '
 16                          || TO_CHAR(rec_emp.salary, '999,999'));
 17     DBMS_OUTPUT.PUT_LINE('입사일 : '
 18                          || TO_CHAR(rec_emp.hire_date, 'YYYY-MM-DD'));
 19     DBMS_OUTPUT.PUT_LINE(c_line);
 20  END;
 21  /
--------------------------------------------

직무   : IT_PROG
급여   :    9,000
입사일 : 2006-01-03
--------------------------------------------

[3] (오류 재현) PUT만 사용하고 줄을 끝내지 않으면
--------------------------------------------------------------------------------

SQL> BEGIN
  2     DBMS_OUTPUT.PUT('이 줄은 화면에 나올까요?');
  3     -- NEW_LINE 도 PUT_LINE 도 호출하지 않았다
  4  END;
  5  /

PL/SQL procedure successfully completed.

-- ▶ 원인 파악
--    PUT은 줄을 마감하지 않으므로 해당 내용은 버퍼에 남아 있을 뿐
--    한 줄로 완성되지 않아 화면에 표시되지 않는다.
--    초급자가 '출력이 사라졌다'고 느끼는 두 번째로 흔한 원인이다.

-- ▶ 수정: 마지막에 NEW_LINE(또는 PUT_LINE)으로 줄을 마감한다
SQL> BEGIN
  2     DBMS_OUTPUT.PUT('이 줄은 화면에 나올까요?');
  3     DBMS_OUTPUT.NEW_LINE;
  4  END;
  5  /
이 줄은 화면에 나올까요?

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
[4] 출력은 '블록 종료 후' 한꺼번에 나타난다
--------------------------------------------------------------------------------

SQL> BEGIN
  2     DBMS_OUTPUT.PUT_LINE('[1] 작업 시작 : '
  3                          || TO_CHAR(SYSTIMESTAMP, 'HH24:MI:SS.FF3'));
  4     DBMS_LOCK.SLEEP(3);                    -- 3초 대기 (19c: DBMS_SESSION.SLEEP)
  5     DBMS_OUTPUT.PUT_LINE('[2] 작업 종료 : '
  6                          || TO_CHAR(SYSTIMESTAMP, 'HH24:MI:SS.FF3'));
  7  END;
  8  /
[1] 작업 시작 : 09:41:02.118
[2] 작업 종료 : 09:41:05.121

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    [1]이 먼저 찍히고 3초 뒤 [2]가 찍히는 것이 아니라,
--    3초를 모두 기다린 뒤 두 줄이 '동시에' 화면에 나타난다.
--    출력이 서버 버퍼에 쌓였다가 블록 종료 후 한꺼번에 전달되기 때문이다.
--    따라서 장시간 작업의 진행 상황을 실시간으로 보는 용도로는 쓸 수 없고,
--    그런 경우에는 로그 테이블 + 자율 트랜잭션(3장/7장)을 사용해야 한다.

  4  BEGIN

  7     FOR i IN 1..2 LOOP
  8        DBMS_OUTPUT.PUT_LINE('조각 ' || i || ' : '
  9                             || SUBSTR(v_long, (i-1)*c_chunk + 1, c_chunk));

--------------------------------------------------------------------------------
[7] 디버깅 도구로서의 DBMS_OUTPUT
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_a   NUMBER := 10;
  3     v_b   NUMBER := 0;
  4     v_c   NUMBER;
  5  BEGIN
  6     DBMS_OUTPUT.PUT_LINE('[진단 1] v_a=' || v_a || ', v_b=' || v_b);
  7     v_c := v_a / v_b;                      -- 0으로 나누기
  8     DBMS_OUTPUT.PUT_LINE('[진단 2] v_c=' || v_c);   -- 여기까지 오지 못한다
  9  EXCEPTION
 10     WHEN ZERO_DIVIDE THEN
 11        DBMS_OUTPUT.PUT_LINE('[예외] SQLCODE = ' || SQLCODE);
 12        DBMS_OUTPUT.PUT_LINE('[예외] SQLERRM = ' || SQLERRM);
 13  END;
 14  /
[진단 1] v_a=10, v_b=0
[예외] SQLCODE = -1476
[예외] SQLERRM = ORA-01476: divisor is equal to zero

PL/SQL procedure successfully completed.

-- ▶ 확인 사항
--    [진단 2]가 출력되지 않은 것으로 7행에서 실패했음을 알 수 있다.
--    예외로 블록이 중단되어도 그때까지 버퍼에 쌓인 출력은 표시되므로
--    '마지막으로 출력된 줄'이 오류 지점을 좁혀 주는 단서가 된다.
--    SQLCODE(오류 번호)와 SQLERRM(오류 메시지)은 3장에서 자세히 다룬다.

--------------------------------------------------------------------------------
[8] 정리
--------------------------------------------------------------------------------
--  · DBMS_OUTPUT은 서버 버퍼에 쌓고 블록 종료 후 클라이언트가 읽어 표시한다
--  · SET SERVEROUTPUT ON은 세션 단위 설정 — 재접속 시 다시 실행해야 한다
--  · PUT은 줄을 마감하지 않는다. NEW_LINE 또는 PUT_LINE으로 끝내야 표시된다
--  · 실시간 진행 상황 확인에는 부적합 → 로그 테이블 방식 사용
--  · 재현한 오류 : SERVEROUTPUT OFF(출력 없음) → SET SERVEROUTPUT ON
--                  PUT 미마감 → NEW_LINE 추가
--                  ORU-10027(버퍼 초과) → SIZE UNLIMITED
--                  ORU-10028(줄 길이 초과) → SUBSTR로 분할 출력
