-- ▶ 원인 파악
--    CONSTANT로 선언한 식별자는 대입의 대상(assignment target)이 될 수 없다.
--    주목할 점은 4행의 '세율(변경 전)' 출력조차 나오지 않았다는 사실이다.
--    PLS-00363은 컴파일 시점 오류이므로 블록이 아예 실행되지 않기 때문이다.
--    상수 위반은 실행 전에 걸러지므로 런타임 사고로 이어지지 않는다.

-- ▶ 수정 1: 값이 바뀌어야 한다면 상수가 아니라 변수로 선언한다

[3] (오류 재현) CONSTANT에 초기값을 주지 않으면 — PLS-00322
--------------------------------------------------------------------------------

SQL> DECLARE
  2     c_rate   CONSTANT NUMBER(4,3);         -- 초기값 없음
  3  BEGIN
  4     DBMS_OUTPUT.PUT_LINE(c_rate);
  5  END;
  6  /
   c_rate   CONSTANT NUMBER(4,3);
            *
ERROR at line 2:
ORA-06550: line 2, column 13:
PLS-00322: declaration of a constant 'C_RATE' must contain an initialization
   assignment

-- ▶ 원인 파악
--    상수는 값을 나중에 바꿀 수 없으므로 선언 시점에 반드시 값을 지정해야 한다.
--    초기값이 없으면 영원히 NULL인 상수가 되어 의미가 없기 때문에
--    Oracle은 이를 PLS-00322 컴파일 오류로 막는다.

-- ▶ 수정: 초기값을 지정한다
SQL> DECLARE
  2     c_rate   CONSTANT NUMBER(4,3) := 0.085;
  3  BEGIN
  4     DBMS_OUTPUT.PUT_LINE('c_rate = ' || c_rate);
  5  END;
  6  /

[5] 중첩 블록의 스코프와 가시성
--------------------------------------------------------------------------------

SQL> <<outer>>                                -- 바깥 블록 레이블
  2  DECLARE
  3     v_sal    NUMBER := 60000;              -- 바깥 v_sal
  4     v_name   VARCHAR2(20) := 'OUTER';
  5  BEGIN
  6     DBMS_OUTPUT.PUT_LINE('[0] 바깥 시작 v_sal = ' || v_sal);
  7
  8     <<inner_blk>>
  9     DECLARE
 10        v_sal   NUMBER := 50000;            -- 안쪽 v_sal (바깥을 가림)
 11        v_bonus NUMBER := 1000;             -- 안쪽에만 존재
 12     BEGIN
 13        DBMS_OUTPUT.PUT_LINE('[1] 안쪽 v_sal         = ' || v_sal);
 14        DBMS_OUTPUT.PUT_LINE('[2] 바깥 v_sal(한정자) = ' || outer.v_sal);
 15        DBMS_OUTPUT.PUT_LINE('[3] 바깥 v_name        = ' || v_name);
 16
 17        outer.v_sal := outer.v_sal + v_bonus;   -- 바깥 변수를 수정할 수 있다
 18        DBMS_OUTPUT.PUT_LINE('[4] 바깥 v_sal 수정 후 = ' || outer.v_sal);
 19     END inner_blk;
 20
 21     DBMS_OUTPUT.PUT_LINE('[5] 바깥 끝 v_sal      = ' || v_sal);
 22  END outer;
 23  /
[0] 바깥 시작 v_sal = 60000
[1] 안쪽 v_sal         = 50000
[2] 바깥 v_sal(한정자) = 60000
[3] 바깥 v_name        = OUTER
[4] 바깥 v_sal 수정 후 = 61000
[5] 바깥 끝 v_sal      = 61000

PL/SQL procedure successfully completed.

-- ▶ 확인 사항 (네 가지 규칙)
--    (1) 안쪽 블록은 바깥 변수를 자유롭게 참조하고 수정할 수 있다  → [3][4]
--    (2) 같은 이름을 안쪽에서 다시 선언하면 안쪽 선언이 우선한다    → [1]
--    (3) 가려진 바깥 변수는 블록 레이블 한정자로 접근한다           → [2]
--    (4) 안쪽에서 바깥 변수를 수정하면 블록을 벗어나도 유지된다     → [5]

--------------------------------------------------------------------------------
[6] (오류 재현) 바깥 블록에서 안쪽 블록의 변수를 참조하면 — PLS-00201
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_outer  NUMBER := 100;
  3  BEGIN
  4     DECLARE
  5        v_inner NUMBER := 200;              -- 안쪽 블록에만 존재
  6     BEGIN
  7        DBMS_OUTPUT.PUT_LINE('안쪽에서: ' || v_inner);
  8     END;
  9
 10     DBMS_OUTPUT.PUT_LINE('바깥에서: ' || v_inner);   -- 스코프를 벗어난 참조
 11  END;
 12  /
      DBMS_OUTPUT.PUT_LINE('바깥에서: ' || v_inner);
                                           *
ERROR at line 10:
ORA-06550: line 10, column 40:
PLS-00201: identifier 'V_INNER' must be declared
ORA-06550: line 10, column 4:
PL/SQL: Statement ignored

-- ▶ 원인 파악
--    v_inner의 스코프는 그것을 선언한 안쪽 블록과 그 하위 블록뿐이다.
--    안쪽 블록의 END를 지나는 순간 변수는 사라지므로
--    바깥 블록에서는 존재하지 않는 식별자가 되어 PLS-00201이 발생한다.
--    '안쪽 → 바깥'은 보이지만 '바깥 → 안쪽'은 보이지 않는다.

-- ▶ 수정: 바깥에서도 써야 하는 값은 바깥 블록에서 선언한다
SQL> DECLARE
  2     v_outer  NUMBER := 100;
  3     v_inner  NUMBER;                       -- 바깥에서 선언
  4  BEGIN
  5     BEGIN
  6        v_inner := 200;                     -- 안쪽에서 값만 채운다
  7        DBMS_OUTPUT.PUT_LINE('안쪽에서: ' || v_inner);
  8     END;
  9
 10     DBMS_OUTPUT.PUT_LINE('바깥에서: ' || v_inner);
 11     DBMS_OUTPUT.PUT_LINE('합계    : ' || (v_outer + v_inner));
 12  END;
 13  /

--------------------------------------------------------------------------------
[9] 정리
--------------------------------------------------------------------------------
--  · CONSTANT는 선언 시 초기값 필수 (없으면 PLS-00322)
--  · 상수 재할당은 컴파일 시점에 PLS-00363으로 차단된다 (실행 자체가 안 됨)
--  · 매직 넘버는 이름 있는 상수로 모아 수정 지점을 한 곳으로 만든다
--  · 스코프 = 변수가 존재하는 영역 / 가시성 = 이름만으로 참조 가능한 영역
--  · 안쪽 → 바깥은 참조·수정 가능, 바깥 → 안쪽은 PLS-00201
--  · 가려진 바깥 변수는 <<레이블>> 과 레이블.변수명 으로 접근한다
--  · 재현한 오류 : PLS-00363(상수 재할당) → 변수로 변경 또는 선언값 수정
--                  PLS-00322(초기값 없음) → 초기값 지정
--                  PLS-00201(스코프 밖 참조) → 바깥 블록에서 선언
