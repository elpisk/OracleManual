[2] 오류 → 원인 → 수정 : 초기화하지 않은 Nested Table (ORA-06531)
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_city IS TABLE OF VARCHAR2(30);
  3     v_city   t_city;                       -- 생성자를 호출하지 않았다
  4  BEGIN
  5     v_city(1) := 'Seoul';
  6     DBMS_OUTPUT.PUT_LINE(v_city(1));
  7  END;
  8  /
DECLARE
*
ERROR at line 1:
ORA-06531: Reference to uninitialized collection
ORA-06512: at line 5

-- [오류 발생]
--   ORA-06531. 컬렉션을 참조할 수 없다는 오류다.
-- [원인 파악]
--   v_city 는 '원소가 0개인 컬렉션' 이 아니라 '컬렉션 자체가 NULL' 인 상태다.
--   Nested Table 과 VARRAY 는 반드시 생성자로 초기화해야 한다.
--   (Associative Array 는 초기화가 필요 없다 — 실습 06 참고)
-- [수정 1단계]
--   생성자 t_city() 로 초기화한다.

SQL> DECLARE
  2     TYPE t_city IS TABLE OF VARCHAR2(30);
  3     v_city   t_city := t_city();           -- 생성자로 초기화(원소 0개)
  4  BEGIN
  5     DBMS_OUTPUT.PUT_LINE('COUNT = ' || v_city.COUNT);
  6     v_city(1) := 'Seoul';
  7  END;
  8  /
COUNT = 0
DECLARE
*
ERROR at line 1:
ORA-06533: Subscript beyond count
ORA-06512: at line 6

-- [오류 발생]
--   이번에는 ORA-06533 이다.
-- [원인 파악]
--   초기화는 되었지만 원소가 0개다. 1번 자리가 아직 존재하지 않는다.
--   Nested Table 은 첨자에 대입하기 전에 EXTEND 로 공간을 확보해야 한다.
-- [수정 2단계]

SQL> DECLARE
  2     TYPE t_city IS TABLE OF VARCHAR2(30);
  3     v_city   t_city := t_city();
  4  BEGIN
  5     v_city.EXTEND;                         -- 공간 1개 확보
  6     v_city(1) := 'Seoul';
  7     v_city.EXTEND(2);                      -- 공간 2개 추가 확보
  8     v_city(2) := 'Busan';
  9     v_city(3) := 'Daegu';
 10
 11     FOR i IN 1 .. v_city.COUNT LOOP
 12        DBMS_OUTPUT.PUT_LINE(i || ' : ' || v_city(i));
 13     END LOOP;
 14  END;
 15  /
1 : Seoul
2 : Busan
3 : Daegu

PL/SQL procedure successfully completed.

-- 생성자 인자로 초기 원소를 지정하면 EXTEND 없이 시작할 수도 있다.
SQL> DECLARE
  2     TYPE t_city IS TABLE OF VARCHAR2(30);
  3     v_city   t_city := t_city('Seoul', 'Busan', 'Daegu');
  4  BEGIN
  5     DBMS_OUTPUT.PUT_LINE('COUNT = ' || v_city.COUNT ||
  6                          ' / LIMIT = ' || NVL(TO_CHAR(v_city.LIMIT), 'NULL'));
  7  END;
  8  /
COUNT = 3 / LIMIT = NULL

PL/SQL procedure successfully completed.

-- Nested Table 은 크기 제한이 없으므로 LIMIT 이 NULL 이다.


--------------------------------------------------------------------------------
[3] 오류 → 원인 → 수정 : VARRAY 최대 크기 초과 (ORA-06532)
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_top3 IS VARRAY(3) OF VARCHAR2(30);
  3     v_top   t_top3 := t_top3('King', 'Kochhar', 'De Haan');
  4  BEGIN
  5     DBMS_OUTPUT.PUT_LINE('COUNT = ' || v_top.COUNT ||
  6                          ' / LIMIT = ' || v_top.LIMIT);
  7     v_top.EXTEND;                          -- 4번째 원소 추가 시도
  8     v_top(4) := 'Hunold';
  9  END;
 10  /
COUNT = 3 / LIMIT = 3
DECLARE
*
ERROR at line 1:
ORA-06532: Subscript outside of limit
ORA-06512: at line 7

-- [오류 발생]
--   ORA-06532. ORA-06533 과 헷갈리기 쉬우므로 구분해서 기억한다.
--     ORA-06533 : 원소 수(COUNT)를 넘어선 첨자 참조
--     ORA-06532 : 선언된 최대 크기(LIMIT)를 넘어선 첨자 참조
-- [원인 파악]
--   VARRAY(3) 은 선언 시점에 최대 크기가 3으로 고정된다.
--   EXTEND 로도 그 한계를 넘을 수 없다.
-- [수정]
--   (A) 상한이 업무 규칙이면 EXTEND 전에 COUNT < LIMIT 를 검사한다
--   (B) 상한이 유동적이면 Nested Table 로 바꾼다

SQL> DECLARE
  2     TYPE t_top3 IS VARRAY(3) OF VARCHAR2(30);
  3     v_top   t_top3 := t_top3('King', 'Kochhar', 'De Haan');
  4  BEGIN
  5     IF v_top.COUNT < v_top.LIMIT THEN
  6        v_top.EXTEND;
  7        v_top(v_top.LAST) := 'Hunold';
  8     ELSE
  9        DBMS_OUTPUT.PUT_LINE('최대 ' || v_top.LIMIT || '개까지만 담을 수 있다.');
 10     END IF;
 11  EXCEPTION
 12     WHEN SUBSCRIPT_OUTSIDE_LIMIT THEN
 13        DBMS_OUTPUT.PUT_LINE('크기 초과 : ' || SQLERRM);
 14  END;
 15  /
최대 3개까지만 담을 수 있다.

PL/SQL procedure successfully completed.

[4] Nested Table 의 DELETE — 희소 상태가 된다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_num IS TABLE OF NUMBER;
  3     v_n   t_num := t_num(10, 20, 30, 40, 50);
  4     v_idx PLS_INTEGER;
  5  BEGIN
  6     DBMS_OUTPUT.PUT_LINE('삭제 전 COUNT = ' || v_n.COUNT);
  7     v_n.DELETE(2);                         -- 가운데 원소 삭제
  8     v_n.DELETE(4);
  9     DBMS_OUTPUT.PUT_LINE('삭제 후 COUNT = ' || v_n.COUNT ||
 10                          ' / LAST = ' || v_n.LAST);
 11
 12     -- 1..COUNT 로 순회하면 실패한다
 13     BEGIN
 14        FOR i IN 1 .. v_n.COUNT LOOP
 15           DBMS_OUTPUT.PUT_LINE(i || ' : ' || v_n(i));
 16        END LOOP;
 17     EXCEPTION
 18        WHEN NO_DATA_FOUND THEN
 19           DBMS_OUTPUT.PUT_LINE('1..COUNT 순회 실패 : ' || SQLERRM);
 20     END;
 21
 22     -- FIRST/NEXT 로는 안전하다
 23     v_idx := v_n.FIRST;
 24     WHILE v_idx IS NOT NULL LOOP
 25        DBMS_OUTPUT.PUT_LINE('OK ' || v_idx || ' : ' || v_n(v_idx));
 26        v_idx := v_n.NEXT(v_idx);
 27     END LOOP;
 28  END;
 29  /
삭제 전 COUNT = 5
삭제 후 COUNT = 3 / LAST = 5
1 : 10
1..COUNT 순회 실패 : ORA-01403: no data found
OK 1 : 10
OK 3 : 30
OK 5 : 50

PL/SQL procedure successfully completed.

-- [관찰]
--   DELETE 후 COUNT 는 3이지만 LAST 는 여전히 5다. 첨자가 1,3,5 로 희소해졌다.
--   따라서 1..COUNT 순회는 두 번째 반복(첨자 2)에서 실패한다.
-- VARRAY 는 항상 조밀하므로 중간 원소를 DELETE 할 수 없다(전체 DELETE 만 가능).

SQL> DECLARE
  2     TYPE t_v IS VARRAY(5) OF NUMBER;
  3     v_v   t_v := t_v(10, 20, 30);
  4  BEGIN
  5     v_v.DELETE(2);
  6  END;
  7  /
   v_v.DELETE(2);
       *
ERROR at line 5:
ORA-06550: line 5, column 8:
PLS-00655: DELETE with parameters not allowed for VARRAY

-- VARRAY 는 중간 삭제가 불가능하다. TRIM 으로 끝에서만 제거할 수 있다.

SQL> DECLARE
  2     TYPE t_v IS VARRAY(5) OF NUMBER;
  3     v_v   t_v := t_v(10, 20, 30);
  4  BEGIN
  5     v_v.TRIM;                              -- 끝에서 1개 제거
  6     DBMS_OUTPUT.PUT_LINE('TRIM 후 COUNT = ' || v_v.COUNT ||
  7                          ' / LAST 값 = ' || v_v(v_v.LAST));
  8  END;
  9  /
TRIM 후 COUNT = 2 / LAST 값 = 20

PL/SQL procedure successfully completed.

[5] 스키마 레벨 타입으로 만들어 SQL 에서 사용하기
--------------------------------------------------------------------------------

  2  /

SQL> DECLARE
  2     v_ids   t_id_list := t_id_list(100, 101, 102);
  3     v_cnt   PLS_INTEGER;
  4  BEGIN
  5     SELECT COUNT(*) INTO v_cnt
  6       FROM employees
  7      WHERE employee_id IN (SELECT column_value FROM TABLE(v_ids));
  8     DBMS_OUTPUT.PUT_LINE('조회된 사원 수 : ' || v_cnt);
  9  END;
 10  /
조회된 사원 수 : 3

PL/SQL procedure successfully completed.

-- 스키마 레벨 Nested Table 타입은 TABLE() 연산자로 SQL 에서 사용할 수 있다.
-- 실습 06 에서 Associative Array 로 시도했을 때 PLS-00642 가 났던 것과 대비된다.

--------------------------------------------------------------------------------
[6] 세 유형 비교 정리
--------------------------------------------------------------------------------

-- +-------------------+------------+----------+-----------+-------------+
-- | 구분              | 초기화     | 크기제한 | 컬럼저장  | 중간 삭제   |
-- +-------------------+------------+----------+-----------+-------------+
-- | Associative Array | 불필요     | 없음     | 불가      | 가능(희소)  |
-- | Nested Table      | 필요       | 없음     | 가능      | 가능(희소)  |
-- | VARRAY            | 필요       | 고정     | 가능      | 불가(조밀)  |
-- +-------------------+------------+----------+-----------+-------------+

-- 주요 오류 번호
--   ORA-06531 : 초기화하지 않은 컬렉션 참조 (COLLECTION_IS_NULL)
--   ORA-06533 : 원소 수를 넘어선 첨자 참조 (SUBSCRIPT_BEYOND_COUNT)
--   ORA-06532 : 선언 최대 크기를 넘어선 첨자 참조 (SUBSCRIPT_OUTSIDE_LIMIT)
--   ORA-01403 : 삭제되었거나 존재하지 않는 첨자 읽기 (NO_DATA_FOUND)
--   PLS-00655 : VARRAY 에 파라미터 있는 DELETE 사용
--   PLS-00642 : 로컬 컬렉션 타입을 SQL 문장에서 사용
