 제3장 실습 05 — COMMIT / ROLLBACK / SAVEPOINT 를 이용한 트랜잭션 제어

 대상 스키마 : HR (EMPLOYEES / DEPARTMENTS / JOBS / JOB_HISTORY /
                   LOCATIONS / COUNTRIES / REGIONS)
 실행 환경   : Oracle Database 19c (19.3) / SQL*Plus
 학습 목표   : 1) 트랜잭션의 시작·종료 시점을 직접 확인한다
               2) SAVEPOINT 와 ROLLBACK TO 로 부분 롤백을 수행한다
               3) 존재하지 않는 세이브포인트 롤백(ORA-01086)을 재현·수정한다
               4) DDL 이 유발하는 암묵적 커밋을 확인한다
               5) 커밋하지 않고 세션을 비정상 종료하면 롤백되는 것을 확인한다

--------------------------------------------------------------------------------
[2] 트랜잭션의 시작과 ROLLBACK
--------------------------------------------------------------------------------

-- 첫 DML 이 실행되는 순간 트랜잭션이 암묵적으로 시작된다.
SQL> INSERT INTO departments(department_id, department_name)
  2  VALUES (900, '임시부서A');

1 row created.

SQL> SELECT department_id, department_name
  2    FROM departments WHERE department_id >= 900;

DEPARTMENT_ID DEPARTMENT_NAME
------------- ------------------------------
          900 임시부서A

1 row selected.

-- 자기 세션에서는 변경이 보이지만 다른 세션에는 아직 보이지 않는다.

SQL> ROLLBACK;

Rollback complete.

SQL> SELECT COUNT(*) FROM departments WHERE department_id >= 900;

  COUNT(*)
----------
         0

-- ROLLBACK 으로 트랜잭션 시작 이후의 모든 변경이 취소되었다.


--------------------------------------------------------------------------------
[3] SAVEPOINT 와 부분 롤백
--------------------------------------------------------------------------------

SQL> BEGIN
  2     INSERT INTO departments(department_id, department_name)
  3     VALUES (910, '1단계부서');
  4     SAVEPOINT sp_step1;
  5
  6     INSERT INTO departments(department_id, department_name)
  7     VALUES (920, '2단계부서');
  8     SAVEPOINT sp_step2;
  9
 10     INSERT INTO departments(department_id, department_name)
 11     VALUES (930, '3단계부서');
 12
 13     ROLLBACK TO sp_step2;          -- 930만 취소
 14     DBMS_OUTPUT.PUT_LINE('3단계만 취소했다. 트랜잭션은 아직 열려 있다.');
 15  END;
 16  /
3단계만 취소했다. 트랜잭션은 아직 열려 있다.

PL/SQL procedure successfully completed.

SQL> SELECT department_id, department_name
  2    FROM departments WHERE department_id >= 900 ORDER BY department_id;

DEPARTMENT_ID DEPARTMENT_NAME
------------- ------------------------------
          910 1단계부서
          920 2단계부서

2 rows selected.

-- ROLLBACK TO 는 트랜잭션을 끝내지 않으므로 아직 확정되지 않았다.

SQL> COMMIT;

Commit complete.

-- 이제 910, 920 두 건이 영구 확정되었다.


--------------------------------------------------------------------------------
[4] 오류 → 원인 → 수정 : 존재하지 않는 세이브포인트 (ORA-01086)
--------------------------------------------------------------------------------

SQL> BEGIN
  2     INSERT INTO departments(department_id, department_name)
  3     VALUES (940, '임시부서B');
  4     SAVEPOINT sp_dept;
  5
  6     INSERT INTO departments(department_id, department_name)
  7     VALUES (950, '임시부서C');
  8
  9     ROLLBACK TO sp_department;     -- 이름 오타
 10     COMMIT;
 11  END;
 12  /
BEGIN
*
ERROR at line 1:
ORA-01086: savepoint 'SP_DEPARTMENT' never established in this session or is
invalid
ORA-06512: at line 9

-- [오류 발생]
--   ORA-01086. 롤백 자체가 실패했다.
-- [원인 파악]
--   선언한 세이브포인트 이름은 sp_dept 인데 sp_department 로 롤백을 시도했다.
--   오타뿐 아니라 조건 분기로 SAVEPOINT 문이 실행되지 않은 경우에도 같은 오류가 난다.
-- [현재 상태 확인]

SQL> SELECT department_id FROM departments
  2   WHERE department_id >= 900 ORDER BY department_id;

DEPARTMENT_ID
-------------
          910
          920
          940
          950

4 rows selected.

-- 940, 950 은 아직 커밋되지 않은 상태로 남아 있다.

SQL> ROLLBACK;

Rollback complete.

-- [수정]
--   이름을 정확히 맞추고, 예외 처리로 롤백 실패에도 대비한다.

SQL> BEGIN
  2     INSERT INTO departments(department_id, department_name)
  3     VALUES (940, '임시부서B');
  4     SAVEPOINT sp_dept;
  5
  6     INSERT INTO departments(department_id, department_name)
  7     VALUES (950, '임시부서C');
  8
  9     ROLLBACK TO sp_dept;           -- 이름 일치
 10     COMMIT;
 11     DBMS_OUTPUT.PUT_LINE('940만 확정되었다.');
 12  EXCEPTION
 13     WHEN OTHERS THEN
 14        DBMS_OUTPUT.PUT_LINE('트랜잭션 오류 : ' || SQLERRM);
 15        ROLLBACK;
 16  END;
 17  /
940만 확정되었다.

PL/SQL procedure successfully completed.

SQL> SELECT department_id FROM departments
  2   WHERE department_id >= 900 ORDER BY department_id;

DEPARTMENT_ID
-------------
          910
          920
          940

3 rows selected.


--------------------------------------------------------------------------------
[5] 같은 이름의 SAVEPOINT 를 다시 선언하면 위치가 이동한다
--------------------------------------------------------------------------------

SQL> BEGIN
  2     INSERT INTO departments(department_id, department_name)
  3     VALUES (960, 'A');
  4     SAVEPOINT sp_x;               -- 첫 번째 선언
  5
  6     INSERT INTO departments(department_id, department_name)
  7     VALUES (970, 'B');
  8     SAVEPOINT sp_x;               -- 같은 이름으로 다시 선언 → 위치 이동
  9
 10     INSERT INTO departments(department_id, department_name)
 11     VALUES (980, 'C');
 12
 13     ROLLBACK TO sp_x;             -- 980만 취소된다 (960, 970 유지)
 14     COMMIT;
 15  END;
 16  /

PL/SQL procedure successfully completed.

SQL> SELECT department_id FROM departments
  2   WHERE department_id >= 950 ORDER BY department_id;

DEPARTMENT_ID
-------------
          960
          970

2 rows selected.

-- 같은 이름을 다시 선언하면 이전 표시는 사라지고 새 위치로 이동한다.
-- 루프 안에서 같은 이름을 반복 사용하면 항상 직전 반복 시작점으로만 되돌아간다.


--------------------------------------------------------------------------------
[6] DDL 이 유발하는 암묵적 커밋
--------------------------------------------------------------------------------

SQL> INSERT INTO departments(department_id, department_name)
  2  VALUES (990, '커밋안한부서');

1 row created.

-- 아직 COMMIT 하지 않았다. 여기서 DDL 을 실행해 본다.
SQL> CREATE TABLE tmp_ddl_test (c1 NUMBER);

Table created.

SQL> ROLLBACK;

Rollback complete.

SQL> SELECT department_id FROM departments WHERE department_id = 990;

DEPARTMENT_ID
-------------
          990

1 row selected.

-- [관찰]
--   ROLLBACK 했는데도 990 이 남아 있다.
-- [원인 파악]
--   CREATE TABLE 같은 DDL 은 실행 앞뒤로 암묵적 커밋을 유발한다.
--   따라서 미확정 상태였던 INSERT 까지 함께 확정되어 버렸다.
-- [교훈]
--   트랜잭션 도중에 DDL 을 실행하지 않는다. 테스트 중에도 특히 주의한다.

SQL> DROP TABLE tmp_ddl_test PURGE;

Table dropped.


--------------------------------------------------------------------------------
[7] 커밋하지 않고 세션을 비정상 종료하면 롤백된다
--------------------------------------------------------------------------------

SQL> INSERT INTO departments(department_id, department_name)
  2  VALUES (995, '세션종료테스트');

1 row created.

-- 여기서 SQL*Plus 창을 강제로 닫거나 세션을 KILL 한다.
-- (EXIT 로 정상 종료하면 커밋되므로 반드시 비정상 종료여야 한다)

$ sqlplus hr/hr@localhost:1521/orclpdb

Connected to:
Oracle Database 19c Enterprise Edition Release 19.0.0.0.0 - Production

SQL> SELECT COUNT(*) FROM departments WHERE department_id = 995;

  COUNT(*)
----------
         0

1 row selected.

-- 비정상 종료 시에는 미확정 트랜잭션이 롤백된다.
-- 반대로 SQL*Plus 를 EXIT 로 정상 종료하면 커밋된다는 점도 기억한다.


--------------------------------------------------------------------------------
[8] 실습 데이터 정리 및 정리
--------------------------------------------------------------------------------

SQL> DELETE FROM departments WHERE department_id >= 900;

6 rows deleted.

SQL> COMMIT;

Commit complete.

-- 1) 트랜잭션은 첫 DML 로 시작하고 COMMIT/ROLLBACK 으로 끝난다.
-- 2) SAVEPOINT + ROLLBACK TO 로 부분 롤백이 가능하며 트랜잭션은 계속 열려 있다.
-- 3) 이름이 틀리면 ORA-01086 이 발생하고 롤백 자체가 실패한다.
-- 4) 같은 이름을 다시 선언하면 세이브포인트 위치가 이동한다.
-- 5) DDL 은 암묵적 커밋을 유발하고, 비정상 세션 종료는 롤백을 유발한다.

SQL> EXIT
Disconnected from Oracle Database 19c Enterprise Edition Release 19.0.0.0.0
