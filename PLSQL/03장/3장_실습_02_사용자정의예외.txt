[2] 오류 → 원인 → 수정 : 선언하지 않은 예외를 RAISE (PLS-00201)
--------------------------------------------------------------------------------

SQL> DECLARE

-- [오류 발생]
--   실행 시 오류가 아니라 컴파일 오류(PLS-00201)다. 블록이 아예 실행되지 않았다.
-- [원인 파악]
--   사용자 정의 예외는 사전 정의 예외와 달리 반드시 선언부에서 선언해야 한다.
--   'e_invalid_salary EXCEPTION;' 선언이 빠져 있어 식별자를 찾을 수 없다.
-- [수정]
--   선언부에 EXCEPTION 선언을 추가한다.

SQL> DECLARE
  2     e_invalid_salary   EXCEPTION;            -- (1) 선언 추가
  3     v_salary   employees.salary%TYPE;
  4  BEGIN
  5     SELECT salary INTO v_salary FROM employees WHERE employee_id = 141;
  6
  7     IF v_salary < 5000 THEN
  8        RAISE e_invalid_salary;               -- (2) 명시적 발생
  9     END IF;
 10     DBMS_OUTPUT.PUT_LINE('검증 통과 : ' || v_salary);
 11  EXCEPTION
 12     WHEN e_invalid_salary THEN               -- (3) 이름으로 포착
 13        DBMS_OUTPUT.PUT_LINE('급여 규칙 위반 : ' || v_salary);
 14  END;

[3] 실제 업무 규칙 검증 — 직무 최소급여 미만 여부
--------------------------------------------------------------------------------

SQL> DECLARE
  2     e_invalid_salary   EXCEPTION;
  3     v_empno    employees.employee_id%TYPE := 141;
  4     v_salary   employees.salary%TYPE;
  5     v_min      jobs.min_salary%TYPE;
  6  BEGIN
  7     SELECT e.salary, j.min_salary
  8       INTO v_salary, v_min
  9       FROM employees e JOIN jobs j ON e.job_id = j.job_id
 10      WHERE e.employee_id = v_empno;
 11
 12     -- 실습을 위해 기준을 인위적으로 올려 규칙 위반을 만든다
 13     IF v_salary < v_min * 2 THEN
 14        RAISE e_invalid_salary;
 15     END IF;
 16     DBMS_OUTPUT.PUT_LINE('검증 통과');
 17  EXCEPTION
 18     WHEN e_invalid_salary THEN
 19        DBMS_OUTPUT.PUT_LINE('사번 ' || v_empno || ' 급여 ' || v_salary ||
 20                             ' 가 기준(' || v_min * 2 || ') 미만이다.');
 21  END;

[4] 오류 → 원인 → 수정 : 사용자 정의 예외의 SQLERRM 이 불친절하다
--------------------------------------------------------------------------------

-- 상위 프로그램이 WHEN OTHERS 로 잡는 상황을 흉내 내 본다.

SQL> DECLARE
  2     e_rule_violation   EXCEPTION;
  3  BEGIN
  4     RAISE e_rule_violation;
  5  EXCEPTION
  6     WHEN OTHERS THEN
  7        DBMS_OUTPUT.PUT_LINE('SQLCODE = ' || SQLCODE);
  8        DBMS_OUTPUT.PUT_LINE('SQLERRM = ' || SQLERRM);
  9  END;
 10  /
SQLCODE = 1
SQLERRM = User-Defined Exception

PL/SQL procedure successfully completed.

-- [문제 확인]
--   SQLCODE 는 일률적으로 1, SQLERRM 은 'User-Defined Exception' 뿐이다.
--   어떤 업무 규칙을 위반했는지 호출자는 전혀 알 수 없다.
-- [원인 파악]
--   사용자 정의 예외에는 오라클 오류 번호와 메시지가 연결되어 있지 않다.
-- [수정]
--   RAISE_APPLICATION_ERROR 로 -20000~-20999 범위의 고유 번호와
--   업무 메시지를 붙여 던진다.

SQL> DECLARE
  2     v_empno    employees.employee_id%TYPE := 141;
  3     v_salary   employees.salary%TYPE;
  4     v_min      jobs.min_salary%TYPE;
  5  BEGIN
  6     SELECT e.salary, j.min_salary
  7       INTO v_salary, v_min
  8       FROM employees e JOIN jobs j ON e.job_id = j.job_id
  9      WHERE e.employee_id = v_empno;
 10
 11     IF v_salary < v_min * 2 THEN
 12        RAISE_APPLICATION_ERROR(-20001,
 13           '사번 ' || v_empno || ' 의 급여 ' || v_salary ||
 14           ' 는 기준 ' || v_min * 2 || ' 미만이다.');
 15     END IF;
 16  END;
 17  /
DECLARE
*

-- 이제 호출자는 SQLCODE 로 -20001 을 얻고, SQLERRM 으로 업무 메시지를 얻는다.


--------------------------------------------------------------------------------
[5] 호출자 입장에서 -20001 을 포착해 본다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     e_salary_rule   EXCEPTION;
  3     PRAGMA EXCEPTION_INIT(e_salary_rule, -20001);
  4     v_salary   employees.salary%TYPE;
  5     v_min      jobs.min_salary%TYPE;
  6  BEGIN
  7     BEGIN
  8        SELECT e.salary, j.min_salary INTO v_salary, v_min
  9          FROM employees e JOIN jobs j ON e.job_id = j.job_id
 10         WHERE e.employee_id = 141;
 11        IF v_salary < v_min * 2 THEN
 12           RAISE_APPLICATION_ERROR(-20001, '급여 기준 미달 : ' || v_salary);
 13        END IF;
 14     EXCEPTION
 15        WHEN e_salary_rule THEN
 16           DBMS_OUTPUT.PUT_LINE('업무 오류 포착 : ' || SQLCODE);
 17           DBMS_OUTPUT.PUT_LINE('메시지        : ' || SQLERRM);
 18     END;
 19  END;
 20  /

[6] 오류 → 원인 → 수정 : 예외의 유효 범위(scope)
--------------------------------------------------------------------------------

SQL> BEGIN
  2     DECLARE
  3        e_inner   EXCEPTION;                 -- 안쪽 블록에서만 선언
  4     BEGIN
  5        RAISE e_inner;
  6     END;
  7  EXCEPTION
  8     WHEN e_inner THEN                       -- 바깥에서 이름 참조 시도
  9        DBMS_OUTPUT.PUT_LINE('바깥에서 포착');
 10  END;
 11  /
   WHEN e_inner THEN
        *
ERROR at line 8:
ORA-06550: line 8, column 9:
PLS-00201: identifier 'E_INNER' must be declared

-- [오류 발생]
--   PLS-00201. 안쪽에서 선언한 예외를 바깥 블록이 이름으로 참조할 수 없다.
-- [원인 파악]
--   예외의 유효 범위는 선언한 블록과 그 안쪽 중첩 블록으로 한정된다.
-- [수정]
--   여러 블록이 공유해야 하는 예외는 가장 바깥 블록에서 선언한다.
--   (여러 프로그램이 공유해야 한다면 패키지 스펙에 선언한다 — 5장)

SQL> DECLARE
  2     e_shared   EXCEPTION;                   -- 바깥 블록에서 선언
  3  BEGIN
  4     BEGIN
  5        RAISE e_shared;
  6     END;
  7  EXCEPTION
  8     WHEN e_shared THEN
  9        DBMS_OUTPUT.PUT_LINE('바깥에서 정상 포착');
 10  END;
 11  /
바깥에서 정상 포착

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[7] 정리
--------------------------------------------------------------------------------

-- 1) 사용자 정의 예외는 선언 → RAISE → WHEN 절 참조의 세 단계가 필요하다.
-- 2) 선언을 빠뜨리면 실행 오류가 아니라 컴파일 오류(PLS-00201)가 난다.
-- 3) RAISE 만 사용하면 SQLCODE=1, SQLERRM='User-Defined Exception' 뿐이다.
-- 4) RAISE_APPLICATION_ERROR(-20000~-20999) 로 의미 있는 오류를 전달한다.
-- 5) 예외의 유효 범위는 선언한 블록과 그 안쪽으로 한정된다.
