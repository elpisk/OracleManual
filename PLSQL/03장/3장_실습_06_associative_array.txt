[2] PLS_INTEGER 첨자 연관 배열 — 초기화 없이 바로 사용
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_emp_name IS TABLE OF employees.last_name%TYPE
  3          INDEX BY PLS_INTEGER;
  4     v_emp   t_emp_name;                    -- 생성자 호출이 필요 없다
  5  BEGIN
  6     FOR rec IN (SELECT employee_id, last_name
  7                   FROM employees WHERE department_id = 90) LOOP
  8        v_emp(rec.employee_id) := rec.last_name;
  9     END LOOP;
 10
 11     DBMS_OUTPUT.PUT_LINE('원소 수  : ' || v_emp.COUNT);
 12     DBMS_OUTPUT.PUT_LINE('첫 첨자  : ' || v_emp.FIRST);
 13     DBMS_OUTPUT.PUT_LINE('마지막   : ' || v_emp.LAST);
 14  END;
 15  /

첫 첨자  : 100

[3] 오류 → 원인 → 수정 : 1..COUNT 순회가 실패한다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_emp_name IS TABLE OF employees.last_name%TYPE
  3          INDEX BY PLS_INTEGER;
  4     v_emp   t_emp_name;
  5  BEGIN
  6     FOR rec IN (SELECT employee_id, last_name
  7                   FROM employees WHERE department_id = 90) LOOP
  8        v_emp(rec.employee_id) := rec.last_name;
  9     END LOOP;
 10
 11     FOR i IN 1 .. v_emp.COUNT LOOP        -- 1, 2, 3 을 읽으려 한다
 12        DBMS_OUTPUT.PUT_LINE(i || ' -> ' || v_emp(i));
 13     END LOOP;
 14  END;
 15  /
DECLARE
*
ERROR at line 1:
ORA-01403: no data found
ORA-06512: at line 12

-- [오류 발생]
--   ORA-01403 (NO_DATA_FOUND). 컬렉션인데 SELECT INTO 와 같은 예외가 났다.
-- [원인 파악]
--   COUNT 는 3이지만 실제 첨자는 100, 101, 102 다.
--   1..3 구간에는 원소가 하나도 없으므로 v_emp(1) 을 읽는 순간 실패한다.
--   Associative Array 는 첨자가 연속될 필요가 없는 희소(sparse) 배열이다.
-- [수정]
--   FIRST 로 시작해 NEXT 로 이동하는 WHILE 루프를 사용한다.

SQL> DECLARE
  2     TYPE t_emp_name IS TABLE OF employees.last_name%TYPE
  3          INDEX BY PLS_INTEGER;
  4     v_emp   t_emp_name;
  5     v_idx   PLS_INTEGER;
  6  BEGIN
  7     FOR rec IN (SELECT employee_id, last_name
  8                   FROM employees WHERE department_id = 90) LOOP
  9        v_emp(rec.employee_id) := rec.last_name;
 10     END LOOP;
 11
 12     v_idx := v_emp.FIRST;
 13     WHILE v_idx IS NOT NULL LOOP
 14        DBMS_OUTPUT.PUT_LINE(v_idx || ' -> ' || v_emp(v_idx));
 15        v_idx := v_emp.NEXT(v_idx);
 16     END LOOP;
 17  END;
 18  /
100 -> King
101 -> Kochhar
102 -> De Haan

PL/SQL procedure successfully completed.

-- 역순으로 순회하려면 LAST 와 PRIOR 를 사용한다.

[4] VARCHAR2 첨자 — 조견표(lookup table) 만들기
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_dept IS TABLE OF departments.department_id%TYPE
  3          INDEX BY VARCHAR2(30);
  4     v_dept   t_dept;
  5  BEGIN
  6     FOR rec IN (SELECT department_id, department_name FROM departments) LOOP
  7        v_dept(UPPER(rec.department_name)) := rec.department_id;   -- 키 정규화
  8     END LOOP;
  9
 10     DBMS_OUTPUT.PUT_LINE('등록된 부서 수 : ' || v_dept.COUNT);
 11     DBMS_OUTPUT.PUT_LINE('IT        : ' || v_dept('IT'));
 12     DBMS_OUTPUT.PUT_LINE('MARKETING : ' || v_dept('MARKETING'));
 13     DBMS_OUTPUT.PUT_LINE('첫 키     : ' || v_dept.FIRST);
 14  END;
 15  /
등록된 부서 수 : 27
IT        : 60
MARKETING : 20
첫 키     : ACCOUNTING

PL/SQL procedure successfully completed.

-- 문자열 첨자는 사전 순으로 정렬되어 FIRST 가 ACCOUNTING 이 된다.


--------------------------------------------------------------------------------
[5] 오류 → 원인 → 수정 : 존재하지 않는 키 조회 (ORA-01403)
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_dept IS TABLE OF departments.department_id%TYPE
  3          INDEX BY VARCHAR2(30);
  4     v_dept   t_dept;
  5  BEGIN
  6     FOR rec IN (SELECT department_id, department_name FROM departments) LOOP
  7        v_dept(rec.department_name) := rec.department_id;   -- 정규화하지 않았다
  8     END LOOP;
  9
 10     DBMS_OUTPUT.PUT_LINE('it : ' || v_dept('it'));         -- 소문자로 조회
 11  END;
 12  /
DECLARE
*
ERROR at line 1:
ORA-01403: no data found
ORA-06512: at line 10

-- [오류 발생]
--   ORA-01403. 분명히 IT 부서가 존재하는데도 조회에 실패했다.
-- [원인 파악]
--   문자열 첨자는 대소문자와 공백까지 정확히 일치해야 한다.
--   저장된 키는 'IT' 이고 조회 키는 'it' 이므로 서로 다른 키다.
--   존재하지 않는 첨자를 '읽으면' NO_DATA_FOUND 가 발생한다.
--   (반면 '쓰기' 는 예외가 나지 않고 새 원소가 만들어진다)
-- [수정]
--   저장과 조회 모두 키를 정규화하고, 읽기 전에 EXISTS 로 확인한다.

SQL> DECLARE
  2     TYPE t_dept IS TABLE OF departments.department_id%TYPE
  3          INDEX BY VARCHAR2(30);
  4     v_dept   t_dept;
  5     v_key    VARCHAR2(30) := UPPER('it');
  6  BEGIN
  7     FOR rec IN (SELECT department_id, department_name FROM departments) LOOP
  8        v_dept(UPPER(rec.department_name)) := rec.department_id;
  9     END LOOP;
 10
 11     IF v_dept.EXISTS(v_key) THEN
 12        DBMS_OUTPUT.PUT_LINE(v_key || ' : ' || v_dept(v_key));
 13     ELSE
 14        DBMS_OUTPUT.PUT_LINE(v_key || ' : 해당 부서가 없다.');
 15     END IF;
 16
 17     -- 정말 없는 키로도 확인해 본다
 18     IF v_dept.EXISTS('NO_SUCH_DEPT') THEN
 19        DBMS_OUTPUT.PUT_LINE('있음');
 20     ELSE
 21        DBMS_OUTPUT.PUT_LINE('NO_SUCH_DEPT : 존재하지 않는 키');
 22     END IF;
 23  EXCEPTION
 24     WHEN NO_DATA_FOUND THEN
 25        DBMS_OUTPUT.PUT_LINE('존재하지 않는 키이다.');
 26  END;
 27  /
IT : 60
NO_SUCH_DEPT : 존재하지 않는 키

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[6] 쓰기는 예외가 나지 않는다 — DELETE 후의 동작 확인
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_num IS TABLE OF NUMBER INDEX BY PLS_INTEGER;
  3     v_n   t_num;
  4  BEGIN
  5     v_n(5)  := 50;                        -- 없는 첨자에 '쓰기' → 새 원소 생성
  6     v_n(10) := 100;
  7     v_n(15) := 150;
  8     DBMS_OUTPUT.PUT_LINE('COUNT=' || v_n.COUNT ||
  9                          ' FIRST=' || v_n.FIRST || ' LAST=' || v_n.LAST);
 10
 11     v_n.DELETE(10);                       -- 가운데 원소 삭제
 12     DBMS_OUTPUT.PUT_LINE('삭제 후 COUNT=' || v_n.COUNT);
 13     DBMS_OUTPUT.PUT_LINE('5의 NEXT=' || v_n.NEXT(5));     -- 15로 건너뛴다
 14     DBMS_OUTPUT.PUT_LINE('EXISTS(10)=' ||
 15        CASE WHEN v_n.EXISTS(10) THEN 'TRUE' ELSE 'FALSE' END);
 16  END;
 17  /
COUNT=3 FIRST=5 LAST=15
삭제 후 COUNT=2
5의 NEXT=15
EXISTS(10)=FALSE

PL/SQL procedure successfully completed.

-- DELETE 후에도 NEXT 는 다음 유효 첨자로 정확히 건너뛴다.
-- 이것이 희소 컬렉션 순회에 FIRST/NEXT 를 써야 하는 이유다.


--------------------------------------------------------------------------------
[7] 오류 → 원인 → 수정 : SQL 문장에서 사용할 수 없다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     TYPE t_id IS TABLE OF NUMBER INDEX BY PLS_INTEGER;
  3     v_ids   t_id;
  4     v_cnt   PLS_INTEGER;
  5  BEGIN
  6     v_ids(1) := 100;
  7     v_ids(2) := 101;
  8
  9     SELECT COUNT(*) INTO v_cnt
 10       FROM employees
 11      WHERE employee_id IN (SELECT * FROM TABLE(v_ids));   -- SQL 에서 사용 시도
 12     DBMS_OUTPUT.PUT_LINE(v_cnt);
 13  END;
 14  /
    WHERE employee_id IN (SELECT * FROM TABLE(v_ids));
                                             *
ERROR at line 11:
ORA-06550: line 11, column 46:
PLS-00642: local collection types not allowed in SQL statements

-- [오류 발생]
--   PLS-00642. 연관 배열을 SQL 문장에서 직접 사용할 수 없다.
-- [원인 파악]
--   Associative Array 는 PL/SQL 전용 타입이며 SQL 엔진이 알지 못한다.
--   TABLE() 연산자를 쓰려면 CREATE TYPE 으로 만든 스키마 레벨 Nested Table 이어야 한다.
-- [수정]
--   (A) 스키마 레벨 타입을 만들어 Nested Table 을 사용하거나
--   (B) 아래처럼 컬렉션을 PL/SQL 안에서 순회하며 처리한다.

SQL> DECLARE
  2     TYPE t_id IS TABLE OF NUMBER INDEX BY PLS_INTEGER;
  3     v_ids   t_id;
  4     v_cnt   PLS_INTEGER;
  5     v_tot   PLS_INTEGER := 0;
  6     v_idx   PLS_INTEGER;
  7  BEGIN
  8     v_ids(1) := 100;
  9     v_ids(2) := 101;
 10
 11     v_idx := v_ids.FIRST;
 12     WHILE v_idx IS NOT NULL LOOP
 13        SELECT COUNT(*) INTO v_cnt
 14          FROM employees WHERE employee_id = v_ids(v_idx);
 15        v_tot := v_tot + v_cnt;
 16        v_idx := v_ids.NEXT(v_idx);
 17     END LOOP;
 18     DBMS_OUTPUT.PUT_LINE('확인된 사원 수 : ' || v_tot);
 19  END;
 20  /
확인된 사원 수 : 2

PL/SQL procedure successfully completed.


--------------------------------------------------------------------------------
[8] 정리
--------------------------------------------------------------------------------

-- 1) Associative Array 는 초기화·EXTEND 없이 바로 사용할 수 있다.
-- 2) 첨자는 PLS_INTEGER 또는 VARCHAR2 이며 연속될 필요가 없다(희소 배열).
-- 3) 존재하지 않는 첨자를 읽으면 ORA-01403, 쓰기는 새 원소를 만든다.
-- 4) 순회는 FIRST/NEXT(역순은 LAST/PRIOR)로 하고 1..COUNT 는 쓰지 않는다.
-- 5) SQL 문장에서는 사용할 수 없다(PLS-00642).
