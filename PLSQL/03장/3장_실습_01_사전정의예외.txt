--   문법에는 아무 문제가 없다. 컴파일은 통과했고 실행 중에 실패했다.

[3] 두 방향의 실패를 각각 다른 핸들러로 처리한다

SQL> DECLARE
  2     v_dept_id   employees.department_id%TYPE := 999;   -- 존재하지 않는 부서
  3     v_name      employees.last_name%TYPE;
  4  BEGIN

  6       FROM employees
  7      WHERE department_id = v_dept_id;

 13        DBMS_OUTPUT.PUT_LINE('부서 ' || v_dept_id || ' : 사원이 여러 명이다.');
 14  END;

--------------------------------------------------------------------------------
[4] 함정 확인 — 집계 함수는 대상이 없어도 NO_DATA_FOUND가 아니다
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_cnt   PLS_INTEGER;
  3     v_sum   NUMBER;
  4  BEGIN
  5     SELECT COUNT(*), SUM(salary) INTO v_cnt, v_sum
  6       FROM employees
  7      WHERE department_id = 999;
  8     DBMS_OUTPUT.PUT_LINE('건수 = ' || v_cnt);
  9     DBMS_OUTPUT.PUT_LINE('합계 = ' || NVL(TO_CHAR(v_sum), 'NULL'));
 10  EXCEPTION
 11     WHEN NO_DATA_FOUND THEN
 12        DBMS_OUTPUT.PUT_LINE('이 핸들러는 실행되지 않는다');
 13  END;
 14  /
건수 = 0
합계 = NULL

PL/SQL procedure successfully completed.

-- 집계 함수만으로 구성된 SELECT INTO 는 대상 행이 없어도 한 행을 반환한다.
-- COUNT(*) 는 0, SUM 은 NULL 이므로 예외가 아니라 "조용한 NULL"이 문제가 된다.
-- 이후 계산에서 v_sum 을 그대로 쓰면 결과가 전부 NULL 이 되어 버린다.


--------------------------------------------------------------------------------
[5] ZERO_DIVIDE(ORA-01476) 재현 — NULL과 0의 차이
--------------------------------------------------------------------------------

SQL> SELECT employee_id, salary, commission_pct
  2    FROM employees
  3   WHERE employee_id IN (145, 178, 200)
  4   ORDER BY employee_id;

EMPLOYEE_ID     SALARY COMMISSION_PCT
----------- ---------- --------------
        145      14000             .4
        178       7000             .15
        200       4400

3 rows selected.

-- 사번 200 은 commission_pct 가 NULL 이다.

SQL> DECLARE
  2     v_result   NUMBER;
  3  BEGIN
  4     -- (1) NULL 로 나누기 : 예외가 아니라 NULL 이 된다
  5     SELECT salary / commission_pct INTO v_result
  6       FROM employees WHERE employee_id = 200;
  7     DBMS_OUTPUT.PUT_LINE('NULL 나누기 결과 : ' || NVL(TO_CHAR(v_result), 'NULL'));
  8
  9     -- (2) 0 으로 나누기 : ZERO_DIVIDE 발생
 10     v_result := 1000 / 0;
 11     DBMS_OUTPUT.PUT_LINE('이 줄은 실행되지 않는다');
 12  EXCEPTION
 13     WHEN ZERO_DIVIDE THEN
 14        DBMS_OUTPUT.PUT_LINE('ZERO_DIVIDE : ' || SQLERRM);
 15  END;
 16  /
NULL 나누기 결과 : NULL
ZERO_DIVIDE : ORA-01476: divisor is equal to zero

PL/SQL procedure successfully completed.

-- [정리]
--   NULL 로 나누면 결과만 NULL 이 될 뿐 예외는 발생하지 않는다.
--   0 으로 나눌 때만 ZERO_DIVIDE(ORA-01476) 가 발생한다.
--   즉 "예외가 나지 않았다"는 사실이 "계산이 옳다"는 뜻은 아니다.


--------------------------------------------------------------------------------
[6] VALUE_ERROR(ORA-06502) 두 가지 형태
--------------------------------------------------------------------------------

SQL> DECLARE
  2     v_result   NUMBER;
  3     v_short    VARCHAR2(5);
  4  BEGIN
  5     BEGIN
  6        v_result := TO_NUMBER('A100');           -- 변환 실패
  7     EXCEPTION
  8        WHEN VALUE_ERROR THEN
  9           DBMS_OUTPUT.PUT_LINE('(1) 변환 : ' || SQLERRM);
 10     END;
 11
 12     BEGIN
 13        v_short := 'ABCDEFGHIJ';                 -- 크기 초과
 14     EXCEPTION
 15        WHEN VALUE_ERROR THEN
 16           DBMS_OUTPUT.PUT_LINE('(2) 크기 : ' || SQLERRM);
 17     END;
 18  END;
 19  /
(1) 변환 : ORA-06502: PL/SQL: numeric or value error: character to number
conversion error
(2) 크기 : ORA-06502: PL/SQL: numeric or value error: character string buffer
too small

PL/SQL procedure successfully completed.

-- 같은 VALUE_ERROR 라도 SQLERRM 뒤쪽 문구가 다르다.
-- 로그에는 SQLERRM 전체를 남겨야 원인을 좁힐 수 있다.


--------------------------------------------------------------------------------
[7] 오류 → 원인 → 수정 : 선언부에서 발생한 예외는 그 블록이 잡지 못한다
--------------------------------------------------------------------------------

-- 앞의 (2) 를 선언부로 옮겨 본다.

SQL> DECLARE
  2     v_short   VARCHAR2(5) := 'ABCDEFGHIJ';      -- 선언과 동시에 크기 초과
  3  BEGIN
  4     DBMS_OUTPUT.PUT_LINE('시작');
  5  EXCEPTION
  6     WHEN VALUE_ERROR THEN
  7        DBMS_OUTPUT.PUT_LINE('여기서 잡히지 않는다 : ' || SQLERRM);
  8  END;
  9  /
DECLARE
*
ERROR at line 1:
ORA-06502: PL/SQL: numeric or value error: character string buffer too small

-- [오류 발생]
--   VALUE_ERROR 핸들러를 분명히 작성했는데도 오류가 그대로 밖으로 나왔다.
-- [원인 파악]
--   예외가 발생한 지점이 선언부(DECLARE)다.
--   선언부에서 발생한 예외는 해당 블록의 EXCEPTION 절이 처리하지 못하고
--   곧바로 바깥 블록 또는 호출 환경으로 전파된다.
-- [수정]
--   선언은 안전한 기본값으로 하고, 값 대입은 실행부에서 수행한다.

SQL> DECLARE
  2     v_short   VARCHAR2(5);                      -- 초기값 없이 선언
  3     v_input   VARCHAR2(30) := 'ABCDEFGHIJ';
  4  BEGIN
  5     v_short := v_input;                         -- 실행부에서 대입
  6     DBMS_OUTPUT.PUT_LINE('대입 성공 : ' || v_short);
  7  EXCEPTION
  8     WHEN VALUE_ERROR THEN
  9        DBMS_OUTPUT.PUT_LINE('크기 초과를 정상적으로 포착 : ' || SQLERRM);
 10  END;
 11  /
크기 초과를 정상적으로 포착 : ORA-06502: PL/SQL: numeric or value error:
character string buffer too small

PL/SQL procedure successfully completed.

-- 같은 오류지만 이번에는 블록이 스스로 처리하고 정상 종료했다.


--------------------------------------------------------------------------------
[8] 정리
--------------------------------------------------------------------------------

-- 1) SELECT INTO 는 0건이면 NO_DATA_FOUND, 2건 이상이면 TOO_MANY_ROWS.
-- 2) 집계 함수만 있는 SELECT INTO 는 대상이 없어도 예외가 발생하지 않는다.
-- 3) 0 으로 나누면 ZERO_DIVIDE, NULL 로 나누면 결과만 NULL.
-- 4) VALUE_ERROR 는 변환 실패와 크기 초과를 모두 포함한다.
-- 5) 선언부의 예외는 그 블록이 잡지 못하므로 대입은 실행부에서 한다.
