[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 첨부파일 보관 테이블(file_store)에 OS 파일을 적재한다.
--   ① 디렉터리 객체와 BFILENAME 으로 서버의 파일을 가리킨다
--   ② LOADBLOBFROMFILE / LOADCLOBFROMFILE 로 DB 안(BLOB/CLOB)으로 옮긴다
--   ③ CONVERTTOBLOB / CONVERTTOCLOB 로 두 타입 사이를 변환한다
--   ④ 파일·디렉터리가 없을 때의 오류를 만나고 선검사로 막는다
-- 작업 유형 : 데이터 적재 (장애 복구 아님)

-- ────────────────────────────────────────────────
-- 4-1. 바이너리 파일 → BLOB
-- ────────────────────────────────────────────────
LOBLAB@orcl> DECLARE
  2    v_b BLOB; v_f BFILE := BFILENAME('LOB_DIR','photo.bin');
  3    d_off INTEGER := 1; s_off INTEGER := 1;
  4  BEGIN
  5    INSERT INTO file_store (id, fname, mime, content)
  6      VALUES (1, 'photo.bin', 'application/octet-stream', EMPTY_BLOB()) RETURNING content INTO v_b;
  7    DBMS_LOB.FILEOPEN(v_f, DBMS_LOB.FILE_READONLY);
  8    DBMS_LOB.LOADBLOBFROMFILE(v_b, v_f, DBMS_LOB.LOBMAXSIZE, d_off, s_off);
  9    DBMS_LOB.FILECLOSE(v_f);
 10    DBMS_OUTPUT.PUT_LINE('BLOB 길이=' || DBMS_LOB.GETLENGTH(v_b) || '  다음 목적지 오프셋=' || d_off
 11      || '  다음 원본 오프셋=' || s_off);
 12    COMMIT;
 13  END;
 14  /
BLOB 길이=12288  다음 목적지 오프셋=12289  다음 원본 오프셋=12289

PL/SQL procedure successfully completed.

-- 읽는 법
--   · BFILENAME('LOB_DIR','photo.bin') 은 디렉터리 객체 이름을 대문자로 받는다.
--     'lob_dir' 처럼 소문자로 주면 ORA-22285 가 난다(따옴표 안은 그대로 쓰인다).
--   · LOBMAXSIZE 는 "파일 끝까지" 라는 뜻이다.
--   · d_off / s_off 는 IN OUT 이다. 호출 뒤에는 '다음에 쓸 위치' 가 들어 있다.
--     12288 바이트를 넣었으니 둘 다 12289 다. 이어붙이기(여러 파일을 한 LOB 으로)에 쓴다.

-- ────────────────────────────────────────────────
-- 4-2. 텍스트 파일 → CLOB (문자집합 변환이 일어난다)
-- ────────────────────────────────────────────────
LOBLAB@orcl> DECLARE
  2    v_c CLOB; v_f BFILE := BFILENAME('LOB_DIR','manual.txt');
  3    d_off INTEGER := 1; s_off INTEGER := 1; lang INTEGER := 0; warn INTEGER;
  4  BEGIN
  5    INSERT INTO file_store (id, fname, mime, text_content)
  6      VALUES (2, 'manual.txt', 'text/plain', EMPTY_CLOB()) RETURNING text_content INTO v_c;
  7    DBMS_LOB.FILEOPEN(v_f, DBMS_LOB.FILE_READONLY);
  8    DBMS_LOB.LOADCLOBFROMFILE(v_c, v_f, DBMS_LOB.LOBMAXSIZE, d_off, s_off,
  9                              NLS_CHARSET_ID('AL32UTF8'), lang, warn);
 10    DBMS_LOB.FILECLOSE(v_f);
 11    DBMS_OUTPUT.PUT_LINE('CLOB 문자수=' || DBMS_LOB.GETLENGTH(v_c) || '  읽은 바이트=' || (s_off-1)
 12      || '  warning=' || warn);
 13    COMMIT;
 14  END;
 15  /
CLOB 문자수=5154  읽은 바이트=7566  warning=0

PL/SQL procedure successfully completed.

-- 읽는 법 — 이 한 줄이 실습 03의 핵심이다
--   · 읽은 바이트 7566 = 파일 크기. 저장된 문자 5154 = wc -m 값.
--     즉 s_off 는 '원본 파일의 바이트 오프셋', GETLENGTH 는 '저장된 문자 수' 다.
--     적재 검증을 파일 크기와 GETLENGTH 로 비교하면 한글이 있는 파일에서 어긋난다.
--   · 여섯 번째 인자는 원본 파일의 문자집합 ID 다. NLS_CHARSET_ID('AL32UTF8') 로 준다.
--     파일이 EUC-KR 이면 NLS_CHARSET_ID('KO16MSWIN949') 를 줘야 한다.
--     틀린 값을 주면 오류 없이 글자가 깨진다(가장 찾기 어려운 사고다).
--   · warn = 0 이면 변환 중 대체 문자가 없었다는 뜻이다.
--     0 이 아니면(예: DBMS_LOB.WARN_INCONVERTIBLE_CHAR = 1) 변환 불가 문자가 있었다는 신호다.

-- ────────────────────────────────────────────────

-- ────────────────────────────────────────────────
-- 4-4. CLOB → BLOB → CLOB 왕복 변환
-- ────────────────────────────────────────────────
LOBLAB@orcl> DECLARE
  2    v_c CLOB; v_b BLOB; d_off INTEGER := 1; s_off INTEGER := 1; lang INTEGER := 0; warn INTEGER;
  3  BEGIN
  4    SELECT text_content INTO v_c FROM file_store WHERE id=2;
  5    INSERT INTO file_store (id, fname, mime, content)
  6      VALUES (3, 'manual.blob', 'application/octet-stream', EMPTY_BLOB()) RETURNING content INTO v_b;
  7    DBMS_LOB.CONVERTTOBLOB(v_b, v_c, DBMS_LOB.LOBMAXSIZE, d_off, s_off, NLS_CHARSET_ID('AL32UTF8'), lang, warn);
  8    DBMS_OUTPUT.PUT_LINE('CONVERTTOBLOB 결과 바이트=' || DBMS_LOB.GETLENGTH(v_b) || '  warning=' || warn);
  9    COMMIT;
 10  END;

--   ORA-22288 : 디렉터리 객체는 있고 '파일' 이 없다. 뒤에 OS 오류 문구가 붙는다.

LOBLAB@orcl> DECLARE
  2    v_f BFILE := BFILENAME('LOB_DIR','nofile.bin');
  3  BEGIN
  4    IF DBMS_LOB.FILEEXISTS(v_f) = 1 THEN
  5      DBMS_LOB.FILEOPEN(v_f, DBMS_LOB.FILE_READONLY);
  6      DBMS_LOB.FILECLOSE(v_f);
  7    ELSE
  8      DBMS_OUTPUT.PUT_LINE('파일이 없다. 적재를 건너뛴다.');

-- 컬럼별 LOB 세그먼트 크기

LOBLAB@orcl> COL column_name FOR a14
LOBLAB@orcl> SELECT l.column_name, s.bytes/1024 AS lob_kb
  2  FROM   user_lobs l JOIN user_segments s ON s.segment_name = l.segment_name
  3  WHERE  l.table_name='FILE_STORE' ORDER BY 1;

COLUMN_NAME        LOB_KB
-------------- ----------
CONTENT              3264
TEXT_CONTENT         1216

-- CONTENT 3264KB : 12KB + 7.4KB + 2MB 를 담았다. 블록·CHUNK 단위로 올림되어 실제보다 크다.
-- TEXT_CONTENT 1216KB : 5154 문자 × 2 바이트 × 2건 ≈ 20KB 인데 1216KB 다.
--   → LOB 세그먼트는 CHUNK(8KB) 단위로 공간을 잡고, 최소 익스텐트가 있다.
--     작은 LOB 을 많이 넣을 때 "데이터보다 세그먼트가 훨씬 크다" 는 현상의 원인이다.
--     작은 값은 인라인(실습 01)으로 두는 것이 공간 면에서 유리하다.

[8] 실무 포인트
--------------------------------------------------------------------------------
-- 1) 파일 적재 검증은 "바이트 = 바이트" 로 한다.
--    BLOB : 파일 크기 = GETLENGTH (일치해야 정상)
--    CLOB : 파일 크기 = s_off - 1 (읽은 바이트), GETLENGTH 는 문자 수 (다른 값이 정상)
--
-- 2) LOADCLOBFROMFILE 의 문자집합 ID 를 반드시 원본 파일에 맞춘다.
--    틀리면 오류가 없고 글자만 깨진다. 파일 인코딩을 먼저 확인한다(file 명령).
--    warn 출력 인자를 확인하는 코드를 넣는다.
--
-- 3) 디렉터리 객체는 서버 경로다. 클라이언트에서 올린 파일은 먼저 서버로 옮겨야 한다.
--    권한은 최소로 준다. /home/oracle 전체나 ORACLE_HOME 을 가리키는 디렉터리 객체에
--    READ·WRITE 를 주면 DB 사용자가 서버 파일을 읽고 쓸 수 있는 통로가 된다.
--
-- 4) 적재 전 FILEEXISTS 선검사를 습관화한다. 배치가 ORA-22288 로 중단되는 것보다
--    "건너뛰고 로그를 남기는" 동작이 운영에 맞는 경우가 많다.
--
-- 5) 큰 파일은 LOADBLOBFROMFILE(서버 측 읽기)이 애플리케이션 스트리밍보다 빠르다(2MB 0.46초).
--    반대로 파일이 클라이언트에만 있다면 이 방법을 쓸 수 없다. 설계 때 경로를 정한다.
--
-- 6) 작은 LOB 을 대량으로 넣으면 세그먼트가 데이터보다 훨씬 커진다(CHUNK 단위 할당).
--    값 분포를 보고 인라인 저장 또는 CHUNK 조정을 검토한다.
