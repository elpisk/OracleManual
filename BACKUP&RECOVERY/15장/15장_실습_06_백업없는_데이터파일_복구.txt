================================================================================
 실습 15-6 : 백업받지 않은 테이블스페이스의 데이터파일 복구
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 30분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 테이블스페이스를 만들고 백업하기 전에 데이터파일이 손상되었다.
-- 백업이 없으므로 RESTORE가 불가능하다.
-- 빈 파일을 만들고 아카이브를 처음부터 적용해 복구한다.
-- NOLOGGING으로 적재한 데이터가 복구되지 않는 것도 확인한다.
-- 복구 유형 : CREATE DATAFILE 후 완전 복구, 무중단

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 15-5까지 완료되어 있어야 한다.
-- 아카이브가 연속이어야 이 복구가 성립한다.

[oracle@oel7v9r1 ~]$ rman target / catalog rcatowner@rcat

RMAN> BACKUP DATABASE PLUS ARCHIVELOG TAG 'BASE_156';
RMAN> EXIT

--------------------------------------------------------------------------------
[3] 초기 상태 확인
--------------------------------------------------------------------------------
[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> CREATE TABLESPACE tbs_nb
  2    DATAFILE '/u03/oradata/ORCL/tbs_nb01.dbf' SIZE 20M;

Tablespace created.

SYS@orcl> SELECT file#, name FROM v$datafile WHERE name LIKE '%tbs_nb%';

     FILE# NAME
---------- ------------------------------------------
        10 /u03/oradata/ORCL/tbs_nb01.dbf

SYS@orcl> CREATE TABLE hr.rec156(id NUMBER, memo VARCHAR2(30)) TABLESPACE tbs_nb;
SYS@orcl> INSERT INTO hr.rec156 VALUES (1, 'logging insert');
SYS@orcl> COMMIT;

-- NOLOGGING으로 적재한 데이터도 만들어 둔다.

SYS@orcl> CREATE TABLE hr.rec156_nl NOLOGGING TABLESPACE tbs_nb
  2  AS SELECT LEVEL AS id, 'nologging' AS memo FROM dual CONNECT BY LEVEL <= 100;

Table created.

SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;

System altered.

SYS@orcl> INSERT INTO hr.rec156 VALUES (2, 'after switch');
SYS@orcl> COMMIT;

-- 이 테이블스페이스는 아직 한 번도 백업되지 않았다.

SYS@orcl> EXIT

[oracle@oel7v9r1 ~]$ rman target / catalog rcatowner@rcat

RMAN> LIST BACKUP OF TABLESPACE tbs_nb SUMMARY;

specification does not match any backup in the repository

RMAN> REPORT UNRECOVERABLE;

Report of files that need backup due to unrecoverable operations
File Type of Backup Required Name
---- ----------------------- ---------------------------------
10   full or incremental     /u03/oradata/ORCL/tbs_nb01.dbf

-- NOLOGGING 작업이 있었으므로 이 목록에 나온다.
-- 즉시 백업하라는 경고이지만 실습을 위해 그대로 둔다.

--------------------------------------------------------------------------------

--------------------------------------------------------------------------------
RMAN> EXIT

[oracle@oel7v9r1 ~]$ rm -f /u03/oradata/ORCL/tbs_nb01.dbf

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> ALTER SYSTEM FLUSH BUFFER_CACHE;

--------------------------------------------------------------------------------
[6] 진단
--------------------------------------------------------------------------------
SYS@orcl> EXIT
[oracle@oel7v9r1 ~]$ rman target / catalog rcatowner@rcat

RMAN> RESTORE DATAFILE 10 PREVIEW;

Starting restore at 16-MAY-25
RMAN-00571: ===========================================================
RMAN-00569: =============== ERROR MESSAGE STACK FOLLOWS ===============
RMAN-00571: ===========================================================
RMAN-03002: failure of restore command at 05/16/2025 15:02:11
RMAN-06023: no backup or copy of datafile 10 found to restore

-- 백업이 없다.

RMAN> LIST ARCHIVELOG ALL SUMMARY;

Key     Thrd Seq     S Low Time
------- ---- ------- - ---------
64      1    34      A 16-MAY-25
65      1    35      A 16-MAY-25

-- 아카이브는 파일 생성 시점부터 연속이다.

-- 판정
--   ① 백업 없음 → RESTORE 불가
--   ② 파일 생성 시점부터 아카이브 연속 → CREATE DATAFILE 후 복구 가능
--   ③ NOLOGGING 적재분은 복구되지 않는다 → 재적재 필요
--   → 빈 파일을 만들고 아카이브를 적용한다.

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 먼저 OFFLINE 없이 CREATE DATAFILE을 시도해 본다.

RMAN> SQL "ALTER DATABASE CREATE DATAFILE 10 AS
  2>   ''/u03/oradata/ORCL/tbs_nb01.dbf''";

RMAN-00571: ===========================================================
RMAN-00569: =============== ERROR MESSAGE STACK FOLLOWS ===============
RMAN-00571: ===========================================================
RMAN-03009: failure of sql command on default channel at 05/16/2025 15:04:22
ORA-01178: file 10 created before last CREATE CONTROLFILE, cannot recreate
ORA-01110: data file 10: '/u03/oradata/ORCL/tbs_nb01.dbf'

-- << 원인 파악 >>
-- 실습 도중 컨트롤파일을 재생성한 이력이 있으면 이 오류가 난다.
-- 이 환경에서는 파일이 ONLINE 상태여서 발생한 문제다.
-- 먼저 OFFLINE으로 내려 데이터베이스가 그 파일을 놓게 해야 한다.

-- << 수정 >>

RMAN> SQL 'ALTER DATABASE DATAFILE 10 OFFLINE';

sql statement: ALTER DATABASE DATAFILE 10 OFFLINE

RMAN> SQL "ALTER DATABASE CREATE DATAFILE 10 AS
  2>   ''/u03/oradata/ORCL/tbs_nb01.dbf''";

sql statement: ALTER DATABASE CREATE DATAFILE 10 AS ...

RMAN> EXIT
[oracle@oel7v9r1 ~]$ ls -l /u03/oradata/ORCL/tbs_nb01.dbf
-rw-r-----. 1 oracle oinstall 20979712 May 16 15:06 tbs_nb01.dbf

-- 빈 파일이 만들어졌다. 크기는 원래와 같다.

[oracle@oel7v9r1 ~]$ rman target / catalog rcatowner@rcat

RMAN> RECOVER DATAFILE 10;

archived log file name=/u02/arch1/1_35_1190530011.dbf thread=1 sequence=35
media recovery complete, elapsed time: 00:00:03
Finished recover at 16-MAY-25

RMAN> SQL 'ALTER DATABASE DATAFILE 10 ONLINE';

sql statement: ALTER DATABASE DATAFILE 10 ONLINE

--------------------------------------------------------------------------------
[8] 검증 및 실무 포인트
--------------------------------------------------------------------------------
RMAN> EXIT

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SELECT * FROM v$recover_file;

no rows selected

SYS@orcl> SELECT id, memo FROM hr.rec156 ORDER BY id;

        ID MEMO
---------- ------------------------------
         1 logging insert
         2 after switch

-- 일반 INSERT로 넣은 데이터는 모두 복구되었다.

SYS@orcl> SELECT COUNT(*) FROM hr.rec156_nl;
SELECT COUNT(*) FROM hr.rec156_nl
                     *
ERROR at line 1:
ORA-01578: ORACLE data block corrupted (file # 10, block # 138)
ORA-01110: data file 10: '/u03/oradata/ORCL/tbs_nb01.dbf'
ORA-26040: Data block was loaded using the NOLOGGING option

-- << 원인 파악 >>
-- ORA-26040은 NOLOGGING으로 적재된 블록이라는 뜻이다.
-- 그 작업은 리두에 기록되지 않아 아카이브로 되살릴 수 없다.
-- 블록이 물리적으로 손상된 것이 아니라 내용이 비어 있는 상태다.
-- 해당 데이터는 재적재하는 수밖에 없다.

SYS@orcl> DROP TABLE hr.rec156_nl PURGE;

Table dropped.

SYS@orcl> CREATE TABLE hr.rec156_nl TABLESPACE tbs_nb
  2  AS SELECT LEVEL AS id, 'reloaded' AS memo FROM dual CONNECT BY LEVEL <= 100;

Table created.

SYS@orcl> SELECT COUNT(*) FROM hr.rec156_nl;

  COUNT(*)
----------
       100

SYS@orcl> EXIT

[oracle@oel7v9r1 ~]$ rman target / catalog rcatowner@rcat

RMAN> BACKUP TABLESPACE tbs_nb TAG 'FIRST_BACKUP';

Starting backup at 16-MAY-25
Finished backup at 16-MAY-25

RMAN> REPORT UNRECOVERABLE;

Report of files that need backup due to unrecoverable operations
File Type of Backup Required Name
---- ----------------------- ---------------------------------

-- 목록이 비었다.

RMAN> EXIT

-- 실무 포인트
-- 1) 백업이 없어도 아카이브가 연속이면 복구할 수 있다.
--    ALTER DATABASE CREATE DATAFILE 로 빈 파일을 만들고
--    파일 생성 시점부터의 아카이브를 적용하는 방식이다.
--
-- 2) 조건은 아카이브가 하나도 빠지지 않는 것이다.
--    중간에 결손이 있으면 그 지점까지만 복구되어 파일을 열 수 없다.
--
-- 3) NOLOGGING으로 적재된 데이터는 복구되지 않는다.
--    조회 시 ORA-01578과 ORA-26040이 함께 나온다.
--    블록 손상이 아니라 내용이 없는 상태이므로 재적재가 답이다.
--
-- 4) REPORT UNRECOVERABLE 이 이 상황을 미리 알려 준다.
--    NOLOGGING 작업 직후 이 목록에 파일이 나타나므로
--    정기 점검 항목으로 넣어 두면 즉시 백업할 근거가 된다.
--
-- 5) 구조를 바꾸면 즉시 백업한다.
--    테이블스페이스를 만들고 백업을 미루면 이 상황이 만들어진다.
--    복구는 되지만 시간이 오래 걸리고 NOLOGGING 부분은 잃는다.
--
-- 6) 대량 적재에 NOLOGGING을 쓴다면 직후 백업을 절차에 넣는다.
--    적재 스크립트 마지막에 BACKUP TABLESPACE 를 붙이는 방식이 확실하다.
