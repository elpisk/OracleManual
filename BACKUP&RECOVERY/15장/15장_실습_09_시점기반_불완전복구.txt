================================================================================
 실습 15-9 : 시점 기반 불완전 복구 — UNTIL TIME / SCN / RESTORE POINT
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 35분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 잘못된 배치 작업으로 데이터가 지워졌다.
-- 그 작업 직전 시점으로 되돌린다.
-- 복원 지점을 만들어 두고 SET UNTIL RESTORE POINT 로 지정한다.
-- SET UNTIL의 순서 규칙과 오픈 전 검증 절차를 확인한다.
-- 복구 유형 : 불완전 복구, 전면 중단, 지정 시점 이후 손실

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 15-8까지 완료되어 있어야 한다.

[oracle@oel7v9r1 ~]$ rman target / catalog rcatowner@rcat

RMAN> BACKUP DATABASE PLUS ARCHIVELOG TAG 'BASE_159';
RMAN> EXIT

--------------------------------------------------------------------------------
[3] 초기 상태 확인
--------------------------------------------------------------------------------
[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> CREATE TABLE hr.orders159(id NUMBER, memo VARCHAR2(30)) TABLESPACE users;

  2  SELECT LEVEL, 'order-' || LEVEL FROM dual CONNECT BY LEVEL <= 1000;

[4] 장애 유발 : 잘못된 배치 작업
--------------------------------------------------------------------------------
SYS@orcl> DELETE FROM hr.orders159 WHERE id > 100;

900 rows deleted.

SYS@orcl> COMMIT;

Commit complete.

SYS@orcl> SELECT COUNT(*) FROM hr.orders159;

  COUNT(*)
----------

-- 조건을 잘못 써서 900건이 지워졌다.
-- 이후에도 정상 업무가 이어졌다고 가정한다.

SYS@orcl> INSERT INTO hr.orders159 VALUES (9001, 'after batch');
SYS@orcl> COMMIT;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;

System altered.

[5] 증상 관찰
--------------------------------------------------------------------------------
SYS@orcl> SELECT COUNT(*) FROM hr.orders159;

  COUNT(*)
----------
       101

-- 파일 손상이 아니므로 v$recover_file 은 비어 있다.

SYS@orcl> SELECT * FROM v$recover_file;

no rows selected

-- 물리적 장애가 아니라 논리적 오류다.
-- 되돌리려면 데이터베이스 전체를 그 시점으로 보내야 한다.

--------------------------------------------------------------------------------
[6] 진단
--------------------------------------------------------------------------------
-- 되돌릴 시점을 확정한다. 세 가지 방법이 있다.

SYS@orcl> SELECT name, scn FROM v$restore_point;

NAME                    SCN
------------------ --------
BEFORE_BATCH_159    2661200

-- ① 복원 지점 : 가장 정확하고 간편하다.

SYS@orcl> SELECT TIMESTAMP_TO_SCN(
  2    TO_TIMESTAMP('2025-05-17 09:13:00','YYYY-MM-DD HH24:MI:SS')) AS scn
  3  FROM dual;

       SCN
----------
   2661280

-- ② 시각으로부터 SCN을 구한다.

-- ③ LogMiner로 그 DELETE의 SCN을 직접 찾을 수도 있다. (10장 참조)

SYS@orcl> EXIT

--   → SET UNTIL RESTORE POINT 로 불완전 복구

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
RMAN> SHUTDOWN IMMEDIATE
RMAN> STARTUP MOUNT

-- 먼저 순서를 잘못 지정해 본다. RESTORE를 먼저 수행한다.

RMAN> RESTORE DATABASE;

Starting restore at 17-MAY-25
channel ORA_DISK_1: restoring datafile 00001 to /u03/oradata/ORCL/system01.dbf
Finished restore at 17-MAY-25

RMAN> RUN {
  2>   SET UNTIL RESTORE POINT before_batch_159;
  3>   RECOVER DATABASE;
  4> }

Starting recover at 17-MAY-25
RMAN-00571: ===========================================================
RMAN-00569: =============== ERROR MESSAGE STACK FOLLOWS ===============
RMAN-00571: ===========================================================
RMAN-03002: failure of recover command at 05/17/2025 09:41:02
RMAN-06556: datafile 1 must be restored from backup older than SCN 2661200

-- << 원인 파악 >>
-- 복원된 백업이 지정한 시점보다 뒤의 것이다.
-- SET UNTIL 없이 RESTORE를 수행해 가장 최신 백업이 선택되었기 때문이다.
-- SET UNTIL 은 RECOVER뿐 아니라 RESTORE의 백업 선택에도 영향을 준다.
-- 따라서 RUN 블록의 첫 줄에 두어야 한다.

-- << 수정 >>

RMAN> RUN {
  2>   SET UNTIL RESTORE POINT before_batch_159;
  3>   RESTORE DATABASE;
  4>   RECOVER DATABASE;
  5> }

[8] 검증 및 실무 포인트
--------------------------------------------------------------------------------
-- RESETLOGS 전에 결과를 확인한다.

RMAN> ALTER DATABASE OPEN READ ONLY;

Statement processed

RMAN> EXIT

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SELECT COUNT(*) FROM hr.orders159;

  COUNT(*)

SYS@orcl> SELECT COUNT(*) FROM hr.orders159 WHERE id = 9001;

----------

-- 배치 이후의 정상 업무도 함께 사라졌다.
-- 이 부분이 손실이며 재처리 대상이다.

SYS@orcl> SELECT TO_CHAR(SCN_TO_TIMESTAMP(checkpoint_change#),
  2    'YYYY-MM-DD HH24:MI:SS') AS recovered_time FROM v$database;

-- 실무 포인트
-- 1) 논리적 오류는 v$recover_file 에 나타나지 않는다.
--    파일은 멀쩡하고 내용만 잘못된 상태이므로 진단 방법이 다르다.
--    무엇이 언제 잘못되었는지를 업무 쪽 정보로 확정해야 한다.
--
-- 2) SET UNTIL 은 RUN 블록의 첫 줄에 둔다.
--    RESTORE의 백업 선택에도 영향을 주기 때문이다.
--    순서를 어기면 RMAN-06556이 나오고 복원을 다시 해야 한다.
--
-- 3) 복원 지점을 미리 만들어 두면 시점 지정이 가장 간편하다.
--    SCN을 찾을 필요 없이 이름으로 지정한다.
--    위험한 배치 작업 전에 만드는 것을 절차로 두면 좋다.
--
-- 4) RESTORE PREVIEW의 최소 복구 SCN보다 앞선 지점은 선택할 수 없다.
--    그보다 앞으로 가려면 더 오래된 백업이 필요하다.
--
-- 5) RESETLOGS 전에 READ ONLY로 열어 확인한다.
--    원하는 시점이 아니면 다시 복구할 수 있다.
--    RESETLOGS를 수행하면 되돌리기가 훨씬 번거로워진다.
--
-- 6) 지정 시점 이후의 정상 업무도 함께 사라진다.
--    이 손실 범위를 복구 전에 현업에 알리고 재처리 계획을 함께 세운다.
--    복구 후에는 즉시 전체 백업을 받는다.
