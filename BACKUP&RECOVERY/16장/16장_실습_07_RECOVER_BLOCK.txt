================================================================================
 실습 16-7 : RECOVER ... BLOCK / CORRUPTION LIST로 손상 블록만 복구
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 30분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 블록 손상을 다시 만들고 블록 단위로만 복구한다.
-- 복구 중에도 다른 객체에 접근할 수 있음을 확인한다.
-- NOLOGGING으로 적재된 블록은 되살릴 수 없음을 확인한다.
-- 복구 유형 : 블록 미디어 복구, 업무 중단 없음

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 16-5·16-6까지 완료되어 tbs16(파일 9)과 hr.blk165(600행)가 있어야 한다.
-- 손상 이전 백업이 반드시 필요하다.

[oracle@oel7v9 ~]$ rman target / catalog rcatowner@rcat

RMAN> BACKUP DATAFILE 9 TAG 'BASE_167';

Starting backup at 18-MAY-25
Finished backup at 18-MAY-25

RMAN> EXIT

--------------------------------------------------------------------------------
[3] 초기 상태 확인
--------------------------------------------------------------------------------
[oracle@oel7v9 ~]$ sqlplus / as sysdba

SYS@orcl> SELECT COUNT(*) FROM hr.blk165;

  COUNT(*)
----------
       600

SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;

System altered.

-- NOLOGGING으로 직접 경로 적재한 테이블을 만든다. 백업 이후에 만든다.

SYS@orcl> CREATE TABLE hr.blk167_nl(id NUMBER, pad VARCHAR2(2000))
  2    TABLESPACE tbs16 NOLOGGING;

Table created.

SYS@orcl> INSERT /*+ APPEND */ INTO hr.blk167_nl
  2  SELECT LEVEL, RPAD('z',2000,'z') FROM dual CONNECT BY LEVEL <= 300;

300 rows created.

SYS@orcl> COMMIT;
SYS@orcl> ALTER SYSTEM CHECKPOINT;

System altered.

-- NOLOGGING 테이블에 /*+ APPEND */ 로 넣으면 직접 경로 적재라 리두를 남기지 않는다.
-- (일반 INSERT나 작은 CTAS는 리두가 남을 수 있어, 직접 경로를 강제하려고 APPEND를 쓴다.)

SYS@orcl> SELECT MIN(DBMS_ROWID.rowid_block_number(ROWID)) AS mn,
  2         MAX(DBMS_ROWID.rowid_block_number(ROWID)) AS mx
  3  FROM   hr.blk167_nl;

        MN         MX
---------- ----------
       387        492

SYS@orcl> EXIT

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
SYS@orcl> EXIT
[oracle@oel7v9 ~]$ rman target / catalog rcatowner@rcat

RMAN> RECOVER CORRUPTION LIST;

Starting recover at 18-MAY-25
allocated channel: ORA_DISK_1
channel ORA_DISK_1: restoring block(s)
restoring blocks of datafile 00009
channel ORA_DISK_1: reading from backup piece /home/oracle/rmanbk/ORCL_20250518_...bkp
channel ORA_DISK_1: restored block(s) from backup piece 1
channel ORA_DISK_1: block restore complete, elapsed time: 00:00:01
starting media recovery
media recovery complete, elapsed time: 00:00:03
Finished recover at 18-MAY-25

-- << 관찰 >>
-- RMAN-06023 같은 오류 없이 "media recovery complete"로 끝난다.
-- 3개 블록을 모두 복원·복구했다고 보고하고 v$database_block_corruption 도 비운다.
-- 그러나 NOLOGGING 블록은 진짜로 복구되지 않았다. 다음에서 드러난다.

RMAN> EXIT

[oracle@oel7v9 ~]$ sqlplus / as sysdba

SYS@orcl> SELECT COUNT(*) FROM hr.blk165;

  COUNT(*)
----------
       600

-- 로깅 테이블은 완전히 복구되었다.

SYS@orcl> SELECT COUNT(*) FROM hr.blk167_nl;
SELECT COUNT(*) FROM hr.blk167_nl
       *
ERROR at line 1:
ORA-01578: ORACLE data block corrupted (file # 9, block # 439)
ORA-01110: data file 9: '/u03/oradata/ORCL/tbs16_01.dbf'
ORA-26040: Data block was loaded using the NOLOGGING option

SYS@orcl> SELECT file#, block#, blocks, corruption_type
  2  FROM   v$database_block_corruption;

no rows selected

-- << 원인 파악 >>
-- 목록은 비었는데 blk167_nl은 여전히 오류다. 이번엔 ORA-26040이 붙었다.
-- RECOVER는 백업(내용 없음)으로 블록을 되돌리고 리두를 적용했지만,
-- NOLOGGING 적재라 리두에 그 데이터가 없어 논리적으로 복구되지 않았다.
-- all-zero 물리 손상이 'NOLOGGING으로 적재된 블록'이라는 논리 손상으로 바뀐 것이다.
-- 어떤 백업·아카이브로도 이 데이터는 되살릴 수 없다.

-- << 수정 >> 복구할 수 없는 객체는 재생성(재적재)한다.

SYS@orcl> DROP TABLE hr.blk167_nl PURGE;

Table dropped.

SYS@orcl> CREATE TABLE hr.blk167_nl TABLESPACE tbs16
  2  AS SELECT LEVEL AS id, RPAD('z',2000,'z') AS pad
  3     FROM dual CONNECT BY LEVEL <= 100;

Table created.

SYS@orcl> SELECT COUNT(*) FROM hr.blk167_nl;

  COUNT(*)
----------
       100

-- NOLOGGING 없이 다시 만들었다.

--------------------------------------------------------------------------------
SYS@orcl> EXIT
[oracle@oel7v9 ~]$ rman target / catalog rcatowner@rcat

RMAN> VALIDATE DATAFILE 9;

List of Datafiles
=================
File Status Marked Corrupt Empty Blocks Blocks Examined High SCN
---- ------ -------------- ------------ --------------- ----------
9    OK     1              7148         7680            7993726
  File Name: /u03/oradata/ORCL/tbs16_01.dbf
  Block Type Blocks Failing Blocks Processed
  ---------- -------------- ----------------
  Data       0              380
  Index      0              0
  Other      0              152

-- Status는 OK다. 다만 'Marked Corrupt'가 1로 남아 있다.
-- 테이블을 지워 그 블록이 사용 중이 아니게 되었지만, 블록 자체에 찍힌 손상 표시는
-- 그 블록이 다시 쓰이거나 재포맷될 때까지 남는다. 사용 중이 아니라 Status는 OK다.

RMAN> BACKUP DATAFILE 9 TAG 'AFTER_BLOCK';
RMAN> EXIT

-- 실무 포인트
-- 1) 블록 복구는 손상 블록만 백업에서 꺼내 되돌린다. 몇 초면 끝난다.
--    파일을 OFFLINE으로 내리지 않아 업무도 멈추지 않는다.
--    같은 파일의 다른 객체는 정상 조회된다. 이것이 파일 복구와의 가장 큰 차이다.
--
-- 2) RECOVER CORRUPTION LIST 는 v$database_block_corruption 목록 전체를 대상으로 한다.
--    수행 후 목록이 비고 "media recovery complete"로 끝나지만,
--    그것이 모든 블록이 살아났다는 뜻은 아니다. 객체를 실제로 조회해 확인해야 한다.
--
-- 3) NOLOGGING(직접 경로)으로 적재된 블록은 복구되지 않는다.
--    물리 손상(ALL ZERO)이 복구 뒤 ORA-26040(논리 손상)으로 바뀐다.
--    리두에 데이터가 없으므로 어떤 방법으로도 되살릴 수 없다.
--
-- 4) NOLOGGING 블록의 유일한 해법은 재적재다.
--    해당 객체를 재생성하거나 원본에서 다시 적재한다. 데이터는 이미 잃은 것이다.
--    복구가 필요한 데이터라면 처음부터 NOLOGGING을 쓰지 않거나,
--    적재 직후 그 테이블스페이스를 백업하는 것이 대비책이다.
--
-- 5) 되살릴 수 없는 블록은 FORCE LOGGING으로 예방한다.
--    ALTER DATABASE FORCE LOGGING 이면 NOLOGGING 지시가 무시되어 늘 리두가 남는다.
--    Data Guard 환경에서는 대개 이 설정을 강제한다.
--
-- 6) 복구 후 VALIDATE로 손상이 없는지 다시 확인한다.
--    재적재 전까지 'Marked Corrupt'가 남을 수 있으나, 사용 중이 아니면 Status는 OK다.
