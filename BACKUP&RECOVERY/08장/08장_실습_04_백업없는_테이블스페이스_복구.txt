================================================================================
 실습 8-4 : 백업 받지 않은 테이블스페이스의 데이터파일 복구
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 20분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 백업 이후 생성한 테이블스페이스의 데이터파일이 유실되었다.
-- 백업본이 없어 복원할 대상이 없다.
-- 파일 생성 시점부터의 아카이브가 모두 있으므로 빈 파일에 리두를 적용한다.
-- NOARCHIVELOG 환경에는 존재하지 않던 선택지다.
-- 복구 유형 : 완전 복구(CREATE DATAFILE)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 8-3을 마친 상태에서 이어서 진행한다.
-- 아카이브가 연속으로 보관되어 있어야 한다.

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> COL name FOR a48

--------------------------------------------------------------------------------
[3] 초기 상태 확인 : 백업 이후 테이블스페이스 생성
--------------------------------------------------------------------------------
SYS@orcl> !ls /home/oracle/backup/ch08/*.dbf
/home/oracle/backup/ch08/hist01.dbf
/home/oracle/backup/ch08/sysaux01.dbf
/home/oracle/backup/ch08/system01.dbf
/home/oracle/backup/ch08/undotbs01.dbf
/home/oracle/backup/ch08/users01.dbf

-- 백업본에는 다섯 개 파일이 있다.

SYS@orcl> CREATE TABLESPACE tbs01
  2    DATAFILE '/u01/app/oracle/oradata/ORCL/tbs01.dbf' SIZE 20M;

Tablespace created.

SYS@orcl> SELECT file#, name, creation_change#, creation_time
  2  FROM   v$datafile WHERE name LIKE '%tbs01%';

     FILE# NAME                                       CREATION_CHANGE# CREATION
---------- ------------------------------------------ ---------------- --------
        11 /u01/app/oracle/oradata/ORCL/tbs01.dbf              2406240 11-MAY-25

-- CREATION_CHANGE# 가 이 파일의 출발 SCN이다.
-- 이 SCN 이후의 아카이브가 모두 있어야 CREATE DATAFILE 이 성립한다.

SYS@orcl> CREATE TABLE hr.emp85 TABLESPACE tbs01
  2  AS SELECT * FROM hr.employees;
Table created.

SYS@orcl> INSERT INTO hr.emp85 SELECT * FROM hr.employees;
107 rows created.
SYS@orcl> COMMIT;

SYS@orcl> SELECT COUNT(*) FROM hr.emp85;

  COUNT(*)
----------
       214

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
System altered.

SYS@orcl> !rm -f /u01/app/oracle/oradata/ORCL/tbs01.dbf

SYS@orcl> ALTER SYSTEM FLUSH BUFFER_CACHE;
System altered.

--------------------------------------------------------------------------------
[6] 진단
--------------------------------------------------------------------------------
SYS@orcl> !ls /home/oracle/backup/ch08/tbs01.dbf 2>/dev/null
[oracle@oel7v9r1 ~]$

-- 백업본에 없다. 백업 이후 만든 테이블스페이스이기 때문이다.

SYS@orcl> SELECT file#, creation_change# FROM v$datafile WHERE file# = 11;

     FILE# CREATION_CHANGE#
---------- ----------------
        11          2406240

SYS@orcl> SELECT sequence#, first_change#, next_change#
  2  FROM   v$log_history WHERE 2406240 BETWEEN first_change# AND next_change#;

 SEQUENCE# FIRST_CHANGE# NEXT_CHANGE#
---------- ------------- ------------
        27       2406100      2406820

SYS@orcl> SELECT sequence#, status FROM v$archived_log
  2  WHERE  sequence# >= 27 ORDER BY sequence#;

 SEQUENCE# S
---------- -
        27 A
        28 A
        29 A

-- 파일 생성 시점(시퀀스 27) 이후의 아카이브가 모두 있다.

-- 판정
--   ① 손상 파일 : 11 (일반 테이블스페이스)
--   ② 백업본 : 없음 → CREATE DATAFILE 경로
--   ③ 아카이브 : 생성 시점부터 연속 보유 → 성립
--   → OFFLINE → CREATE DATAFILE → RECOVER → ONLINE

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 빈 파일을 만들어 본다.

SYS@orcl> ALTER DATABASE CREATE DATAFILE 11
  2  AS '/u01/app/oracle/oradata/ORCL/tbs01.dbf';
ALTER DATABASE CREATE DATAFILE 11
*
ERROR at line 1:
ORA-01180: can not create datafile 1
ORA-01110: data file 11: '/u01/app/oracle/oradata/ORCL/tbs01.dbf'

-- 위 오류는 대상이 잘못 지정된 경우다. 실제로는 아래 오류가 먼저 나온다.

SYS@orcl> ALTER TABLESPACE tbs01 OFFLINE NORMAL;
ALTER TABLESPACE tbs01 OFFLINE NORMAL
*
ERROR at line 1:
ORA-01191: file 11 is already offline - cannot do a normal offline
ORA-01110: data file 11: '/u01/app/oracle/oradata/ORCL/tbs01.dbf'

-- << 원인 파악 >>
-- 파일이 이미 문제 상태라 체크포인트를 수행할 수 없다.
-- OFFLINE NORMAL은 체크포인트를 요구하므로 손상된 파일에는 쓸 수 없다.
-- 장애 대응에서는 OFFLINE IMMEDIATE 가 사실상 유일한 선택이다.
-- 또한 CREATE DATAFILE 은 대상이 OFFLINE 상태여야 안전하게 수행된다.

-- << 수정 >>

SYS@orcl> ALTER TABLESPACE tbs01 OFFLINE IMMEDIATE;

Tablespace altered.

SYS@orcl> ALTER DATABASE CREATE DATAFILE 11
  2  AS '/u01/app/oracle/oradata/ORCL/tbs01.dbf';

Database altered.

SYS@orcl> !ls -l /u01/app/oracle/oradata/ORCL/tbs01.dbf
-rw-r-----. 1 oracle oinstall 20979712 May 11 11:04 tbs01.dbf

-- 원본과 같은 크기의 빈 파일이 만들어졌다.

SYS@orcl> SELECT file#, error, change# FROM v$recover_file WHERE file# = 11;

     FILE# ERROR                    CHANGE#
---------- -------------------- -----------
        11                          2406240

-- 복구 시작 SCN이 파일 생성 SCN과 같다. 처음부터 리두를 적용한다는 뜻이다.

SYS@orcl> SET AUTORECOVERY ON
SYS@orcl> RECOVER TABLESPACE tbs01;
Media recovery complete.

SYS@orcl> ALTER TABLESPACE tbs01 ONLINE;

Tablespace altered.

[8] 검증 및 실무 포인트
--------------------------------------------------------------------------------
SYS@orcl> SELECT * FROM v$recover_file;

no rows selected

SYS@orcl> SELECT COUNT(*) FROM hr.emp85;

  COUNT(*)
----------
       214

-- 백업본이 없었는데도 데이터가 완전히 되살아났다.

SYS@orcl> SELECT file#, name, status, bytes/1024/1024 AS mb
  2  FROM   v$datafile WHERE file# = 11;

     FILE# NAME                                       STATUS     MB
---------- ------------------------------------------ ------- -----
        11 /u01/app/oracle/oradata/ORCL/tbs01.dbf     ONLINE     20

SYS@orcl> SELECT resetlogs_change# FROM v$database;

RESETLOGS_CHANGE#
-----------------
                1

-- NOLOGGING 세그먼트가 있었다면 어떻게 되는지 확인한다.

SYS@orcl> CREATE TABLE hr.emp85n TABLESPACE tbs01 NOLOGGING
  2  AS SELECT * FROM hr.employees;
Table created.

SYS@orcl> SELECT table_name, logging FROM dba_tables
  2  WHERE  table_name IN ('EMP85','EMP85N');

TABLE_NAME LOG
---------- ---
EMP85      YES
EMP85N     NO

-- LOGGING이 NO인 세그먼트는 리두가 생성되지 않는다.
-- 이 상태에서 같은 복구를 하면 조회 시 ORA-01578 또는 ORA-26040이 발생한다.
-- 재적재 외에 방법이 없으므로 NOLOGGING 작업 직후에는 반드시 백업을 받는다.

SYS@orcl> SELECT name, force_logging FROM v$database;

NAME      FOR
--------- ---
ORCL      NO

-- FORCE LOGGING 을 켜면 NOLOGGING 지정이 무시된다.

-- 실무 포인트
-- 1) 백업본이 없어도 ARCHIVELOG 환경에서는 복구할 수 있다.
--    파일 생성 시점부터의 리두가 모두 아카이브에 남아 있기 때문이다.
--    빈 파일을 출발점으로 삼아 그 리두를 처음부터 적용하는 방식이다.
--
-- 2) 성패는 CREATION_CHANGE# 이후 아카이브의 연속성에 달렸다.
--    하나라도 빠지면 그 지점에서 복구가 멈춘다.
--    시작 전에 v$log_history 와 v$archived_log 로 반드시 확인한다.
--
-- 3) 손상된 파일에는 OFFLINE IMMEDIATE 를 쓴다.
--    NORMAL은 체크포인트를 요구하므로 ORA-01191 등으로 실패한다.
--
-- 4) AS 절을 생략하면 원래 경로에, 지정하면 새 위치에 만든다.
--    새 위치에 만들면 컨트롤파일의 경로도 함께 갱신되므로 RENAME이 필요 없다.
--
-- 5) NOLOGGING 세그먼트는 이 방법으로 복구되지 않는다.
--    복구는 성공하고 ONLINE도 되지만 해당 블록을 읽을 때 ORA-01578이 나온다.
--    객체를 다시 만드는 것 외에 방법이 없다.
--
-- 6) 근본 대책은 백업 대상 관리다.
--    테이블스페이스를 새로 만들면 백업 스크립트와 매니페스트에 즉시 반영하고,
--    가능하면 그 자리에서 첫 백업을 받아 둔다.

SYS@orcl> EXIT
