[6] 진행 로그 : Stored Script 등록
--------------------------------------------------------------------------------
[oracle@oel7v9 ~]$ rman target / catalog rcatowner@rcat

RMAN> CREATE SCRIPT full_backup COMMENT 'full backup + archivelog + cleanup' {
  2>   BACKUP AS COMPRESSED BACKUPSET DATABASE
  3>     FORMAT '/home/oracle/rmanbk/full_%d_%T_%U'
  4>     TAG 'FULL_DAILY';
  5>   BACKUP ARCHIVELOG ALL DELETE INPUT;
  6>   DELETE NOPROMPT OBSOLETE;
  7> }

created script full_backup

RMAN> LIST SCRIPT NAMES;

List of Stored Scripts in Recovery Catalog


    Scripts of Target Database ORCL

       Script Name
       Description
       -----------------------------------------------------------------------
       full_backup
       full backup + archivelog + cleanup

RMAN> PRINT SCRIPT full_backup;

printing stored script: full_backup
{
  BACKUP AS COMPRESSED BACKUPSET DATABASE
    FORMAT '/home/oracle/rmanbk/full_%d_%T_%U'
    TAG 'FULL_DAILY';

  DELETE NOPROMPT OBSOLETE;
}

-- 입력한 줄 모양 그대로 저장된다. COMMENT 는 ASCII 로 쓰는 편이 안전하다.

-- 크론 연계용 셸 스크립트를 만든다.

[oracle@oel7v9 ~]$ cat > /home/oracle/rman/full_backup.sh <<'EOS'
#!/bin/bash
export ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_1
export ORACLE_SID=orcl
export PATH=$ORACLE_HOME/bin:$PATH
LOG=/home/oracle/rman/log/full_$(date +%Y%m%d_%H%M).log

rman target / catalog rcatowner/oracle_4U@rcat log=$LOG <<EOF
RUN { EXECUTE SCRIPT full_backup; }
EXIT;
EOF

if grep -q 'RMAN-' $LOG; then
  echo "[ORCL] RMAN backup FAILED - see $LOG"
  exit 1
fi
echo "[ORCL] RMAN backup OK - $LOG"
EOS

-- 실무 포인트
-- 1) 구조 변경만 있고 백업이 없으면 카탈로그에 반영되지 않는다.
--    RESYNC CATALOG를 수동으로 수행해야 최신 상태가 된다.
--    백업이나 복구 명령을 수행하면 자동으로 일어난다.
--
-- 2) 자동 백업이 켜져 있으면 구조 변경 시에도 컨트롤파일 백업이 만들어진다.
--    다만 몇 분 뒤에 만들어지므로 직후에는 보이지 않는다. 파일 목록의 시각으로 확인한다.
--
-- 3) EXECUTE SCRIPT는 RUN 블록 안에서만 쓸 수 있다.
--    프롬프트에서 직접 실행하면 RMAN-01009 구문 오류가 난다.
--
-- 4) Stored Script는 절차를 문서가 아니라 시스템에 고정한다.
--    담당자가 바뀌어도 같은 절차가 실행되며 실수 여지가 줄어든다.
--    PRINT SCRIPT ... TO FILE 로 내보내 형상 관리에 포함할 수 있다.
--
-- 5) 크론에서는 프로필이 읽히지 않는다.
--    ORACLE_HOME, ORACLE_SID, PATH를 스크립트 안에서 반드시 설정한다.
--
-- 6) 로그에서 RMAN- 문자열을 찾는 것이 가장 단순한 실패 판정이다.
--    종료 코드만으로는 부분 실패를 놓칠 수 있다.
--    로그를 날짜별로 남기고 실패 시 알림이 가도록 연결한다.
