================================================================================
 실습 19-10 : DBMS_METADATA DDL 추출과 Transportable Tablespace
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 40분
================================================================================

--------------------------------------------------------------------------------
[1] 실습 개요
--------------------------------------------------------------------------------
-- DBMS_METADATA로 객체 정의를 텍스트로 뽑는다.
-- transform 파라미터로 저장 절을 제거해 다른 환경에서 쓸 수 있게 만든다.
-- 전송 가능 테이블스페이스 방식으로 대용량 이관 절차를 확인한다.
-- 자기 완결성 조건을 위반했을 때의 증상을 재현한다.
-- 작업 유형 : 구조 문서화와 대용량 이관

--------------------------------------------------------------------------------

[4] 작업 수행 : DBMS_METADATA로 DDL 추출
--------------------------------------------------------------------------------
SYS@orcl> SET LONG 200000 LONGCHUNKSIZE 200000 PAGESIZE 0 LINESIZE 300 TRIMSPOOL ON
SYS@orcl> SET FEEDBACK OFF VERIFY OFF ECHO OFF

-- LONGCHUNKSIZE 를 같이 키운다. 빠뜨리면 긴 줄이 80자 단위로 잘려 STORAGE 절이 두 줄로 찍힌다.

SYS@orcl> SELECT DBMS_METADATA.get_ddl('TABLE','SALES','TTS19') FROM dual;

  CREATE TABLE "TTS19"."SALES"
   (    "SALE_ID" NUMBER,
        "CUST_ID" NUMBER,
        "AMT" NUMBER(12,2),
        "MEMO" VARCHAR2(200),
         PRIMARY KEY ("SALE_ID")
  USING INDEX PCTFREE 10 INITRANS 2 MAXTRANS 255
  STORAGE(INITIAL 65536 NEXT 1048576 MINEXTENTS 1 MAXEXTENTS 2147483645
  PCTINCREASE 0 FREELISTS 1 FREELIST GROUPS 1
  BUFFER_POOL DEFAULT FLASH_CACHE DEFAULT CELL_FLASH_CACHE DEFAULT)
  TABLESPACE "TTS_SRC"  ENABLE
   ) SEGMENT CREATION IMMEDIATE
  PCTFREE 10 PCTUSED 40 INITRANS 1 MAXTRANS 255
 NOCOMPRESS LOGGING
  STORAGE(INITIAL 65536 NEXT 1048576 MINEXTENTS 1 MAXEXTENTS 2147483645
  PCTINCREASE 0 FREELISTS 1 FREELIST GROUPS 1
  BUFFER_POOL DEFAULT FLASH_CACHE DEFAULT CELL_FLASH_CACHE DEFAULT)
  TABLESPACE "TTS_SRC"

-- 저장 절이 길어 읽기 불편하고 다른 환경에서 그대로 쓰기도 어렵다.
-- 끝에 세미콜론이 없어 이대로는 스크립트로 못 돌린다.

SYS@orcl> EXEC DBMS_METADATA.set_transform_param(
  2    DBMS_METADATA.session_transform, 'SQLTERMINATOR', TRUE);
SYS@orcl> EXEC DBMS_METADATA.set_transform_param(
  2    DBMS_METADATA.session_transform, 'PRETTY', TRUE);
SYS@orcl> EXEC DBMS_METADATA.set_transform_param(
  2    DBMS_METADATA.session_transform, 'STORAGE', FALSE);
SYS@orcl> EXEC DBMS_METADATA.set_transform_param(
  2    DBMS_METADATA.session_transform, 'SEGMENT_ATTRIBUTES', FALSE);

SYS@orcl> SELECT DBMS_METADATA.get_ddl('TABLE','SALES','TTS19') FROM dual;

  CREATE TABLE "TTS19"."SALES"
   (    "SALE_ID" NUMBER,
        "CUST_ID" NUMBER,
        "AMT" NUMBER(12,2),
        "MEMO" VARCHAR2(200),
         PRIMARY KEY ("SALE_ID")
  USING INDEX  ENABLE
   ) ;

-- 훨씬 간결해졌고 테이블스페이스 지정도 빠졌다. USING INDEX 는 남되 속성은 비었다.

[6] 확인 : 자기 완결성 검사
--------------------------------------------------------------------------------
-- 인덱스 하나를 다른 테이블스페이스로 옮겨 보고 검사한다.

SYS@orcl> ALTER INDEX tts19.idx_sales_cust REBUILD TABLESPACE users;

Index altered.

SYS@orcl> EXEC DBMS_TTS.transport_set_check('TTS_SRC', TRUE);

PL/SQL procedure successfully completed.

SYS@orcl> SELECT violations FROM transport_set_violations;

no rows selected

-- 위반이 아니다. 전송 집합(TTS_SRC) 안의 객체는 밖을 가리키지 않는다.
-- USERS 의 인덱스가 안을 가리키지만 그것은 집합 밖의 사정이다. 그 인덱스는 안 옮겨질 뿐이다.
-- 세 번째 인자 full_check 를 TRUE 로 주면 이런 "밖에서 안을 가리키는" 경우까지 잡는다.

SYS@orcl> EXEC DBMS_TTS.transport_set_check('TTS_SRC', TRUE, TRUE);
SYS@orcl> SELECT violations FROM transport_set_violations;

VIOLATIONS
----------------------------------------------------------------------------------------------------
ORA-39907: Index TTS19.IDX_SALES_CUST in tablespace USERS points to table TTS19.SALES in tablespace
TTS_SRC.

-- 옮긴 뒤 인덱스를 잃지 않으려면 이 검사도 돌려 본다.

-- 진짜 위반은 반대 방향이다. 집합 안의 객체가 밖의 객체를 가리키는 경우다.

SYS@orcl> ALTER INDEX tts19.idx_sales_cust REBUILD TABLESPACE tts_src;

Index altered.

SYS@orcl> COMMIT;
SYS@orcl> CREATE INDEX tts19.idx_cust_name
  2    ON tts19.cust(cname) TABLESPACE tts_src;

Index created.

SYS@orcl> EXEC DBMS_TTS.transport_set_check('TTS_SRC', TRUE);
SYS@orcl> SELECT violations FROM transport_set_violations;

-- 앞에서 AND DATAFILES 없이 내렸던 원래 파일은 OS 에 남아 있으므로 따로 지운다.

-- 실무 포인트
-- 1) DBMS_METADATA는 현재 상태를 즉시 추출한다.
--    덤프를 만들 필요가 없어 구조 문서화에 간편하다.
--    SQLFILE은 덤프의 내용을 추출한다는 점에서 대상이 다르다.
--
-- 2) transform 파라미터를 설정하지 않으면 저장 절이 그대로 따라오고 세미콜론도 없다.
--    STORAGE와 SEGMENT_ATTRIBUTES를 FALSE, SQLTERMINATOR 를 TRUE 로 두면 바로 돌릴 수 있는 형태가 된다.
--    여러 객체를 한 파일로 뽑을 때는 생성 순서(테이블 → 인덱스 → 뷰)를 직접 정한다.
--
-- 3) 전송 가능 테이블스페이스는 파일을 그대로 옮긴다.
--    100MB 테이블스페이스의 메타데이터 덤프가 196KB에 불과하다.
--    데이터 양이 커질수록 이 방식의 이점이 커진다.
--
-- 4) 자기 완결성이 절대 조건이다.
--    위반하면 ORA-39187 로 거부되며 덤프가 만들어지지 않는다.
--    transport_set_check를 먼저 돌려 위반 목록(ORA-39907)을 확인한다.
--    기본 검사는 "안에서 밖을 가리키는" 것만 잡는다. 밖의 인덱스가 안의 표를 가리키는 경우는
--    위반이 아니라 그 인덱스가 빠지는 것이므로, 놓치기 싫으면 full_check=TRUE 로 한 번 더 돌린다.
--
-- 5) 내보내는 동안 읽기 전용이어야 하고, 데이터파일 복사도 그 상태에서 끝낸다.
--    READ WRITE 로 돌린 뒤 복사한 파일은 메타데이터와 어긋나 ORA-19722 로 반입이 거부된다.
--    그 시간만큼 쓰기가 막히므로 업무와 사전 협의가 필요하다.
--    메타데이터만 뽑으므로 시간 자체는 짧다.
--
-- 6) 플랫폼 엔디안이 다르면 RMAN CONVERT가 필요하다.
--    v$transportable_platform 으로 양쪽을 미리 확인한다.
--    같은 엔디안이면 파일을 그대로 복사하면 된다.
