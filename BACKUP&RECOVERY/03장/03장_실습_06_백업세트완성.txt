--------------------------------------------------------------------------------
[1] 실습 개요
--------------------------------------------------------------------------------
-- 데이터 파일 복사만으로는 백업이 완성되지 않는다.
-- 제어 파일 백업, 백업 구간을 덮는 아카이브, 파라미터 파일을 채워 세트를 완성한다.
-- 마지막으로 매니페스트를 작성해 무엇을 언제 받았는지 기록한다.
-- 복구 유형 : 해당 없음(백업 세트 구성)

[4] 작업 수행 : 제어 파일 트레이스와 파라미터 파일
--------------------------------------------------------------------------------
SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO TRACE
  2    AS '/home/oracle/backup/hot/20250509/control.txt';

SYS@orcl> !grep 'CREATE CONTROLFILE' /home/oracle/backup/hot/20250509/control.txt

CREATE CONTROLFILE REUSE DATABASE "ORCL" RESETLOGS  ARCHIVELOG

-- 두 버전이 들어 있다. 온라인 리두가 온전하면 NORESETLOGS, 유실되었으면 RESETLOGS 를 쓴다.

-- 같은 경로로 트레이스 백업을 한 번 더 받아 본다.

SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO TRACE
  2    AS '/home/oracle/backup/hot/20250509/control.txt';
ALTER DATABASE BACKUP CONTROLFILE TO TRACE
*
ERROR at line 1:
ORA-27038: created file already exists
Additional information: 1

-- << 원인 파악 >>
-- 대상 파일이 이미 있으면 덮어쓰지 않고 오류를 낸다.
-- 실수로 기존 백업을 지우는 것을 막기 위한 동작이다.
-- 덮어쓰려면 REUSE 를 붙이고, 이력을 남기려면 파일명에 날짜를 넣는다.

-- << 수정 >>

SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO TRACE
  2    AS '/home/oracle/backup/hot/20250509/control.txt' REUSE;

Database altered.

-- 파라미터 파일 사본

--------------------------------------------------------------------------------
[5] 진행 로그 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 백업 구간을 덮는 아카이브를 찾아 복사하려 한다.

SYS@orcl> SELECT sequence#, name, first_change#, next_change#
  2  FROM   v$archived_log
  3  WHERE  first_change# > 2342683
  4  ORDER  BY sequence#;

no rows selected

-- << 오류 발생 >>
-- 결과가 비었다. 조건이 잘못되었다.

-- << 원인 파악 >>
-- 각 아카이브 로그는 [FIRST_CHANGE#, NEXT_CHANGE#) 구간을 담는다.
-- 백업 시작 SCN 2342683 을 포함하는 로그는 FIRST_CHANGE# 가 그보다 작고
-- NEXT_CHANGE# 가 그보다 큰 것이다.
-- FIRST_CHANGE# > 2342683 조건은 그 로그를 걸러내 버린다.
-- 올바른 조건은 NEXT_CHANGE# > 2342683 이다.

SYS@orcl> SELECT sequence#, first_change#, next_change#
  2  FROM   v$archived_log WHERE dest_id = 1 ORDER BY sequence# DESC FETCH FIRST 5 ROWS ONLY;

 SEQUENCE# FIRST_CHANGE# NEXT_CHANGE#
---------- ------------- ------------
        41       2343405      2343910
        40       2342806      2343405
        39       2342683      2342806
        38       2341598      2342683
        37       2341425      2341598

-- << 수정 >>

SYS@orcl> SELECT sequence#, name, first_change#, next_change#
  2  FROM   v$archived_log
  3  WHERE  dest_id = 1 AND next_change# > 2342683
  4  ORDER  BY sequence#;

 SEQUENCE# NAME                                          FIRST_CHANGE# NEXT_CHANGE#

[6] 매니페스트 작성
--------------------------------------------------------------------------------
SYS@orcl> SET PAGESIZE 0 FEEDBACK OFF HEADING OFF ECHO OFF TRIMSPOOL ON
SYS@orcl> SPOOL /home/oracle/backup/hot/20250509/backup_manifest.txt

SYS@orcl> SELECT 'BACKUP TIME  : ' || TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') FROM dual
  2  UNION ALL SELECT 'BACKUP TYPE  : HOT / WHOLE / INCONSISTENT' FROM dual
  3  UNION ALL SELECT 'DB NAME      : ' || name FROM v$database
  4  UNION ALL SELECT 'DBID         : ' || TO_CHAR(dbid) FROM v$database
  5  UNION ALL SELECT 'LOG MODE     : ' || log_mode FROM v$database
  6  UNION ALL SELECT 'BACKUP SCN   : ' || TO_CHAR(MIN(change#)) FROM v$backup WHERE change# > 0
  7  UNION ALL SELECT 'DATAFILE     : ' || name FROM v$datafile
  8  UNION ALL SELECT 'ARCHIVE FROM : seq 39' FROM dual
  9  UNION ALL SELECT 'CONTROLFILE  : control.ctl (binary), control.txt (trace)' FROM dual
 10  UNION ALL SELECT 'PFILE        : initorcl.ora' FROM dual;

SYS@orcl> SPOOL OFF
SYS@orcl> SET PAGESIZE 100 FEEDBACK ON HEADING ON

--------------------------------------------------------------------------------
[8] 실무 포인트
--------------------------------------------------------------------------------
-- 1) 백업 세트는 네 요소로 구성된다.
--    데이터 파일 / 제어 파일 백업 / 백업 구간을 덮는 아카이브 / 파라미터 파일.
--    Cold Backup 에서는 온라인 리두 로그가 추가된다.
--
-- 2) 아카이브 범위를 찾을 때 조건은 NEXT_CHANGE# > 백업시작SCN 이다.
--    FIRST_CHANGE# 기준으로 잡으면 백업 시작 SCN 을 포함하는 로그가 빠진다.
--    이 실수는 복구 시점에 "첫 아카이브가 없다"는 형태로 드러난다.
--
-- 3) 매니페스트에 DBID 를 반드시 기록한다.
--    제어 파일을 모두 잃으면 RMAN 이 카탈로그를 읽지 못하므로
--    SET DBID 로 자동 백업을 찾아야 한다. 15·16장에서 실제로 사용한다.
--
-- 4) 제어 파일 백업은 두 종류를 모두 받는다.
--    바이너리는 SCN 정보를, 트레이스는 최신 구조를 갖는다.
--    구조가 바뀔 때마다 새로 받는다.
--
-- 5) 검증 조회 한 줄을 백업 스크립트 끝에 넣어 로그에 남긴다.
--    데이터 파일 수, 백업 모드 잔류 수, 마지막 아카이브 시퀀스 세 값이면
--    백업이 정상인지 대부분 판별된다.
--
-- 6) 이 백업 세트는 8장 이후 ARCHIVELOG 복구 시나리오에서 사용한다.
--    날짜 디렉터리째 보관하고 다음 세대를 받을 때 덮어쓰지 않는다.

SYS@orcl> EXIT
