================================================================================
 실습 3-3 : Hot Backup 수행
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL
 소요 시간 : 약 15분
================================================================================

--------------------------------------------------------------------------------
[1] 실습 개요
--------------------------------------------------------------------------------
-- 데이터베이스를 연 채로 테이블스페이스 단위 백업을 수행한다.
-- BEGIN BACKUP 이 헤더 SCN 을 동결하고 데이터 파일 쓰기는 계속된다는 점을 직접 확인한다.
-- 여기서 만든 백업본은 8장 이후 ARCHIVELOG 시나리오의 출발점이 되므로 삭제하지 않는다.
-- 복구 유형 : 해당 없음(백업 작업)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- ARCHIVELOG 모드여야 한다. NOARCHIVELOG 에서는 BEGIN BACKUP 이 거부된다.
-- 데이터베이스가 OPEN 상태여야 한다.
-- 백업 디스크에 데이터 파일 전체 크기 이상의 여유가 있어야 한다.

[oracle@oel7v9r1 ~]$ mkdir -p /home/oracle/backup/hot/20250509
[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> COL file_name FOR a50

SYS@orcl> SELECT name, log_mode, open_mode FROM v$database;

NAME      LOG_MODE     OPEN_MODE
--------- ------------ --------------------
ORCL      ARCHIVELOG   READ WRITE

--------------------------------------------------------------------------------
[3] 초기 상태 확인
--------------------------------------------------------------------------------
SYS@orcl> SELECT tablespace_name, file_id, file_name FROM dba_data_files ORDER BY file_id;

TABLESPACE_NAME    FILE_ID FILE_NAME
--------------- ---------- --------------------------------------------------
SYSTEM                   1 /u01/app/oracle/oradata/ORCL/system01.dbf
SYSAUX                   3 /u01/app/oracle/oradata/ORCL/sysaux01.dbf
UNDOTBS1                 4 /u01/app/oracle/oradata/ORCL/undotbs01.dbf
USERS                    7 /u01/app/oracle/oradata/ORCL/users01.dbf

SYS@orcl> SELECT file#, status, change#, time FROM v$backup ORDER BY file#;

     FILE# STATUS               CHANGE# TIME
---------- ------------------ --------- -------------------
         1 NOT ACTIVE                 0
         3 NOT ACTIVE                 0
         4 NOT ACTIVE                 0
         7 NOT ACTIVE                 0

-- 모두 NOT ACTIVE 다. 백업 모드에 들어간 파일이 없다.

SYS@orcl> SELECT checkpoint_change# FROM v$database;

CHECKPOINT_CHANGE#
------------------
           2342510

--------------------------------------------------------------------------------
[4] 작업 수행 : USERS 테이블스페이스 백업
--------------------------------------------------------------------------------
SYS@orcl> ALTER TABLESPACE users BEGIN BACKUP;

Tablespace altered.

SYS@orcl> SELECT file#, status, change#, time FROM v$backup WHERE file# = 7;

     FILE# STATUS               CHANGE# TIME
---------- ------------------ --------- -------------------
         7 ACTIVE               2342683 2025-05-09 13:22:14

-- STATUS 가 ACTIVE 로 바뀌었고 CHANGE# 에 동결된 SCN 이 기록되었다.

-- 백업 모드에서도 데이터 파일 쓰기가 계속되는지 확인한다.

SYS@orcl> SELECT file#, checkpoint_change# FROM v$datafile_header WHERE file# = 7;

     FILE# CHECKPOINT_CHANGE#
---------- ------------------
         7            2342683

SYS@orcl> !ls -l --time-style=+%H:%M:%S /u01/app/oracle/oradata/ORCL/users01.dbf
-rw-r-----. 1 oracle oinstall 5251072 13:22:14 /u01/app/oracle/oradata/ORCL/users01.dbf

-- 백업 모드에서 DML 을 수행한다.

SYS@orcl> UPDATE hr.employees SET salary = salary + 10 WHERE department_id = 50;
45 rows updated.

SYS@orcl> COMMIT;
Commit complete.

SYS@orcl> ALTER SYSTEM CHECKPOINT;
System altered.

SYS@orcl> !ls -l --time-style=+%H:%M:%S /u01/app/oracle/oradata/ORCL/users01.dbf
-rw-r-----. 1 oracle oinstall 5251072 13:24:07 /u01/app/oracle/oradata/ORCL/users01.dbf

-- 파일 갱신 시각이 바뀌었다. 데이터 파일 쓰기는 계속되고 있다.

SYS@orcl> SELECT file#, checkpoint_change# FROM v$datafile_header WHERE file# = 7;

     FILE# CHECKPOINT_CHANGE#
---------- ------------------
         7            2342683

-- 헤더의 SCN 은 그대로다. 동결된 것은 헤더뿐이라는 점이 확인된다.

SYS@orcl> !cp -pv /u01/app/oracle/oradata/ORCL/users01.dbf /home/oracle/backup/hot/20250509/
'/u01/app/oracle/oradata/ORCL/users01.dbf' -> '/home/oracle/backup/hot/20250509/users01.dbf'

SYS@orcl> ALTER TABLESPACE users END BACKUP;

Tablespace altered.

SYS@orcl> SELECT file#, status FROM v$backup WHERE file# = 7;

     FILE# STATUS
---------- ------------------
         7 NOT ACTIVE

--------------------------------------------------------------------------------
[5] 진행 로그 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 나머지 테이블스페이스도 같은 방식으로 처리하려다 이름을 틀린다.

SYS@orcl> ALTER TABLESPACE undotbs BEGIN BACKUP;
ALTER TABLESPACE undotbs BEGIN BACKUP
*
ERROR at line 1:
ORA-00959: tablespace 'UNDOTBS' does not exist

-- << 원인 파악 >>
-- 실제 이름은 UNDOTBS1 이다. UNDO 테이블스페이스는 생성 시 숫자가 붙는 경우가 많다.
-- 이름을 손으로 적지 말고 dba_tablespaces 에서 확인해 쓴다.

SYS@orcl> SELECT tablespace_name, contents, status FROM dba_tablespaces ORDER BY tablespace_name;

TABLESPACE_NAME CONTENTS  STATUS
--------------- --------- ---------
SYSAUX          PERMANENT ONLINE
SYSTEM          PERMANENT ONLINE
TEMP            TEMPORARY ONLINE
UNDOTBS1        UNDO      ONLINE
USERS           PERMANENT ONLINE

-- << 수정 >>

SYS@orcl> ALTER TABLESPACE undotbs1 BEGIN BACKUP;

Tablespace altered.

SYS@orcl> !cp -pv /u01/app/oracle/oradata/ORCL/undotbs01.dbf /home/oracle/backup/hot/20250509/

SYS@orcl> ALTER TABLESPACE undotbs1 END BACKUP;

Tablespace altered.

-- TEMP 는 TEMPORARY 이므로 백업 대상이 아니다.
-- BEGIN BACKUP 을 시도하면 오류가 난다.

SYS@orcl> ALTER TABLESPACE temp BEGIN BACKUP;
ALTER TABLESPACE temp BEGIN BACKUP
*
ERROR at line 1:
ORA-03217: invalid option for alter of TEMPORARY TABLESPACE

-- 임시 테이블스페이스는 복구 대상이 아니므로 백업 모드 자체가 없다.

--------------------------------------------------------------------------------

-- 모든 파일의 백업 모드가 해제되었다.

--------------------------------------------------------------------------------
[7] 검증
--------------------------------------------------------------------------------
-- 백업 구간의 리두를 아카이브로 확정한다.

SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
System altered.

SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO
  2    '/home/oracle/backup/hot/20250509/control.ctl' REUSE;
Database altered.

SYS@orcl> !ls -l /home/oracle/backup/hot/20250509/
total 1541632
-rw-r-----. 1 oracle oinstall  10600448 May  9 13:38 control.ctl
-rw-r-----. 1 oracle oinstall 587210752 May  9 13:35 sysaux01.dbf
-rw-r-----. 1 oracle oinstall 943726592 May  9 13:34 system01.dbf
-rw-r-----. 1 oracle oinstall 110108672 May  9 13:31 undotbs01.dbf
-rw-r-----. 1 oracle oinstall   5251072 May  9 13:25 users01.dbf

SYS@orcl> !ls /home/oracle/backup/hot/20250509/*.log 2>/dev/null | wc -l
0

-- 온라인 리두 로그가 없다. Hot Backup 에서는 의도적으로 제외한 것이다.

SYS@orcl> SELECT (SELECT COUNT(*) FROM v$datafile)                        AS datafiles,
  2         (SELECT COUNT(*) FROM v$backup WHERE status = 'ACTIVE')       AS still_in_backup,
  3         (SELECT MAX(sequence#) FROM v$archived_log WHERE dest_id = 1) AS last_seq
  4  FROM   dual;

 DATAFILES STILL_IN_BACKUP   LAST_SEQ
---------- --------------- ----------
         4               0         41

-- 데이터 파일 4개, 백업 모드 잔류 0, 아카이브 41까지 생성되었다.

--------------------------------------------------------------------------------
[8] 실무 포인트
--------------------------------------------------------------------------------
-- 1) BEGIN BACKUP 은 데이터 파일 쓰기를 멈추지 않는다.
--    파일 갱신 시각이 계속 바뀌는데도 헤더 SCN 은 동결된 것을 이 실습에서 직접 확인했다.
--    쓰기를 멈춘다면 무중단 백업이라는 목적 자체가 성립하지 않는다.
--
-- 2) 테이블스페이스 이름은 조회해서 쓴다.
--    UNDO 테이블스페이스는 UNDOTBS1 처럼 숫자가 붙는 경우가 많아 자주 틀린다.
--
-- 3) TEMP 는 백업 모드 대상이 아니다(ORA-03217).
--    임시 데이터만 담아 복구 대상이 아니기 때문이며, 유실 시 재생성으로 끝난다.
--
-- 4) 테이블스페이스 단위로 나눠 받으면 백업 모드 체류 시간이 짧아 리두 증가가 최소화된다.
--    ALTER DATABASE BEGIN BACKUP 은 간단하지만 복사 시간 내내 모든 파일이 영향을 받는다.
--
-- 5) 백업 디렉터리에 .log 파일이 없는지 반드시 확인한다.
--    Hot Backup 백업본에 온라인 리두 로그가 섞이면 복구 시 CURRENT 리두를 덮어쓸 위험이 있다.
--
-- 6) 이 백업본은 8장 이후 시나리오에서 사용한다. 아카이브와 함께 보관한다(실습 3-6).

SYS@orcl> EXIT
