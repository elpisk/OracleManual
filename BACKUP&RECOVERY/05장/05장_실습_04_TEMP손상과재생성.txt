================================================================================
 실습 5-4 : TEMP 파일 손상과 재생성
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / NOARCHIVELOG
 소요 시간 : 약 12분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- TEMP 파일이 유실되는 두 경우를 비교한다.
--   종료된 상태에서 유실 → 기동 시 Oracle이 자동 재생성
--   열린 상태에서 유실   → 정렬 작업만 실패, 수동으로 ADD 후 DROP
-- TEMP는 백업 대상이 아니며 복구가 아니라 재생성으로 끝난다.
-- 복구 유형 : 해당 없음(재생성)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 데이터베이스가 OPEN 상태여야 한다.
-- TEMP 파일은 백업본에 없어도 무방하다.

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> COL name FOR a48

[5] 작업 수행 (2) : 열린 상태에서 유실
--------------------------------------------------------------------------------
SYS@orcl> !rm -f /u01/app/oracle/oradata/ORCL/temp01.dbf

-- 작은 정렬은 PGA 안에서 끝나므로 성공한다.

SYS@orcl> SELECT * FROM (SELECT object_name FROM dba_objects ORDER BY object_name)
  2  WHERE ROWNUM <= 3;

OBJECT_NAME
------------------------------
ACCESS$
ALERT_QT
ALL_ALL_TABLES

-- 큰 정렬은 TEMP를 요구해 실패한다.

SYS@orcl> SELECT COUNT(*) FROM (
  2    SELECT object_name FROM dba_objects
  3    UNION
  4    SELECT object_name FROM dba_objects);
SELECT COUNT(*) FROM (
                     *
ERROR at line 1:
ORA-01116: error in opening database file 201
ORA-01110: data file 201: '/u01/app/oracle/oradata/ORCL/temp01.dbf'
ORA-27041: unable to open file

-- 파일 번호가 201이다. TEMP는 201번부터 시작하는 별도 번호 체계를 쓴다.

--------------------------------------------------------------------------------
[6] 진행 로그 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 없어진 파일을 먼저 제거해 본다.

SYS@orcl> ALTER DATABASE TEMPFILE '/u01/app/oracle/oradata/ORCL/temp01.dbf' DROP;

Database altered.

SYS@orcl> SELECT COUNT(*) FROM v$tempfile;

  COUNT(*)
----------
         0

-- 임시 테이블스페이스에 파일이 하나도 없는 상태가 되었다.

SYS@orcl> SELECT COUNT(*) FROM (
  2    SELECT object_name FROM dba_objects
  3    UNION
  4    SELECT object_name FROM dba_objects);
SELECT COUNT(*) FROM (
                     *
ERROR at line 1:
ORA-01652: unable to extend temp segment by 128 in tablespace TEMP

-- << 원인 파악 >>
-- 오류가 ORA-01116에서 ORA-01652로 바뀌었다.
-- 이제는 "파일을 열 수 없다"가 아니라 "확장할 공간이 없다"는 뜻이다.
-- 제거를 먼저 한 탓에 정렬에 쓸 파일이 하나도 없는 공백이 생겼다.
-- 추가를 먼저 하고 제거를 나중에 해야 이런 공백이 생기지 않는다.

-- << 수정 >>

SYS@orcl> ALTER TABLESPACE temp ADD TEMPFILE
  2    '/u01/app/oracle/oradata/ORCL/temp02.dbf' SIZE 60M;

Tablespace altered.

SYS@orcl> SELECT file#, name, status, bytes/1024/1024 AS mb FROM v$tempfile;

     FILE# NAME                                          STATUS      MB
---------- --------------------------------------------- ------- ------
         2 /u01/app/oracle/oradata/ORCL/temp02.dbf       ONLINE      60

SYS@orcl> SELECT COUNT(*) FROM (
  2    SELECT object_name FROM dba_objects
  3    UNION
  4    SELECT object_name FROM dba_objects);

  COUNT(*)
----------
     73204

-- 정렬이 정상 수행된다.

[7] 검증 : 사용량 점검과 테이블스페이스 교체
--------------------------------------------------------------------------------
SYS@orcl> SELECT tablespace_name, total_blocks, used_blocks, max_used_blocks
  2  FROM   v$sort_segment;

TABLESPACE_NAME TOTAL_BLOCKS USED_BLOCKS MAX_USED_BLOCKS
--------------- ------------ ----------- ---------------
TEMP                    7680           0            4096

-- MAX_USED_BLOCKS가 최대 사용 이력이다. 크기 산정의 근거가 된다.

SYS@orcl> SELECT s.sid, s.username, u.tablespace, u.blocks
  2  FROM   v$tempseg_usage u, v$session s WHERE u.session_addr = s.saddr;

no rows selected

-- 현재 TEMP를 쓰는 세션이 없다.

-- 임시 테이블스페이스 자체를 교체하는 방법도 확인한다.

SYS@orcl> CREATE TEMPORARY TABLESPACE temp2
  2    TEMPFILE '/u01/app/oracle/oradata/ORCL/temp03.dbf' SIZE 100M;
Tablespace created.

SYS@orcl> DROP TABLESPACE temp INCLUDING CONTENTS AND DATAFILES;
DROP TABLESPACE temp INCLUDING CONTENTS AND DATAFILES
*
ERROR at line 1:
ORA-12906: cannot drop default temporary tablespace

SYS@orcl> ALTER DATABASE DEFAULT TEMPORARY TABLESPACE temp2;
Database altered.

SYS@orcl> DROP TABLESPACE temp INCLUDING CONTENTS AND DATAFILES;
Tablespace dropped.

SYS@orcl> SELECT property_value FROM database_properties
  2  WHERE  property_name = 'DEFAULT_TEMP_TABLESPACE';

PROPERTY_VALUE
---------------
TEMP2

-- 실습을 원래대로 되돌리려면 TEMP를 다시 만들고 기본으로 전환한 뒤 TEMP2를 지운다.

--------------------------------------------------------------------------------
[8] 실무 포인트
--------------------------------------------------------------------------------
-- 1) 종료 중 유실과 열린 중 유실의 처리가 다르다.
--    전자는 STARTUP만 하면 자동 재생성되고, 후자는 수동 처리가 필요하다.
--    자동 재생성은 기동 시점에만 일어난다는 점이 이 차이의 이유다.
--
-- 2) 순서가 중요하다. ADD 먼저, DROP 나중이다.
--    반대로 하면 파일이 하나도 없는 공백이 생겨 ORA-01652가 발생한다.
--    오류 코드가 ORA-01116에서 ORA-01652로 바뀌는 것으로 이 차이를 확인할 수 있다.
--
-- 3) TEMP 파일 번호는 201부터 시작한다.
--    오류 메시지의 file 201을 보고 데이터 파일을 찾다가 헤매는 경우가 있다.
--
-- 4) 작은 정렬은 PGA 안에서 끝나 성공한다.
--    "큰 쿼리만 실패한다"는 신고가 들어오면 TEMP를 의심한다.
--
-- 5) 기본 임시 테이블스페이스는 DROP 할 수 없다(ORA-12906).
--    반드시 ALTER DATABASE DEFAULT TEMPORARY TABLESPACE로 전환한 뒤 지운다.
--    사용자별로 다른 임시 테이블스페이스가 지정되어 있으면 dba_users도 함께 확인한다.
--
-- 6) TEMP는 백업 대상이 아니며 복구 계획의 고려 대상도 아니다.
--    다만 조치가 늦으면 업무에 즉각 영향이 가므로 대응은 빨라야 한다.

SYS@orcl> EXIT
