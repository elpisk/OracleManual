================================================================================
 실습 11-3 : 운영 중 CURRENT 리두 로그 삭제
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 22분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 데이터베이스가 열린 상태에서 CURRENT 그룹의 멤버가 사라진다.
-- 멤버 하나가 남으면 서비스가 이어지고, 멤버 교체로 손실 없이 고친다.
-- 멤버가 모두 사라져도 Linux 에서는 LGWR 가 열어 둔 파일 핸들로 계속 쓴다.
-- 인스턴스가 살아 있는 동안 대응하면 아카이브 하나만 잃고 데이터는 지킨다.
-- 재기동해 버리면 실습 11-2 와 같은 불완전 복구가 된다.
-- 복구 유형 : 손실 회피(멤버 교체) 또는 아카이브 구간 손실(CLEAR UNARCHIVED)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 11-2를 마친 상태에서 이어서 진행한다.

[oracle@oel7v9 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> COL member FOR a44

[3] 초기 상태 확인
--------------------------------------------------------------------------------
SYS@orcl> CREATE TABLE hr.emp113(id NUMBER, memo VARCHAR2(30)) TABLESPACE users;
SYS@orcl> INSERT INTO hr.emp113 VALUES (1, 'running');
SYS@orcl> COMMIT;
Commit complete.

SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
System altered.

[4] 장애 유발 (1) : CURRENT 그룹의 멤버 하나 삭제
--------------------------------------------------------------------------------
SYS@orcl> !rm -f /u02/oradata/orcl/redo03b.log

SYS@orcl> INSERT INTO hr.emp113 VALUES (2, 'one member lost');
SYS@orcl> COMMIT;
Commit complete.

-- 서비스가 계속된다. 남은 멤버로 기록되기 때문이다.

SYS@orcl> SELECT group#, member, status FROM v$logfile WHERE group# = 3;

    GROUP# MEMBER                                       STATUS
---------- -------------------------------------------- -------
         3 /u01/app/oracle/oradata/orcl/redo03.log
         3 /u02/oradata/orcl/redo03b.log

-- v$logfile 은 여전히 비어 있다. 이 뷰는 파일 존재를 확인하지 않는다.

SYS@orcl> !grep -n -A3 'orcl_lg00' \
          /u01/app/oracle/diag/rdbms/orcl/orcl/trace/alert_orcl.log | tail -4
20702:Errors in file /u01/app/oracle/diag/rdbms/orcl/orcl/trace/orcl_lg00_31627.trc:
20703-ORA-27041: unable to open file
20704-Linux-x86_64 Error: 2: No such file or directory
20705-Additional information: 3

-- LGWR 의 작업 프로세스(lg00)가 그 멤버를 다시 열려다 실패한 기록만 남는다.
-- alert 로그 감시가 없으면 아무도 모른 채 지나간다.

-- << 올바른 처리 >> 스위치로 그룹을 비운 뒤 멤버를 교체한다.
-- 남은 멤버를 cp 로 복사해 넣는 것은 안 된다. LGWR 는 이미 지워진 원본의 핸들에
-- 계속 쓰고 있어 복사본은 곧 낡은 파일이 되고, 다음 기동 때 헤더가 어긋난다.

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
System altered.

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          1 YES INACTIVE
         3          2 YES ACTIVE
         4          3 NO  CURRENT

SYS@orcl> ALTER DATABASE DROP LOGFILE MEMBER '/u02/oradata/orcl/redo03b.log';

Database altered.

SYS@orcl> ALTER DATABASE ADD LOGFILE MEMBER '/u02/oradata/orcl/redo03b.log' TO GROUP 3;

Database altered.

SYS@orcl> SELECT group#, member, status FROM v$logfile WHERE group# = 3;

    GROUP# MEMBER                                       STATUS
---------- -------------------------------------------- -------
         3 /u01/app/oracle/oradata/orcl/redo03.log
         3 /u02/oradata/orcl/redo03b.log                INVALID

-- 새 멤버는 처음 쓰일 때까지 INVALID 로 표시된다. 다음 순환에서 정상이 된다.
-- 손실도 중단도 없었다.

--------------------------------------------------------------------------------
[5] 장애 유발 (2) : CURRENT 그룹의 멤버 전부 삭제
--------------------------------------------------------------------------------
SYS@orcl> !rm -f /u01/app/oracle/oradata/orcl/redo04.log /u02/oradata/orcl/redo04b.log

SYS@orcl> INSERT INTO hr.emp113 VALUES (3, 'all members lost');
SYS@orcl> COMMIT;
Commit complete.

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
System altered.

SYS@orcl> SELECT COUNT(*) FROM hr.emp113;

  COUNT(*)
----------
         3

SYS@orcl> SELECT status FROM v$instance;

STATUS
------------
OPEN

-- 멤버가 하나도 없는데 커밋도 스위치도 되고 인스턴스도 살아 있다.
-- Linux 는 열려 있는 파일을 지워도 핸들을 가진 프로세스가 닫을 때까지 내용을 유지한다.
-- LGWR 가 기동 때 열어 둔 핸들로 계속 쓰고 있는 것이다.

--------------------------------------------------------------------------------
[6] 진단
--------------------------------------------------------------------------------
SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          4 NO  CURRENT
         3          2 YES INACTIVE
         4          3 NO  ACTIVE

-- 스위치가 일어나 그룹 1이 CURRENT가 되었다.
-- 지운 그룹 4는 ACTIVE 이고 ARCHIVED가 NO다. 아카이버는 파일 이름으로 열기 때문에
-- 핸들이 없어 시퀀스 3을 아카이브하지 못한다.

SYS@orcl> !grep -n 'ORA-00313' \
          /u01/app/oracle/diag/rdbms/orcl/orcl/trace/alert_orcl.log | tail -1
20740:ORA-00313: open failed for members of log group 4 of thread 1

SYS@orcl> COL error FOR a40
SYS@orcl> SELECT dest_id, status, error FROM v$archive_dest WHERE dest_id = 1;

   DEST_ID STATUS    ERROR
---------- --------- ----------------------------------------
         1 VALID

-- 대상은 정상이다. 원본이 없는 것이다.

-- 판정
--   ① 멤버 : 그룹 4 전손 (핸들로만 살아 있음)
--   ② 그룹 상태 : ACTIVE, ARCHIVED NO → 아카이브 불가
--   ③ 인스턴스 : 열려 있음. 데이터는 버퍼와 데이터파일에 있다.
--   ④ 방치하면 : 순환이 그룹 4 차례에 오면 "cannot allocate new log, archival required" 로 멈춘다.
--                재기동하면 핸들이 사라져 실습 11-2 와 같은 불완전 복구가 된다.
--   → 지금 CLEAR UNARCHIVED 로 처리한다. 잃는 것은 시퀀스 3 의 아카이브뿐이다.

--------------------------------------------------------------------------------
[7] 복구 절차
--------------------------------------------------------------------------------
SYS@orcl> ALTER DATABASE CLEAR UNARCHIVED LOGFILE GROUP 4;

Database altered.

-- 열린 상태에서는 ACTIVE 그룹도 비울 수 있다. 체크포인트를 먼저 끝내고 비운다.
-- (MOUNT 에서 크래시 복구에 필요한 그룹이면 ORA-01624 로 거부된다. 11-2)

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          4 NO  CURRENT
         3          2 YES INACTIVE
         4          0 YES UNUSED

SYS@orcl> !ls -l /u01/app/oracle/oradata/orcl/redo04.log /u02/oradata/orcl/redo04b.log
-rw-r-----. 1 oracle dba 209715712 May 13 11:12 /u01/app/oracle/oradata/orcl/redo04.log
-rw-r-----. 1 oracle dba 209715712 May 13 11:12 /u02/oradata/orcl/redo04b.log

-- 멤버 파일이 새로 만들어졌다.

SYS@orcl> SELECT COUNT(*) FROM hr.emp113;

  COUNT(*)
----------
         3

-- 3번 행도 그대로다. 시퀀스 3 의 변경은 이미 데이터파일에 반영되었다.

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
System altered.

-- 교체한 멤버도 한 번 쓰인 뒤라 INVALID 가 사라졌다.

SYS@orcl> SELECT COUNT(*) FROM v$tempfile;

  COUNT(*)
----------
         1

-- 아카이브 사슬이 끊겼으므로 즉시 전체 백업을 받는다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> !rm -rf /home/oracle/backup/ch11; mkdir -p /home/oracle/backup/ch11
SYS@orcl> !cp -p /u01/app/oracle/oradata/orcl/*.dbf /home/oracle/backup/ch11/
SYS@orcl> !rm -f /home/oracle/backup/ch11/temp01.dbf
SYS@orcl> STARTUP
SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO
  2    '/home/oracle/backup/control_ch11.bkp' REUSE;

-- 실무 포인트
-- 1) 멤버 하나 손상은 서비스에 영향이 없다.
--    alert 로그에 lg00 의 ORA-27041 만 남고 남은 멤버로 기록이 계속된다.
--    스위치로 그룹을 비운 뒤 DROP/ADD LOGFILE MEMBER 로 교체하면 손실이 없다.
--    남은 멤버를 cp 로 복사해 넣는 것은 해결이 아니다.
--
-- 2) 그룹 전손도 Linux 에서는 인스턴스를 바로 죽이지 않는다.
--    LGWR 가 기동 때 연 핸들로 계속 쓰기 때문이다. 다만 아카이버는 이름으로 열므로
--    그 시퀀스를 아카이브하지 못하고, 순환이 그 그룹에 다시 오면 멈춘다.
--
-- 3) 살아 있는 동안이 기회다.
--    열린 상태의 CLEAR UNARCHIVED 는 그 시퀀스의 아카이브만 잃고 데이터는 지킨다.
--    재기동하면 핸들이 사라져 크래시 복구가 그 그룹을 찾고(ORA-00313),
--    비우지도 못해(ORA-01624) 실습 11-2 의 전체 복원·불완전 복구가 된다.
--
-- 4) alert 로그 감시가 그 기회를 만든다.
--    ORA-27041, ORA-00313 이 보이면 v$log 의 ARCHIVED 와 STATUS 를 바로 본다.
--
-- 5) 근본 대책은 멤버를 서로 다른 디스크에 두는 것이다.
--    이 실습에서는 두 멤버를 순차로 지웠지만 실제로는 디스크 하나가 사라지면
--    같은 디스크의 멤버가 동시에 없어진다.

SYS@orcl> EXIT
