================================================================================
 실습 11-1 : CURRENT 리두 로그 삭제 후 정상 종료
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 20분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 정상 종료 후 CURRENT 리두 로그 그룹의 멤버가 모두 유실되었다.
-- 종료 시 체크포인트가 완료되어 리두에만 있는 변경이 없다.
-- CLEAR UNARCHIVED 로 그룹을 다시 만들고 그대로 연다.
-- 데이터 손실도 RESETLOGS 도 없지만, 그 시퀀스의 아카이브가 빠져 백업 기준점이 바뀐다.
-- 복구 유형 : 복구 불필요(CLEAR UNARCHIVED)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- ARCHIVELOG 모드이며 실습 직전에 전체 백업을 새로 받는다.
-- 리두 로그는 그룹 세 개, 그룹마다 멤버 두 개다.

[oracle@oel7v9 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> COL member FOR a44

SYS@orcl> SELECT file#, checkpoint_change#, fuzzy FROM v$datafile_header ORDER BY file#;

     FILE# CHECKPOINT_CHANGE# FUZ
---------- ------------------ ---
         1            2431000 NO
         3            2431000 NO
         4            2431000 NO

-- 모든 파일이 같은 SCN이고 FUZZY가 NO다. 적용할 리두가 없다.

-- 판정
--   ① 멤버 : 전손
--   ② 그룹 상태 : CURRENT, ARCHIVED NO
--   ③ 종료 : 정상
--   ④ FUZZY : 전부 NO → 손실 없음
--   → CLEAR UNARCHIVED 후 OPEN

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 일반 CLEAR로 처리해 본다.

SYS@orcl> ALTER DATABASE CLEAR LOGFILE GROUP 1;
ALTER DATABASE CLEAR LOGFILE GROUP 1
*
ERROR at line 1:
ORA-00350: log 1 of instance orcl (thread 1) needs to be archived
ORA-00312: online log 1 thread 1: '/u01/app/oracle/oradata/orcl/redo01.log'
ORA-00312: online log 1 thread 1: '/u02/oradata/orcl/redo01b.log'

-- << 원인 파악 >>
-- ARCHIVED가 NO이므로 아직 아카이브되지 않은 그룹이다.
-- Oracle은 아카이브되지 않은 리두를 그냥 버리지 못하게 막는다.
-- 그런데 이 그룹은 파일이 이미 없어 아카이브할 원본조차 없다.
-- 아카이브 대상 문제가 아니라 파일 부재가 원인임을 먼저 확인해야 한다.
-- 확인 후에만 UNARCHIVED 옵션을 붙이는 것이 정당하다.

SYS@orcl> COL destination FOR a20
SYS@orcl> COL error FOR a40
SYS@orcl> SELECT dest_id, destination, status, error FROM v$archive_dest
  2  WHERE  destination IS NOT NULL;

   DEST_ID DESTINATION          STATUS    ERROR
---------- -------------------- --------- ----------------------------------------
         1 /home/oracle/arch1   VALID

-- 아카이브 대상은 정상이다. 파일이 없어서 아카이브할 수 없는 것이다.

SYS@orcl> ALTER SYSTEM ARCHIVE LOG ALL;
ALTER SYSTEM ARCHIVE LOG ALL
*
ERROR at line 1:
ORA-00271: there are no logs that need archiving

-- CURRENT 그룹은 ARCHIVE LOG ALL 의 대상이 아니다. 정상 종료로 스레드가 닫혀
-- "아카이브할 것이 없다" 고 답한다. 그 그룹의 리두는 이미 데이터파일에 다 반영되어 있다.

-- << 수정 >>

SYS@orcl> ALTER DATABASE CLEAR UNARCHIVED LOGFILE GROUP 1;

Database altered.

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log WHERE group# = 1;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          0 YES UNUSED

SYS@orcl> !ls -l /u01/app/oracle/oradata/orcl/redo01.log
-rw-r-----. 1 oracle dba 209715712 May 13 09:52 /u01/app/oracle/oradata/orcl/redo01.log

SYS@orcl> ALTER DATABASE OPEN;

Database altered.

-- RESETLOGS 없이 열렸다. 정상 종료로 스레드가 닫혀 있어 리두 연속성이 끊기지 않았고,
-- 비운 그룹은 다음 순서의 그룹이 대신한다.

--------------------------------------------------------------------------------
[8] 검증 및 실무 포인트
--------------------------------------------------------------------------------
SYS@orcl> SELECT * FROM v$recover_file;

no rows selected

SYS@orcl> SELECT id, memo FROM hr.emp111 ORDER BY id;

        ID MEMO
---------- ------------------------------
         1 before shutdown
         2 in current redo

-- CURRENT 리두에만 있던 2번 행까지 살아 있다. 손실이 없다.
-- 정상 종료 시 체크포인트로 데이터파일에 이미 반영되었기 때문이다.

SYS@orcl> SELECT incarnation#, resetlogs_change#, status FROM v$database_incarnation
  2  WHERE  status = 'CURRENT';

INCARNATION# RESETLOGS_CHANGE# STATUS
------------ ----------------- -------
           7           2428020 CURRENT

-- 인카네이션도 그대로다.

SYS@orcl> SELECT group#, sequence#, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# STATUS
---------- ---------- ----------------
         1          0 UNUSED
         3          8 CURRENT
         4          6 INACTIVE

-- 시퀀스 7 을 건너뛰고 8 이 그룹 3 에서 시작했다.

SYS@orcl> SELECT sequence#, status FROM v$archived_log
  2  WHERE  resetlogs_change# = (SELECT resetlogs_change# FROM v$database)
  3  AND    sequence# >= 5 ORDER BY sequence#;

 SEQUENCE# S
---------- -
         5 A
         6 A

-- 시퀀스 7 의 아카이브는 없다. 데이터는 잃지 않았지만 이 구간을 건너뛰는 복구는 불가능하다.
-- 실습 직전에 받은 백업본은 시퀀스 7 을 지나야 현재에 닿으므로 더 이상 쓸 수 없다.

SYS@orcl> SELECT COUNT(*) FROM v$tempfile;

  COUNT(*)
----------
         1

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
System altered.

-- 아카이브 사슬이 끊겼으므로 즉시 전체 백업을 받는다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> !rm -rf /home/oracle/backup/ch11; mkdir -p /home/oracle/backup/ch11
SYS@orcl> !cp -p /u01/app/oracle/oradata/orcl/*.dbf /home/oracle/backup/ch11/
SYS@orcl> !rm -f /home/oracle/backup/ch11/temp01.dbf
SYS@orcl> STARTUP
SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO
  2    '/home/oracle/backup/control_ch11.bkp' REUSE;

-- 실무 포인트
-- 1) 정상 종료 상태에서는 CURRENT 그룹을 잃어도 손실이 없다.
--    종료 시 체크포인트가 완료되어 리두에만 있는 변경이 남지 않기 때문이다.
--    v$datafile_header 의 FUZZY가 모두 NO인 것이 그 증거다.
--
-- 2) 그룹이 전손된 채 STARTUP 하면 OPEN 단계에서 인스턴스가 내려간다.
--    클라이언트에는 ORA-03113 만 보이고 ORA-00313 은 alert 로그에 있다.
--    STARTUP MOUNT 로 올린 뒤 처리한다.
--
-- 3) ORA-00350이 나오면 ARCHIVED가 NO인 이유를 먼저 확인한다.
--    아카이브 대상 문제로 밀린 것이면 대상을 고쳐 처리하면 손실이 없다.
--    파일이 이미 없어 아카이브할 수 없는 경우에만 UNARCHIVED가 정당하다.
--    이 실습에서는 v$archive_dest 가 VALID 였고 ARCHIVE LOG ALL 은 ORA-00271 로
--    "아카이브할 것이 없다" 고 했다. 파일 부재가 원인임을 확정한 뒤 UNARCHIVED를 붙였다.
--
-- 4) 정상 종료 뒤의 CLEAR UNARCHIVED 는 RESETLOGS 를 요구하지 않는다.
--    스레드가 닫혀 있어 리두 연속성이 끊기지 않기 때문이다. 인카네이션도 그대로다.
--
-- 5) 그래도 시퀀스 하나의 아카이브가 빠진다.
--    그 구간을 지나는 복구가 불가능해지므로 이전 백업본은 무효다. 즉시 전체 백업을 받는다.
--
-- 6) 이 실습이 손쉬웠던 이유는 종료 방식 하나 때문이다.
--    다음 실습에서 같은 장애를 비정상 종료 상태에서 다시 본다.

SYS@orcl> EXIT
