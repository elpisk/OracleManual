================================================================================
 실습 11-4 : CLEAR UNARCHIVED LOGFILE 을 이용한 복구
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 20분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 아카이브 대상 문제로 아카이브가 밀린 상태에서 리두 그룹이 손상되었다.
-- ARCHIVED가 NO인 이유를 확인하지 않고 UNARCHIVED를 붙이면 손실이 생긴다.
-- 대상을 먼저 고치면 손실 없이 처리할 수 있음을 확인한다.
-- CLEAR UNARCHIVED 의 효과와 대가를 아카이브 연속성으로 검증한다.
-- 복구 유형 : 손실 회피(대상 복구) 또는 리두 구간 손실

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 11-3을 마친 상태에서 이어서 진행한다.
-- 아카이브 대상은 /u02/arch1 이다.

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> COL destination FOR a20

SYS@orcl> COMMIT;

SYS@orcl> !rm -f /u01/app/oracle/oradata/ORCL/redo03.log /u02/oradata/ORCL/redo03b.log

[6] 진단
--------------------------------------------------------------------------------
SYS@orcl> SELECT dest_id, destination, status, error FROM v$archive_dest
  2  WHERE  destination IS NOT NULL;

   DEST_ID DESTINATION          STATUS  ERROR
---------- -------------------- ------- ----------------------------------
         1 /u02/arch1           ERROR   ORA-19504: failed to create file

SYS@orcl> !ls -ld /u02/arch1
dr-x------. 2 oracle oinstall 4096 May 13 11:40 /u02/arch1

-- 대상 디렉터리에 쓸 수 없다. 이것이 ARCHIVED NO의 원인이다.

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          6 NO  CURRENT
         3          4 NO  INACTIVE
         4          5 NO  INACTIVE

-- 판정
--   ① 손상 그룹 : 3 (INACTIVE, ARCHIVED NO)
--   ② ARCHIVED NO의 원인 : 대상 권한 문제(그룹 4·1은 파일이 멀쩡)
--   ③ 그룹 3은 파일이 없어 아카이브할 원본이 없다
--   → 대상을 먼저 고쳐 나머지를 아카이브하고, 그룹 3만 UNARCHIVED로 처리

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 확인 없이 세 그룹을 모두 UNARCHIVED로 처리하려 한다(잘못된 접근).

SYS@orcl> ALTER DATABASE CLEAR UNARCHIVED LOGFILE GROUP 4;

Database altered.

SYS@orcl> SELECT MIN(sequence#) AS oldest, MAX(sequence#) AS newest, COUNT(*) AS cnt,
  2         MAX(sequence#) - MIN(sequence#) + 1 AS expected
  3  FROM   v$archived_log WHERE status = 'A';

    OLDEST     NEWEST        CNT   EXPECTED
---------- ---------- ---------- ----------
         1          3          3          3

-- 시퀀스 5의 아카이브가 영원히 만들어지지 않게 되었다.

-- << 원인 파악 >>
-- 그룹 4는 파일이 멀쩡했다. 아카이브가 밀린 것은 대상 권한 문제였다.
-- 권한을 고치고 ARCHIVE LOG ALL 만 했으면 손실 없이 아카이브되었을 것이다.
-- UNARCHIVED로 비운 탓에 시퀀스 5 구간의 리두를 영구히 잃었다.
-- ARCHIVED가 NO인 이유를 먼저 확인해야 하는 이유가 이것이다.
-- 명령이 성공한다는 것과 그것이 옳은 선택이라는 것은 다른 문제다.

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          6 NO  CURRENT
         3          4 NO  INACTIVE
         4          0 YES UNUSED

-- << 수정 >>
-- 남은 그룹은 대상을 고쳐 정상 처리한다.

SYS@orcl> !chmod 700 /u02/arch1

SYS@orcl> ALTER SYSTEM ARCHIVE LOG ALL;
ALTER SYSTEM ARCHIVE LOG ALL
*
ERROR at line 1:
ORA-16038: log 3 sequence# 4 cannot be archived
ORA-00312: online log 3 thread 1: '/u01/app/oracle/oradata/ORCL/redo03.log'

-- 그룹 3은 파일이 없어 여전히 아카이브할 수 없다.

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          6 YES CURRENT
         3          4 NO  INACTIVE
         4          0 YES UNUSED

-- 그룹 1은 아카이브되었다. 대상을 고친 효과다.

SYS@orcl> ALTER DATABASE CLEAR UNARCHIVED LOGFILE GROUP 3;

Database altered.

-- 그룹 3만 UNARCHIVED로 처리했다. 파일이 없어 다른 방법이 없었다.

SYS@orcl> SELECT group#, sequence#, archived, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# ARC STATUS
---------- ---------- --- ----------------
         1          6 YES CURRENT
         3          0 YES UNUSED
         4          0 YES UNUSED

--------------------------------------------------------------------------------
[8] 검증 및 실무 포인트
--------------------------------------------------------------------------------
SYS@orcl> SELECT MIN(sequence#) AS oldest, MAX(sequence#) AS newest, COUNT(*) AS cnt,
  2         MAX(sequence#) - MIN(sequence#) + 1 AS expected
  3  FROM   v$archived_log WHERE status = 'A';

    OLDEST     NEWEST        CNT   EXPECTED
---------- ---------- ---------- ----------
         1          6          4          6

-- 개수 4, 기대값 6이다. 두 구간이 빠졌다.

SYS@orcl> SELECT sequence# FROM v$archived_log WHERE status = 'A' ORDER BY sequence#;

 SEQUENCE#
----------
         1
         2
         3
         6

-- 시퀀스 4와 5가 없다.
--   4 : 파일이 없어 불가피하게 버린 구간
--   5 : 확인 없이 UNARCHIVED로 처리해 불필요하게 버린 구간

SYS@orcl> SELECT COUNT(*) FROM hr.emp114;

  COUNT(*)
----------
         3

-- 데이터는 그대로다. 커밋된 변경은 이미 데이터파일에 반영되었다.
-- 리두를 버리는 것은 미래의 복구 수단을 잃는 것이지 현재 데이터를 잃는 것이 아니다.

SYS@orcl> SELECT dest_id, status, error FROM v$archive_dest WHERE dest_id = 1;

   DEST_ID STATUS  ERROR
---------- ------- --------------------
         1 VALID

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
System altered.

-- 버린 구간이 있으므로 즉시 전체 백업을 받는다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> !rm -rf /home/oracle/backup/ch11 && mkdir -p /home/oracle/backup/ch11
SYS@orcl> !cp -p /u02/oradata/ORCL/*.dbf /home/oracle/backup/ch11/
SYS@orcl> !rm -f /home/oracle/backup/ch11/temp01.dbf
SYS@orcl> STARTUP

SYS@orcl> SELECT TO_CHAR(SYSDATE, 'YYYY-MM-DD HH24:MI:SS') AS new_baseline FROM dual;

NEW_BASELINE
-------------------
2025-05-13 11:58:20

-- 이 백업이 새로운 복구 기준점이다. 이전 백업본은 무효다.

-- 실무 포인트
-- 1) ARCHIVED가 NO인 이유를 먼저 확인한다.
--    대상 문제로 밀린 것이면 대상을 고쳐 처리하면 손실이 없다.
--    파일이 이미 없어 아카이브할 수 없는 경우에만 UNARCHIVED가 정당하다.
--
-- 2) 확인 없이 UNARCHIVED를 붙이면 불필요한 손실이 생긴다.
--    이 실습에서 그룹 4는 파일이 멀쩡했는데도 리두를 버렸다.
--    권한 한 줄만 고쳤으면 정상 아카이브되었을 구간이다.
--
-- 3) 확인 순서는 세 가지다.
--    v$archive_dest 의 STATUS와 ERROR, OS 레벨의 권한과 공간, 그리고 파일 존재 여부다.
--    ARCHIVE LOG ALL 을 시도해 어느 그룹이 실패하는지 보는 것도 방법이다.
--
-- 4) 버린 구간은 연속성 점검에서 드러난다.
--    개수와 기대값이 달라지며 그 시점 이전의 백업본이 무효가 된다.
--    작업 기록에 남기지 않으면 나중에 원인을 오판한다.
--
-- 5) 리두를 버려도 현재 데이터는 사라지지 않는다.
--    커밋된 변경은 이미 데이터파일에 반영되어 있다.
--    잃는 것은 그 구간을 재현할 수 있는 미래의 복구 수단이다.
--
-- 6) CLEAR UNARCHIVED 직후 전체 백업이 필수다.
--    새 백업이 새로운 복구 기준점이 되며 이전 백업본은 라벨을 붙여 구분한다.

SYS@orcl> EXIT
