================================================================================
 실습 18-8 : Flashback Database와 Guaranteed Restore Point
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 40분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 배포 작업 전에 되돌릴 준비를 한다.
-- FRA를 구성하고 FLASHBACK ON으로 로그 생성을 켠다.
-- 보장 복원 지점을 만들고 배포 실패를 가정해 되돌린다.
-- 되돌린 뒤 다시 앞으로 진행하는 절차도 확인한다.
-- 복구 유형 : 데이터베이스 전체 되돌리기, 전면 중단

--------------------------------------------------------------------------------

[4] 작업 수행 : FRA 구성과 FLASHBACK ON
--------------------------------------------------------------------------------
-- FRA 없이 FLASHBACK ON을 시도해 본다.

SYS@orcl> ALTER DATABASE FLASHBACK ON;

ALTER DATABASE FLASHBACK ON
*
ERROR at line 1:
ORA-38706: Cannot turn on FLASHBACK DATABASE logging.
ORA-38709: Recovery Area is not enabled.

-- << 원인 파악 >>
-- Flashback 로그는 FRA에만 저장된다.
-- FRA가 구성되어 있지 않으면 로그를 만들 곳이 없다.
-- db_recovery_file_dest 와 크기를 먼저 설정해야 한다.

-- << 수정 >>

SYS@orcl> ALTER SYSTEM SET db_recovery_file_dest_size = 8G SCOPE = BOTH;

System altered.

SYS@orcl> ALTER SYSTEM SET db_recovery_file_dest = '/u03/fra' SCOPE = BOTH;

System altered.

SYS@orcl> ALTER SYSTEM SET db_flashback_retention_target = 2880 SCOPE = BOTH;

System altered.

-- 보존 목표를 48시간으로 잡았다.

SYS@orcl> ALTER DATABASE FLASHBACK ON;

Database altered.

SYS@orcl> SELECT flashback_on FROM v$database;

FLASHBACK_ON
------------------
YES

[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 열린 상태에서 되돌리기를 시도해 본다.

SYS@orcl> FLASHBACK DATABASE TO RESTORE POINT before_deploy;

FLASHBACK DATABASE TO RESTORE POINT before_deploy
*
ERROR at line 1:
ORA-38757: Database must be mounted and not open to FLASHBACK.

-- << 원인 파악 >>
-- Flashback Database는 데이터베이스 전체를 되감으므로
-- 열린 상태에서는 수행할 수 없다.
-- MOUNT 상태로 내려야 한다. 10장의 불완전 복구와 같은 조건이다.

-- << 수정 >>

SYS@orcl> SHUTDOWN IMMEDIATE

Database closed.
Database dismounted.
ORACLE instance shut down.

SYS@orcl> STARTUP MOUNT

ORACLE instance started.
Database mounted.

SYS@orcl> FLASHBACK DATABASE TO RESTORE POINT before_deploy;

Flashback complete.

SYS@orcl> ALTER DATABASE OPEN READ ONLY;

Database altered.

SYS@orcl> SELECT COUNT(*) FROM hr.orders188;

  COUNT(*)
----------
      1000

SYS@orcl> SELECT memo FROM hr.orders188 WHERE id = 1;

MEMO
------------------------------
order-1

SYS@orcl> SELECT COUNT(*) FROM dba_tables
  2  WHERE  owner = 'HR' AND table_name = 'NEW_FEATURE188';

  COUNT(*)
----------
         0

-- 배포 전 상태로 돌아왔다. 새로 만든 테이블도 사라졌다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> STARTUP MOUNT
SYS@orcl> ALTER DATABASE OPEN RESETLOGS;

Database altered.

--------------------------------------------------------------------------------
SYS@orcl> SELECT COUNT(*) FROM hr.orders188;

-- RESETLOGS로 열었으므로 인카네이션이 올라갔다.

-- 복원 지점을 지우지 않으면 어떻게 되는지 확인한다.

SYS@orcl> SELECT name, ROUND(storage_size/1024/1024) AS mb FROM v$restore_point;

-- 로그가 계속 쌓여 640MB가 되었다.

  2  FROM   v$flash_recovery_area_usage WHERE percent_space_used > 0;

------------------ ------------------ ---------------
ARCHIVED LOG                     12.4              18
FLASHBACK LOG                    18.2              12

SYS@orcl> DROP RESTORE POINT before_deploy;

Restore point dropped.

SYS@orcl> SELECT file_type, percent_space_used FROM v$flash_recovery_area_usage
  2  WHERE  file_type = 'FLASHBACK LOG';

FILE_TYPE          PERCENT_SPACE_USED
------------------ ------------------
FLASHBACK LOG                     4.1

-- 복원 지점을 지우자 보존 의무가 사라져 로그가 정리되었다.

-- 되돌린 뒤 앞으로 가는 절차도 확인한다.

SYS@orcl> CREATE RESTORE POINT rp_test GUARANTEE FLASHBACK DATABASE;
SYS@orcl> SELECT current_scn FROM v$database;

CURRENT_SCN
-----------
    2821800

SYS@orcl> UPDATE hr.orders188 SET memo = 'v2-' || id WHERE id <= 100;
SYS@orcl> COMMIT;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;

System altered.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> STARTUP MOUNT
SYS@orcl> FLASHBACK DATABASE TO RESTORE POINT rp_test;

Flashback complete.

SYS@orcl> ALTER DATABASE OPEN READ ONLY;
SYS@orcl> SELECT memo FROM hr.orders188 WHERE id = 1;

MEMO
------------------------------
order-1

-- 너무 많이 되돌렸다고 판단하고 앞으로 진행한다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> STARTUP MOUNT
SYS@orcl> RECOVER DATABASE UNTIL SCN 2822400;

Media recovery complete.

SYS@orcl> ALTER DATABASE OPEN RESETLOGS;

Database altered.

SYS@orcl> SELECT memo FROM hr.orders188 WHERE id = 1;

MEMO
------------------------------
v2-1

-- 아카이브를 적용해 앞으로 진행할 수 있었다.
-- RESETLOGS 전이라면 이 유연성이 있다.

SYS@orcl> DROP RESTORE POINT rp_test;
SYS@orcl> DROP TABLE hr.orders188 PURGE;

Table dropped.

SYS@orcl> SELECT name, space_limit/1024/1024/1024 AS limit_gb,
  2         ROUND(space_used/space_limit*100, 1) AS pct
  3  FROM   v$recovery_file_dest;

NAME       LIMIT_GB    PCT
---------- -------- ------
/u03/fra        8.0   16.5

-- 실무 포인트
-- 1) Flashback Database는 FRA가 전제다.
--    FRA 없이 FLASHBACK ON을 시도하면 ORA-38709가 나온다.
--    로그를 저장할 곳이 없기 때문이다.
--
-- 2) 설정 이전 시점으로는 되돌릴 수 없다.
--    oldest_flashback_time 이 그 한계이며 설정 시점부터 로그가 쌓인다.
--    장애가 난 뒤에 켜는 것은 의미가 없다.
--
-- 3) MOUNT 상태에서만 수행할 수 있다.
--    열린 상태에서는 ORA-38757로 거부된다.
--    되돌린 뒤 RESETLOGS로 열어야 하는 것도 불완전 복구와 같다.
--
-- 4) READ ONLY로 확인한 뒤 RESETLOGS로 여는 절차를 지킨다.
--    확인 없이 RESETLOGS를 수행하면 되돌리기가 번거로워진다.
--
-- 5) 너무 많이 되돌렸다면 앞으로 갈 수 있다.
--    RESETLOGS 전이라면 RECOVER DATABASE UNTIL 로 진행한다.
--    아카이브가 있어야 가능하며 이 유연성이 큰 이점이다.
--
-- 6) 보장 복원 지점은 반드시 지운다.
--    지우지 않으면 로그가 계속 쌓여 FRA가 가득 찬다.
--    이 실습에서도 640MB까지 늘었다가 삭제 후 정리되었다.
--    배포 절차서에 삭제 단계를 명시해 둔다.
