================================================================================
 실습 10-3 : SCN-based 불완전 복구
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL / ARCHIVELOG
 소요 시간 : 약 25분
================================================================================

--------------------------------------------------------------------------------
[1] 시나리오 개요
--------------------------------------------------------------------------------
-- 같은 시각에 여러 작업이 몰려 시각만으로는 복구 지점을 특정할 수 없다.
-- LogMiner로 리두를 분석해 문제 DML의 정확한 SCN을 찾는다.
-- UNTIL CHANGE 로 그 SCN 직전까지 복구한다.
-- 네 기준 중 가장 정밀한 방식이다.
-- 복구 유형 : 불완전 복구(SCN-based)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 실습 10-2를 마친 상태에서 이어서 진행한다. 10-2 끝에 받은 전체 백업을 쓴다.
-- LogMiner 사용 권한(SYSDBA)이 필요하다.
-- LogMiner 가 리두를 해독하려면 최소 보충 로깅이 켜져 있어야 한다.

[oracle@oel7v9 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> ALTER SESSION SET nls_date_format = 'YYYY-MM-DD HH24:MI:SS';
Session altered.

SYS@orcl> SELECT supplemental_log_data_min FROM v$database;

SUPPLEME
--------
NO

SYS@orcl> ALTER DATABASE ADD SUPPLEMENTAL LOG DATA;

Database altered.

SYS@orcl> SELECT supplemental_log_data_min FROM v$database;

SUPPLEME
--------
YES

-- 꺼진 채로 만든 리두는 LogMiner 가 온전히 읽지 못한다. 작은 트랜잭션은
-- In-Memory Undo 방식으로 기록되어 v$logmnr_contents 에서 통째로 빠지고,
-- 큰 DML 도 일부만 보인다. 사후에 켜도 이미 만들어진 리두에는 소용이 없으므로
-- 분석할 리두가 생기기 전에 켜 둔다.

[6] 진단 : LogMiner로 SCN 확보
--------------------------------------------------------------------------------
SYS@orcl> COL name FOR a44
SYS@orcl> SELECT sequence#, name FROM v$archived_log
  2  WHERE  resetlogs_change# = (SELECT resetlogs_change# FROM v$database)
  3  ORDER BY sequence# DESC FETCH FIRST 2 ROWS ONLY;

 SEQUENCE# NAME
---------- --------------------------------------------
         2 /home/oracle/arch1/arch_1_2_1200548120.arc
         1 /home/oracle/arch1/arch_1_1_1200548120.arc

-- 10-2 의 RESETLOGS 로 시퀀스가 1부터 다시 시작했다. 이전 인카네이션의 아카이브가
-- 섞이지 않도록 현재 인카네이션(resetlogs_change#)으로 거른다.
-- [4] 의 작업은 시퀀스 2 에 있다.

SYS@orcl> BEGIN
  2    DBMS_LOGMNR.ADD_LOGFILE('/home/oracle/arch1/arch_1_2_1200548120.arc', DBMS_LOGMNR.NEW);
  3  END;
  4  /
PL/SQL procedure successfully completed.

-- 분석을 시작하지 않고 조회하면 실패한다.

SYS@orcl> SELECT COUNT(*) FROM v$logmnr_contents;
SELECT COUNT(*) FROM v$logmnr_contents
                     *
ERROR at line 1:
ORA-01306: dbms_logmnr.start_logmnr() must be invoked before selecting from v$logmnr_contents

-- 파일 등록과 분석 시작은 별개의 단계다.

SYS@orcl> EXEC DBMS_LOGMNR.START_LOGMNR( -
>   OPTIONS => DBMS_LOGMNR.DICT_FROM_ONLINE_CATALOG);
PL/SQL procedure successfully completed.

SYS@orcl> COL seg_owner FOR a9
SYS@orcl> COL seg_name FOR a10
SYS@orcl> COL operation FOR a10
SYS@orcl> SELECT scn, TO_CHAR(timestamp, 'HH24:MI:SS') AS t,
  2         seg_owner, seg_name, operation
  3  FROM   v$logmnr_contents
  4  WHERE  seg_owner = 'HR' AND operation IN ('INSERT','DELETE')
  5  ORDER  BY scn FETCH FIRST 5 ROWS ONLY;

       SCN T        SEG_OWNER SEG_NAME   OPERATION
---------- -------- --------- ---------- ----------
   2423412 11:02:18 HR        INSA_2025  INSERT
   2423417 11:02:18 HR        EMP103     DELETE
   2423417 11:02:18 HR        EMP103     DELETE
   2423417 11:02:18 HR        EMP103     DELETE
   2423417 11:02:18 HR        EMP103     DELETE

-- DELETE 는 지운 행마다 한 줄씩 나온다. 한 문장이라 SCN 은 모두 같다.

SYS@orcl> SELECT seg_name, operation, COUNT(*) AS cnt, MIN(scn) AS min_scn, MAX(scn) AS max_scn
  2  FROM   v$logmnr_contents
  3  WHERE  seg_owner = 'HR' AND operation IN ('INSERT','DELETE')
  4  GROUP  BY seg_name, operation ORDER BY min_scn;

SEG_NAME   OPERATION         CNT    MIN_SCN    MAX_SCN
---------- ---------- ---------- ---------- ----------
INSA_2025  INSERT              2    2423412    2423420
EMP103     DELETE             50    2423417    2423417

-- 같은 시각이지만 SCN이 다르다.
-- 정상 INSERT(20번)는 2423412, 잘못된 DELETE는 2423417, 21번 INSERT 는 2423420 이다.

SYS@orcl> SELECT MIN(scn) AS delete_start FROM v$logmnr_contents
  2  WHERE  seg_owner = 'HR' AND seg_name = 'EMP103' AND operation = 'DELETE';

DELETE_START
------------
     2423417

SYS@orcl> EXEC DBMS_LOGMNR.END_LOGMNR;
PL/SQL procedure successfully completed.

-- 시각을 SCN으로 환산하면 어디를 가리키는지도 본다.

SYS@orcl> SELECT TIMESTAMP_TO_SCN(
  2    TO_TIMESTAMP('2025-05-12 11:02:18', 'YYYY-MM-DD HH24:MI:SS')) AS scn FROM dual;

       SCN
----------
   2423412

-- 그 초의 첫 SCN, 즉 20번 INSERT 의 SCN 이다. 초 안에서 DELETE 를 구분하지 못한다.
-- 이 환산은 복구 전에 해 둔다. RESETLOGS 뒤에는 SCN-시각 매핑 표가 백업 시점으로
-- 돌아가 엉뚱한 값을 낸다.

-- 판정
--   ① 목표 : SCN 2423417 직전
--   ② 시각 기준으로는 20번 INSERT까지 사라지므로 부적합
--   ③ UNTIL CHANGE 2423417 로 지정하면 그 직전까지 적용된다.

--------------------------------------------------------------------------------
[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 먼저 시각 기준으로 시도해 차이를 확인한다.
-- READ ONLY 로 검증할 것이므로 처음부터 USING BACKUP CONTROLFILE 을 붙인다(10-2).

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> !cp -p /home/oracle/backup/ch10/*.dbf /u01/app/oracle/oradata/orcl/
SYS@orcl> STARTUP MOUNT

SYS@orcl> SET AUTORECOVERY ON
SYS@orcl> RECOVER DATABASE UNTIL TIME '2025-05-12 11:02:18' USING BACKUP CONTROLFILE;

no rows selected

-- << 원인 파악 >>
-- DELETE는 되돌렸지만 정상 INSERT였던 20번도 함께 사라졌다.
-- UNTIL TIME 은 지정한 시각의 첫 리두 레코드 직전에서 멈춘다.
-- 11:02:18 의 첫 레코드가 20번 INSERT(2423412)라 거기서 멈춘 것이다.
-- 초 단위까지만 구분하므로 같은 초의 작업을 나눌 수 없다.
-- 초 단위보다 정밀한 기준이 필요하며 그것이 SCN이다.
-- (READ ONLY 에서 SCN_TO_TIMESTAMP 는 ORA-08181 로 실패한다. 멈춘 지점은 헤더로 본다.)

-- << 수정 >>
-- SCN 기준으로 다시 지정한다. 멈춘 2423412 가 목표 2423417 보다 앞이므로
-- 이어서 적용하면 되고 복원은 필요 없다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> STARTUP MOUNT

SYS@orcl> SET AUTORECOVERY ON
SYS@orcl> RECOVER DATABASE UNTIL CHANGE 2423417 USING BACKUP CONTROLFILE;
ORA-00279: change 2423412 generated at 05/12/2025 11:02:18 needed for thread 1
ORA-00289: suggestion : /home/oracle/arch1/arch_1_2_1200548120.arc
ORA-00280: change 2423412 for thread 1 is in sequence #2


Log applied.
Media recovery complete.

SYS@orcl> SELECT checkpoint_change#, TO_CHAR(checkpoint_time, 'HH24:MI:SS') AS stopped_at
  2  FROM   v$datafile_header WHERE file# = 1;

CHECKPOINT_CHANGE# STOPPED_AT
------------------ ----------
           2423417 11:02:18

-- 지정한 SCN 이 그대로 헤더에 찍혔다. 그 SCN 의 변경(DELETE)은 적용되지 않았다.

SYS@orcl> ALTER DATABASE OPEN READ ONLY;

Database altered.

SYS@orcl> SELECT COUNT(*) FROM hr.emp103;

  COUNT(*)
----------
       107

SYS@orcl> SELECT id, TO_CHAR(order_date, 'HH24:MI:SS') AS t FROM hr.insa_2025
  2  WHERE  id >= 20 ORDER BY id;

        ID T
---------- --------
        20 11:02:18

-- 정상 INSERT는 살아 있고 잘못된 DELETE만 사라졌다.

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> STARTUP MOUNT
SYS@orcl> ALTER DATABASE OPEN RESETLOGS;

-- 템프파일은 그대로다. 컨트롤파일을 되돌리거나 다시 만들지 않았기 때문이다.
-- RESETLOGS 오픈은 템프파일 기록을 지우지 않는다.
-- 여기서 ADD TEMPFILE 을 하면 ORA-01537(file already part of database)로 거부된다.
-- 컨트롤파일을 재생성한 경우에만 템프파일이 사라지며 그때 다시 만든다(11-6, 11-7).

SYS@orcl> SHUTDOWN IMMEDIATE
SYS@orcl> !rm -rf /home/oracle/backup/ch10; mkdir -p /home/oracle/backup/ch10
SYS@orcl> !cp -p /u01/app/oracle/oradata/orcl/*.dbf /home/oracle/backup/ch10/
SYS@orcl> !rm -f /home/oracle/backup/ch10/temp01.dbf
SYS@orcl> STARTUP
SYS@orcl> ALTER DATABASE BACKUP CONTROLFILE TO
  2    '/home/oracle/backup/control_ch10.bkp' REUSE;

-- 실무 포인트
-- 1) 시각 기준은 초 단위까지만 구분한다.
--    지정한 초의 첫 레코드 직전에서 멈추므로 같은 초의 작업은 모두 빠진다.
--    이 실습의 첫 시도가 그 경우다.
--
-- 2) SCN 기준은 트랜잭션 단위로 정밀하다.
--    UNTIL CHANGE 는 지정한 SCN 직전까지 적용하므로
--    되돌릴 DML의 SCN을 그대로 지정하면 그 직전 상태가 된다.
--
-- 3) LogMiner로 SCN을 확보한다.
--    ADD_LOGFILE 로 아카이브를 등록하고 START_LOGMNR 로 분석을 시작한다.
--    v$logmnr_contents 에서 소유자·객체·작업 유형으로 필터링한다.
--    최소 보충 로깅이 꺼져 있던 구간의 리두는 온전히 읽히지 않는다. 미리 켜 둔다.
--
-- 4) LogMiner는 복구 전에 수행해야 한다.
--    복구 후에는 그 리두가 이미 적용되었거나 RESETLOGS로 끊긴 상태다.
--    END_LOGMNR 로 세션을 정리하는 것도 잊지 않는다.
--
-- 5) TIMESTAMP_TO_SCN 은 그 초의 첫 SCN을 돌려준다.
--    초 안의 위치를 구분하지 못하므로 정밀한 지점에는 LogMiner가 필요하다.
--    RESETLOGS 뒤에는 매핑 표가 어긋나므로 환산도 복구 전에 한다.
--
-- 6) 사고 이후의 작업은 어느 기준을 쓰든 사라진다.
--    이 실습의 21번 INSERT가 그렇다. 손실 구간을 정확히 통보해야 한다.

SYS@orcl> EXIT
