================================================================================
 실습 1-3 : 체크포인트와 SCN 흐름 추적
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL
 소요 시간 : 약 12분
================================================================================

--------------------------------------------------------------------------------
[1] 실습 개요
--------------------------------------------------------------------------------
-- "커밋은 리두 기록만 보장한다"는 명제를 SCN 값의 변화로 직접 확인한다.
-- 커밋만으로는 체크포인트 SCN 이 변하지 않고, 로그 스위치가 있어야 갱신된다는 점,
-- 그리고 제어 파일과 데이터 파일 헤더의 SCN 이 어떻게 맞춰지는지를 관찰한다.
-- 복구 유형 : 해당 없음(확인 실습)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 데이터베이스가 OPEN 상태여야 한다.
-- HR 스키마가 설치되어 있어야 한다.
-- 이 실습은 데이터를 변경하지만 되돌릴 수 있는 범위이며 파일을 손상시키지 않는다.

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> SET PAGESIZE 100
SYS@orcl> COL name FOR a48

--------------------------------------------------------------------------------
[3] 초기 상태 확인
--------------------------------------------------------------------------------
-- 제어 파일이 기억하는 전체 체크포인트 SCN

SYS@orcl> SELECT checkpoint_change# FROM v$database;

CHECKPOINT_CHANGE#
------------------
           2097189

-- 각 데이터 파일 헤더의 체크포인트 SCN

SYS@orcl> SELECT file#, name, checkpoint_change#, status
  2  FROM   v$datafile ORDER BY file#;

     FILE# NAME                                             CHECKPOINT_CHANGE# STATUS
---------- ------------------------------------------------ ------------------ -------
         1 /u01/app/oracle/oradata/ORCL/system01.dbf                   2097189 SYSTEM
         3 /u01/app/oracle/oradata/ORCL/sysaux01.dbf                   2097189 ONLINE
         4 /u01/app/oracle/oradata/ORCL/undotbs01.dbf                  2097189 ONLINE
         7 /u01/app/oracle/oradata/ORCL/users01.dbf                    2097189 ONLINE

-- 모든 값이 2097189 로 같다. 일관성 있는(consistent) 상태다.
-- 이 상태라면 복원만으로 복구가 끝난다.

-- SCN 을 대략적인 시각으로 환산

SYS@orcl> SELECT SCN_TO_TIMESTAMP(2097189) AS ts FROM dual;

TS
---------------------------------------------------------
02-MAY-25 02.08.31.000000000 AM

--------------------------------------------------------------------------------
[4] 작업 수행 : 커밋이 체크포인트 SCN 에 미치는 영향
--------------------------------------------------------------------------------
SYS@orcl> SELECT salary FROM hr.employees WHERE employee_id = 100;

    SALARY
----------
     24000

SYS@orcl> UPDATE hr.employees SET salary = salary * 1.1 WHERE employee_id = 100;
1 row updated.

SYS@orcl> COMMIT;
Commit complete.

SYS@orcl> SELECT salary FROM hr.employees WHERE employee_id = 100;

    SALARY
----------
     26400

-- 커밋이 끝났다. 이제 체크포인트 SCN 을 다시 본다.

SYS@orcl> SELECT checkpoint_change# FROM v$database;

CHECKPOINT_CHANGE#
------------------
           2097189

-- 값이 변하지 않았다.
-- 커밋은 LGWR 가 리두를 기록했다는 것만 보장하며,
-- 변경 블록이 데이터 파일에 반영되었다는 뜻이 아니다.

-- 현재 SCN 은 이미 올라가 있다. 커밋 자체는 SCN 을 증가시킨다.

SYS@orcl> SELECT current_scn FROM v$database;

CURRENT_SCN
-----------
    2098102

-- CURRENT_SCN(2098102) 과 CHECKPOINT_CHANGE#(2097189) 의 차이가
-- "리두에는 있으나 데이터 파일에는 아직 없는" 구간이다.
-- 인스턴스가 지금 죽으면 이 구간이 롤포워드 대상이 된다.

--------------------------------------------------------------------------------
[5] 진행 로그 : 로그 스위치가 체크포인트를 유발한다
--------------------------------------------------------------------------------
SYS@orcl> SELECT group#, sequence#, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# STATUS
---------- ---------- ----------------
         1         97 INACTIVE
         2         98 CURRENT
         3         96 INACTIVE

SYS@orcl> ALTER SYSTEM SWITCH LOGFILE;
System altered.

SYS@orcl> SELECT group#, sequence#, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# STATUS
---------- ---------- ----------------
         1         97 INACTIVE
         2         98 ACTIVE
         3         99 CURRENT

-- CURRENT 가 3번으로 넘어갔고 2번은 ACTIVE 가 되었다.
-- ACTIVE 는 기록은 끝났으나 아직 체크포인트가 완료되지 않은 상태다.

SYS@orcl> SELECT checkpoint_change# FROM v$database;

CHECKPOINT_CHANGE#
------------------
           2098460

-- 값이 2097189 에서 2098460 으로 올라갔다.
-- 로그 스위치가 체크포인트를 유발했기 때문이다.

-- 잠시 후 다시 조회하면 2번 그룹이 INACTIVE 로 바뀐다.

SYS@orcl> SELECT group#, sequence#, status FROM v$log ORDER BY group#;

    GROUP#  SEQUENCE# STATUS
---------- ---------- ----------------
         1         97 INACTIVE
         2         98 INACTIVE
         3         99 CURRENT

-- << 오류 발생 → 원인 파악 → 수정 >>
-- 체크포인트를 강제로 일으키려다 명령을 잘못 쓴다.

SYS@orcl> ALTER DATABASE CHECKPOINT;
ALTER DATABASE CHECKPOINT
*
ERROR at line 1:
ORA-02231: missing or invalid option to ALTER DATABASE

-- << 원인 파악 >>
-- 체크포인트 강제는 ALTER DATABASE 가 아니라 ALTER SYSTEM 의 기능이다.
-- ALTER DATABASE 는 데이터베이스 자체의 상태(마운트/오픈/로그 모드 등)를 다룬다.

-- << 수정 >>

SYS@orcl> ALTER SYSTEM CHECKPOINT;
System altered.

SYS@orcl> SELECT checkpoint_change# FROM v$database;

CHECKPOINT_CHANGE#
------------------
           2098533

-- 로그 스위치 없이도 체크포인트 SCN 이 갱신되었다.

--------------------------------------------------------------------------------
[6] 진단 : 제어 파일이 아는 값과 파일에 적힌 값
--------------------------------------------------------------------------------
-- v$datafile 은 제어 파일이 기억하는 값이다.

SYS@orcl> SELECT file#, checkpoint_change# FROM v$datafile ORDER BY file#;

     FILE# CHECKPOINT_CHANGE#
---------- ------------------
         1            2098533
         3            2098533
         4            2098533
         7            2098533

-- v$datafile_header 는 파일 헤더에서 직접 읽은 값이다.

SYS@orcl> SELECT file#, checkpoint_change#, status, error
  2  FROM   v$datafile_header ORDER BY file#;

     FILE# CHECKPOINT_CHANGE# STATUS  ERROR
---------- ------------------ ------- ------------
         1            2098533 ONLINE
         3            2098533 ONLINE
         4            2098533 ONLINE
         7            2098533 ONLINE

-- 정상 상태에서는 두 값이 같다.
-- 파일이 유실되었거나 백업본으로 복원된 직후에는 두 값이 달라지며,
-- 그 차이가 곧 "제어 파일이 아는 상태와 실제 파일 상태가 어긋났다"는 진단이 된다.
-- ERROR 컬럼은 정상일 때 비어 있고, 문제가 있으면 FILE NOT FOUND 등이 표시된다.

--------------------------------------------------------------------------------
[7] 검증
--------------------------------------------------------------------------------
-- 실습 전후의 SCN 변화를 정리하면 다음과 같다.
--
--   시작 시점          CHECKPOINT_CHANGE# = 2097189
--   UPDATE + COMMIT 후 CHECKPOINT_CHANGE# = 2097189  (변화 없음)
--   SWITCH LOGFILE 후  CHECKPOINT_CHANGE# = 2098460  (갱신됨)
--   ALTER SYSTEM CHECKPOINT 후            = 2098533  (갱신됨)
--
-- 변경 데이터를 원래대로 되돌린다.

SYS@orcl> UPDATE hr.employees SET salary = 24000 WHERE employee_id = 100;
1 row updated.

SYS@orcl> COMMIT;
Commit complete.

--------------------------------------------------------------------------------
[8] 실무 포인트
--------------------------------------------------------------------------------
-- 1) 커밋과 체크포인트는 별개다. 커밋은 리두를, 체크포인트는 데이터 파일 반영을 다룬다.
--    이 구분이 서지 않으면 "커밋했는데 왜 복구가 필요한가"라는 의문이 풀리지 않는다.
--
-- 2) CURRENT_SCN 과 CHECKPOINT_CHANGE# 의 차이가 인스턴스 복구 시 롤포워드할 양이다.
--    FAST_START_MTTR_TARGET 을 설정하면 Oracle 이 이 간격을 목표 안으로 억제한다.
--
-- 3) 복구 작업의 첫 조회는 언제나 v$database 와 v$datafile 의 CHECKPOINT_CHANGE# 비교다.
--    전부 같으면 복원만, 특정 파일만 작으면 그 SCN 부터의 리두 적용이 필요하다.
--
-- 4) 마운트 상태에서 데이터베이스가 열리지 않을 때는 v$datafile 이 아니라
--    v$datafile_header 를 봐야 한다. 실제 파일의 상태를 알려 주는 쪽이 후자다.
--
-- 5) SCN_TO_TIMESTAMP 는 근사치를 돌려준다. 보관 기간을 넘은 SCN 에는 ORA-08181 이 발생한다.
--    불완전 복구 시점을 정할 때 참고용으로만 쓰고, 정밀한 지정에는 SCN 을 직접 쓴다.

SYS@orcl> EXIT
