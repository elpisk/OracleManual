================================================================================
 실습 1-4 : 백업 스크립트 생성
 과정      : Oracle Database 19c Backup & Recovery
 환경      : Oracle 19.3.0 / Oracle Linux 7 / non-CDB ORCL
 소요 시간 : 약 12분
================================================================================

--------------------------------------------------------------------------------
[1] 실습 개요
--------------------------------------------------------------------------------
-- 실습 1-1 에서 얻은 파일 목록을 가공해 OS 복사 명령을 자동 생성한다.
-- 경로를 손으로 옮겨 적으면 반드시 빠뜨리는 파일이 생기므로, 조회 결과에서 만들어 낸다.
-- 함께 파라미터 파일 백업과 제어 파일 트레이스도 확보한다.
-- 여기서 만든 산출물은 3장 Cold Backup 실습에서 그대로 재사용한다.
-- 복구 유형 : 해당 없음(백업 준비 실습)

--------------------------------------------------------------------------------
[2] 사전 조건
--------------------------------------------------------------------------------
-- 데이터베이스가 OPEN 상태여야 한다.
-- 백업 디렉터리를 만들 수 있는 OS 권한이 있어야 한다.
-- 이 실습은 스크립트를 생성만 하고 실제 백업은 수행하지 않는다.

[oracle@oel7v9r1 ~]$ mkdir -p /home/oracle/backup/cold
[oracle@oel7v9r1 ~]$ ls -ld /home/oracle/backup/cold
drwxr-xr-x. 2 oracle oinstall 6 May  6 09:48 /home/oracle/backup/cold

[oracle@oel7v9r1 ~]$ sqlplus / as sysdba

--------------------------------------------------------------------------------
[3] 초기 상태 확인
--------------------------------------------------------------------------------
SYS@orcl> SET LINESIZE 200
SYS@orcl> SET PAGESIZE 0
SYS@orcl> SET FEEDBACK OFF
SYS@orcl> SET HEADING OFF
SYS@orcl> SET TRIMSPOOL ON

SYS@orcl> SELECT COUNT(*) FROM v$datafile;
         4

SYS@orcl> SELECT COUNT(*) FROM v$controlfile;
         2

SYS@orcl> SELECT COUNT(*) FROM v$logfile;
         3

-- 백업 대상은 데이터 파일 4 + 제어 파일 2 + 리두 로그 3 = 총 9개다.

--------------------------------------------------------------------------------
[4] 작업 수행 : 복사 명령 자동 생성
--------------------------------------------------------------------------------
SYS@orcl> SPOOL /home/oracle/backup/cold/cold_backup.sh

SYS@orcl> SELECT '#!/bin/bash' FROM dual;
#!/bin/bash

SYS@orcl> SELECT '# Cold Backup script - generated ' ||
  2         TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') FROM dual;
# Cold Backup script - generated 2025-05-06 09:52:11

SYS@orcl> SELECT 'BKP=/home/oracle/backup/cold' FROM dual;
BKP=/home/oracle/backup/cold

SYS@orcl> SELECT 'cp -v ' || name || ' $BKP/' FROM v$datafile
  2  UNION ALL
  3  SELECT 'cp -v ' || name || ' $BKP/' FROM v$controlfile
  4  UNION ALL
  5  SELECT 'cp -v ' || member || ' $BKP/' FROM v$logfile;
cp -v /u01/app/oracle/oradata/ORCL/system01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/sysaux01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/undotbs01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/users01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/control01.ctl $BKP/
cp -v /u01/app/oracle/oradata/ORCL/control02.ctl $BKP/
cp -v /u01/app/oracle/oradata/ORCL/redo01.log $BKP/
cp -v /u01/app/oracle/oradata/ORCL/redo02.log $BKP/
cp -v /u01/app/oracle/oradata/ORCL/redo03.log $BKP/

SYS@orcl> SPOOL OFF

--------------------------------------------------------------------------------
[5] 진행 로그 : 오류 발생 → 원인 파악 → 수정
--------------------------------------------------------------------------------
-- 생성된 스크립트를 확인한다.

SYS@orcl> !cat /home/oracle/backup/cold/cold_backup.sh
SQL> SELECT '#!/bin/bash' FROM dual;
#!/bin/bash

SQL> SELECT '# Cold Backup script - generated ' ||
  2         TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') FROM dual;
# Cold Backup script - generated 2025-05-06 09:52:11
...

-- << 오류 발생 >>

[oracle@oel7v9r1 ~]$ bash /home/oracle/backup/cold/cold_backup.sh
/home/oracle/backup/cold/cold_backup.sh: line 1: SQL>: command not found
/home/oracle/backup/cold/cold_backup.sh: line 2: syntax error near unexpected token `'#!/bin/bash''

-- << 원인 파악 >>
-- SPOOL 이 명령 프롬프트와 입력한 SQL 문까지 함께 파일에 기록했다.
-- SET ECHO OFF 가 적용되지 않은 상태이며, 스크립트 실행 시 이 줄들이 셸 명령으로 해석된다.

SYS@orcl> SHOW ECHO
echo ON

-- << 수정 >>
-- ECHO 를 끄고 다시 생성한다. 스풀 파일에 결과만 남는다.

SYS@orcl> SET ECHO OFF
SYS@orcl> SET VERIFY OFF
SYS@orcl> SPOOL /home/oracle/backup/cold/cold_backup.sh

SYS@orcl> SELECT '#!/bin/bash' FROM dual
  2  UNION ALL
  3  SELECT '# Cold Backup script - generated ' ||
  4         TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') FROM dual
  5  UNION ALL
  6  SELECT 'BKP=/home/oracle/backup/cold' FROM dual
  7  UNION ALL
  8  SELECT 'cp -v ' || name || ' $BKP/' FROM v$datafile
  9  UNION ALL
 10  SELECT 'cp -v ' || name || ' $BKP/' FROM v$controlfile
 11  UNION ALL
 12  SELECT 'cp -v ' || member || ' $BKP/' FROM v$logfile;

SYS@orcl> SPOOL OFF

SYS@orcl> !cat /home/oracle/backup/cold/cold_backup.sh
#!/bin/bash
# Cold Backup script - generated 2025-05-06 09:55:40
BKP=/home/oracle/backup/cold
cp -v /u01/app/oracle/oradata/ORCL/system01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/sysaux01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/undotbs01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/users01.dbf $BKP/
cp -v /u01/app/oracle/oradata/ORCL/control01.ctl $BKP/
cp -v /u01/app/oracle/oradata/ORCL/control02.ctl $BKP/
cp -v /u01/app/oracle/oradata/ORCL/redo01.log $BKP/
cp -v /u01/app/oracle/oradata/ORCL/redo02.log $BKP/
cp -v /u01/app/oracle/oradata/ORCL/redo03.log $BKP/

-- 이제 셸에서 실행 가능한 형태다.

--------------------------------------------------------------------------------
[8] 실무 포인트
--------------------------------------------------------------------------------
-- 1) 백업 스크립트는 손으로 쓰지 않고 조회 결과에서 만든다.
--    데이터 파일이 추가되어도 다시 생성만 하면 반영되므로 누락이 생기지 않는다.
--
-- 2) 이 스크립트는 데이터베이스를 종료한 뒤 실행해야 한다.
--    NOARCHIVELOG 모드에서 열린 상태로 복사하면 일관성 없는 백업이 되어 사용할 수 없다.
--    실제 실행 절차는 3장에서 다룬다.
--
-- 3) SPOOL 로 스크립트를 만들 때는 SET ECHO OFF, SET HEADING OFF, SET FEEDBACK OFF,
--    SET PAGESIZE 0, SET TRIMSPOOL ON 을 함께 적용한다. 하나라도 빠지면 실행되지 않는다.
--
-- 4) initorcl.ora 안의 control_files 항목을 확인해 둔다.
--    제어 파일을 모두 잃었을 때 어느 경로에 몇 개를 복원해야 하는지 알려 주는 유일한 근거가 된다.
--
-- 5) 트레이스로 받은 CREATE CONTROLFILE 스크립트는 NORESETLOGS 와 RESETLOGS 두 형태가
--    한 파일에 함께 들어 있다. 상황에 맞는 쪽을 골라 써야 하며, 6장에서 그 선택 기준을 다룬다.

SYS@orcl> EXIT
