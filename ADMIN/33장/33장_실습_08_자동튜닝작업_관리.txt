================================================================================
 33장 실습 08 — 자동 튜닝 작업(Automated SQL Tuning Task) 관리 실습
 목표: 자동 SQL 튜닝 작업(SYS_AUTO_SQL_TUNING_TASK)의 상태를 확인하고,
       자동 프로파일 구현(ACCEPT_SQL_PROFILES)을 켜고, 최근 결과를 조회한다.
       자동 유지관리 작업 프레임워크로 활성/비활성을 제어한다.
       잘못된 파라미터 값 설정 오류를 진단·수정한다.
 환경: CDB=cdb1, PDB=orclpdb1 (SYS로 수행)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] 자동 유지관리 작업 클라이언트의 상태를 확인한다.
SYS@cdb1> COLUMN client_name FORMAT A28
SYS@cdb1> SELECT client_name, status
  2    FROM dba_autotask_client
  3   WHERE client_name IN ('sql tuning advisor',
  4                         'auto optimizer stats collection');

CLIENT_NAME                  STATUS
---------------------------- --------
auto optimizer stats collect ENABLED
sql tuning advisor           ENABLED

-- 자동 SQL 튜닝은 자동 통계 수집과 함께 유지관리 윈도우에서 실행된다.

--------------------------------------------------------------------------------
-- [2] 자동 SQL 튜닝 태스크의 파라미터를 조회한다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT parameter_name, parameter_value
  2    FROM dba_advisor_parameters
  3   WHERE task_name = 'SYS_AUTO_SQL_TUNING_TASK'
  4     AND parameter_name IN ('ACCEPT_SQL_PROFILES','MAX_SQL_PROFILES_PER_EXEC',
  5                            'EXECUTION_DAYS_TO_EXPIRE');

PARAMETER_NAME              PARAMETER_VALUE
-------------------------- ----------------
ACCEPT_SQL_PROFILES        FALSE
MAX_SQL_PROFILES_PER_EXEC  20
EXECUTION_DAYS_TO_EXPIRE   30

-- 기본적으로 ACCEPT_SQL_PROFILES=FALSE(보고만, 자동 구현 안 함).

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정)
--     자동 프로파일 구현을 켜려다 값에 오타(TRUEE)를 넣어 오류가 난다.
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2    DBMS_AUTO_SQLTUNE.SET_AUTO_TUNING_TASK_PARAMETER(
  3      parameter => 'ACCEPT_SQL_PROFILES', value => 'TRUEE');
  4  END;
  5  /
BEGIN
*
ERROR at line 1:
ORA-13714: Invalid value "TRUEE" for parameter "ACCEPT_SQL_PROFILES".
ORA-06512: at "SYS.DBMS_AUTO_SQLTUNE", line 210
ORA-06512: at line 2

SYS@cdb1> -- 원인 파악: 이 파라미터는 TRUE/FALSE만 허용한다. 'TRUEE'는 잘못된 값.
SYS@cdb1> -- 수정: 올바른 값 'TRUE'로 다시 설정한다.
SYS@cdb1> BEGIN
  2    DBMS_AUTO_SQLTUNE.SET_AUTO_TUNING_TASK_PARAMETER(
  3      parameter => 'ACCEPT_SQL_PROFILES', value => 'TRUE');
  4  END;
  5  /

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT parameter_value FROM dba_advisor_parameters
  2   WHERE task_name='SYS_AUTO_SQL_TUNING_TASK'
  3     AND parameter_name='ACCEPT_SQL_PROFILES';

PARAMETER_VALUE
----------------
TRUE

-- 이제 검증상 3배 이상 개선되는 프로파일은 야간에 자동 구현된다.

--------------------------------------------------------------------------------
-- [4] 최근 자동 튜닝 결과 요약 보고서를 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> SET LONG 200000 LINESIZE 140 PAGESIZE 0
SYS@cdb1> SELECT DBMS_AUTO_SQLTUNE.REPORT_AUTO_TUNING_TASK FROM dual;

GENERAL INFORMATION SECTION
-------------------------------------------------------------------------------
 Task Name : SYS_AUTO_SQL_TUNING_TASK
 Analysis Period : last 7 days
-------------------------------------------------------------------------------
 Number of SQLs Analyzed        : 214
 Number of SQL Profiles Created : 6
 Number auto-implemented        : 4   (>=3x improvement, ACCEPT=TRUE)
-------------------------------------------------------------------------------

--------------------------------------------------------------------------------
-- [5] 자동 구현된 프로파일을 확인한다(TYPE=AUTO).
--------------------------------------------------------------------------------
SYS@cdb1> SELECT name, type, status FROM dba_sql_profiles
  2   WHERE type='AUTO' ORDER BY created DESC FETCH FIRST 3 ROWS ONLY;

NAME                       TYPE  STATUS
-------------------------- ----- --------
SYS_SQLPROF_017a2f...      AUTO  ENABLED
SYS_SQLPROF_017a2e...      AUTO  ENABLED

--------------------------------------------------------------------------------
-- [6] 유지보수 창에서 자동 SQL 튜닝을 잠시 비활성화/재활성화한다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.DISABLE( -
>   client_name => 'sql tuning advisor', operation => NULL, window_name => NULL);

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT status FROM dba_autotask_client
  2   WHERE client_name='sql tuning advisor';
STATUS
--------
DISABLED

SYS@cdb1> -- 실습 정리: 다시 활성화한다.
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.ENABLE( -
>   client_name => 'sql tuning advisor', operation => NULL, window_name => NULL);

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 자동 SQL 튜닝(SYS_AUTO_SQL_TUNING_TASK)은 야간 유지관리 윈도우에서 실행된다.
-- 2) ACCEPT_SQL_PROFILES=TRUE면 3배 이상 개선 프로파일을 자동 구현한다.
-- 3) 파라미터 값은 TRUE/FALSE만 허용 — 오타는 ORA-13714. 올바른 값으로 수정.
-- 4) 결과는 REPORT_AUTO_TUNING_TASK와 DBA_ADVISOR_*/DBA_SQL_PROFILES로 확인.
-- 5) 프레임워크 제어는 DBMS_AUTO_TASK_ADMIN.ENABLE/DISABLE로 수행한다.

SYS@cdb1> EXIT
