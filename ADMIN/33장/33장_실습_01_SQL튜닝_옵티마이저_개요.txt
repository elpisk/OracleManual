================================================================================
 33장 실습 01 — SQL 튜닝 프로세스 및 옵티마이저 개요 확인
 목표: 문제 SQL을 식별하고(V$SQL/V$SQLSTATS), 실행 계획을 확보해(DBMS_XPLAN)
       예상 카디널리티(E-Rows)와 실제 카디널리티(A-Rows)를 비교하는
       기본 튜닝 흐름을 익힌다. EXPLAIN PLAN과 DISPLAY_CURSOR의 차이도 확인한다.
 환경: CDB=cdb1, PDB=orclpdb1 (예제 스키마 SH/HR 사용, SYS로 수행)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] 현재 옵티마이저 모드와 관련 파라미터를 확인한다.
SYS@cdb1> SHOW PARAMETER optimizer_mode
NAME                                 TYPE        VALUE
------------------------------------ ----------- ------------------------------
optimizer_mode                       string      ALL_ROWS

SYS@cdb1> SHOW PARAMETER optimizer_features_enable
NAME                                 TYPE        VALUE
------------------------------------ ----------- ------------------------------
optimizer_features_enable            string      19.1.0

-- ALL_ROWS = 전체 처리량 최적화(기본). CBO가 통계 기반으로 최소 비용 계획을 고른다.

--------------------------------------------------------------------------------
-- [2] 자원을 많이 쓰는 상위 SQL을 식별한다(경과 시간 기준 상위 5).
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN sql_text FORMAT A44 TRUNCATE
SYS@cdb1> -- (오류 → 원인 → 수정) 존재하지 않는 컬럼명 elapsed_secs 로 조회한다.
SYS@cdb1> SELECT sql_id, executions, elapsed_secs, sql_text
  2    FROM v$sqlstats
  3   WHERE parsing_schema_name = 'SH'
  4   ORDER BY elapsed_time DESC;
       SELECT sql_id, executions, elapsed_secs, sql_text
                                  *
ERROR at line 1:
ORA-00904: "ELAPSED_SECS": invalid identifier

SYS@cdb1> -- 원인 파악: V$SQLSTATS 에는 ELAPSED_SECS 컬럼이 없다(ELAPSED_TIME, 마이크로초).
SYS@cdb1> -- 수정: 실제 컬럼 ELAPSED_TIME 을 초로 환산해 조회한다.
SYS@cdb1> SELECT sql_id, executions,
  2         ROUND(elapsed_time/1000000,2) AS elapsed_s,
  3         buffer_gets, sql_text
  4    FROM v$sqlstats
  5   WHERE parsing_schema_name = 'SH'
  6   ORDER BY elapsed_time DESC
  7   FETCH FIRST 5 ROWS ONLY;

9bxarq4k2u1pp       9821       8.04      302118 SELECT amount_sold FROM sales W...

-- 단일 실행이 느린 SQL(f8ts...)과, 개별은 짧지만 자주 실행되는 SQL(9bxa...) 모두 후보.

--------------------------------------------------------------------------------
-- [3] EXPLAIN PLAN으로 '예상' 계획을 본다(실제 실행하지 않음).
--------------------------------------------------------------------------------
SYS@cdb1> EXPLAIN PLAN FOR
  2  SELECT c.cust_last_name, SUM(s.amount_sold)
  3    FROM sh.sales s JOIN sh.customers c ON s.cust_id = c.cust_id
  4   WHERE c.cust_state_province = 'CA' AND c.country_id = 52790
  5   GROUP BY c.cust_last_name;

Explained.

SYS@cdb1> SELECT * FROM TABLE(DBMS_XPLAN.DISPLAY(NULL,NULL,'BASIC +ROWS'));

--------------------------------------------------------------------
| Id | Operation                    | Name      | Rows  |
--------------------------------------------------------------------
|  0 | SELECT STATEMENT             |           |    12 |
|  1 |  HASH GROUP BY               |           |    12 |
|  2 |   NESTED LOOPS               |           |    35 |
|  3 |    TABLE ACCESS FULL         | CUSTOMERS |     1 |
|  4 |    TABLE ACCESS BY INDEX ROW | SALES     |    35 |
--------------------------------------------------------------------

-- 예상: CUSTOMERS 조건이 1행이라고 추정 → NESTED LOOPS 선택.
-- 그러나 EXPLAIN PLAN에는 실제 실행 통계가 없어 이 추정이 맞는지 알 수 없다.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정)
--     실제 A-Rows를 보려고 방금 EXPLAIN PLAN한 것을 DISPLAY_CURSOR로 조회한다.
--     그러나 EXPLAIN PLAN은 커서를 실행한 것이 아니므로 실제 통계가 없다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT * FROM TABLE(
  2    DBMS_XPLAN.DISPLAY_CURSOR('does_not_exist_id', NULL, 'ALLSTATS LAST'));

SQL_ID: does_not_exist_id, child number:
cannot fetch plan for SQL_ID: does_not_exist_id, CHILD_NUMBER:
Please verify value of SQL_ID and CHILD_NUMBER;

-- 원인 파악: EXPLAIN PLAN은 '추정'만 만들고 커서를 실행하지 않는다.
--            실제 A-Rows는 SQL을 실제로 실행한 뒤 그 커서를 조회해야 나온다.
-- 수정: GATHER_PLAN_STATISTICS 힌트를 주고 실제로 실행한 뒤 DISPLAY_CURSOR 사용.

SYS@cdb1> SELECT /*+ GATHER_PLAN_STATISTICS */ c.cust_last_name, SUM(s.amount_sold)
  2    FROM sh.sales s JOIN sh.customers c ON s.cust_id = c.cust_id
  3   WHERE c.cust_state_province = 'CA' AND c.country_id = 52790
  4   GROUP BY c.cust_last_name;

... (rows returned) ...

SYS@cdb1> SELECT * FROM TABLE(
  2    DBMS_XPLAN.DISPLAY_CURSOR(NULL, NULL, 'ALLSTATS LAST'));

---------------------------------------------------------------------------------
| Id | Operation             | Name      | Starts | E-Rows | A-Rows | A-Time     |
---------------------------------------------------------------------------------
|  1 |  HASH GROUP BY        |           |      1 |     12 |    140 | 00:00:03.9 |
|  2 |   NESTED LOOPS        |           |      1 |     35 |    812K| 00:00:03.8 |
|  3 |    TABLE ACCESS FULL  | CUSTOMERS |      1 |      1 |   9800 | 00:00:00.1 |
|  4 |    TABLE ACCESS ...   | SALES     |   9800 |     35 |    812K| 00:00:03.5 |
---------------------------------------------------------------------------------

-- 핵심: CUSTOMERS를 E-Rows=1로 추정했으나 A-Rows=9800(약 1만 배 오차)!
--       두 조건 컬럼(state_province, country_id)의 상관관계를 옵티마이저가
--       독립으로 가정해 카디널리티를 심하게 과소평가 → 잘못된 NESTED LOOPS.
--       (실습 04에서 이 문제를 SQL Plan Directive/확장 통계로 다룬다.)
