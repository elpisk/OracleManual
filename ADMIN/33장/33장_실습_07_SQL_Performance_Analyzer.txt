================================================================================
 33장 실습 07 — SQL Performance Analyzer(SPA) 개요 확인
 목표: DBMS_SQLPA로 SPA 분석 태스크를 만들어, 환경 변경(옵티마이저 파라미터)
       전후의 SQL 성능을 before/after로 측정하고 비교 보고서를 확인한다.
       존재하지 않는 STS 참조 오류를 진단·수정한다.
 환경: CDB=cdb1, PDB=orclpdb1 (실습 05의 STS 'prod_sts' 재사용, SYS로 수행)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] SPA 분석 태스크를 STS 기반으로 생성한다.
--     (오류 → 원인 → 수정) 먼저 존재하지 않는 STS 이름으로 시도해 본다.
SYS@cdb1> DECLARE t VARCHAR2(64);
  2  BEGIN
  3    t := DBMS_SQLPA.CREATE_ANALYSIS_TASK(
  4           sqlset_name => 'prod_stss',   -- 오타: 존재하지 않는 STS
  5           task_name   => 'spa_param');
  6  END;
  7  /
DECLARE t VARCHAR2(64);
*
ERROR at line 1:
ORA-13757: "SQL Tuning Set" "prod_stss" owned by user "SYS" does not exist.
ORA-06512: at "SYS.DBMS_SQLPA", line 1024
ORA-06512: at line 3

SYS@cdb1> -- 원인 파악: STS 이름 오타. 실제 STS 목록을 확인한다.
SYS@cdb1> SELECT name, statement_count FROM dba_sqlset WHERE owner='SYS';

NAME       STATEMENT_COUNT
---------- ---------------
prod_sts                12

SYS@cdb1> -- 수정: 올바른 STS 이름 'prod_sts'로 다시 생성한다.
SYS@cdb1> DECLARE t VARCHAR2(64);
  2  BEGIN
  3    t := DBMS_SQLPA.CREATE_ANALYSIS_TASK(
  4           sqlset_name => 'prod_sts',
  5           task_name   => 'spa_param');
  6  END;
  7  /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [2] 변경 전(before) 성능을 측정한다(TEST EXECUTE: 실제 실행 후 통계 수집).
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SQLPA.EXECUTE_ANALYSIS_TASK( -
>   task_name      => 'spa_param', -
>   execution_type => 'TEST EXECUTE', -
>   execution_name => 'before');

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [3] 환경을 변경한다(예: 세션 옵티마이저 파라미터 조정).
--------------------------------------------------------------------------------
SYS@cdb1> ALTER SESSION SET optimizer_index_cost_adj = 20;

Session altered.

--------------------------------------------------------------------------------
-- [4] 변경 후(after) 성능을 측정한다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SQLPA.EXECUTE_ANALYSIS_TASK( -
>   task_name      => 'spa_param', -
>   execution_type => 'TEST EXECUTE', -
>   execution_name => 'after');

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [5] 두 실행을 비교한다(COMPARE PERFORMANCE, 지표 = ELAPSED_TIME).
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SQLPA.EXECUTE_ANALYSIS_TASK( -
>   task_name      => 'spa_param', -
>   execution_type => 'COMPARE PERFORMANCE', -
>   execution_params => DBMS_ADVISOR.ARGLIST( -
>     'comparison_metric','elapsed_time'));

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [6] 비교 보고서를 출력한다(개선/회귀 SQL 집계).
--------------------------------------------------------------------------------
SYS@cdb1> SET LONG 200000 LINESIZE 140 PAGESIZE 0
SYS@cdb1> SELECT DBMS_SQLPA.REPORT_ANALYSIS_TASK('spa_param', -
>   type=>'TEXT', section=>'SUMMARY') FROM dual;

-- 실제 프로덕션에 변경을 적용하기 전에, 회귀 2건만 골라 대응할 수 있다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) SPA는 DBMS_SQLPA로 STS 기반 before/after를 측정·비교한다(프로덕션 무영향).
-- 2) 실행 유형: EXPLAIN PLAN(계획만) / TEST EXECUTE(실제 실행 통계).
-- 3) 존재하지 않는 STS 참조는 ORA-13757. DBA_SQLSET에서 실제 이름을 확인한다.
-- 4) 보고서는 개선/회귀/불변을 집계 → 회귀 SQL만 SPM 등으로 부분 대응한다.

SYS@cdb1> EXIT
