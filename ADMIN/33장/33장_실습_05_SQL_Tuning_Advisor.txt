 33장 실습 05 — SQL Tuning Advisor 개요 확인
 목표: DBMS_SQLTUNE로 SQL Tuning Advisor 태스크를 생성·실행하고 권고를 조회한다.
       SQL 튜닝 세트(STS)의 개념을 확인하고, 프로파일 수락 흐름을 익힌다.
       잘못된 태스크 이름 조회(ORA-13780)와 권한 부족(ORA-01031)을 진단한다.
 환경: CDB=cdb1, PDB=orclpdb1 (SYS로 수행, 후반부 일반계정 예시 포함)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] 튜닝 대상 SQL_ID를 커서 캐시에서 확인한다.
SYS@cdb1> COLUMN sql_text FORMAT A50 TRUNCATE
SYS@cdb1> SELECT sql_id, sql_text FROM v$sql
  2   WHERE sql_text LIKE 'SELECT /*+ demo */%' FETCH FIRST 1 ROWS ONLY;

SQL_ID        SQL_TEXT
------------- --------------------------------------------------
f8tsjfq3n0ksx SELECT /*+ demo */ c.cust_last_name, SUM(s.amount_

--------------------------------------------------------------------------------
-- [2] SQL Tuning Advisor 태스크를 생성하고 실행한다(COMPREHENSIVE).
--------------------------------------------------------------------------------
SYS@cdb1> DECLARE
  2    t VARCHAR2(64);
  3  BEGIN
  4    t := DBMS_SQLTUNE.CREATE_TUNING_TASK(
  5           sql_id    => 'f8tsjfq3n0ksx',
  6           task_name => 'tune_sales_q',
  7           scope     => 'COMPREHENSIVE',
  8           time_limit=> 120);
  9    DBMS_SQLTUNE.EXECUTE_TUNING_TASK('tune_sales_q');
 10  END;
 11  /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정)
--     보고서를 조회하는데 태스크 이름을 오타(tune_sales)로 넣어 오류가 난다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT DBMS_SQLTUNE.REPORT_TUNING_TASK('tune_sales') FROM dual;
SELECT DBMS_SQLTUNE.REPORT_TUNING_TASK('tune_sales') FROM dual
       *
ERROR at line 1:
ORA-13780: SQL statement does not exist.
ORA-06512: at "SYS.DBMS_SQLTUNE", line 8224
ORA-06512: at "SYS.PRVT_ADVISOR", line 2892

SYS@cdb1> -- 원인 파악: 'tune_sales'라는 태스크가 없다(실제 이름은 tune_sales_q).
SYS@cdb1> -- 실제 태스크 이름을 확인한다.
SYS@cdb1> SELECT task_name, status FROM dba_advisor_tasks
  2   WHERE task_name LIKE 'tune_sales%';

TASK_NAME       STATUS
--------------- ----------
tune_sales_q    COMPLETED

SYS@cdb1> -- 수정: 올바른 이름으로 다시 조회한다.
SYS@cdb1> SET LONG 100000 LINESIZE 130 PAGESIZE 0
SYS@cdb1> SELECT DBMS_SQLTUNE.REPORT_TUNING_TASK('tune_sales_q') FROM dual;

-------------------------------------------------------------------------------

--------------------------------------------------------------------------------
-- [4] 권고된 SQL 프로파일을 수락한다(관리자 검토 후 적용).
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SQLTUNE.ACCEPT_SQL_PROFILE( -
>   task_name => 'tune_sales_q', name => 'prof_sales_q', force_match => TRUE);

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT name, type, status FROM dba_sql_profiles
  2   WHERE name='prof_sales_q';

NAME          TYPE   STATUS
------------- ------ --------
prof_sales_q  MANUAL ENABLED

-- 프로파일은 SQL 텍스트를 수정하지 않고 딕셔너리에 저장되어 계획을 개선한다.

--------------------------------------------------------------------------------
-- [5] SQL 튜닝 세트(STS) 개념 확인: 여러 SQL을 묶어 재사용하는 객체.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SQLTUNE.CREATE_SQLSET(sqlset_name => 'prod_sts', -
>   description => 'top SQL for tuning/SPA');

PL/SQL procedure successfully completed.

SYS@cdb1> -- 커서 캐시에서 SQL을 STS로 적재(개념).
SYS@cdb1> DECLARE
  2    cur DBMS_SQLTUNE.SQLSET_CURSOR;
  3  BEGIN
  4    OPEN cur FOR
  5      SELECT VALUE(p) FROM TABLE(DBMS_SQLTUNE.SELECT_CURSOR_CACHE(
  6        'parsing_schema_name = ''SH''')) p;
  7    DBMS_SQLTUNE.LOAD_SQLSET('prod_sts', cur);
  8  END;
  9  /

PL/SQL procedure successfully completed.

-- 이 STS는 실습 06(Access Advisor)·실습 07(SPA)의 입력으로 재사용된다.

--------------------------------------------------------------------------------
-- [6] (참고) 일반 계정에서 튜닝 태스크를 만들면 권한 부족이 날 수 있다.
--------------------------------------------------------------------------------
SYS@cdb1> CONNECT hr/oracle@orclpdb1
Connected.
HR@cdb1> DECLARE t VARCHAR2(64);
  2  BEGIN t := DBMS_SQLTUNE.CREATE_TUNING_TASK(sql_id=>'f8tsjfq3n0ksx'); END;
  3  /
DECLARE t VARCHAR2(64);
*
ERROR at line 1:
ORA-01031: insufficient privileges
ORA-06512: at "SYS.DBMS_SQLTUNE", line 1234

HR@cdb1> -- 원인: SQL Tuning Advisor 사용에는 ADVISOR 권한이 필요하다.
HR@cdb1> CONNECT sys@orclpdb1 as sysdba
Connected.
SYS@cdb1> GRANT ADVISOR TO hr;

Grant succeeded.

-- 이제 HR도 튜닝 태스크를 생성할 수 있다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) SQL Tuning Advisor는 DBMS_SQLTUNE로 CREATE/EXECUTE/REPORT_TUNING_TASK 사용.
-- 2) 권고: 통계·SQL 프로파일·인덱스·재작성. 프로파일은 ACCEPT_SQL_PROFILE로 적용.
-- 3) 잘못된 태스크 이름 조회는 ORA-13780. DBA_ADVISOR_TASKS에서 실제 이름 확인.
-- 4) 일반 계정은 ADVISOR 권한이 없으면 ORA-01031. GRANT ADVISOR로 해결.
-- 5) STS는 여러 SQL을 묶어 Tuning/Access Advisor/SPA에 재사용하는 자산이다.

SYS@cdb1> EXIT
