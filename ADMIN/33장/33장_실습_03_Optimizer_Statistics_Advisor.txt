================================================================================
 33장 실습 03 — Optimizer Statistics Advisor로 통계 문제 분석 실습
 목표: 19c의 Optimizer Statistics Advisor로 통계 '관리 방식'의 문제를
       규칙 기반으로 진단한다. 태스크 생성·실행, 보고서 출력, 스크립트 생성,
       그리고 자동 태스크(AUTO_STATS_ADVISOR_TASK) 결과 조회를 실습한다.
 환경: CDB=cdb1, PDB=orclpdb1 (SYS로 수행)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] 통계 어드바이저의 내장 규칙을 확인한다.
SYS@cdb1> COLUMN name FORMAT A32
SYS@cdb1> COLUMN rule_type FORMAT A10
SYS@cdb1> SELECT rule_id, name, rule_type
  2    FROM v$stats_advisor_rules
  3   WHERE rule_id BETWEEN 1 AND 6
  4   ORDER BY rule_id;

   RULE_ID NAME                             RULE_TYPE
---------- -------------------------------- ----------
         1 UseAutoJob                       SYSTEM
         2 CompleteAutoJob                  SYSTEM
         3 MaintainStatsHistory             SYSTEM
         4 UseConcurrent                    SYSTEM
         5 UseDefaultPreference             SYSTEM
         6 TurnOnSQLPlanDirective           SYSTEM

-- 규칙은 '통계 값'이 아니라 '통계를 수집·관리하는 방식'을 점검한다.

--------------------------------------------------------------------------------
-- [2] 사용자 정의 어드바이저 태스크를 생성하고 실행한다.
--------------------------------------------------------------------------------
SYS@cdb1> DECLARE
  2    v_task VARCHAR2(64);
  3  BEGIN
  4    v_task := DBMS_STATS.CREATE_ADVISOR_TASK('my_stats_task');
  5    v_task := DBMS_STATS.EXECUTE_ADVISOR_TASK('my_stats_task');
  6  END;
  7  /

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT task_name, status, created
  2    FROM dba_advisor_tasks
  3   WHERE task_name = 'my_stats_task';

TASK_NAME       STATUS      CREATED
--------------- ----------- -------------------
my_stats_task   COMPLETED   2026-08-12 11:05:44

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정)
--     같은 이름으로 태스크를 다시 만들려다 이름 중복 오류가 난다.
--------------------------------------------------------------------------------
SYS@cdb1> DECLARE v_task VARCHAR2(64);
  2  BEGIN v_task := DBMS_STATS.CREATE_ADVISOR_TASK('my_stats_task'); END;
  3  /
DECLARE v_task VARCHAR2(64);
*
ERROR at line 1:
ORA-20000: Unable to create task: task name is already used
ORA-06512: at "SYS.DBMS_STATS", line 41590
ORA-06512: at line 2

SYS@cdb1> -- 원인 파악: 동일 이름의 태스크가 이미 DBA_ADVISOR_TASKS에 존재한다.
SYS@cdb1> -- 수정: 기존 태스크를 삭제하거나 다른 이름을 사용한다.
SYS@cdb1> EXEC DBMS_STATS.DROP_ADVISOR_TASK('my_stats_task');

PL/SQL procedure successfully completed.

SYS@cdb1> DECLARE v_task VARCHAR2(64);
  2  BEGIN
  3    v_task := DBMS_STATS.CREATE_ADVISOR_TASK('my_stats_task');
  4    v_task := DBMS_STATS.EXECUTE_ADVISOR_TASK('my_stats_task');
  5  END;
  6  /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [4] TEXT 보고서를 출력해 발견 사항(findings)과 권고를 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> SET LONG 100000 LINESIZE 130 PAGESIZE 0
SYS@cdb1> SELECT DBMS_STATS.REPORT_STATS_ADVISOR_TASK('my_stats_task') FROM dual;

GENERAL INFORMATION

 Execution Name : EXEC_1

Rule Name: UseDefaultPreference

-- 실습 02에서 임의로 바꾼 프리퍼런스가 규칙 위반으로 잡혔다(진단 성공).

--------------------------------------------------------------------------------
-- [5] 권고를 실행 가능한 스크립트로 변환한다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT DBMS_STATS.SCRIPT_ADVISOR_TASK('my_stats_task') FROM dual;

-- Script generated for task my_stats_task
BEGIN
  DBMS_STATS.SET_TABLE_PREFS('SH','SALES','ESTIMATE_PERCENT',
    'DBMS_STATS.AUTO_SAMPLE_SIZE');
END;
/

--------------------------------------------------------------------------------
-- [6] 자동 태스크(AUTO_STATS_ADVISOR_TASK) 결과도 동일 방식으로 조회 가능하다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT DBMS_STATS.REPORT_STATS_ADVISOR_TASK('AUTO_STATS_ADVISOR_TASK')
  2    FROM dual;

... (매일 자동 실행된 최신 진단 보고서) ...

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) Optimizer Statistics Advisor는 통계 '관리 방식'을 규칙 기반으로 점검한다(19c).
-- 2) CREATE/EXECUTE_ADVISOR_TASK로 실행, REPORT_STATS_ADVISOR_TASK로 보고,
--    SCRIPT_ADVISOR_TASK로 조치 스크립트를 생성한다.
-- 3) 태스크 이름 중복은 ORA-20000. DROP_ADVISOR_TASK로 정리 후 재생성한다.
-- 4) 자동 태스크 AUTO_STATS_ADVISOR_TASK 결과도 같은 함수로 조회한다.

SYS@cdb1> EXIT
