================================================================================
 33장 실습 02 — 옵티마이저 통계 수집 및 환경설정(Preferences) 실습
 목표: DBMS_STATS로 통계를 수집하고, 프리퍼런스(GLOBAL/TABLE)를 설정·조회한다.
       통계 이력 조회, 잘못 설정한 프리퍼런스 이름 오류를 진단·수정한다.
 환경: CDB=cdb1, PDB=orclpdb1 (SH 스키마 사용, SYS로 수행)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] 현재 전역 프리퍼런스 기본값을 확인한다.
SYS@cdb1> SELECT DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT') AS est,
  2         DBMS_STATS.GET_PREFS('METHOD_OPT')       AS method,
  3         DBMS_STATS.GET_PREFS('STALE_PERCENT')    AS stale
  4    FROM dual;

EST                  METHOD                       STALE
-------------------- ---------------------------- -----
DBMS_STATS.AUTO_SAM  FOR ALL COLUMNS SIZE AUTO    10

-- 기본: AUTO_SAMPLE_SIZE + SIZE AUTO + stale 10%. 19c 권장 기본값.

--------------------------------------------------------------------------------
-- [2] SH.SALES 통계를 수집한다(자동 표본, 히스토그램 자동, 인덱스 통계 포함).
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_STATS.GATHER_TABLE_STATS( -
>   ownname => 'SH', tabname => 'SALES', -
>   estimate_percent => DBMS_STATS.AUTO_SAMPLE_SIZE, -
>   method_opt => 'FOR ALL COLUMNS SIZE AUTO', cascade => TRUE);

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT num_rows, blocks, last_analyzed
  2    FROM dba_tables WHERE owner='SH' AND table_name='SALES';

  NUM_ROWS     BLOCKS LAST_ANALYZED
---------- ---------- -------------------
    918843       1769 2026-08-12 10:22:07

--------------------------------------------------------------------------------
-- [3] 특정 테이블만 다른 정책을 적용한다(개별 프리퍼런스가 전역을 재정의).
--     SALES는 실시간성이 중요해 stale 임계를 5%로 낮춘다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_STATS.SET_TABLE_PREFS('SH','SALES','STALE_PERCENT','5');

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT DBMS_STATS.GET_PREFS('STALE_PERCENT','SH','SALES') AS tbl_stale,
  2         DBMS_STATS.GET_PREFS('STALE_PERCENT')            AS global_stale
  3    FROM dual;

TBL_STALE  GLOBAL_STALE
---------- ------------
5          10

-- 계층 확인: 테이블 수준 5%가 전역 10%를 재정의했다.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정)
--     프리퍼런스 이름을 오타(ESTIMATE_PCT)로 설정하려다 오류가 난다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_STATS.SET_TABLE_PREFS('SH','SALES','ESTIMATE_PCT','100');
BEGIN DBMS_STATS.SET_TABLE_PREFS('SH','SALES','ESTIMATE_PCT','100'); END;

*
ERROR at line 1:
ORA-20001: Invalid or Illegal input value for parameter - ESTIMATE_PCT
ORA-06512: at "SYS.DBMS_STATS", line 41590
ORA-06512: at line 1

SYS@cdb1> -- 원인 파악: 프리퍼런스 이름은 ESTIMATE_PCT가 아니라 ESTIMATE_PERCENT.
SYS@cdb1> -- 유효한 프리퍼런스 이름을 다시 확인하고 올바른 이름으로 설정한다.
SYS@cdb1> EXEC DBMS_STATS.SET_TABLE_PREFS('SH','SALES','ESTIMATE_PERCENT','100');

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SH','SALES') FROM dual;

DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SH','SALES')
-----------------------------------------------------
100

-- 수정 완료. (참고: 대부분의 경우 AUTO_SAMPLE_SIZE가 더 낫다. 여기선 실습용.)

--------------------------------------------------------------------------------
-- [5] 통계 이력을 확인한다(자동 보존, 기본 31일).
--------------------------------------------------------------------------------
SYS@cdb1> SELECT table_name, stats_update_time
  2    FROM dba_tab_stats_history
  3   WHERE owner='SH' AND table_name='SALES'
  4   ORDER BY stats_update_time DESC FETCH FIRST 3 ROWS ONLY;

TABLE_NAME STATS_UPDATE_TIME
---------- ---------------------------------
SALES      12-AUG-26 10.22.07.482000 AM +09:00
SALES      11-AUG-26 22.00.15.130000 PM +09:00

-- 필요 시 DBMS_STATS.RESTORE_TABLE_STATS(...,timestamp)로 특정 시점 통계 복원 가능.

--------------------------------------------------------------------------------
-- [6] 실습 정리: 개별 프리퍼런스를 기본으로 되돌린다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_STATS.DELETE_TABLE_PREFS('SH','SALES','ESTIMATE_PERCENT');

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 통계 수집 표준은 DBMS_STATS(ANALYZE 아님). AUTO_SAMPLE_SIZE + SIZE AUTO 권장.
-- 2) 프리퍼런스 계층: GLOBAL < DATABASE < SCHEMA < TABLE (좁은 범위가 우선).
-- 3) 프리퍼런스 이름 오타는 ORA-20001. GET_PREFS로 유효 이름/값을 확인한다.
-- 4) 통계 이력은 자동 보존되며 RESTORE_*_STATS로 복원한다.

SYS@cdb1> EXIT
