 33장 실습 06 — SQL Access Advisor 개요 확인
 목표: DBMS_ADVISOR로 SQL Access Advisor 태스크를 구성해 워크로드 기반
       인덱스/구체화 뷰/파티셔닝 권고를 생성하고, DDL 스크립트를 확인한다.
       존재하지 않는 태스크 이름 조회 오류를 진단·수정한다.
 환경: CDB=cdb1, PDB=orclpdb1 (SH 스키마, 실습 05의 STS 'prod_sts' 재사용)
================================================================================

$ sqlplus sys@orclpdb1 as sysdba

-- [1] 간편 실행: QUICK_TUNE으로 단일 SQL에 대한 액세스 권고를 받아 본다.
SYS@cdb1> VAR sql VARCHAR2(200)
SYS@cdb1> EXEC :sql := 'SELECT * FROM sh.sales WHERE prod_id=136 AND cust_id=1000';
SYS@cdb1> EXEC DBMS_ADVISOR.QUICK_TUNE( -
>   advisor_name => DBMS_ADVISOR.SQLACCESS_ADVISOR, -
>   task_name    => 'qt_sales', -
>   attr1        => :sql);

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [2] 워크로드 기반 정식 태스크를 생성한다(STS를 워크로드로 사용).
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2    DBMS_ADVISOR.CREATE_TASK(
  3      advisor_name => 'SQL Access Advisor',
  4      task_name    => 'acc_task');
  5  END;
  6  /

PL/SQL procedure successfully completed.

SYS@cdb1> -- STS 'prod_sts'를 워크로드로 연결한다.
SYS@cdb1> BEGIN
  2    DBMS_ADVISOR.ADD_STS_REF(
  3      task_name    => 'acc_task',
  4      sts_owner    => 'SYS',
  5      workload_name=> 'prod_sts');
  6  END;
  7  /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [3] 권고 유형을 인덱스/구체화 뷰 모두로 지정하고 실행한다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_ADVISOR.SET_TASK_PARAMETER('acc_task', -
>   'ANALYSIS_SCOPE', 'ALL');   -- INDEX, MVIEW, PARTITION 모두 고려
SYS@cdb1> EXEC DBMS_ADVISOR.EXECUTE_TASK('acc_task');

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT status FROM dba_advisor_tasks WHERE task_name='acc_task';
STATUS
-----------
COMPLETED

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정)
--     권고를 조회하는데 태스크 이름을 오타(acc_tasks)로 넣어 오류가 난다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT DBMS_ADVISOR.GET_TASK_SCRIPT('acc_tasks') FROM dual;
SELECT DBMS_ADVISOR.GET_TASK_SCRIPT('acc_tasks') FROM dual
       *
ERROR at line 1:
ORA-13605: The specified task or object acc_tasks does not exist for the
current user.
ORA-06512: at "SYS.PRVT_ADVISOR", line 2892

SYS@cdb1> -- 원인 파악: 'acc_tasks'라는 태스크가 없다(실제 이름은 acc_task).
SYS@cdb1> SELECT task_name FROM dba_advisor_tasks WHERE task_name LIKE 'acc_%';
TASK_NAME
-----------
acc_task

SYS@cdb1> -- 수정: 올바른 이름으로 권고 DDL 스크립트를 조회한다.
SYS@cdb1> SET LONG 100000 LINESIZE 130 PAGESIZE 0
SYS@cdb1> SELECT DBMS_ADVISOR.GET_TASK_SCRIPT('acc_task') FROM dual;

Rem  SQL Access Advisor: Recommendations
Rem
CREATE INDEX "SH"."SALES_IDX$$_ACC1"
    ON "SH"."SALES" ("PROD_ID","CUST_ID")
    COMPUTE STATISTICS;

Rem  Estimated benefit: 78%  Storage: ~120 MB

--------------------------------------------------------------------------------
-- [5] 권고 상세를 뷰로 확인한다(무조건 생성하지 말고 검토한다).
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN benefit FORMAT 999
SYS@cdb1> SELECT rec_id, rank, benefit
  2    FROM dba_advisor_recommendations
  3   WHERE task_name='acc_task' ORDER BY rank;

    REC_ID       RANK    BENEFIT
---------- ---------- ----------
         1          1         78
         2          2         31

-- Best Practice: 기존 인덱스와 중복되지 않는지, DML 부담이 크지 않은지 확인 후
--                이득(benefit)이 높은 항목부터 선별 적용한다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) SQL Access Advisor는 DBMS_ADVISOR로 워크로드 기반 액세스 구조를 권고한다.
-- 2) 권고: 인덱스/구체화 뷰/파티셔닝 — GET_TASK_SCRIPT로 DDL 스크립트 산출.
-- 3) 잘못된 태스크 이름은 ORA-13605. DBA_ADVISOR_TASKS에서 실제 이름을 확인한다.
-- 4) 권고는 즉시 반영하지 말고 이득/비용·기존 인덱스 중복을 검토해 선별 적용한다.

SYS@cdb1> EXIT
