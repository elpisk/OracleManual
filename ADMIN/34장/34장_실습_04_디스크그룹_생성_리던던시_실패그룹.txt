================================================================================
 34장 실습 04 — 디스크 그룹 생성 및 리던던시/실패 그룹 구성
 목표: 후보 디스크를 확인하고 NORMAL 리던던시 디스크 그룹을 실패 그룹 2개로 생성한다.
       (오류→원인→수정) 실패 그룹이 하나뿐이라 NORMAL 생성 실패 → 실패 그룹 2개로 수정.
 환경: ASM 인스턴스 +ASM, grid(SYSASM)
================================================================================

$ sqlplus / as sysasm

-- [1] 사용 가능한 후보 디스크를 확인한다(HEADER_STATUS).
SQL> SELECT path, header_status, os_mb FROM v$asm_disk
  2   WHERE header_status IN ('CANDIDATE','PROVISIONED') ORDER BY path;
PATH                          HEADER_STATUS  OS_MB
----------------------------- -------------- ------
/dev/oracleasm/disks/DISK10   CANDIDATE      10240
/dev/oracleasm/disks/DISK11   CANDIDATE      10240
/dev/oracleasm/disks/DISK12   CANDIDATE      10240
/dev/oracleasm/disks/DISK13   CANDIDATE      10240

--------------------------------------------------------------------------------
-- [2] (오류 → 원인 → 수정) NORMAL 리던던시인데 실패 그룹을 하나만 지정한다.
--------------------------------------------------------------------------------
SQL> CREATE DISKGROUP dgnorm NORMAL REDUNDANCY
  2    FAILGROUP fg1 DISK '/dev/oracleasm/disks/DISK10',
  3                       '/dev/oracleasm/disks/DISK11'
  4    ATTRIBUTE 'compatible.asm'='19.0.0.0';
CREATE DISKGROUP dgnorm NORMAL REDUNDANCY
*
ERROR at line 1:
ORA-15018: diskgroup cannot be created
ORA-15072: command requires at least 2 regular failure groups, discovered only 1

-- 원인 파악: NORMAL 은 미러를 다른 실패 그룹에 두어야 하므로 실패 그룹이 최소 2개 필요하다.
-- 수정: 실패 그룹을 fg1, fg2 두 개로 나누어 다시 생성한다.
SQL> CREATE DISKGROUP dgnorm NORMAL REDUNDANCY
  2    FAILGROUP fg1 DISK '/dev/oracleasm/disks/DISK10' NAME d_0001,
  3                       '/dev/oracleasm/disks/DISK11' NAME d_0002
  4    FAILGROUP fg2 DISK '/dev/oracleasm/disks/DISK12' NAME d_0003,
  5                       '/dev/oracleasm/disks/DISK13' NAME d_0004
  6    ATTRIBUTE 'au_size'='4M',
  7              'compatible.asm'='19.0.0.0',
  8              'compatible.rdbms'='19.0.0.0';
Diskgroup created.

-- [3] 생성 결과와 실패 그룹 구성을 확인한다.
SQL> SELECT name, type, total_mb, free_mb, usable_file_mb FROM v$asm_diskgroup
  2   WHERE name='DGNORM';
NAME    TYPE    TOTAL_MB  FREE_MB  USABLE_FILE_MB
------- ------- --------- -------- --------------
DGNORM  NORMAL     40960    40760           17800

SQL> SELECT d.name disk, d.failgroup, d.state FROM v$asm_disk d
  2   JOIN v$asm_diskgroup g ON d.group_number=g.group_number
  3   WHERE g.name='DGNORM' ORDER BY d.name;
DISK    FAILGROUP  STATE
------- ---------- --------
D_0001  FG1        NORMAL
D_0002  FG1        NORMAL
D_0003  FG2        NORMAL
D_0004  FG2        NORMAL

-- [4] USABLE_FILE_MB(약 17800)가 TOTAL 의 절반 수준인 이유: NORMAL 2중 미러 + 예비 공간.
-- 정리: NORMAL 은 실패 그룹 2개가 필수이며, 미러는 서로 다른 실패 그룹에 배치된다.
SQL> EXIT
