================================================================================
 34장 실습 05 — 디스크 추가/삭제/리밸런싱/스크러빙
 목표: DGNORM 디스크 그룹에 디스크를 추가/삭제하며 리밸런싱을 관찰하고,
       스크러빙으로 무결성을 검사한다.
       (오류→원인→수정) 존재하지 않는 디스크 이름 DROP → ORA-15054 → 정확한 이름으로 수정.
 환경: ASM 인스턴스 +ASM, grid(SYSASM), 디스크 그룹 DGNORM
================================================================================

$ sqlplus / as sysasm

-- [1] 새 디스크를 추가하고 다른 디스크를 같은 문장에서 삭제한다(리밸런싱 1회).
SQL> ALTER DISKGROUP dgnorm
  2    ADD  FAILGROUP fg1 DISK '/dev/oracleasm/disks/DISK14' NAME d_0005
  3    DROP DISK d_0002
  4    REBALANCE POWER 6;
Diskgroup altered.

-- [2] 리밸런싱 진행 상황을 모니터링한다.
SQL> SELECT operation, state, power, est_minutes FROM v$asm_operation;
OPERA STATE   POWER EST_MINUTES
----- ------- ----- -----------
REBAL RUN         6           2

-- [3] DROP 된 디스크의 상태를 확인한다(리밸런싱 끝나면 FORMER).
SQL> SELECT name, state, header_status FROM v$asm_disk WHERE name='D_0002';
NAME    STATE     HEADER_STATUS
------- --------- -------------
D_0002  DROPPING  MEMBER

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정) 존재하지 않는 디스크 이름으로 DROP 을 시도한다.
--------------------------------------------------------------------------------
SQL> ALTER DISKGROUP dgnorm DROP DISK d_9999;
ALTER DISKGROUP dgnorm DROP DISK d_9999
*
ERROR at line 1:
ORA-15054: disk "D_9999" does not exist in diskgroup "DGNORM"

-- 원인 파악: 지정한 디스크 이름이 이 디스크 그룹에 없다(오타/착오).
-- 수정: V$ASM_DISK 로 실제 이름을 확인 후 올바른 이름을 사용한다.
SQL> SELECT name FROM v$asm_disk d JOIN v$asm_diskgroup g
  2   ON d.group_number=g.group_number WHERE g.name='DGNORM';
NAME
-------
D_0001
D_0003
D_0004
D_0005

--------------------------------------------------------------------------------
-- [5] 리밸런싱 완료를 기다린 뒤(WAIT) 디스크 스크러빙을 수행한다.
--------------------------------------------------------------------------------
-- 리밸런싱이 끝났는지 확인(V$ASM_OPERATION 이 비어야 함)
SQL> SELECT count(*) FROM v$asm_operation;
  COUNT(*)
----------
         0

-- 디스크 그룹 전체를 검사하고 손상 시 미러로 복구(REPAIR), 완료까지 대기(WAIT)
SQL> ALTER DISKGROUP dgnorm SCRUB POWER HIGH REPAIR WAIT;
Diskgroup altered.

-- 특정 파일만 검사(복구 없이 진단)
SQL> ALTER DISKGROUP dgnorm SCRUB FILE '+DGNORM/orcl/datafile/users.259.1049'
  2    POWER MAX;
Diskgroup altered.

-- 정리: 추가/삭제를 한 문장에 모아 리밸런싱을 1회로 제한했고, 잘못된 디스크명(ORA-15054)을
-- 실제 이름으로 수정했으며, 스크러빙으로 무결성 검사·복구를 수행했다.
SQL> EXIT
