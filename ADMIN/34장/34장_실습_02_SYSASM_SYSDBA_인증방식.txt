================================================================================
 34장 실습 02 — SYSASM / SYSDBA 권한 차이 및 ASM 인증 방식
 목표: 동일 작업을 SYSDBA 와 SYSASM 으로 각각 시도하여 권한 차이를 체감하고,
       OS 인증과 비밀번호 파일 인증(V$PWFILE_USERS)을 확인한다.
       (오류→원인→수정) SYSDBA 로 디스크 그룹 생성 시도 → 권한 부족 → SYSASM 재접속.
 환경: ASM 인스턴스 +ASM, grid(OSASM) 계정
================================================================================

-- [1] 먼저 SYSDBA(for ASM) 로 접속한다.
$ sqlplus / as sysdba
Connected to:
Oracle Database 19c Enterprise Edition Release 19.0.0.0.0

SQL> SHOW PARAMETER instance_type
NAME            TYPE     VALUE
--------------- -------- --------
instance_type   string   asm

--------------------------------------------------------------------------------
-- [2] (오류 → 원인 → 수정) SYSDBA 로 디스크 그룹 생성을 시도한다.
--------------------------------------------------------------------------------
SQL> CREATE DISKGROUP test EXTERNAL REDUNDANCY
  2    DISK '/dev/oracleasm/disks/DISK9' NAME test_0001;
CREATE DISKGROUP test EXTERNAL REDUNDANCY
*
ERROR at line 1:
ORA-15260: permission denied on ASM disk group

-- 원인 파악: 디스크 그룹 생성은 ASM 관리 작업으로 SYSASM 권한이 필요하다.
--            SYSDBA(for ASM)는 데이터베이스의 파일 접근용이며 관리 권한이 부족하다.
SQL> EXIT

-- 수정: SYSASM 으로 재접속하여 동일 작업을 수행한다.
$ sqlplus / as sysasm
SQL> CREATE DISKGROUP test EXTERNAL REDUNDANCY
  2    DISK '/dev/oracleasm/disks/DISK9' NAME test_0001;
Diskgroup created.

--------------------------------------------------------------------------------
-- [3] 비밀번호 파일에 등록된 관리 사용자를 확인한다(비밀번호 파일 인증 대상).
--------------------------------------------------------------------------------
SQL> SELECT username, sysasm, sysdba, sysoper FROM v$pwfile_users ORDER BY username;
USERNAME   SYSASM SYSDBA SYSOPER
---------- ------ ------ -------
ASMSNMP    FALSE  TRUE   FALSE
SYS        TRUE   TRUE   TRUE

-- [4] ASM 전용 관리 사용자를 만들고 SYSASM 권한을 부여한다.
SQL> CREATE USER asm_op IDENTIFIED BY "Welcome1#";
User created.
SQL> GRANT SYSASM TO asm_op;
Grant succeeded.

SQL> SELECT username, sysasm FROM v$pwfile_users WHERE username='ASM_OP';
USERNAME   SYSASM
---------- ------
ASM_OP     TRUE

-- [5] OS 인증과 비밀번호 파일 인증의 차이 확인
--     · OS 인증:    sqlplus / as sysasm        (asmadmin 그룹 소속 근거, 로컬 전용)
--     · 비번 인증:  sqlplus asm_op@+ASM as sysasm   (원격/도구 접속에 사용)

-- [6] 실습 정리를 위해 방금 만든 test 디스크 그룹을 삭제한다.
SQL> DROP DISKGROUP test INCLUDING CONTENTS;
Diskgroup dropped.

-- 정리: 같은 명령이라도 SYSDBA 로는 실패(ORA-15260), SYSASM 으로는 성공했다.
-- ASM 관리 작업은 반드시 SYSASM 으로 수행한다는 원칙을 확인했다.
SQL> EXIT
