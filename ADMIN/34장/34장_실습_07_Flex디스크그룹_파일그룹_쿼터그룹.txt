================================================================================
 34장 실습 07 — Oracle ASM Flex 디스크 그룹 및 파일 그룹/쿼터 그룹 구성
 목표: FLEX REDUNDANCY 디스크 그룹을 만들고 쿼터 그룹/파일 그룹으로 PDB별 공간을 관리한다.
       (오류→원인→수정) 실패 그룹 2개로 FLEX 생성 실패 → 3개로 수정.
 환경: ASM 인스턴스 +ASM, grid(SYSASM), 후보 디스크 DISK16~19
================================================================================

$ sqlplus / as sysasm

--------------------------------------------------------------------------------
-- [1] (오류 → 원인 → 수정) FLEX 인데 실패 그룹을 2개만 지정한다.
--------------------------------------------------------------------------------
SQL> CREATE DISKGROUP flexdg FLEX REDUNDANCY
  2    FAILGROUP fg1 DISK '/dev/oracleasm/disks/DISK16'
  3    FAILGROUP fg2 DISK '/dev/oracleasm/disks/DISK17'
  4    ATTRIBUTE 'compatible.asm'='19.0.0.0','compatible.rdbms'='19.0.0.0';
CREATE DISKGROUP flexdg FLEX REDUNDANCY
*
ERROR at line 1:
ORA-15427: FLEX redundancy disk group requires at least 3 failure groups

-- 원인 파악: FLEX 디스크 그룹은 최소 3개의 실패 그룹이 필요하다.
-- 수정: 실패 그룹을 3개로 늘려 생성한다.
SQL> CREATE DISKGROUP flexdg FLEX REDUNDANCY
  2    FAILGROUP fg1 DISK '/dev/oracleasm/disks/DISK16' NAME f_0001
  3    FAILGROUP fg2 DISK '/dev/oracleasm/disks/DISK17' NAME f_0002
  4    FAILGROUP fg3 DISK '/dev/oracleasm/disks/DISK18' NAME f_0003
  5    ATTRIBUTE 'compatible.asm'='19.0.0.0','compatible.rdbms'='19.0.0.0';
Diskgroup created.

-- [2] 쿼터 그룹을 만들고 상한을 지정한다.
SQL> ALTER DISKGROUP flexdg ADD QUOTAGROUP sales_qg SET 'quota'='5G';
Diskgroup altered.

-- [3] PDB(salespdb)용 파일 그룹을 만들고 리던던시(HIGH)와 쿼터 그룹을 지정한다.
SQL> ALTER DISKGROUP flexdg ADD FILEGROUP salespdb
  2    DATABASE salespdb
  3    SET 'redundancy'='high'
  4    SET 'quota_group'='sales_qg';
Diskgroup altered.

-- [4] 파일 그룹/쿼터 그룹 구성을 확인한다.
SQL> SELECT name, redundancy FROM v$asm_filegroup
  2   WHERE group_number=(SELECT group_number FROM v$asm_diskgroup WHERE name='FLEXDG');
NAME       REDUNDANCY
---------- ----------
DEFAULT_FILEGROUP  MIRROR
SALESPDB           HIGH

SQL> SELECT name, quota_limit_mb, used_quota_mb FROM v$asm_quotagroup
  2   WHERE group_number=(SELECT group_number FROM v$asm_diskgroup WHERE name='FLEXDG');
NAME       QUOTA_LIMIT_MB  USED_QUOTA_MB
---------- --------------- -------------
GENERIC              0                0
SALES_QG          5120                0

-- [5] 정리를 위해 실습용 FLEXDG 를 삭제한다.
SQL> DROP DISKGROUP flexdg INCLUDING CONTENTS;
Diskgroup dropped.

-- 정리: FLEX 는 실패 그룹 3개가 필수(ORA-15427)이며, 파일 그룹으로 PDB별 리던던시(HIGH)를,
-- 쿼터 그룹으로 합계 사용량 상한(5G)을 통제했다.
SQL> EXIT
