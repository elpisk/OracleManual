-- 17장 실습_06: Deferred Segment Creation 활성화/비활성화 실습

[oracle@dbsvr ~]$ sqlplus / as sysdba
...
SYS@cdb1> alter session set container = salespdb;

Session altered.

======================================================================
[1단계] 전역 파라미터 확인 (기본 TRUE)
======================================================================

SYS@cdb1> show parameter deferred_segment_creation

NAME                             TYPE      VALUE
-------------------------------- --------- -----
deferred_segment_creation        boolean   TRUE

======================================================================
[2단계] 지연 생성 관찰: 빈 테이블은 세그먼트가 없다
======================================================================

SYS@cdb1> create table hr.d_demo (id number, name varchar2(50))
  2  tablespace users;

Table created.

- 아직 INSERT 전 -> 세그먼트가 없어 DBA_SEGMENTS에 나타나지 않음
SYS@cdb1> select segment_name from dba_segments
  2  where owner='HR' and segment_name='D_DEMO';

no rows selected

- 딕셔너리에는 존재하며 SEGMENT_CREATED=NO 로 표시됨
SYS@cdb1> select table_name, segment_created from dba_tables
  2  where owner='HR' and table_name='D_DEMO';

TABLE_NAME  SEGMENT_CREATED
----------- ---------------
D_DEMO      NO

- 첫 행을 삽입하면 그 순간 세그먼트가 생성됨
SYS@cdb1> insert into hr.d_demo values (1,'first');

1 row created.

SYS@cdb1> commit;

Commit complete.

SYS@cdb1> select segment_name, bytes/1024 as kb from dba_segments
  2  where owner='HR' and segment_name='D_DEMO';

SEGMENT_NAME    KB
--------------- -----
D_DEMO          64

======================================================================
[3단계] 개별 오버라이드: SEGMENT CREATION IMMEDIATE / DEFERRED
======================================================================

- 전역이 TRUE라도 이 테이블만 즉시 세그먼트 생성
SYS@cdb1> create table hr.d_now (id number)
  2  segment creation immediate tablespace users;

Table created.

SYS@cdb1> select segment_name from dba_segments
  2  where owner='HR' and segment_name='D_NOW';

SEGMENT_NAME
---------------
D_NOW              -- INSERT 전인데도 이미 존재

======================================================================
[4단계] 전역 비활성화 후 동작 비교
======================================================================

- (실수) DEFERRED_SEGMENT_CREATION은 boolean인데 ON/OFF로 지정 시도
SYS@cdb1> alter session set deferred_segment_creation = OFF;
ERROR:
ORA-00096: invalid value OFF for parameter deferred_segment_creation,
must be from among true, false

- 원인: 값은 true/false 여야 함. false로 정정
SYS@cdb1> alter session set deferred_segment_creation = false;

Session altered.

- 이제 절을 안 붙여도 생성 즉시 세그먼트가 만들어짐
SYS@cdb1> create table hr.d_off (id number) tablespace users;

Table created.

SYS@cdb1> select table_name, segment_created from dba_tables
  2  where owner='HR' and table_name='D_OFF';

TABLE_NAME  SEGMENT_CREATED
----------- ---------------
D_OFF       YES

======================================================================
[5단계] 빈 테이블에 세그먼트 강제 생성: ALLOCATE EXTENT
======================================================================

- 세션을 다시 지연(true)으로 돌리고, 지연 생성된 빈 테이블을 준비
SYS@cdb1> alter session set deferred_segment_creation = true;

Session altered.

SYS@cdb1> create table hr.d_later (id number) tablespace users;

Table created.

- 데이터 없이 세그먼트만 강제로 생성
SYS@cdb1> alter table hr.d_later allocate extent;

Table altered.

SYS@cdb1> select segment_name, extents, bytes/1024 as kb
  2  from   dba_segments
  3  where  owner='HR' and segment_name='D_LATER';

SEGMENT_NAME    EXTENTS    KB
--------------- ------- -----
D_LATER               1    64

======================================================================
[6단계] 정리 및 원복
======================================================================

- [정리]
-   전역: DEFERRED_SEGMENT_CREATION (기본 true)
-   개별: SEGMENT CREATION IMMEDIATE / DEFERRED
-   강제: ALTER TABLE ... ALLOCATE EXTENT
SYS@cdb1> drop table hr.d_demo purge;
SYS@cdb1> drop table hr.d_now purge;
SYS@cdb1> drop table hr.d_off purge;
SYS@cdb1> drop table hr.d_later purge;

Table dropped.

SYS@cdb1> exit
