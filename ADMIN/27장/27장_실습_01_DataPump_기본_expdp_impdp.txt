================================================================================
 27장 실습 01 — Data Pump Export/Import 기본 (expdp/impdp)
 목표: 디렉터리 객체 생성 → HR 스키마 내보내기(expdp) → 가져오기(impdp)
 환경: CDB=cdb1, PDB=pdb1, OS=Oracle Linux 7 (oracle 계정)
================================================================================

-- [1] PDB1에 SYS로 접속하여 디렉터리 객체를 만든다.
$ sqlplus sys@pdb1 as sysdba

SQL*Plus: Release 19.0.0.0.0 - Production on Wed Aug 5 09:12:04 2026
Copyright (c) 1982, 2019, Oracle.  All rights reserved.

SYS@pdb1> -- Data Pump 파일은 클라이언트가 아니라 DB 서버 디스크에 생성된다.
SYS@pdb1> -- 그 위치를 '디렉터리 객체'라는 논리 이름으로 지정한다.
SYS@pdb1> CREATE DIRECTORY dp_dir AS '/u01/app/oracle/dpdump';

Directory created.

SYS@pdb1> -- 내보내기를 수행할 HR 사용자에게 READ/WRITE 권한 부여
SYS@pdb1> GRANT READ, WRITE ON DIRECTORY dp_dir TO hr;

Grant succeeded.

SYS@pdb1> -- 디렉터리 객체가 어느 컨테이너에 만들어졌는지 확인
SYS@pdb1> SELECT directory_name, directory_path FROM dba_directories
  2       WHERE directory_name = 'DP_DIR';

DIRECTORY_NAME   DIRECTORY_PATH
---------------- ------------------------------
DP_DIR           /u01/app/oracle/dpdump

SYS@pdb1> EXIT

--------------------------------------------------------------------------------
-- [2] 첫 시도: expdp 실행 → 오류 발생 (OS 경로가 실제로 없음)
--------------------------------------------------------------------------------
$ expdp hr@pdb1 SCHEMAS=hr DIRECTORY=dp_dir DUMPFILE=hr.dmp LOGFILE=hr_exp.log

ORA-29283: invalid file operation: nonexistent file or path [/u01/app/oracle/dpdump]

-- 원인 파악:
--   디렉터리 객체(DP_DIR)는 만들어졌지만, 그 객체가 가리키는 OS 경로
--   /u01/app/oracle/dpdump 가 서버 파일 시스템에 실제로 존재하지 않는다.
--   디렉터리 객체 생성은 논리 이름만 등록할 뿐 실제 디렉터리를 만들지 않는다.

--------------------------------------------------------------------------------
-- [4] 가져오기 테스트: HR_COPY 스키마로 REMAP 하여 impdp
--------------------------------------------------------------------------------
$ sqlplus system@pdb1

SYSTEM@pdb1> CREATE USER hr_copy IDENTIFIED BY oracle
  2          DEFAULT TABLESPACE users QUOTA UNLIMITED ON users;
User created.
SYSTEM@pdb1> GRANT CREATE SESSION, CREATE TABLE TO hr_copy;
Grant succeeded.
SYSTEM@pdb1> EXIT

$ impdp system@pdb1 DIRECTORY=dp_dir DUMPFILE=hr.dmp LOGFILE=hr_imp.log \
         REMAP_SCHEMA=hr:hr_copy TABLE_EXISTS_ACTION=REPLACE

-- 결론:
--   1) 디렉터리 객체는 논리 이름일 뿐, OS 경로가 실제 존재해야 한다(ORA-29283).
--   2) expdp는 SCHEMAS/DIRECTORY/DUMPFILE/LOGFILE 조합으로 스키마를 내보낸다.
--   3) impdp는 REMAP_SCHEMA로 소유자를 바꿔 가져올 수 있다.
