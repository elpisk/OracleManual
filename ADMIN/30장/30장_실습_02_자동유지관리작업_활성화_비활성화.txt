================================================================================
 30장 실습 02 — 자동 유지관리 작업 활성화/비활성화 실습
 목표: 프레임워크 전체·클라이언트 단위·윈도우 단위로 자동 유지관리 작업을
       활성화/비활성화하고, 계층적 제어의 의미를 확인한다.
 환경: CDB=cdb1
================================================================================

-- [1] 세 작업의 현재 상태를 먼저 확인한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> COLUMN client_name FORMAT A34
SYS@cdb1> SELECT client_name, status FROM dba_autotask_client ORDER BY client_name;

CLIENT_NAME                        STATUS
---------------------------------- ---------
auto optimizer stats collection    ENABLED
auto space advisor                 ENABLED
sql tuning advisor                 ENABLED

--------------------------------------------------------------------------------
-- [2] 클라이언트 단위: SQL 튜닝 어드바이저만 비활성화한다(가장 무거운 작업).
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.DISABLE(client_name=>'sql tuning advisor', operation=>NULL, window_name=>NULL);

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT client_name, status FROM dba_autotask_client ORDER BY client_name;

CLIENT_NAME                        STATUS
---------------------------------- ---------
auto optimizer stats collection    ENABLED
auto space advisor                 ENABLED
sql tuning advisor                 DISABLED

-- 다른 두 작업은 그대로 두고 SQL 튜닝 어드바이저만 껐다.

--------------------------------------------------------------------------------
-- [3] 윈도우 단위: 통계 수집을 SATURDAY_WINDOW 에서만 비활성화한다.
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2         DBMS_AUTO_TASK_ADMIN.DISABLE(
  3           client_name => 'auto optimizer stats collection',
  4           operation   => NULL,
  5           window_name => 'SATURDAY_WINDOW');
  6       END;
  7       /

PL/SQL procedure successfully completed.

SYS@cdb1> -- 클라이언트 전체 STATUS 는 여전히 ENABLED 로 보인다(윈도우 단위 제어이므로).
SYS@cdb1> SELECT client_name, status FROM dba_autotask_client
  2      WHERE client_name = 'auto optimizer stats collection';

CLIENT_NAME                        STATUS
---------------------------------- ---------
auto optimizer stats collection    ENABLED

-- 즉 SATURDAY_WINDOW 에서만 건너뛰고, 다른 요일 윈도우에서는 정상 수집된다.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정) 존재하지 않는 윈도우 이름을 지정한다.
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2         DBMS_AUTO_TASK_ADMIN.DISABLE(
  3           client_name => 'auto optimizer stats collection',
  4           operation   => NULL,
  5           window_name => 'SAT_WINDOW');   -- 오타: 정확한 이름은 SATURDAY_WINDOW
  6       END;
  7       /
BEGIN
*
ERROR at line 1:
ORA-27476: "SYS.SAT_WINDOW" does not exist
ORA-06512: at "SYS.DBMS_ISCHED", line 5810
ORA-06512: at "SYS.DBMS_AUTO_TASK_ADMIN", line 157
ORA-06512: at line 2

SYS@cdb1> -- 원인 파악: window_name 에 준 'SAT_WINDOW' 라는 윈도우가 존재하지 않는다.
SYS@cdb1> -- 존재하는 윈도우 이름을 먼저 확인한다.
SYS@cdb1> COLUMN window_name FORMAT A18
SYS@cdb1> SELECT window_name FROM dba_scheduler_windows
  2      WHERE window_name LIKE '%SATURDAY%';

WINDOW_NAME
------------------
SATURDAY_WINDOW

SYS@cdb1> -- 수정: 정확한 이름 'SATURDAY_WINDOW' 를 사용한다(이미 [3]에서 반영됨).

--------------------------------------------------------------------------------
-- [5] 계층적 제어 확인: 프레임워크 전체를 끄면 개별 ENABLE 도 무효다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.DISABLE;    -- 인자 없음 = 프레임워크 전체

PL/SQL procedure successfully completed.

SYS@cdb1> -- 이 상태에서는 개별 클라이언트가 ENABLED 여도 실제로는 아무 것도 큐잉되지 않는다.
SYS@cdb1> -- 원복: 프레임워크 전체를 다시 켠다.
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.ENABLE;

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [6] 실습에서 변경한 항목을 원상 복구한다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.ENABLE(client_name=>'sql tuning advisor', operation=>NULL, window_name=>NULL);
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.ENABLE(client_name=>'auto optimizer stats collection', operation=>NULL, window_name=>'SATURDAY_WINDOW');

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 클라이언트 단위 DISABLE 은 STATUS 를 DISABLED 로 바꾼다.
-- 2) 윈도우 단위 DISABLE 은 클라이언트 STATUS 를 바꾸지 않고 그 윈도우에서만 제외한다.
-- 3) 존재하지 않는 윈도우 지정 시 ORA-27476 → DBA_SCHEDULER_WINDOWS 로 이름 확인.
-- 4) 프레임워크 전체(인자 없는 DISABLE)가 꺼지면 하위 설정은 효력이 없다(계층적).

SYS@cdb1> EXIT
