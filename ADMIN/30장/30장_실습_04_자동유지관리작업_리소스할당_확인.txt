================================================================================
 30장 실습 04 — 자동 유지관리 작업 리소스 할당 확인
 목표: DEFAULT_MAINTENANCE_PLAN 의 자원 배분을 DBA_RSRC_PLAN_DIRECTIVES 로
       조회하고, ORA$AUTOTASK 서브플랜의 세부 구조와 활성 플랜을 확인한다.
 환경: CDB=cdb1
================================================================================

-- [1] 유지관리 윈도우가 사용하는 리소스 플랜을 확인한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> COLUMN window_name   FORMAT A16
SYS@cdb1> COLUMN resource_plan FORMAT A26
SYS@cdb1> SELECT window_name, resource_plan
  2       FROM  dba_scheduler_windows
  3      WHERE window_name = 'MONDAY_WINDOW';

WINDOW_NAME      RESOURCE_PLAN
---------------- --------------------------
MONDAY_WINDOW    DEFAULT_MAINTENANCE_PLAN

-- 윈도우가 열리면 DEFAULT_MAINTENANCE_PLAN 이 자동 활성화된다.

--------------------------------------------------------------------------------
-- [2] DEFAULT_MAINTENANCE_PLAN 의 최상위 자원 배분을 조회한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN plan             FORMAT A26
SYS@cdb1> COLUMN group_or_subplan FORMAT A22
SYS@cdb1> COLUMN type             FORMAT A14
SYS@cdb1> SELECT plan, group_or_subplan, type, mgmt_p1
  2       FROM  dba_rsrc_plan_directives
  3      WHERE plan = 'DEFAULT_MAINTENANCE_PLAN'
  4      ORDER BY group_or_subplan;

PLAN                       GROUP_OR_SUBPLAN       TYPE           MGMT_P1
-------------------------- ---------------------- -------------- -------
DEFAULT_MAINTENANCE_PLAN   ORA$AUTOTASK           PLAN                25
DEFAULT_MAINTENANCE_PLAN   ORA$DIAGNOSTICS        CONSUMER_GROUP       5
DEFAULT_MAINTENANCE_PLAN   OTHER_GROUPS           CONSUMER_GROUP      70

-- ORA$AUTOTASK 는 TYPE=PLAN(서브플랜)으로 25% 상한을 받고,
-- 나머지 70% 는 OTHER_GROUPS(그 시간대의 배치/사용자)에 남긴다.

--------------------------------------------------------------------------------
-- [3] ORA$AUTOTASK 서브플랜 내부의 세부 배분을 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT plan, group_or_subplan, mgmt_p1, mgmt_p2, mgmt_p3
  2       FROM  dba_rsrc_plan_directives
  3      WHERE plan = 'ORA$AUTOTASK'
  4      ORDER BY group_or_subplan;

PLAN            GROUP_OR_SUBPLAN            MGMT_P1  MGMT_P2  MGMT_P3
--------------- -------------------------- -------  -------  -------
ORA$AUTOTASK    ORA$AUTOTASK_HEALTH_CHECK        0        0       25
ORA$AUTOTASK    ORA$AUTOTASK_MEDIUM             0        0       25
ORA$AUTOTASK    ORA$AUTOTASK_SPACE              0        0       25
ORA$AUTOTASK    ORA$AUTOTASK_SQL_TUNE           0        0       25
ORA$AUTOTASK    ORA$AUTOTASK_STATS              0        0       25
ORA$AUTOTASK    ORA$AUTOTASK_URGENT             0      100        0

-- 급박함(URGENT)이 상위 레벨을 먼저 받고, 일반 작업들은 하위 레벨에서 균등 분배된다.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정) 뷰 이름을 헷갈려 잘못 조회한다.
--------------------------------------------------------------------------------
SYS@cdb1> SELECT plan, group_or_subplan
  2       FROM  dba_rsrc_plan_directive          -- 오타: 끝의 S 누락
  3      WHERE plan = 'DEFAULT_MAINTENANCE_PLAN';
  FROM  dba_rsrc_plan_directive
        *
ERROR at line 2:
ORA-00942: table or view does not exist

SYS@cdb1> -- 원인 파악: 뷰 이름이 틀렸다. 정확한 이름은 복수형 DBA_RSRC_PLAN_DIRECTIVES 다.
SYS@cdb1> -- 사용 가능한 리소스 관리자 뷰를 확인한다.
SYS@cdb1> SELECT view_name FROM dba_views
  2      WHERE view_name LIKE 'DBA_RSRC_PLAN%';

VIEW_NAME
------------------------------
DBA_RSRC_PLANS
DBA_RSRC_PLAN_DIRECTIVES

SYS@cdb1> -- 수정: DBA_RSRC_PLAN_DIRECTIVES 로 다시 조회하면 [2]와 동일하게 정상 조회된다.

--------------------------------------------------------------------------------
-- [5] 현재 활성 상태인 리소스 플랜을 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN name FORMAT A26
SYS@cdb1> SELECT name, is_top_plan FROM v$rsrc_plan;

NAME                       IS_TOP_PLAN
-------------------------- -----------
INTERNAL_PLAN              FALSE
DEFAULT_MAINTENANCE_PLAN   TRUE

-- 지금 유지관리 윈도우가 열려 있어 DEFAULT_MAINTENANCE_PLAN 이 최상위 활성 플랜이다.
-- (윈도우가 닫히면 이전 플랜 또는 플랜 없음으로 복귀한다.)

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 윈도우의 resource_plan 속성으로 유지관리 시간대 자원 정책이 결정된다.
-- 2) DBA_RSRC_PLAN_DIRECTIVES 로 플랜별 배분(MGMT_P1~P8)을 조회한다(복수형 주의).
-- 3) TYPE 컬럼으로 서브플랜(PLAN)과 소비자 그룹(CONSUMER_GROUP)을 구분한다.
-- 4) V$RSRC_PLAN 으로 현재 활성 플랜을 확인한다.

SYS@cdb1> EXIT
