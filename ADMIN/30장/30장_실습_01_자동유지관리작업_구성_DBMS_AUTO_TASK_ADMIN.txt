================================================================================
 30장 실습 01 — 자동 유지관리 작업 구성(DBMS_AUTO_TASK_ADMIN) 실습
 목표: DBMS_AUTO_TASK_ADMIN 패키지로 자동 유지관리 작업의 구성 상태를 파악하고,
       클라이언트/오퍼레이션/윈도우 계층과 제어 범위를 이해한다.
 환경: CDB=cdb1 (루트 컨테이너에서 수행)
================================================================================

-- [1] 현재 자동 유지관리 작업(클라이언트)의 구성 상태를 조회한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> SET LINESIZE 160
SYS@cdb1> COLUMN client_name FORMAT A34
SYS@cdb1> COLUMN status      FORMAT A9
SYS@cdb1> SELECT client_name, status, attributes
  2       FROM  dba_autotask_client
  3      ORDER BY client_name;

CLIENT_NAME                        STATUS    ATTRIBUTES
---------------------------------- --------- ---------------------------
auto optimizer stats collection    ENABLED   ON BY DEFAULT, VOLATILE, ...
auto space advisor                 ENABLED   ON BY DEFAULT, VOLATILE, ...
sql tuning advisor                 ENABLED   ONCE PER WINDOW, ON BY DEF...

-- 세 가지 사전 정의 작업이 모두 ENABLED 상태다. client_name 문자열을 그대로
-- DBMS_AUTO_TASK_ADMIN 의 인자로 사용한다.

--------------------------------------------------------------------------------
-- [2] 각 클라이언트가 어떤 오퍼레이션으로 구성되는지 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN client_name    FORMAT A34
SYS@cdb1> COLUMN operation_name FORMAT A28
SYS@cdb1> SELECT client_name, operation_name
  2       FROM  dba_autotask_operation
  3      ORDER BY client_name;

CLIENT_NAME                        OPERATION_NAME
---------------------------------- ----------------------------
auto optimizer stats collection    auto optimizer stats job
auto space advisor                 auto space advisor job
sql tuning advisor                 automatic sql tuning task

-- 클라이언트(기능 단위) 아래에 오퍼레이션이 있고, 윈도우가 열릴 때 이 오퍼레이션이
-- 실제 태스크/잡으로 인스턴스화되어 실행된다.

--------------------------------------------------------------------------------
-- [3] 어떤 윈도우에서 어떤 클라이언트가 참여하는지 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN window_name FORMAT A16
SYS@cdb1> COLUMN client_name FORMAT A34
SYS@cdb1> SELECT window_name, client_name, optimizer_stats, segment_advisor,
  2              sql_tune_advisor
  3       FROM  dba_autotask_window_clients
  4      WHERE window_name = 'MONDAY_WINDOW';

WINDOW_NAME      CLIENT_NAME                        OPT SEG SQL
---------------- ---------------------------------- --- --- ---
MONDAY_WINDOW    ENABLED                            ENA ENA ENA

-- MONDAY_WINDOW 에서 세 작업이 모두 참여(ENABLED)하도록 구성되어 있다.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정) 잘못된 client_name 으로 DISABLE 을 시도한다.
--------------------------------------------------------------------------------
SYS@cdb1> -- 이름을 임의로 'optimizer stats' 라고 잘못 지정했다.
SYS@cdb1> BEGIN
  2         DBMS_AUTO_TASK_ADMIN.DISABLE(
  3           client_name => 'optimizer stats',
  4           operation   => NULL,
  5           window_name => NULL);
  6       END;
  7       /
BEGIN
*
ERROR at line 1:
ORA-20001: Invalid client name: optimizer stats
ORA-06512: at "SYS.DBMS_AUTO_TASK_ADMIN", line 108
ORA-06512: at line 2

SYS@cdb1> -- 원인 파악: client_name 은 DBA_AUTOTASK_CLIENT 에 등록된 정확한 이름이어야 한다.
SYS@cdb1> --   'optimizer stats' 는 존재하지 않는 이름이라 검증에 걸렸다.
SYS@cdb1> -- 수정: 정확한 이름 'auto optimizer stats collection' 을 사용한다.
SYS@cdb1> BEGIN
  2         DBMS_AUTO_TASK_ADMIN.DISABLE(
  3           client_name => 'auto optimizer stats collection',
  4           operation   => NULL,
  5           window_name => NULL);
  6       END;
  7       /

PL/SQL procedure successfully completed.

SYS@cdb1> -- 상태 확인
SYS@cdb1> SELECT client_name, status FROM dba_autotask_client
  2      WHERE client_name = 'auto optimizer stats collection';

CLIENT_NAME                        STATUS
---------------------------------- ---------
auto optimizer stats collection    DISABLED

--------------------------------------------------------------------------------
-- [5] 원상 복구: 다시 활성화한다(멱등적으로 안전).
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_AUTO_TASK_ADMIN.ENABLE(client_name=>'auto optimizer stats collection', operation=>NULL, window_name=>NULL);

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT client_name, status FROM dba_autotask_client
  2      WHERE client_name = 'auto optimizer stats collection';

CLIENT_NAME                        STATUS
---------------------------------- ---------
auto optimizer stats collection    ENABLED

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) DBMS_AUTO_TASK_ADMIN 은 자동 유지관리 작업의 관리 진입점이다.
-- 2) 계층: 클라이언트(DBA_AUTOTASK_CLIENT) → 오퍼레이션(DBA_AUTOTASK_OPERATION)
--    → 윈도우 참여(DBA_AUTOTASK_WINDOW_CLIENTS).
-- 3) client_name 은 반드시 정확한 등록명을 사용한다(오타 시 ORA-20001).
-- 4) ENABLE/DISABLE 은 멱등에 가까워 반복 호출해도 안전하다.

SYS@cdb1> EXIT
