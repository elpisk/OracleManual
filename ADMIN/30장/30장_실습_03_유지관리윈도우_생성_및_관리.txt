================================================================================
 30장 실습 03 — 유지관리 윈도우 생성 및 관리 실습
 목표: DBMS_SCHEDULER 로 사용자 정의 윈도우를 생성·수정·그룹추가·삭제하고,
       윈도우 그룹(MAINTENANCE_WINDOW_GROUP) 멤버를 관리한다.
 환경: CDB=cdb1 (루트에서 수행)
================================================================================

-- [1] 매일 새벽 1시에 열려 3시간 지속되는 사용자 정의 윈도우를 생성한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> BEGIN
  2         DBMS_SCHEDULER.CREATE_WINDOW(
  3           window_name     => 'FINANCE_MAINT_WINDOW',
  4           resource_plan   => 'DEFAULT_MAINTENANCE_PLAN',
  5           repeat_interval => 'freq=daily;byhour=1;byminute=0;bysecond=0',
  6           duration        => INTERVAL '3' HOUR,
  7           comments        => '회계 배치 이후 야간 유지관리용');
  8       END;
  9       /

PL/SQL procedure successfully completed.

SYS@cdb1> COLUMN window_name     FORMAT A22
SYS@cdb1> COLUMN repeat_interval FORMAT A38
SYS@cdb1> SELECT window_name, repeat_interval, duration, enabled
  2       FROM  dba_scheduler_windows
  3      WHERE window_name = 'FINANCE_MAINT_WINDOW';

WINDOW_NAME            REPEAT_INTERVAL                        DURATION      ENA
---------------------- -------------------------------------- ------------- ---
FINANCE_MAINT_WINDOW   freq=daily;byhour=1;byminute=0;bysec... +000 03:00:00 TRUE

-- 생성 직후 기본적으로 ENABLED=TRUE 상태다.

--------------------------------------------------------------------------------
-- [2] (오류 → 원인 → 수정) 같은 이름으로 다시 생성해 본다.
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2         DBMS_SCHEDULER.CREATE_WINDOW(
  3           window_name     => 'FINANCE_MAINT_WINDOW',
  4           resource_plan   => 'DEFAULT_MAINTENANCE_PLAN',
  5           repeat_interval => 'freq=daily;byhour=3',
  6           duration        => INTERVAL '2' HOUR);
  7       END;
  8       /
BEGIN
*
ERROR at line 1:
ORA-27477: "SYS"."FINANCE_MAINT_WINDOW" already exists
ORA-06512: at "SYS.DBMS_ISCHED", line 3138
ORA-06512: at "SYS.DBMS_SCHEDULER", line 2802
ORA-06512: at line 2

SYS@cdb1> -- 원인 파악: 동일한 이름의 윈도우가 이미 존재한다(ORA-27477).
SYS@cdb1> -- 수정: 속성을 바꾸려면 새로 만들지 말고 SET_ATTRIBUTE 로 변경한다.
SYS@cdb1> BEGIN
  2         DBMS_SCHEDULER.SET_ATTRIBUTE(
  3           name      => 'FINANCE_MAINT_WINDOW',
  4           attribute => 'DURATION',
  5           value     => INTERVAL '4' HOUR);
  6       END;
  7       /

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT window_name, duration FROM dba_scheduler_windows
  2      WHERE window_name = 'FINANCE_MAINT_WINDOW';

WINDOW_NAME            DURATION
---------------------- -------------
FINANCE_MAINT_WINDOW   +000 04:00:00

-- 지속 시간이 3시간 → 4시간으로 변경되었다.

--------------------------------------------------------------------------------
-- [3] 윈도우를 유지관리 윈도우 그룹에 추가해 자동 작업에 참여시킨다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SCHEDULER.ADD_WINDOW_GROUP_MEMBER('MAINTENANCE_WINDOW_GROUP','FINANCE_MAINT_WINDOW');

PL/SQL procedure successfully completed.

SYS@cdb1> COLUMN window_group_name FORMAT A26
SYS@cdb1> COLUMN window_name       FORMAT A22
SYS@cdb1> SELECT window_group_name, window_name
  2       FROM  dba_scheduler_wingroup_members
  3      WHERE window_group_name = 'MAINTENANCE_WINDOW_GROUP'
  4      ORDER BY window_name;

WINDOW_GROUP_NAME          WINDOW_NAME
-------------------------- ----------------------
MAINTENANCE_WINDOW_GROUP   FINANCE_MAINT_WINDOW
MAINTENANCE_WINDOW_GROUP   FRIDAY_WINDOW
MAINTENANCE_WINDOW_GROUP   MONDAY_WINDOW
... (요일별 윈도우 계속) ...
MAINTENANCE_WINDOW_GROUP   THURSDAY_WINDOW
MAINTENANCE_WINDOW_GROUP   TUESDAY_WINDOW
MAINTENANCE_WINDOW_GROUP   WEDNESDAY_WINDOW

-- FINANCE_MAINT_WINDOW 가 그룹 멤버로 추가되었다.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정) 그룹 멤버인 윈도우를 곧바로 삭제 시도한다.
--------------------------------------------------------------------------------
SYS@cdb1> EXEC DBMS_SCHEDULER.DROP_WINDOW('FINANCE_MAINT_WINDOW');
BEGIN DBMS_SCHEDULER.DROP_WINDOW('FINANCE_MAINT_WINDOW'); END;
*
ERROR at line 1:
ORA-27479: Cannot "DROP" "FINANCE_MAINT_WINDOW" because other objects depend on it
ORA-06512: at "SYS.DBMS_ISCHED", line 3401
ORA-06512: at "SYS.DBMS_SCHEDULER", line 2846

SYS@cdb1> -- 원인 파악: 윈도우가 윈도우 그룹의 멤버라서 의존성 때문에 삭제가 막혔다.
SYS@cdb1> -- 수정: 먼저 그룹에서 제거한 뒤 삭제한다.
SYS@cdb1> EXEC DBMS_SCHEDULER.REMOVE_WINDOW_GROUP_MEMBER('MAINTENANCE_WINDOW_GROUP','FINANCE_MAINT_WINDOW');

PL/SQL procedure successfully completed.

SYS@cdb1> EXEC DBMS_SCHEDULER.DROP_WINDOW('FINANCE_MAINT_WINDOW');

PL/SQL procedure successfully completed.

SYS@cdb1> SELECT window_name FROM dba_scheduler_windows
  2      WHERE window_name = 'FINANCE_MAINT_WINDOW';

no rows selected

-- 그룹에서 제거 후 삭제하여 정상적으로 제거되었다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) CREATE_WINDOW 로 생성, 속성 변경은 SET_ATTRIBUTE(같은 이름 재생성은 ORA-27477).
-- 2) 자동 작업 참여는 ADD_WINDOW_GROUP_MEMBER 로 MAINTENANCE_WINDOW_GROUP 에 추가.
-- 3) 그룹 멤버 윈도우 삭제 시 ORA-27479 → 먼저 REMOVE_WINDOW_GROUP_MEMBER 후 DROP.
-- 4) 존재하지 않는 윈도우 참조는 ORA-27476 으로 구분된다.

SYS@cdb1> EXIT
