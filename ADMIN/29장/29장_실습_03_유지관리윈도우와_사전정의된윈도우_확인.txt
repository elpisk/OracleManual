================================================================================
 29장 실습 03 — 유지관리 윈도우 개념 및 사전 정의된 윈도우 확인
 목표: 사전 정의된 요일별/주말 윈도우와, 이를 묶은 MAINTENANCE_WINDOW_GROUP,
       그리고 윈도우 열림 시 활성화되는 DEFAULT_MAINTENANCE_PLAN 을 확인
 환경: CDB=cdb1
================================================================================

-- [1] CDB 루트에 접속해 사전 정의된 유지관리 윈도우 목록을 확인한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> SET LINESIZE 160
SYS@cdb1> COLUMN window_name     FORMAT A18
SYS@cdb1> COLUMN repeat_interval FORMAT A42
SYS@cdb1> COLUMN duration        FORMAT A14
SYS@cdb1> SELECT window_name, repeat_interval, duration, enabled
  2       FROM  dba_scheduler_windows
  3      ORDER BY window_name;

WINDOW_NAME        REPEAT_INTERVAL                            DURATION       ENA
------------------ ------------------------------------------ -------------- ---
FRIDAY_WINDOW      freq=daily;byday=FRI;byhour=22;byminute=0  +000 04:00:00  TRUE
MONDAY_WINDOW      freq=daily;byday=MON;byhour=22;byminute=0  +000 04:00:00  TRUE
SATURDAY_WINDOW    freq=daily;byday=SAT;byhour=6;byminute=0   +000 20:00:00  TRUE
SUNDAY_WINDOW      freq=daily;byday=SUN;byhour=6;byminute=0   +000 20:00:00  TRUE
THURSDAY_WINDOW    freq=daily;byday=THU;byhour=22;byminute=0  +000 04:00:00  TRUE
TUESDAY_WINDOW     freq=daily;byday=TUE;byhour=22;byminute=0  +000 04:00:00  TRUE
WEDNESDAY_WINDOW   freq=daily;byday=WED;byhour=22;byminute=0  +000 04:00:00  TRUE

7 rows selected.

-- 평일(월~금): 밤 10시(byhour=22) 시작, 4시간 / 주말(토·일): 오전 6시 시작, 20시간.

--------------------------------------------------------------------------------
-- [2] 이 윈도우들을 묶은 윈도우 그룹을 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN window_group_name FORMAT A28
SYS@cdb1> SELECT window_group_name, number_of_windows, enabled
  2       FROM  dba_scheduler_window_groups;

WINDOW_GROUP_NAME            NUMBER_OF_WINDOWS ENA
---------------------------- ----------------- ---
MAINTENANCE_WINDOW_GROUP                     7 TRUE

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정) 그룹 멤버를 잘못된 그룹명으로 조회해 결과가 없다.
--------------------------------------------------------------------------------
SYS@cdb1> -- 멤버를 보려는데 그룹명을 'MAINTENANCE_WINDOWS_GROUP'(WINDOWS 복수형)로 잘못 입력했다.
SYS@cdb1> SELECT window_name FROM dba_scheduler_wingroup_members
  2       WHERE window_group_name = 'MAINTENANCE_WINDOWS_GROUP';

no rows selected

SYS@cdb1> -- 원인 파악: 그룹명 철자가 틀렸다. 올바른 이름은 MAINTENANCE_WINDOW_GROUP(단수 WINDOW)이다.
SYS@cdb1> SELECT window_group_name FROM dba_scheduler_window_groups;

WINDOW_GROUP_NAME
----------------------------
MAINTENANCE_WINDOW_GROUP

SYS@cdb1> -- 수정: 올바른 그룹명으로 멤버를 조회한다.
SYS@cdb1> SELECT window_group_name, window_name
  2       FROM  dba_scheduler_wingroup_members
  3      WHERE window_group_name = 'MAINTENANCE_WINDOW_GROUP'
  4      ORDER BY window_name;

WINDOW_GROUP_NAME            WINDOW_NAME
---------------------------- ------------------
MAINTENANCE_WINDOW_GROUP     FRIDAY_WINDOW
MAINTENANCE_WINDOW_GROUP     MONDAY_WINDOW
MAINTENANCE_WINDOW_GROUP     SATURDAY_WINDOW
MAINTENANCE_WINDOW_GROUP     SUNDAY_WINDOW
MAINTENANCE_WINDOW_GROUP     THURSDAY_WINDOW
MAINTENANCE_WINDOW_GROUP     TUESDAY_WINDOW
MAINTENANCE_WINDOW_GROUP     WEDNESDAY_WINDOW

7 rows selected.

--------------------------------------------------------------------------------
-- [4] 각 윈도우가 어떤 리소스 플랜을 사용하는지 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN resource_plan FORMAT A28
SYS@cdb1> SELECT window_name, resource_plan
  2       FROM  dba_scheduler_windows
  3      ORDER BY window_name;

WINDOW_NAME        RESOURCE_PLAN
------------------ ----------------------------
FRIDAY_WINDOW      DEFAULT_MAINTENANCE_PLAN
MONDAY_WINDOW      DEFAULT_MAINTENANCE_PLAN
SATURDAY_WINDOW    DEFAULT_MAINTENANCE_PLAN
SUNDAY_WINDOW      DEFAULT_MAINTENANCE_PLAN
THURSDAY_WINDOW    DEFAULT_MAINTENANCE_PLAN
TUESDAY_WINDOW     DEFAULT_MAINTENANCE_PLAN
WEDNESDAY_WINDOW   DEFAULT_MAINTENANCE_PLAN

7 rows selected.

-- 모든 윈도우가 DEFAULT_MAINTENANCE_PLAN 을 사용한다.
-- 윈도우가 열리면 이 플랜이 자동 활성화되어 유지관리 작업의 자원을 통제한다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 사전 정의 윈도우: 요일별(월~일) 7개. 평일 밤10시 4h, 주말 오전6시 20h.
-- 2) 7개 윈도우는 MAINTENANCE_WINDOW_GROUP(단수 WINDOW!)으로 묶여 있다.
-- 3) 자동 유지관리 작업은 이 그룹에 연결되어, 어떤 멤버 윈도우가 열려도 실행된다.
-- 4) 모든 윈도우는 DEFAULT_MAINTENANCE_PLAN 을 리소스 플랜으로 사용한다.

SYS@cdb1> EXIT
