================================================================================
 29장 실습 02 — 사전 정의된 자동 유지관리 작업 확인
 목표: 세 가지 작업(옵티마이저 통계/Segment Advisor/SQL Tuning Advisor)의
       활성 상태와 오퍼레이션 정의를 DBA_AUTOTASK_* 뷰로 확인
 환경: CDB=cdb1, PDB=pdb1
================================================================================

-- [1] CDB 루트에 접속해 세 작업(클라이언트)의 활성 상태를 확인한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> SET LINESIZE 140
SYS@cdb1> COLUMN client_name FORMAT A34
SYS@cdb1> COLUMN status      FORMAT A10
SYS@cdb1> SELECT client_name, status, attributes
  2       FROM  dba_autotask_client
  3      ORDER BY client_name;

CLIENT_NAME                        STATUS     ATTRIBUTES
---------------------------------- ---------- ------------------------------------
auto optimizer stats collection    ENABLED    ON BY DEFAULT, VOLATILE, SAFE TO KILL
auto space advisor                 ENABLED    ON BY DEFAULT, VOLATILE, SAFE TO KILL
sql tuning advisor                 ENABLED    ONCE PER WINDOW, ON BY DEFAULT, ...

-- 세 작업 모두 기본적으로 ENABLED 상태다.

--------------------------------------------------------------------------------
-- [2] 각 클라이언트가 수행하는 오퍼레이션(작업 정의)을 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN operation_name FORMAT A34
SYS@cdb1> SELECT client_name, operation_name
  2       FROM  dba_autotask_operation
  3      ORDER BY client_name;

CLIENT_NAME                        OPERATION_NAME
---------------------------------- ----------------------------------
auto optimizer stats collection    auto optimizer stats job
auto space advisor                 auto space advisor job
sql tuning advisor                 automatic sql tuning task

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정) 클라이언트 이름을 부정확하게 조회해 결과가 없다.
--------------------------------------------------------------------------------
SYS@cdb1> -- 통계 수집 작업만 보려는데 이름을 'auto stats' 로 대충 입력했다.
SYS@cdb1> SELECT client_name, status FROM dba_autotask_client
  2       WHERE client_name = 'auto stats';

no rows selected

SYS@cdb1> -- 원인 파악: 클라이언트 이름은 정확한 문자열이어야 한다.
SYS@cdb1> --   정확한 이름은 'auto optimizer stats collection' 이다.
SYS@cdb1> --   부분 문자열로 찾을 때는 LIKE 를 쓴다.
SYS@cdb1> SELECT client_name, status FROM dba_autotask_client
  2       WHERE client_name LIKE '%stats%';

CLIENT_NAME                        STATUS
---------------------------------- ----------
auto optimizer stats collection    ENABLED

SYS@cdb1> -- 수정: 정확한 이름으로 조회하면 정상적으로 한 행이 나온다.
SYS@cdb1> SELECT client_name, status FROM dba_autotask_client
  2       WHERE client_name = 'auto optimizer stats collection';

CLIENT_NAME                        STATUS
---------------------------------- ----------
auto optimizer stats collection    ENABLED

--------------------------------------------------------------------------------
-- [4] PDB 수준에서도 세 작업의 상태를 확인한다(작업 활성화는 PDB별 제어 가능).
--------------------------------------------------------------------------------
SYS@cdb1> ALTER SESSION SET CONTAINER = pdb1;

Session altered.

SYS@cdb1> SHOW CON_NAME

CON_NAME
------------------------------
PDB1

SYS@cdb1> SELECT client_name, status FROM dba_autotask_client ORDER BY client_name;

CLIENT_NAME                        STATUS
---------------------------------- ----------
auto optimizer stats collection    ENABLED
auto space advisor                 ENABLED
sql tuning advisor                 ENABLED

-- PDB1 에서도 세 작업이 활성 상태다. 각 PDB는 독립적으로 상태를 가진다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 세 작업의 클라이언트 이름:
--    auto optimizer stats collection / auto space advisor / sql tuning advisor
-- 2) 상태 조회: DBA_AUTOTASK_CLIENT.STATUS (ENABLED/DISABLED)
-- 3) 작업 정의: DBA_AUTOTASK_OPERATION.OPERATION_NAME
-- 4) 이름은 정확한 문자열로 비교하거나 LIKE 로 부분 검색한다(오타 시 no rows).
-- 5) 작업 활성 상태는 컨테이너(CDB/PDB)별로 개별 관리된다.

SYS@pdb1> EXIT
