================================================================================
 29장 실습 04 — 유지관리 윈도우 세부정보 조회 실습
 목표: 특정 윈도우의 스케줄·지속시간·다음 시작 시각·활성여부·리소스 플랜을
       상세 조회하고, DEFAULT_MAINTENANCE_PLAN 의 소비자 그룹 구성을 확인
 환경: CDB=cdb1
================================================================================

-- [1] MONDAY_WINDOW 의 세부 속성을 조회한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> SET LINESIZE 160
SYS@cdb1> COLUMN window_name   FORMAT A16
SYS@cdb1> COLUMN resource_plan FORMAT A26
SYS@cdb1> SELECT window_name, resource_plan, duration, enabled, active,
  2              TO_CHAR(next_start_date,'YYYY-MM-DD HH24:MI') AS next_start
  3       FROM  dba_scheduler_windows
  4      WHERE window_name = 'MONDAY_WINDOW';

WINDOW_NAME      RESOURCE_PLAN              DURATION       ENA ACT NEXT_START
---------------- -------------------------- -------------- --- --- ----------------
MONDAY_WINDOW    DEFAULT_MAINTENANCE_PLAN   +000 04:00:00  TRUE FAL 2026-08-17 22:00

-- ACTIVE=FALSE: 지금은 열려 있지 않다. NEXT_START: 다음 월요일 밤 10시에 열린다.

--------------------------------------------------------------------------------
-- [2] 윈도우 열림/닫힘 로그로 최근 실제 개폐 이력을 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN window_name FORMAT A16
SYS@cdb1> COLUMN operation   FORMAT A10
SYS@cdb1> SELECT window_name, operation,
  2              TO_CHAR(log_date,'MM-DD HH24:MI') AS log_time
  3       FROM  dba_scheduler_window_log
  4      WHERE window_name = 'MONDAY_WINDOW'
  5      ORDER BY log_date DESC FETCH FIRST 4 ROWS ONLY;

WINDOW_NAME      OPERATION  LOG_TIME
---------------- ---------- ----------
MONDAY_WINDOW    CLOSE      08-11 02:00
MONDAY_WINDOW    OPEN       08-10 22:00
MONDAY_WINDOW    CLOSE      08-04 02:00
MONDAY_WINDOW    OPEN       08-03 22:00

-- 밤 10시 OPEN → 새벽 2시 CLOSE. 4시간 윈도우가 정상 동작하고 있다.

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정) DATE 컬럼을 문자열로 오해해 잘못 비교하다 실패한다.
--------------------------------------------------------------------------------
SYS@cdb1> -- 특정 날짜 이후 로그만 보려고 문자열로 직접 비교했다.
SYS@cdb1> SELECT window_name, operation FROM dba_scheduler_window_log
  2       WHERE window_name = 'MONDAY_WINDOW'
  3         AND log_date > '2026-08-01';
       AND log_date > '2026-08-01'
                       *
ERROR at line 3:
ORA-01861: literal does not match format string

SYS@cdb1> -- 원인 파악: log_date 는 TIMESTAMP/DATE 타입이라 문자열 '2026-08-01' 과
SYS@cdb1> --   직접 비교하면 NLS 포맷이 맞지 않아 변환 오류가 난다.
SYS@cdb1> -- 수정: TO_DATE 로 명시적으로 변환해 비교한다.
SYS@cdb1> SELECT window_name, operation,
  2              TO_CHAR(log_date,'MM-DD HH24:MI') AS log_time
  3       FROM  dba_scheduler_window_log
  4      WHERE window_name = 'MONDAY_WINDOW'
  5        AND log_date > TO_DATE('2026-08-01','YYYY-MM-DD')
  6      ORDER BY log_date;

WINDOW_NAME      OPERATION  LOG_TIME
---------------- ---------- ----------
MONDAY_WINDOW    OPEN       08-03 22:00
MONDAY_WINDOW    CLOSE      08-04 02:00
MONDAY_WINDOW    OPEN       08-10 22:00
MONDAY_WINDOW    CLOSE      08-11 02:00

--------------------------------------------------------------------------------
-- [4] DEFAULT_MAINTENANCE_PLAN 의 소비자 그룹 구성(ORA$AUTOTASK 등)을 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN plan             FORMAT A26
SYS@cdb1> COLUMN group_or_subplan FORMAT A22
SYS@cdb1> SELECT plan, group_or_subplan, mgmt_p1
  2       FROM  dba_rsrc_plan_directives
  3      WHERE plan = 'DEFAULT_MAINTENANCE_PLAN'
  4      ORDER BY group_or_subplan;

PLAN                       GROUP_OR_SUBPLAN       MGMT_P1
-------------------------- ---------------------- ----------
DEFAULT_MAINTENANCE_PLAN   ORA$AUTOTASK                   25
DEFAULT_MAINTENANCE_PLAN   ORA$DIAGNOSTICS                 5
DEFAULT_MAINTENANCE_PLAN   OTHER_GROUPS                   70

-- ORA$AUTOTASK(유지관리 작업 전용 그룹)에 CPU 상한이 배정되어 있다(예: 25%).
-- 나머지는 그 시간대의 다른 작업(OTHER_GROUPS)을 위해 남겨 둔다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 윈도우 세부: DBA_SCHEDULER_WINDOWS 의 duration/enabled/active/next_start_date.
-- 2) 실제 개폐 이력: DBA_SCHEDULER_WINDOW_LOG (OPEN/CLOSE).
-- 3) DATE/TIMESTAMP 비교에는 TO_DATE 로 명시적 변환을 쓴다(ORA-01861 회피).
-- 4) DEFAULT_MAINTENANCE_PLAN 은 ORA$AUTOTASK 그룹으로 유지관리 작업 자원을 통제.

SYS@cdb1> EXIT
