================================================================================
 28장 실습 01 — 외부 테이블(External Table) 생성 개념 확인
 목표: 디렉터리 객체 준비 → ORGANIZATION EXTERNAL 기본 구조 이해
       → 데이터가 DB가 아닌 OS 파일에 남는다는 개념을 눈으로 확인
 환경: CDB=cdb1, PDB=pdb1, OS=Oracle Linux 7 (oracle 계정)
================================================================================

-- [1] PDB1에 SYS로 접속하여 디렉터리 객체를 만든다.
$ sqlplus sys@pdb1 as sysdba

SQL*Plus: Release 19.0.0.0.0 - Production on Mon Aug 10 10:02:11 2026
Copyright (c) 1982, 2019, Oracle.  All rights reserved.

SYS@pdb1> -- 외부 테이블은 OS 파일을 참조한다. 그 위치를 '디렉터리 객체'로 등록한다.
SYS@pdb1> CREATE OR REPLACE DIRECTORY data_dir AS '/u01/app/oracle/extdata';

Directory created.

SYS@pdb1> -- 외부 테이블을 만들고 조회할 HR에게 READ/WRITE 권한을 준다.
SYS@pdb1> -- READ는 데이터 파일 읽기, WRITE는 로그/불량 파일 기록에 필요하다.
SYS@pdb1> GRANT READ, WRITE ON DIRECTORY data_dir TO hr;

Grant succeeded.

SYS@pdb1> EXIT

--------------------------------------------------------------------------------
-- [2] 입력 데이터 파일(dept.csv)을 서버 디렉터리에 만든다.
--------------------------------------------------------------------------------
$ mkdir -p /u01/app/oracle/extdata
$ cat > /u01/app/oracle/extdata/dept.csv <<'EOF'
10,ACCOUNTING,NEW YORK
20,RESEARCH,DALLAS
30,SALES,CHICAGO
40,OPERATIONS,BOSTON
EOF
$ ls -l /u01/app/oracle/extdata/dept.csv
-rw-r--r--. 1 oracle oinstall 66 Aug 10 10:04 /u01/app/oracle/extdata/dept.csv

--------------------------------------------------------------------------------
-- [3] (오류 → 원인 → 수정) 디렉터리 객체 없이 외부 테이블을 만들려다 실패
--------------------------------------------------------------------------------
$ sqlplus hr@pdb1

HR@pdb1> -- 처음엔 실수로 존재하지 않는 디렉터리 이름(EXT_DIR)을 지정했다.
HR@pdb1> CREATE TABLE dept_ext
  2    ( deptno NUMBER(2), dname VARCHAR2(20), loc VARCHAR2(20) )
  3    ORGANIZATION EXTERNAL
  4    ( TYPE ORACLE_LOADER
  5      DEFAULT DIRECTORY ext_dir
  6      ACCESS PARAMETERS
  7      ( RECORDS DELIMITED BY NEWLINE
  8        FIELDS TERMINATED BY ','
  9        ( deptno, dname, loc ) )
 10      LOCATION ('dept.csv') )
 11    REJECT LIMIT UNLIMITED;
  ORGANIZATION EXTERNAL
  *
ERROR at line 3:
ORA-00959: tablespace 'EXT_DIR' does not exist
-- (일부 버전에서는 조회 시점에 ORA-29913/KUP-04040: file dept.csv not found 형태로 드러난다)

HR@pdb1> -- 원인 파악:
HR@pdb1> --   외부 테이블 정의는 만들어질 수 있어도, 참조한 디렉터리 객체 EXT_DIR 가
HR@pdb1> --   존재하지 않으면 조회 시 파일을 열지 못해 실패한다.
HR@pdb1> --   앞 단계에서 만든 디렉터리 객체 이름은 DATA_DIR 였다.
HR@pdb1> SELECT directory_name FROM all_directories WHERE directory_name IN ('EXT_DIR','DATA_DIR');

DIRECTORY_NAME
------------------------------
DATA_DIR

HR@pdb1> -- 수정: 올바른 디렉터리 객체 DATA_DIR 를 사용해 다시 만든다.
HR@pdb1> CREATE TABLE dept_ext
  2    ( deptno NUMBER(2), dname VARCHAR2(20), loc VARCHAR2(20) )
  3    ORGANIZATION EXTERNAL
  4    ( TYPE ORACLE_LOADER
  5      DEFAULT DIRECTORY data_dir
  6      ACCESS PARAMETERS
  7      ( RECORDS DELIMITED BY NEWLINE
  8        FIELDS TERMINATED BY ','
  9        ( deptno, dname, loc ) )
 10      LOCATION ('dept.csv') )
 11    REJECT LIMIT UNLIMITED;

Table created.

--------------------------------------------------------------------------------
-- [4] 외부 테이블을 일반 테이블처럼 조회한다 (SELECT 시점에 파일을 읽는다).
--------------------------------------------------------------------------------
HR@pdb1> SELECT * FROM dept_ext ORDER BY deptno;

    DEPTNO DNAME                LOC
---------- -------------------- --------------------
        10 ACCOUNTING           NEW YORK
        20 RESEARCH             DALLAS
        30 SALES                CHICAGO
        40 OPERATIONS           BOSTON

-- 4행이 조회되었다. 이 데이터는 DB 안에 저장된 것이 아니라 dept.csv 를 읽은 것이다.

--------------------------------------------------------------------------------
-- [5] '데이터가 DB에 없다'는 개념을 세그먼트 조회로 확인한다.
--------------------------------------------------------------------------------
HR@pdb1> -- 일반 테이블과 달리 외부 테이블은 세그먼트를 갖지 않는다.
HR@pdb1> SELECT segment_name, segment_type FROM user_segments
  2       WHERE segment_name = 'DEPT_EXT';

no rows selected

HR@pdb1> -- 대신 외부 테이블 메타데이터는 딕셔너리에 있다.
HR@pdb1> SELECT table_name, type_name, default_directory_name
  2       FROM user_external_tables WHERE table_name = 'DEPT_EXT';

TABLE_NAME   TYPE_NAME       DEFAULT_DIRECTORY_NAME
------------ --------------- ------------------------
DEPT_EXT     ORACLE_LOADER   DATA_DIR

--------------------------------------------------------------------------------
-- [6] 파일을 바꾸면 조회 결과도 바뀐다 = 읽기 전용 '창(窓)'임을 확인.
--------------------------------------------------------------------------------
HR@pdb1> HOST echo '50,MARKETING,SEOUL' >> /u01/app/oracle/extdata/dept.csv

HR@pdb1> SELECT COUNT(*) FROM dept_ext;

  COUNT(*)
----------
         5

-- 파일에 한 줄 추가했더니 별도 적재 없이 조회 결과가 5건으로 늘었다.
-- 외부 테이블은 '조회 시점의 파일 상태'를 그대로 반영한다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 외부 테이블은 CREATE TABLE ... ORGANIZATION EXTERNAL 로 만든다.
-- 2) 참조하는 디렉터리 객체가 반드시 존재해야 하며, READ/WRITE 권한이 필요하다.
-- 3) 데이터는 DB 세그먼트가 아니라 OS 파일에 있으므로 USER_SEGMENTS 에 나타나지 않는다.
-- 4) 메타데이터는 USER_EXTERNAL_TABLES 에 저장된다.
-- 5) 파일이 바뀌면 조회 결과가 바뀐다(읽기 전용, 스냅샷 아님).

HR@pdb1> DROP TABLE dept_ext;

Table dropped.

HR@pdb1> EXIT
$ rm -f /u01/app/oracle/extdata/dept.csv
