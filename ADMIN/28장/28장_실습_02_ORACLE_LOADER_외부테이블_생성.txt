================================================================================
 28장 실습 02 — ORACLE_LOADER 액세스 드라이버로 외부 테이블 생성
 목표: ACCESS PARAMETERS(RECORDS/FIELDS/BADFILE/LOGFILE) 를 직접 작성하고,
       거부 레코드를 만나 원인을 파악한 뒤 파싱 규칙을 수정하여 재조회
 환경: CDB=cdb1, PDB=pdb1, OS=Oracle Linux 7 (oracle 계정)
================================================================================

-- 사전 준비: 디렉터리 객체와 권한은 실습 01에서 만든 DATA_DIR 를 재사용한다.
--            (/u01/app/oracle/extdata, HR 에게 READ/WRITE 부여됨)

--------------------------------------------------------------------------------
-- [1] 입력 CSV(emp.csv) 작성 — 값 안에 쉼표가 있는 직무, 급여 누락 레코드 포함
--------------------------------------------------------------------------------
$ cat > /u01/app/oracle/extdata/emp.csv <<'EOF'
7369,SMITH,CLERK,800,20
7499,ALLEN,"SALES, JR",1600,30
7521,WARD,SALESMAN,,30
7566,JONES,MANAGER,2975,20
7654,MARTIN,SALESMAN,1250,30
EOF
$ cat /u01/app/oracle/extdata/emp.csv
7369,SMITH,CLERK,800,20
7499,ALLEN,"SALES, JR",1600,30
7521,WARD,SALESMAN,,30
7566,JONES,MANAGER,2975,20
7654,MARTIN,SALESMAN,1250,30

-- 2행 직무 'SALES, JR' 안에 쉼표가 있고, 3행은 급여(sal)가 비어 있다.

--------------------------------------------------------------------------------
-- [2] (오류 → 원인 → 수정) 1차 외부 테이블: 인용/누락 처리 없이 생성
--------------------------------------------------------------------------------
$ sqlplus hr@pdb1

HR@pdb1> CREATE TABLE emp_ext
  2    ( empno NUMBER(4), ename VARCHAR2(20), job VARCHAR2(15),
  3      sal NUMBER(7,2), deptno NUMBER(2) )
  4    ORGANIZATION EXTERNAL
  5    ( TYPE ORACLE_LOADER
  6      DEFAULT DIRECTORY data_dir
  7      ACCESS PARAMETERS
  8      ( RECORDS DELIMITED BY NEWLINE
  9        BADFILE data_dir:'emp.bad'
 10        LOGFILE data_dir:'emp.log'
 11        FIELDS TERMINATED BY ','
 12        ( empno, ename, job, sal, deptno ) )
 13      LOCATION ('emp.csv') )
 14    REJECT LIMIT UNLIMITED;

Table created.

HR@pdb1> SELECT empno, ename, job, sal, deptno FROM emp_ext ORDER BY empno;

     EMPNO ENAME      JOB                SAL     DEPTNO
---------- ---------- --------------- ------- ----------
      7369 SMITH      CLERK               800         20
      7566 JONES      MANAGER            2975         20
      7654 MARTIN     SALESMAN           1250         30

-- 5행 중 3행만 조회된다. 7499(ALLEN), 7521(WARD)이 사라졌다.

HR@pdb1> -- 원인 파악: 거부 레코드는 BADFILE 로 빠진다. 로그를 확인한다.
HR@pdb1> EXIT
$ cat /u01/app/oracle/extdata/emp.bad
7499,ALLEN,"SALES, JR",1600,30
7521,WARD,SALESMAN,,30
$ grep -i "error\|KUP-" /u01/app/oracle/extdata/emp.log | head
KUP-04021: field formatting error for field DEPTNO
KUP-04023: field start is after end of record
-- 7499: 'SALES, JR' 안의 쉼표가 구분자로 오인되어 필드가 하나 밀렸다.
-- 7521: sal 이 비어 있어 뒤 필드(deptno) 위치가 어긋났다.

--------------------------------------------------------------------------------
-- [3] 수정: OPTIONALLY ENCLOSED BY 로 인용 처리 + 누락 필드 NULL 처리
--------------------------------------------------------------------------------
$ sqlplus hr@pdb1

HR@pdb1> DROP TABLE emp_ext;

Table dropped.

HR@pdb1> CREATE TABLE emp_ext
  2    ( empno NUMBER(4), ename VARCHAR2(20), job VARCHAR2(15),
  3      sal NUMBER(7,2), deptno NUMBER(2) )
  4    ORGANIZATION EXTERNAL
  5    ( TYPE ORACLE_LOADER
  6      DEFAULT DIRECTORY data_dir
  7      ACCESS PARAMETERS
  8      ( RECORDS DELIMITED BY NEWLINE
  9        BADFILE data_dir:'emp.bad'
 10        LOGFILE data_dir:'emp.log'
 11        FIELDS TERMINATED BY ',' OPTIONALLY ENCLOSED BY '"'
 12        MISSING FIELD VALUES ARE NULL
 13        ( empno, ename, job, sal, deptno ) )
 14      LOCATION ('emp.csv') )
 15    REJECT LIMIT UNLIMITED;

Table created.

HR@pdb1> SELECT empno, ename, job, sal, deptno FROM emp_ext ORDER BY empno;

     EMPNO ENAME      JOB                SAL     DEPTNO
---------- ---------- --------------- ------- ----------
      7369 SMITH      CLERK               800         20
      7499 ALLEN      SALES, JR          1600         30
      7521 WARD       SALESMAN                        30
      7566 JONES      MANAGER            2975         20
      7654 MARTIN     SALESMAN           1250         30

-- 5행 모두 조회된다. 7499의 'SALES, JR' 는 하나의 값으로, 7521의 sal 은 NULL 로 들어갔다.

--------------------------------------------------------------------------------
-- [4] 외부 테이블을 변환·필터하여 내부 테이블로 적재(ETL 패턴)
--------------------------------------------------------------------------------
HR@pdb1> CREATE TABLE emp_stage AS
  2    SELECT empno, INITCAP(ename) AS ename, job,
  3           NVL(sal,0) AS sal, deptno
  4    FROM   emp_ext
  5    WHERE  deptno = 30;

Table created.

HR@pdb1> SELECT empno, ename, sal FROM emp_stage ORDER BY empno;

     EMPNO ENAME          SAL
---------- ---------- -------
      7499 Allen         1600
      7521 Ward             0
      7654 Martin        1250

-- 읽으면서 변환(INITCAP, NVL)하고 필터(deptno=30)하여 한 문장으로 적재했다.

--------------------------------------------------------------------------------
-- [5] DML 은 불가함을 확인 (외부 테이블은 읽기 전용)
--------------------------------------------------------------------------------
HR@pdb1> DELETE FROM emp_ext WHERE deptno = 30;
DELETE FROM emp_ext WHERE deptno = 30
            *
ERROR at line 1:
ORA-30657: operation not supported on external organized table

-- 데이터를 바꾸려면 내부 테이블(emp_stage)에서 DML 을 수행해야 한다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) OPTIONALLY ENCLOSED BY '"' 는 값 내부 구분자(쉼표)를 무시하게 한다.
-- 2) MISSING FIELD VALUES ARE NULL 은 누락된 후행 필드를 NULL 로 채워 거부를 막는다.
-- 3) 거부 레코드는 BADFILE 로, 사유는 LOGFILE 로 확인한다.
-- 4) 외부 테이블은 SELECT 로 변환·필터하며 CTAS 로 내부 테이블에 적재할 수 있다.
-- 5) 외부 테이블에 DML 을 시도하면 ORA-30657 이 발생한다(읽기 전용).

HR@pdb1> DROP TABLE emp_stage PURGE;
Table dropped.
HR@pdb1> DROP TABLE emp_ext;
Table dropped.
HR@pdb1> EXIT
$ rm -f /u01/app/oracle/extdata/emp.csv /u01/app/oracle/extdata/emp.bad /u01/app/oracle/extdata/emp.log
