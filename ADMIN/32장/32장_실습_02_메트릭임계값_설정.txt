================================================================================
 32장 실습 02 — 메트릭 임계값(Metric Thresholds) 설정
 목표: DBMS_SERVER_ALERT.SET_THRESHOLD로 경고/위험 임계값을 설정하고,
       GET_THRESHOLD/DBA_THRESHOLDS로 확인하며, 임계값을 초기화한다.
 환경: CDB=cdb1 (루트에서 SYS로 수행)
================================================================================

-- [1] 특정 테이블스페이스(USERS)에 대해 사용률 임계값을 개별 지정한다.
--     기본값(85/97) 대신 이 테이블스페이스만 80/90 으로 더 엄격하게 감시한다.
$ sqlplus sys@cdb1 as sysdba

SYS@cdb1> BEGIN
  2         DBMS_SERVER_ALERT.SET_THRESHOLD(
  3           metrics_id        => DBMS_SERVER_ALERT.TABLESPACE_PCT_FULL,
  4           warning_operator  => DBMS_SERVER_ALERT.OPERATOR_GE,
  5           warning_value     => '80',
  6           critical_operator => DBMS_SERVER_ALERT.OPERATOR_GE,
  7           critical_value    => '90',
  8           observation_period      => 1,
  9           consecutive_occurrences => 1,
 10           instance_name     => NULL,
 11           object_type       => DBMS_SERVER_ALERT.OBJECT_TYPE_TABLESPACE,
 12           object_name       => 'USERS');
 13       END;
 14       /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [2] 설정된 임계값을 DBA_THRESHOLDS 로 확인한다.
--------------------------------------------------------------------------------
SYS@cdb1> COLUMN metrics_name FORMAT A26
SYS@cdb1> SELECT metrics_name, warning_value, critical_value, object_name
  2       FROM  dba_thresholds
  3      WHERE  object_name = 'USERS';

METRICS_NAME               WARNING_VALUE CRITICAL_VALUE OBJECT_NAME
-------------------------- ------------- -------------- -----------
Tablespace Space Usage     80            90             USERS

-- USERS 테이블스페이스만 80/90 으로 개별 감시된다.

--------------------------------------------------------------------------------
-- [3] GET_THRESHOLD 로 프로그램적으로 현재 임계값을 조회한다.
--------------------------------------------------------------------------------
SYS@cdb1> SET SERVEROUTPUT ON
SYS@cdb1> DECLARE
  2         w VARCHAR2(60); wo NUMBER;
  3         c VARCHAR2(60); co NUMBER;
  4         op NUMBER; oc NUMBER;
  5       BEGIN
  6         DBMS_SERVER_ALERT.GET_THRESHOLD(
  7           metrics_id => DBMS_SERVER_ALERT.TABLESPACE_PCT_FULL,
  8           warning_operator=>wo, warning_value=>w,
  9           critical_operator=>co, critical_value=>c,
 10           observation_period=>op, consecutive_occurrences=>oc,
 11           instance_name=>NULL,
 12           object_type=>DBMS_SERVER_ALERT.OBJECT_TYPE_TABLESPACE,
 13           object_name=>'USERS');
 14         DBMS_OUTPUT.PUT_LINE('warning='||w||'  critical='||c);
 15       END;
 16       /
warning=80  critical=90

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [4] (오류 → 원인 → 수정)
--     경고/위험 값을 숫자 리터럴로 전달하려다 타입 불일치 오류를 만난다.
--     SET_THRESHOLD 의 value 인자는 VARCHAR2(문자열)이어야 한다.
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2         DBMS_SERVER_ALERT.SET_THRESHOLD(
  3           metrics_id        => DBMS_SERVER_ALERT.TABLESPACE_PCT_FULL,
  4           warning_operator  => DBMS_SERVER_ALERT.OPERATOR_GE,
  5           warning_value     => 80,          -- 숫자로 전달 (오류 유발)
  6           critical_operator => DBMS_SERVER_ALERT.OPERATOR_GE,
  7           critical_value    => 90,
  8           observation_period=>1, consecutive_occurrences=>1,
  9           instance_name=>NULL,
 10           object_type=>DBMS_SERVER_ALERT.OBJECT_TYPE_TABLESPACE,
 11           object_name=>'USERS');
 12       END;
 13       /
         warning_value     => 80,
                              *
ERROR at line 5:
ORA-06550: line 5, column 30:
PLS-00306: wrong number or types of arguments in call to 'SET_THRESHOLD'

SYS@cdb1> -- 원인 파악: warning_value/critical_value 는 VARCHAR2 타입이다.
SYS@cdb1> -- 숫자 80 은 암묵적 변환 규칙상 오버로드를 찾지 못해 PLS-00306 이 난다.
SYS@cdb1> -- 수정: 반드시 작은따옴표로 감싼 문자열('80','90')로 전달한다.
SYS@cdb1> BEGIN
  2         DBMS_SERVER_ALERT.SET_THRESHOLD(
  3           metrics_id        => DBMS_SERVER_ALERT.TABLESPACE_PCT_FULL,
  4           warning_operator  => DBMS_SERVER_ALERT.OPERATOR_GE,
  5           warning_value     => '80',
  6           critical_operator => DBMS_SERVER_ALERT.OPERATOR_GE,
  7           critical_value    => '90',
  8           observation_period=>1, consecutive_occurrences=>1,
  9           instance_name=>NULL,
 10           object_type=>DBMS_SERVER_ALERT.OBJECT_TYPE_TABLESPACE,
 11           object_name=>'USERS');
 12       END;
 13       /

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [5] 임계값을 데이터베이스 기본값으로 되돌린다(값을 NULL 로 지정).
--------------------------------------------------------------------------------
SYS@cdb1> BEGIN
  2         DBMS_SERVER_ALERT.SET_THRESHOLD(
  3           metrics_id        => DBMS_SERVER_ALERT.TABLESPACE_PCT_FULL,
  4           warning_operator  => DBMS_SERVER_ALERT.OPERATOR_GE,
  5           warning_value     => NULL,
  6           critical_operator => DBMS_SERVER_ALERT.OPERATOR_GE,
  7           critical_value    => NULL,
  8           observation_period=>1, consecutive_occurrences=>1,
  9           instance_name=>NULL,
 10           object_type=>DBMS_SERVER_ALERT.OBJECT_TYPE_TABLESPACE,
 11           object_name=>'USERS');
 12       END;
 13       /

PL/SQL procedure successfully completed.

-- USERS 개별 임계값이 제거되어 다시 데이터베이스 기본(85/97)으로 감시된다.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) DBMS_SERVER_ALERT.SET_THRESHOLD 로 경고/위험 두 단계 임계값을 설정한다.
-- 2) value 인자는 반드시 VARCHAR2(문자열)로 전달한다(PLS-00306 방지).
-- 3) 설정 확인은 DBA_THRESHOLDS 또는 GET_THRESHOLD 로 한다.
-- 4) warning_value/critical_value 를 NULL 로 주면 기본값으로 되돌아간다.

SYS@cdb1> EXIT
