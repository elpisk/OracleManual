================================================================================
 4장 실습 02 — Undo 보존과 ORA-01555 재현 → UNDO_RETENTION·테이블스페이스 조정
 목표: 짧은 보존·NOGUARANTEE 상태에서 '긴 조회 + 동시 대량 변경'으로
       ORA-01555 snapshot too old를 재현한다. 원인을 조회 길이·동시 변경량·
       Undo 공간의 함수로 규명하고, UNDO_RETENTION 상향과 RETENTION GUARANTEE로
       해소한 뒤 V$UNDOSTAT로 재발이 없음을 확인한다.
 환경: non-CDB=orcl 19.3. 구성 변경은 SYS, 부하는 SQLT 터미널 2개(A: 긴 조회,
       B: 대량 변경). 대상은 복제 테이블 REVIEW_LOG_WL(원본 훼손 금지).
       재현을 확실히 하려고 16MB 짜리 작은 Undo 테이블스페이스를 임시로 쓴다.
 주의: 개인 실습 환경 전용. 파라미터·테이블스페이스 변경이 포함된다. 마지막에
       변경값을 원복하는 절차를 반드시 수행한다. 공용 DB에서 수행하지 않는다.
================================================================================

$ sqlplus / as sysdba

--------------------------------------------------------------------------------
-- [1] (조치 전 상태) 현재 Undo 구성을 캡처한다. 근거 없는 조치를 경계한다.
--------------------------------------------------------------------------------
SYS@orcl> COLUMN tablespace_name FORMAT A16

SYS@orcl> COLUMN contents FORMAT A10

SYS@orcl> COLUMN retention FORMAT A12

SYS@orcl> SELECT tablespace_name, contents, retention
  2       FROM  dba_tablespaces WHERE contents='UNDO';

TABLESPACE_NAME  CONTENTS   RETENTION
---------------- ---------- ------------
UNDOTBS1         UNDO       NOGUARANTEE

SYS@orcl> COLUMN value FORMAT A10

SYS@orcl> SELECT value FROM v$parameter WHERE name='undo_retention';

VALUE
----------
900

SYS@orcl> SELECT tablespace_name, ROUND(SUM(bytes)/1024/1024) mb
  2       FROM  dba_data_files WHERE tablespace_name='UNDOTBS1'
  3       GROUP BY tablespace_name;

TABLESPACE_NAME          MB
---------------- ----------
UNDOTBS1               1055

-- 조치 전 요약: RETENTION=NOGUARANTEE, UNDO_RETENTION=3600(이 랩의 현재값; 기본은 900),
--   Undo 1,055MB(자동 확장). 이 크기에서는 ORA-01555 를 만나기 어려우므로, 아래 [2]에서
--   '작은 Undo + 짧은 보존'이라는 조건을 일부러 만든다.

--------------------------------------------------------------------------------
-- [2] (재현 준비) 작은 Undo 테이블스페이스로 바꾸고 보존을 아주 짧게 낮춘다(개인 실습 전용).
--     자동 확장이 꺼진 16MB Undo = 공간 상한이 곧 '과거 이미지가 덮어써지는 속도'다.
--------------------------------------------------------------------------------
SYS@orcl> CREATE UNDO TABLESPACE undo_small
  2       DATAFILE '/u03/oradata/ORCL/undo_small01.dbf' SIZE 16M AUTOEXTEND OFF;

Tablespace created.

SYS@orcl> ALTER SYSTEM SET undo_tablespace = undo_small SCOPE=MEMORY;

System altered.

SYS@orcl> ALTER SYSTEM SET undo_retention = 1 SCOPE=MEMORY;

System altered.

SYS@orcl> -- 재현용 복제 테이블(원본 REVIEW_LOG 를 훼손하지 않는다). 50만 행, 약 36MB.
SYS@orcl> CREATE TABLE sqlt.review_log_wl AS SELECT * FROM sqlt.review_log;

Table created.

--------------------------------------------------------------------------------
-- [3] (이상 징후 발생) 세션 A 에서 '느린 긴 조회'를 시작한다.
--     1만 행마다 1초씩 쉬는 커서 루프 = 약 50초 동안 표를 천천히 훑는 조회.
--     (읽기 시작 시점의 이미지로 끝까지 읽어야 하므로, 뒤쪽 블록을 읽을 때는
--      그 사이 바뀐 블록의 과거 이미지를 Undo 에서 되살려야 한다.)
--------------------------------------------------------------------------------
-- (터미널 A) 긴 조회
$ sqlplus sqlt/sqlt@orcl

SQLT@orcl> SET SERVEROUTPUT ON

SQLT@orcl> DECLARE
  2         n NUMBER := 0;
  3       BEGIN
  4         FOR r IN (SELECT log_id FROM review_log_wl) LOOP
  5           n := n + 1;
  6           IF MOD(n, 10000) = 0 THEN DBMS_SESSION.SLEEP(1); END IF;
  7         END LOOP;
  8         DBMS_OUTPUT.PUT_LINE('완주: ' || n || ' 행');
  9       END;
 10       /

DECLARE
*
ERROR at line 1:
ORA-01555: snapshot too old: rollback segment number  with name "" too small
ORA-06512: at line 4
ORA-06512: at line 4

-- (A 가 도는 동안, 다른 터미널에서 세션 B 를 실행한다)

-- (터미널 B) 동시 대량 변경 + 잦은 커밋 — 같은 표를 200조각으로 나눠 갱신·커밋(약 40초)
$ sqlplus sqlt/sqlt@orcl

SQLT@orcl> BEGIN
  2         FOR i IN 0..199 LOOP
  3           UPDATE review_log_wl SET action_msg = 'WL-' || i WHERE MOD(log_id, 200) = i;
  4           COMMIT;                                          -- Undo 빠른 순환
  5         END LOOP;
  6       END;
  7       /

PL/SQL procedure successfully completed.

-- (터미널 A) 로 돌아오면 — 조회가 중간에 죽어 있다.
-- ▶ 대기: 터미널 A

-- 이상 징후: 세션 A 의 조회가 ORA-01555 로 중단됐다. A 의 SQL 자체는 문제없다.
--   (PL/SQL 안에서 난 ORA-01555 는 세그먼트 번호·이름이 비어 보인다.)

--------------------------------------------------------------------------------
-- [4] (원인 파악) 왜 났는가 — V$UNDOSTAT 로 근거를 확인한다.
--------------------------------------------------------------------------------
$ sqlplus / as sysdba

SYS@orcl> SELECT TO_CHAR(begin_time,'HH24:MI') begin, maxquerylen mql,
  2         tuned_undoretention tuned, ssolderrcnt ora1555, nospaceerrcnt nospace,
  3         undoblks
  4       FROM  v$undostat
  5       ORDER BY begin_time DESC FETCH FIRST 3 ROWS ONLY;

BEGIN        MQL      TUNED    ORA1555    NOSPACE   UNDOBLKS
----- ---------- ---------- ---------- ---------- ----------
19:14       1093       1754          1          0       6793
19:04        791       1572          2          0      30206
18:54        187       3600          0          0        331

-- 원인 확정(맨 윗줄 = 현재 10분 구간):
--   SSOLDERRCNT ≥ 1 → 이 구간에 ORA-01555 가 발생했다.
--   UNDOBLKS(이 구간에 소비한 Undo 블록) × 8KB 가 16MB 를 훌쩍 넘는다(이번 실측 6,793블록 ≈ 53MB)
--   → 세션 A 가 시작 시점 이미지를 되살리려 찾은 과거 블록이 이미 여러 바퀴 덮어써졌다.
--   TUNED_UNDORETENTION 은 구간 안의 최대값이라 UNDOTBS1 시절 값(3600)이 섞여 크게 보일 수
--   있다 — 고정 크기 16MB Undo 에서는 보존 시간이 얼마든 '공간'이 먼저 바닥난다.
--   => '조회 길이 > 실제 보존' + '동시 변경이 Undo 를 덮어씀'의 전형적 결과다.
--      SQL 을 고칠 문제가 아니라 Undo 구성 문제임이 수치로 확인된다.

--------------------------------------------------------------------------------
-- [5] (조치) Undo 공간을 넉넉히 확보하고, 보존을 조회 길이보다 길게 늘려 보장 모드로 전환한다.
--------------------------------------------------------------------------------
SYS@orcl> ALTER SYSTEM SET undo_tablespace = undotbs1 SCOPE=MEMORY;   -- 1GB·자동 확장 Undo 로 복귀

System altered.

SYS@orcl> ALTER SYSTEM SET undo_retention = 3600 SCOPE=MEMORY;        -- 1시간 > 조회 길이

System altered.

SYS@orcl> ALTER TABLESPACE undotbs1 RETENTION GUARANTEE;              -- 보존 보장

Tablespace altered.

--------------------------------------------------------------------------------
-- [6] (조치 후 상태 + 재확인) 같은 재현을 다시 돌린다.
--------------------------------------------------------------------------------

  2       WHERE contents='UNDO';

------------------------------ -----------

UNDO_SMALL                     NOGUARANTEE

-- 재확인: 조치 후에는 A 가 '완주: 500000 행'으로 끝났다. 1GB Undo 에 UNDO_RETENTION=3600

-- [정리]

-- 1) ORA-01555는 Undo 세그먼트가 '작아서'가 아니라, 긴 조회가 필요로 하는
--    과거 이미지를 동시 변경이 덮어써서 난다. SQL 문제가 아니다.
-- 2) 근거는 V$UNDOSTAT: MAXQUERYLEN > (TUNED_)UNDORETENTION 이고 SSOLDERRCNT>0.
-- 3) 조치는 셋 중 하나 이상: Undo 확장 / UNDO_RETENTION↑ / RETENTION GUARANTEE.
--    GUARANTEE는 강력하지만 공간 부족 시 DML이 ORA-30036으로 막히므로,
--    Undo 크기가 최대 DML을 감당하는지 먼저 확인한다.
-- 4) 조치 전→재현→원인(수치)→조치→재확인→원복의 순서를 그대로 남긴다.
-- ※ V$UNDOSTAT·DBA_TABLESPACES 조회는 관리 팩 라이선스가 필요 없다.

SYS@orcl> EXIT
