================================================================================
 7장 실습 06 — 통계 Preferences 설정과 적용 범위 확인 (ADMIN 33장 심화)
 목표: SET_TABLE_PREFS로 테이블 단위 통계 정책을 바꾸고, GET_PREFS로 계층
       우선순위(TABLE > SCHEMA > DATABASE > GLOBAL)를 확인한 뒤 원복한다.
 환경: non-CDB=orcl 19.3 (SYS로 수행 — SCHEMA/DATABASE Prefs 확인 포함)
 주의: ★개인 실습 환경 전용★. Preferences는 자동 통계 수집 동작을 바꾼다.
       실습 끝에 DELETE_TABLE_PREFS로 반드시 원복한다.
================================================================================

$ sqlplus / as sysdba

SYS@orcl> SET LINESIZE 200
SYS@orcl> SET PAGESIZE 100

--------------------------------------------------------------------------------
-- [1] (조치 전 상태) MEDICAL_CLAIMS의 현재 실효 Preferences를 기록한다.
--------------------------------------------------------------------------------
SYS@orcl> SELECT
  2    DBMS_STATS.GET_PREFS('METHOD_OPT','SQLT','MEDICAL_CLAIMS') method_opt,
  3    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SQLT','MEDICAL_CLAIMS') est,
  4    DBMS_STATS.GET_PREFS('STALE_PERCENT','SQLT','MEDICAL_CLAIMS') stale
  5    FROM dual;

METHOD_OPT                 EST                          STALE
-------------------------- ---------------------------- -----
FOR ALL COLUMNS SIZE AUTO  DBMS_STATS.AUTO_SAMPLE_SIZE  10

-- 해석: 테이블에 별도 설정이 없으면 상위 계층(GLOBAL 등)의 값이 실효값으로
--   나타난다. METHOD_OPT=AUTO, EST=AUTO_SAMPLE_SIZE, STALE_PERCENT=10(기본).
--   이 세 값이 Before 상태다.

--------------------------------------------------------------------------------
-- [2] (조치) 테이블 단위 Preferences를 변경한다. (TABLE 범위 = 최우선)
--------------------------------------------------------------------------------
SYS@orcl> EXEC DBMS_STATS.SET_TABLE_PREFS('SQLT','MEDICAL_CLAIMS', -
>            'ESTIMATE_PERCENT','100');

PL/SQL procedure successfully completed.

SYS@orcl> EXEC DBMS_STATS.SET_TABLE_PREFS('SQLT','MEDICAL_CLAIMS', -
>            'STALE_PERCENT','5');

PL/SQL procedure successfully completed.

SYS@orcl> EXEC DBMS_STATS.SET_TABLE_PREFS('SQLT','MEDICAL_CLAIMS', -
>            'METHOD_OPT', -
>            'FOR ALL COLUMNS SIZE AUTO FOR COLUMNS SIZE 254 HOSP_ID');

PL/SQL procedure successfully completed.

SYS@orcl> -- 재확인: 실효값이 바뀌었는지 본다.
SYS@orcl> SELECT
  2    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SQLT','MEDICAL_CLAIMS') est,
  3    DBMS_STATS.GET_PREFS('STALE_PERCENT','SQLT','MEDICAL_CLAIMS') stale
  4    FROM dual;

EST   STALE
----- -----
100   5

-- 해석: 테이블 설정이 상위 계층을 이겨 EST=100, STALE=5로 실효값이 바뀌었다.

--------------------------------------------------------------------------------
-- [3] (검증) 우선순위 확인 — 같은 항목의 DATABASE 값과 TABLE 값을 대조한다.
--------------------------------------------------------------------------------
SYS@orcl> -- 인자 없이 GET_PREFS를 부르면 상위(GLOBAL/DATABASE) 값이 나온다.
SYS@orcl> SELECT
  2    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT') global_or_db,
  3    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SQLT','MEDICAL_CLAIMS') tbl
  4    FROM dual;

GLOBAL_OR_DB                 TBL
---------------------------- -----
DBMS_STATS.AUTO_SAMPLE_SIZE  100

-- 해석: 상위 계층은 여전히 AUTO_SAMPLE_SIZE지만, 테이블 실효값은 100이다.
--   좁은 범위(TABLE)가 넓은 범위(GLOBAL/DATABASE)를 이긴다는 우선순위 확인.

--------------------------------------------------------------------------------
-- [4] (이상 징후 → 원인 → 조치 → 재확인)
--     이상 징후: SET_SCHEMA_PREFS로 스키마 전체를 바꿨더니 특정 테이블만
--       기대와 다른 값이 나온다.
--------------------------------------------------------------------------------
SYS@orcl> EXEC DBMS_STATS.SET_SCHEMA_PREFS('SQLT','ESTIMATE_PERCENT','50');

PL/SQL procedure successfully completed.

SYS@orcl> SELECT
  2    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SQLT','MEDICAL_CLAIMS') mc,
  3    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SQLT','PATIENTS') pt
  4    FROM dual;

MC    PT
----- -----
100   50

-- 이상 징후: 같은 스키마인데 MEDICAL_CLAIMS는 100, PATIENTS는 50이다.
-- 원인: MEDICAL_CLAIMS에는 TABLE 범위 설정(100)이 있어 SCHEMA 설정(50)을
--   덮어쓴다. PATIENTS는 TABLE 설정이 없어 SCHEMA 값(50)을 따른다.
--   즉 계층 우선순위(TABLE > SCHEMA)에 따른 정상 동작이다.
-- 조치: 실습이 끝났으므로 SCHEMA·TABLE 설정을 모두 원복한다.
SYS@orcl> EXEC DBMS_STATS.DELETE_SCHEMA_PREFS('SQLT','ESTIMATE_PERCENT');

PL/SQL procedure successfully completed.

--------------------------------------------------------------------------------
-- [5] (조치) 실습 원복 — 테이블 Preferences를 삭제해 기본으로 되돌린다.
--------------------------------------------------------------------------------
SYS@orcl> EXEC DBMS_STATS.DELETE_TABLE_PREFS('SQLT','MEDICAL_CLAIMS', -
>            'ESTIMATE_PERCENT');
PL/SQL procedure successfully completed.
SYS@orcl> EXEC DBMS_STATS.DELETE_TABLE_PREFS('SQLT','MEDICAL_CLAIMS', -
>            'STALE_PERCENT');
PL/SQL procedure successfully completed.
SYS@orcl> EXEC DBMS_STATS.DELETE_TABLE_PREFS('SQLT','MEDICAL_CLAIMS', -
>            'METHOD_OPT');
PL/SQL procedure successfully completed.

SYS@orcl> -- 재확인: Before 값으로 돌아왔는지 확인한다.
SYS@orcl> SELECT
  2    DBMS_STATS.GET_PREFS('METHOD_OPT','SQLT','MEDICAL_CLAIMS') method_opt,
  3    DBMS_STATS.GET_PREFS('ESTIMATE_PERCENT','SQLT','MEDICAL_CLAIMS') est,
  4    DBMS_STATS.GET_PREFS('STALE_PERCENT','SQLT','MEDICAL_CLAIMS') stale
  5    FROM dual;

METHOD_OPT                 EST                          STALE
-------------------------- ---------------------------- -----
FOR ALL COLUMNS SIZE AUTO  DBMS_STATS.AUTO_SAMPLE_SIZE  10

-- 재확인 완료: [1]의 Before 상태로 원복됨.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) Preferences는 GLOBAL·DATABASE·SCHEMA·TABLE 네 계층, 좁은 범위가 우선.
-- 2) SET_TABLE_PREFS로 테이블만 정책을 두고, GET_PREFS(스키마,테이블)로 실효값 확인.
-- 3) TABLE 설정은 SCHEMA/DATABASE/GLOBAL 값을 덮어쓴다(우선순위 검증).
-- 4) 자동 통계 수집도 이 Preferences를 따르므로 변경은 신중히, 원복은 필수.
-- ※ 원복 완료: DELETE_TABLE_PREFS·DELETE_SCHEMA_PREFS로 기본값 복원.

SYS@orcl> EXIT
