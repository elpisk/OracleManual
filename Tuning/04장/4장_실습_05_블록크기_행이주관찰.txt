================================================================================
 4장 실습 05 — 블록 크기와 행 이주(row migration) 관찰
 목표: DB_BLOCK_SIZE(8192)를 확인하고, 큰 PCTFREE 없이 만든 테이블에 대량
       UPDATE로 행을 키워 행 이주를 유발한다. ANALYZE ... LIST CHAINED ROWS와
       DBA_TABLES.CHAIN_CNT로 이주 행을 찾아내고, PCTFREE 상향 + MOVE + 인덱스
       REBUILD로 해소해 이주가 사라졌음을 확인한다.
 환경: non-CDB=orcl 19.3 (부하·객체는 SQLT, 통계·조회는 SYS 또는 SQLT)
 주의: 개인 실습 환경 전용. 재현용 테이블(MIG_WL)만 사용하며 원본을 훼손하지
       않는다. 마지막에 재현 객체를 제거한다. 공용 DB에서 수행하지 않는다.
================================================================================

$ sqlplus / as sysdba

--------------------------------------------------------------------------------
-- [1] (조치 전 상태) 블록 크기를 확인한다. 8KB가 이주 관찰의 전제다.
--------------------------------------------------------------------------------
SYS@orcl> SELECT value FROM v$parameter WHERE name='db_block_size';
VALUE
----------
8192

-- 해석: 8KB 블록. 한 블록에 들어갈 수 있는 데이터 양의 상한이 정해진다.
--   UPDATE로 행이 커져 남은 자리를 넘으면 다른 블록으로 '이주'한다.

$ sqlplus sqlt/sqlt

--------------------------------------------------------------------------------
-- [2] (재현 준비) PCTFREE를 아주 낮게(1) 준 테이블을 만들고, 처음엔 작은
--     행으로 블록을 빽빽이 채운다. 나중 UPDATE가 자리를 못 찾게 한다.
--------------------------------------------------------------------------------
SQLT@orcl> CREATE TABLE mig_wl (
  2         id     NUMBER PRIMARY KEY,
  3         v      VARCHAR2(4000))
  4       PCTFREE 1;                       -- 예약 공간 거의 없음
Table created.

SQLT@orcl> INSERT INTO mig_wl
  2       SELECT LEVEL, RPAD('x', 30, 'x')  -- 처음엔 30바이트짜리 작은 행
  3       FROM   dual CONNECT BY LEVEL <= 50000;
50000 rows created.
SQLT@orcl> COMMIT;

SQLT@orcl> EXEC DBMS_STATS.GATHER_TABLE_STATS(USER,'MIG_WL');
PL/SQL procedure successfully completed.

SQLT@orcl> SELECT chain_cnt, num_rows, blocks
  2       FROM  user_tables WHERE table_name='MIG_WL';
 CHAIN_CNT   NUM_ROWS     BLOCKS
---------- ---------- ----------
         0      50000        <환경값>

-- 조치 전 요약: CHAIN_CNT=0. 아직 이주 행이 없다(작은 행으로 꽉 채운 상태).

--------------------------------------------------------------------------------
-- [3] (이상 징후 유발) 모든 행을 크게 UPDATE한다. 30바이트 → 3000바이트.
--     원래 블록에 자리가 없으니 대거 이주가 일어난다.
--------------------------------------------------------------------------------
SQLT@orcl> UPDATE mig_wl SET v = RPAD('y', 3000, 'y');   -- 행을 100배로 키움
50000 rows updated.
SQLT@orcl> COMMIT;

SQLT@orcl> EXEC DBMS_STATS.GATHER_TABLE_STATS(USER,'MIG_WL');
PL/SQL procedure successfully completed.

SQLT@orcl> SELECT chain_cnt, num_rows, blocks
  2       FROM  user_tables WHERE table_name='MIG_WL';
 CHAIN_CNT   NUM_ROWS     BLOCKS
---------- ---------- ----------
     <환경값>      50000        <환경값>

-- 이상 징후: CHAIN_CNT가 0에서 크게 증가했다(전체 5만 행 대비 상당수).
--   커진 행이 원 블록에 안 들어가 이주된 것이다.

--------------------------------------------------------------------------------
-- [4] (원인 파악) 이주 행을 실제로 찾아낸다. ANALYZE ... LIST CHAINED ROWS.
--------------------------------------------------------------------------------
SQLT@orcl> @?/rdbms/admin/utlchain.sql        -- CHAINED_ROWS 결과 테이블 생성
Table created.

SQLT@orcl> ANALYZE TABLE mig_wl LIST CHAINED ROWS INTO chained_rows;
Table analyzed.

SQLT@orcl> SELECT COUNT(*) chained_cnt FROM chained_rows
  2       WHERE table_name='MIG_WL';
CHAINED_CNT
-----------
      <환경값>

-- 원인 확정: chained_rows에 이주된 행의 ROWID가 담겼다. 이 행들은 3000바이트로
--   8KB 블록의 원 자리에 못 들어가 이동했고, 원 자리엔 포인터만 남았다.
--   행 크기(3000B)가 블록(8192B)보다 작으므로 '체이닝'이 아니라 '이주'다.
--   => 인덱스로 이 행을 읽으면 원 블록→포인터→새 블록으로 한 번 더 점프한다.

--------------------------------------------------------------------------------
-- [5] (조치) PCTFREE를 높이고 테이블을 재구성(MOVE)해 이주를 해소한다.
--------------------------------------------------------------------------------
SQLT@orcl> ALTER TABLE mig_wl PCTFREE 30;      -- 앞으로의 UPDATE 여유 확보
Table altered.
SQLT@orcl> ALTER TABLE mig_wl MOVE;            -- 재구성(이미 커진 행을 재배치)
Table altered.

-- 주의: MOVE는 ROWID를 바꾸므로 인덱스가 UNUSABLE이 된다. 반드시 리빌드한다.
SQLT@orcl> SELECT index_name, status FROM user_indexes
  2       WHERE table_name='MIG_WL';
INDEX_NAME          STATUS
------------------- --------
SYS_C00XXXXX        UNUSABLE

SQLT@orcl> ALTER INDEX SYS_C00XXXXX REBUILD;
Index altered.

--------------------------------------------------------------------------------
-- [6] (조치 후 상태 + 재확인) 통계 재수집 후 이주가 사라졌는지 확인한다.
--------------------------------------------------------------------------------
SQLT@orcl> EXEC DBMS_STATS.GATHER_TABLE_STATS(USER,'MIG_WL');
PL/SQL procedure successfully completed.

SQLT@orcl> SELECT chain_cnt FROM user_tables WHERE table_name='MIG_WL';
 CHAIN_CNT
----------
         0

SQLT@orcl> TRUNCATE TABLE chained_rows;
SQLT@orcl> ANALYZE TABLE mig_wl LIST CHAINED ROWS INTO chained_rows;
SQLT@orcl> SELECT COUNT(*) FROM chained_rows WHERE table_name='MIG_WL';
  COUNT(*)
----------
         0

-- 재확인: CHAIN_CNT=0, chained_rows도 0건. 이주가 해소됐다.
--   PCTFREE 30으로 재구성하니 커진 행이 각자 블록에 자리를 잡았다.

--------------------------------------------------------------------------------
-- [7] (원복) 재현 객체를 정리한다.
--------------------------------------------------------------------------------
SQLT@orcl> DROP TABLE mig_wl PURGE;
Table dropped.
SQLT@orcl> DROP TABLE chained_rows PURGE;
Table dropped.

--------------------------------------------------------------------------------
-- [정리]
--------------------------------------------------------------------------------
-- 1) 행 이주는 UPDATE로 커진 행이 원 블록에 안 들어가 이동하고 포인터만 남는
--    현상이다. 인덱스 조회 시 한 번 더 점프해 논리·물리 읽기가 는다.
-- 2) 찾는 법: DBA/USER_TABLES.CHAIN_CNT(이주+체이닝 합산) + ANALYZE ...
--    LIST CHAINED ROWS(실제 ROWID 목록). 행 크기 < 블록 크기이면 이주다.
-- 3) 해소: PCTFREE 상향 → ALTER TABLE ... MOVE → 인덱스 REBUILD.
--    MOVE는 ROWID를 바꿔 인덱스를 UNUSABLE로 만들므로 리빌드가 필수다.
-- 4) 예방이 최선이다: UPDATE로 커지는 테이블은 생성 시 PCTFREE를 넉넉히 준다.
--    행이 블록보다 커서 나는 '체이닝'은 PCTFREE로 못 막고 더 큰 블록 TS가 답이다.
-- ※ 이 작업들은 관리 팩 라이선스가 필요 없다.

SQLT@orcl> EXIT
