--------------------------------------------------------------------------------
 [1] 조회 전용 속성이란
--------------------------------------------------------------------------------

-- ★ 13장의 SHOW DATABASE VERBOSE 출력에서 (monitor) 로 표시된 것들이 있었다.
--
--     InconsistentProperties          = '(monitor)'
--     InconsistentLogXptProps         = '(monitor)'
--     LogXptStatus                    = '(monitor)'
--     SendQEntries                    = '(monitor)'
--     RecvQEntries                    = '(monitor)'
--     TopWaitEvents                   = '(monitor)'
--     SidName                         = '(monitor)'
--
-- ★ 값을 설정하는 것이 아니라 조회하면 계산해 주는 속성이다.
--   이름을 지정해 하나씩 본다.

--------------------------------------------------------------------------------
 [2] 상태 보고
--------------------------------------------------------------------------------

DGMGRL> show database 'boston' 'StatusReport'

STATUS REPORT
       INSTANCE_NAME   SEVERITY ERROR_TEXT

-- ★ 비어 있으면 정상이다. 머리글만 나오고 행이 없다.
--
-- ★ 문제가 있으면 인스턴스별 오류가 나열된다.
--   13장에서 ORA-16766 이 났을 때는 여기에 내용이 있었을 것이다.
--
-- ★ SHOW DATABASE 의 Database Error(s) / Database Warning(s) 와
--   같은 정보를 표 형태로 준다.
--   RAC 처럼 인스턴스가 여럿이면 어느 인스턴스인지 구별할 수 있다.

--------------------------------------------------------------------------------
 [3] 전송 상태
--------------------------------------------------------------------------------

DGMGRL> show database 'chicago' 'LogXptStatus'

LOG TRANSPORT STATUS
PRIMARY_INSTANCE_NAME STANDBY_DATABASE_NAME     STATUS      ERROR
             chicago               boston      VALID

-- ★ Primary 에서 조회한다. Standby 에서 조회하면 보낼 곳이 없다.
--
-- ★ 어느 Standby 로 보내고 있고 그 상태가 어떤지 알려 준다.
--   V$ARCHIVE_DEST_STATUS 의 STATUS·ERROR 에 대응한다.
--
-- ★ 구성원이 여럿이면 행이 여러 개 나온다.
--   10장의 Far Sync 구성이라면 farsync 와 boston 이 각각 나온다.

--------------------------------------------------------------------------------
 [4] 두 개의 큐
--------------------------------------------------------------------------------

DGMGRL> show database 'chicago' 'SendQEntries'

PRIMARY_SEND_QUEUE
  STANDBY_NAME  STATUS   RESETLOGS_ID  THREAD  LOG_SEQ  TIME_GENERATED       FIRST_CHANGE#  SIZE (KBs)
                CURRENT    1242322892       1       69  08/27/2026 18:56:36        2264780        6911

-- ★ 보낼 것이 남아 있는가를 보여 준다.
--
--   STATUS = CURRENT     지금 쓰고 있는 로그. 정상이다
--   RESETLOGS_ID         5장부터 계속 본 그 값
--   LOG_SEQ 69           현재 시퀀스
--   SIZE (KBs) 6911      현재 로그에 쌓인 양
--
-- ★ 전송이 밀리면 CURRENT 가 아닌 항목이 쌓인다.

DGMGRL> show database 'boston' 'RecvQEntries'

STANDBY_RECEIVE_QUEUE
  STATUS  RESETLOGS_ID  THREAD  LOG_SEQ  TIME_GENERATED  FIRST_CHANGE#  SIZE (KBs)

-- ★ 받았지만 아직 적용하지 못한 것을 보여 준다.
--   비어 있으면 밀린 것이 없다는 뜻이다.
--
-- ★ 두 큐를 함께 보면 어디서 막혔는지 알 수 있다.
--
--   SendQ            RecvQ      해석
--   ---------------  ---------  ------------------------------
--   비었거나 CURRENT  비었음     정상
--   쌓임             비었음     전송이 막혔다
--   비었음           쌓임       적용이 막혔다
--   쌓임             쌓임       둘 다 문제이거나 Standby 가 멈췄다
--
-- ★ 7장에서 transport lag 과 apply lag 을 조합해 진단했던 것과 같다.
--   큐를 보면 양까지 알 수 있다는 점이 다르다.
--   "얼마나 밀렸는가" 를 KB 단위로 볼 수 있다.

--------------------------------------------------------------------------------
 [5] 대기 이벤트
--------------------------------------------------------------------------------

DGMGRL> show database 'boston' 'TopWaitEvents'

TOP SYSTEM WAIT EVENTS
               Event            Wait Time
   rdbms ipc message              8006598
        PX Idle Wait              2912212
Space Manager: slave idle wait    2728772
    class slave wait              1639532
parallel recovery slave next change  1451003

-- ★ 목록에 보이는 것 대부분이 유휴 대기다.
--
--   rdbms ipc message                   유휴. 메시지를 기다린다
--   PX Idle Wait                        유휴. 병렬 슬레이브가 논다
--   Space Manager: slave idle wait      유휴
--   class slave wait                    유휴
--   parallel recovery slave next change 적용 관련이나 역시 대기
--
-- ★ 마지막 항목이 적용의 성격을 보여 준다.
--   적용할 리두가 없어 기다리는 것이므로 유휴에 가깝다.
--   값이 크다고 문제가 아니다.
--
-- ★ 적용이 병목이라면 유휴가 아닌 I/O 계열이 상위로 올라온다.
--
-- ★ 이 속성을 쓰는 방법은 하나다. 평소 값을 알아 두는 것이다.
--   절대값보다 평소와의 차이가 신호다.
--   정기 점검에서 이 목록을 기록해 두면 나중에 비교할 수 있다.

--------------------------------------------------------------------------------
 [6] 어긋난 속성
--------------------------------------------------------------------------------

DGMGRL> show database 'boston' 'InconsistentProperties'

INCONSISTENT PROPERTIES
   INSTANCE_NAME        PROPERTY_NAME         MEMORY_VALUE         SPFILE_VALUE         BROKER_VALUE

-- ★ 비어 있다. 13장에서 SQL 로 파라미터를 바꿨을 때도 비어 있었다.
--
-- ★ 세 값을 비교하는 구조라는 점을 읽어 둔다.
--
--   MEMORY_VALUE   지금 메모리에 있는 값
--   SPFILE_VALUE   SPFILE 에 있는 값
--   BROKER_VALUE   Broker 구성 파일에 있는 값
--
--   셋이 갈라질 수 있다는 것을 전제한 설계다.
--   예를 들어 SCOPE=MEMORY 로만 바꾸면 앞의 둘이 달라진다.
--
-- ★ 13장에서 확인한 대로 이 뷰가 비어 있다고 안심할 수는 없다.
--   감지되지 않는 변경이 있었다.

--------------------------------------------------------------------------------
 [7] 오류 재현 — 조회 전용 속성에 값을 주면
--------------------------------------------------------------------------------

DGMGRL> edit database 'boston' set property 'TopWaitEvents'='5';

Error: ORA-16568: cannot set property

Failed.

>>> 원인 파악

-- ★ 조회 전용 속성이므로 설정할 수 없다.
--
-- ★ 13장에서 본 오류들과 구별한다.
--
--   ORA-16538  속성 이름 자체가 없다
--   ORA-16790  속성은 있는데 값이 틀렸다
--   ORA-16568  속성은 있는데 설정할 수 없다 (조회 전용)
--
--   세 번호를 구별하면 무엇을 고쳐야 할지 바로 알 수 있다.

>>> 조치

-- 조회만 한다. SET 이 아니라 SHOW 를 쓴다.

>>> 재확인

DGMGRL> show database 'boston' 'TopWaitEvents'

TOP SYSTEM WAIT EVENTS
               Event            Wait Time
   rdbms ipc message              8006598

-- ★ 조회는 정상이다.

--------------------------------------------------------------------------------
 [8] 정리
--------------------------------------------------------------------------------

-- ★ 이 실습에서 확인한 것
--
--   ① (monitor) 로 표시된 속성은 조회 전용이다
--   ② StatusReport 는 인스턴스별 오류를 표로 준다
--   ③ LogXptStatus 는 Primary 에서 조회한다
--   ④ SendQ / RecvQ 조합으로 전송이 막혔는지 적용이 막혔는지 구별
--   ⑤ 큐는 양(KB)까지 알려 준다
--   ⑥ TopWaitEvents 상위는 대부분 유휴 대기. 평소 값과 비교한다
--   ⑦ InconsistentProperties 는 메모리·SPFILE·Broker 세 값을 비교한다
--   ⑧ 조회 전용 속성에 값을 주면 ORA-16568
--
-- ★ 다음 실습에서 지연을 만들고 임계값을 조절한다.
