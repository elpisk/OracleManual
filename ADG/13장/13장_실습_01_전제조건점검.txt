--

 [4] tnsnames 별칭
--------------------------------------------------------------------------------

[oracle@chicago-srv ~]$ cat $ORACLE_HOME/network/admin/tnsnames.ora
CHICAGO =
  (DESCRIPTION =
    (ADDRESS_LIST =
      (ADDRESS = (PROTOCOL = TCP)(HOST = chicago-srv)(PORT = 1521))
    )
    (CONNECT_DATA =
      (SERVICE_NAME = chicago)
    )
  )

BOSTON =
  (DESCRIPTION =
    (ADDRESS_LIST =
      (ADDRESS = (PROTOCOL = TCP)(HOST = boston-srv.localdomain)(PORT = 1521))
    )
    (CONNECT_DATA =
      (SERVICE_NAME = boston)
    )
  )
...

-- ★ 두 별칭이 있으면 된다.
--   Broker 의 CONNECT IDENTIFIER 로 이 이름을 쓴다.
--
-- ★ 헷갈리기 쉬운 지점이다.
--   CONNECT IDENTIFIER 는 tnsnames 별칭이지 DB_UNIQUE_NAME 이 아니다.
--   실습 환경에서는 우연히 같은 이름이라 구별이 안 보인다.
--   별칭이 chicago_dg 였다면 connect identifier is chicago_dg 로 적어야 한다.

>>> 원인 파악

-- ★ 접속 자체는 됐다. Connected to "chicago" 가 나왔다.
--   그러나 DMON 프로세스가 없으므로 구성을 물어볼 상대가 없다.
--
-- ★ 이 구별이 진단에 도움이 된다.
--
--   접속조차 안 된다        → Oracle Net 문제. tnsping 부터 본다
--   Connected 후 ORA-16525  → DMON 이 없다. DG_BROKER_START 확인
--   Connected 후 ORA-16532  → DMON 은 있는데 구성이 없다. 만들면 된다
--
--   오류 번호 하나로 어느 단계에서 막혔는지 알 수 있다.

>>> 조치

-- DG_BROKER_START 를 TRUE 로 바꾼다. 실습 02 에서 한다.

 [6] 정리
--------------------------------------------------------------------------------

-- ★ 이 실습에서 확인한 것
--
--   ① 손으로 만든 구성은 대칭이며 그 대칭을 사람이 유지해야 한다
--   ② 표기가 갈린 흔적이 있다 (online_logfiles vs online_logfile)
--   ③ Broker 파라미터는 셋. dg_broker_start 는 FALSE
--   ④ 구성 파일 이름은 dr1<이름>.dat / dr2<이름>.dat
--   ⑤ 정적 등록 <이름>_DGMGRL 이 양쪽에 있어야 한다
--   ⑥ CONNECT IDENTIFIER 는 tnsnames 별칭이다
--   ⑦ Broker 없이 접속하면 ORA-16525
--
-- ★ 다음 실습에서 Broker 를 켠다.
