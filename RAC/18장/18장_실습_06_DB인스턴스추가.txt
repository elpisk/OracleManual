 [5] dbca 가 만든 것
--------------------------------------------------------------------------------

SQL> select inst_id, instance_name, host_name, status, thread# from gv$instance order by 1;

   INST_ID INSTANCE_NAME    HOST_NAME          STATUS          THREAD#
---------- ---------------- ------------------ ------------ ----------
         1 racdb1           rac1.localdomain   OPEN                  1
         2 racdb2           rac2.localdomain   OPEN                  2

SQL> select thread#, status, enabled, instance from v$thread order by 1;

   THREAD# STATUS ENABLED  INSTANCE
---------- ------ -------- ----------
         1 OPEN   PUBLIC   racdb1
         2 OPEN   PUBLIC   racdb2

SQL> select thread#, group#, bytes/1024/1024 mb, status from v$log order by 1,2;

   THREAD#     GROUP#         MB STATUS
---------- ---------- ---------- ----------------
         1          1        200 INACTIVE
         1          2        200 CURRENT
         2          3        200 INACTIVE
         2          4        200 CURRENT

SQL> select inst_id, value from gv$parameter where name='undo_tablespace' order by 1;

   INST_ID VALUE
---------- ----------
         1 UNDOTBS1
         2 UNDOTBS2

-- 실습 02 에서 dbca -deleteInstance 가 지운 것을 dbca -addInstance 가 그대로 되살렸다.
--
--   스레드 2           PUBLIC 으로 활성화
--   리두 그룹 3, 4     스레드 2 용, 200MB — 스레드 1 과 같은 크기
--   UNDOTBS2           racdb2 전용
--   파라미터           racdb2 의 thread=2, undo_tablespace=UNDOTBS2
--
-- 리두는 스레드 1 과 같은 200MB 로 만들어졌다.
-- 손으로 인스턴스를 추가하면 이 네 가지를 직접 해야 한다.
--   alter database add logfile thread 2 group 3 ... size 200m;
--   alter database enable public thread 2;
--   create undo tablespace undotbs2 ...;
--   alter system set ... sid='racdb2';
-- 그리고 srvctl add instance 로 CRS 에 등록한다.
-- dbca 는 이것을 2분 안에 순서대로 해 준다.
