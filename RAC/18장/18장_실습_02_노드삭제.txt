================================================================================
 제18장 실습 02 — rac2 를 클러스터에서 제거하는 전체 절차  [주의]
 대상   : rac1, rac2
 목표   : 인스턴스 → DB 홈 → Clusterware → 그리드 홈 순서로 rac2 를 빼고,
          단계마다 rac1 의 racdb1 이 영향을 받지 않는지 확인한다
 소요   : 약 1시간 (명령 실행만 합계 약 9분)
 선행   : 실습 01 완료. 두 노드 전원을 끄고 양쪽 스냅샷을 뜬 뒤 시작한다
 계정   : oracle, grid, root
 주의   : 되돌릴 수 없다. OCR 에서 rac2 가 지워지고 rac2 의 소프트웨어가 삭제된다.
          공유 디스크는 VM 스냅샷에 들어가지 않으므로 스냅샷만으로는 원복되지 않는다
================================================================================

--------------------------------------------------------------------------------
 [1] 순서
--------------------------------------------------------------------------------

-- 설치의 역순이다. 위에 올린 것부터 내린다.
--
--   1. DB 인스턴스 삭제          dbca -deleteInstance          (rac1, oracle)
--   2. DB 홈 노드 목록 축소      runInstaller -updateNodeList -local  (rac2)
--   3. DB 홈 삭제                deinstall -local              (rac2, oracle)
--   4. DB 홈 노드 목록 축소      runInstaller -updateNodeList  (rac1)
--   5. Clusterware 해제          rootcrs.sh -deconfig -force   (rac2, root)
--   6. OCR 에서 노드 삭제        crsctl delete node -n rac2    (rac1, root)
--   7. 그리드 홈 노드 목록 축소  -updateNodeList -local CRS=TRUE (rac2)
--   8. 그리드 홈 삭제            deinstall -local + rootdeinstall.sh (rac2)
--   9. 그리드 홈 노드 목록 축소  -updateNodeList CRS=TRUE      (rac1)
--
-- 2·3·4 와 7·8·9 가 같은 모양으로 반복된다.
-- 지울 노드에서 "나는 나 혼자다" 라고 목록을 줄이고(-local), 지운 다음,
-- 남는 노드에서 "이제 나 혼자다" 라고 목록을 줄인다.
-- 이 순서를 어기면 deinstall 이 남는 노드의 홈까지 지우려 든다.

--------------------------------------------------------------------------------

--------------------------------------------------------------------------------

[oracle@rac1 ~]$ dbca -silent -deleteInstance -gdbName racdb -instanceName racdb2 \
>   -nodeName rac2 -sysDBAUserName sys -sysDBAPassword <암호>

-- 78.7초. 실행은 살아남는 rac1 에서 한다.
-- dbca 가 rac2 의 인스턴스를 내리고, 스레드 2 와 UNDOTBS2 를 지우고,
-- CRS 의 DB 자원에서 rac2 를 뺀다.

[oracle@rac1 ~]$ srvctl config database -d racdb | grep -E 'instances|nodes'
Database instances: racdb1
Configured nodes: rac1

SQL> select thread#, status, enabled from v$thread;

   THREAD# STATUS ENABLED
---------- ------ --------
         1 OPEN   PUBLIC

SQL> select thread#, group#, status from v$log order by 1,2;

   THREAD#     GROUP# STATUS
---------- ---------- ----------------
         1          1 INACTIVE
         1          2 CURRENT

SQL> select tablespace_name from dba_tablespaces where contents='UNDO';

TABLESPACE_NAME
------------------------------
UNDOTBS1

SQL> select count(*) from gv$active_services;    →  5

-- 스레드 2, 리두 그룹 3·4, UNDOTBS2 가 모두 사라졌다.
-- 서비스도 10행에서 5행이 됐다.
--
-- CRS 자원 개수는 17 그대로다. ora.racdb.db 는 자원 하나에 인스턴스가
-- CARDINALITY_ID 로 붙는 구조라 인스턴스가 줄어도 자원 수는 같다.

[grid@rac1 ~]$ crsctl stat res ora.racdb.db -v | grep CARDINALITY_ID
CARDINALITY_ID=1

 [3] DB 홈 — 응답 파일부터 만든다
--------------------------------------------------------------------------------

[oracle@rac2 ~]$ ls /u01/app/oracle/diag/rdbms/racdb/
i_1.mif  racdb2
[oracle@rac2 ~]$ ps -ef | grep -c [o]ra_pmon_racdb2
0
[oracle@rac2 ~]$ grep racdb /etc/oratab
(없음)

-- rac2 에서 인스턴스는 사라졌다. oratab 에도 없다.
-- 진단 디렉터리(diag/.../racdb2)는 남는다. 장애 분석 기록이라 dbca 가 지우지 않는다.

[oracle@rac2 ~]$ $ORACLE_HOME/oui/bin/runInstaller -updateNodeList -silent \
>   ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_rac "CLUSTER_NODES={rac2}" -local
'UpdateNodeList' was successful.

-- 5.1초. rac2 의 인벤토리만 바뀐다.

[oracle@rac2 ~]$ grep -A3 OraDB19Home2 /u01/app/oraInventory/ContentsXML/inventory.xml
<HOME NAME="OraDB19Home2" LOC="/u01/app/oracle/product/19.3.0/dbhome_rac" TYPE="O" IDX="3">
   <NODE_LIST>
      <NODE NAME="rac2"/>
   </NODE_LIST>

[oracle@rac1 ~]$ grep -A4 OraDB19Home2 /u01/app/oraInventory/ContentsXML/inventory.xml
<HOME NAME="OraDB19Home2" LOC="/u01/app/oracle/product/19.3.0/dbhome_rac" TYPE="O" IDX="3">
   <NODE_LIST>
      <NODE NAME="rac1"/>
      <NODE NAME="rac2"/>
   </NODE_LIST>

-- rac1 은 아직 rac1,rac2 다. 이 상태가 맞다.

[oracle@rac2 ~]$ $ORACLE_HOME/deinstall/deinstall -local -checkonly
...
The following nodes are part of this cluster: rac2,rac1
...
Specify the list of database names that are configured locally on this node for this Oracle home. Local configurations of the discovered databases will be removed []:
...
####################### DECONFIG CHECK OPERATION SUMMARY #######################
The cluster node(s) on which the Oracle home deinstallation will be performed are:rac2
Oracle Home selected for deinstall is: /u01/app/oracle/product/19.3.0/dbhome_rac
Location of response file generated: '/tmp/deinstall2026-10-16_06-27-27PM/response/deinstall_OraDB19Home2.rsp'

-- 87.4초. -checkonly 는 아무것도 지우지 않고 응답 파일만 만든다.
-- 두 줄을 꼭 본다.
--
--   deinstallation will be performed are:rac2
--     rac2 에서만 지운다. 여기에 rac1 이 보이면 멈춘다.
--
--   database names ... []
--     지울 DB 목록이 비어 있다. 인스턴스를 [2] 에서 먼저 지웠기 때문이다.

[oracle@rac2 ~]$ grep -E 'CLUSTER_NODES|LOCAL_NODE|DB_UNIQUE|HOME_TYPE|local=' \
>   /tmp/deinstall2026-10-16_06-27-27PM/response/deinstall_OraDB19Home2.rsp
CLUSTER_NODES=rac2
DB_UNIQUE_NAME_LIST=
HOME_TYPE=RACDB
LOCAL_NODE=rac2
local=true

-- DB_UNIQUE_NAME_LIST 가 비어 있는 것이 중요하다.
-- 순서를 거꾸로 해서 인스턴스를 남긴 채 deinstall 을 했다면
-- 여기에 racdb 가 잡히고 deinstall 이 데이터베이스 구성까지 건드리려 한다.

 [4] DB 홈 삭제
--------------------------------------------------------------------------------

[oracle@rac2 ~]$ $ORACLE_HOME/deinstall/deinstall -local -silent \

-- 50.4초. 7GB 가 지워졌다.
-- ORACLE_BASE 는 dbhome_1 이 쓰고 있어서 남긴다. 판단을 deinstall 이 알아서 한다.

[oracle@rac1 ~]$ srvctl status database -d racdb -v
Instance racdb1 is running on node rac1. Instance status: Open.

-- 단계마다 rac1 을 본다. racdb1 은 그대로 Open 이다.

[oracle@rac1 ~]$ $ORACLE_HOME/oui/bin/runInstaller -updateNodeList -silent \
>   ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_rac "CLUSTER_NODES={rac1}"
'UpdateNodeList' was successful.

-- 23.4초. 이번에는 -local 을 붙이지 않는다. rac1 에서 "이 홈의 노드는 rac1 뿐" 이라고
-- 선언한다. rac2 에는 이미 이 홈이 없으므로 고칠 것도 없다.

 [5] Clusterware 해제 — rac2 에서
--------------------------------------------------------------------------------

[root@rac2 ~]# /u01/app/19.3.0/grid/crs/install/rootcrs.sh -deconfig -force
Using configuration parameter file: /u01/app/19.3.0/grid/crs/install/crsconfig_params
The log of current session can be found at:
  /u01/app/grid/crsdata/rac2/crsconfig/crsdeconfig_rac2_2026-10-16_06-31-40PM.log
2026/10/16 18:32:31 CLSRSC-4006: Removing Oracle Trace File Analyzer (TFA) Collector.
2026/10/16 18:34:30 CLSRSC-4007: Successfully removed Oracle Trace File Analyzer (TFA) Collector.
2026/10/16 18:34:31 CLSRSC-336: Successfully deconfigured Oracle Clusterware stack on this node

-- 172.0초. 그중 2분은 TFA 제거다.
--
-- -force 는 다른 노드와 통신이 안 되거나 자원이 남아 있어도 진행하라는 뜻이다.
-- 마지막 노드가 아니므로 -lastnode 는 붙이지 않는다. 붙이면 OCR·보팅 디스크까지
-- 지우는 "클러스터 전체 해체" 가 된다.

[grid@rac1 ~]$ olsnodes -s -t
rac1    Active    Unpinned
rac2    Inactive  Unpinned

[root@rac2 ~]# crsctl check crs
CRS-4047: No Oracle Clusterware components configured.
CRS-4000: Command Check failed, or completed with errors.

-- rac2 는 Inactive 가 됐지만 아직 목록에 있다. OCR 에는 rac2 의 항목이 남아 있다.
-- rac2 쪽에는 Clusterware 구성이 없다. 그리드 홈 6.4GB 는 그대로다.
-- CRS 자원은 17 → 16. rac2-vip 가 빠졌다.

--------------------------------------------------------------------------------
 [6] OCR 에서 노드 삭제 — rac1 에서
--------------------------------------------------------------------------------

[root@rac1 ~]# /u01/app/19.3.0/grid/bin/crsctl delete node -n rac2
CRS-4661: Node rac2 successfully deleted.

-- 1.2초. OCR 에서 rac2 의 노드 항목을 지운다. root 로 실행해야 한다.
-- 이 단계부터 rac2 는 클러스터가 아는 노드가 아니다.

[grid@rac1 ~]$ olsnodes -s -t
rac1    Active    Unpinned

[grid@rac1 ~]$ crsctl stat res -t | grep -A3 'ora.asm('
ora.asm(ora.asmgroup)
      1        ONLINE  ONLINE       rac1                     Started,STABLE
      2        ONLINE  OFFLINE                               Instance Shutdown,ST
                                                             ABLE
      3        ONLINE  OFFLINE                               STABLE

-- ASM 자원의 3번(rac2 에서 돌던 +ASM)이 OFFLINE 으로 남는다.
-- Flex ASM 의 카디널리티는 3 으로 고정이라 노드가 줄어도 칸은 그대로다.
-- 오류가 아니다.

 [7] 그리드 홈 — 노드 목록 축소와 응답 파일
--------------------------------------------------------------------------------

[grid@rac2 ~]$ /u01/app/19.3.0/grid/oui/bin/runInstaller -updateNodeList -silent \
>   ORACLE_HOME=/u01/app/19.3.0/grid "CLUSTER_NODES={rac2}" CRS=TRUE -local
'UpdateNodeList' was successful.

-- 69.8초. 홈 유형이 "Standalone Server" 로 나온다.
-- Clusterware 를 [5] 에서 이미 해제했으므로 deinstall 은 이 홈을
-- 클러스터 홈이 아니라 Oracle Restart 홈으로 본다. 응답 파일에도
-- HOME_TYPE=SIHA, CRS_HOME=false 로 적힌다.
--
-- 그 덕분에 지울 범위가 rac2 하나로 좁혀진다. 클러스터 홈으로 보았다면
-- 원격 노드까지 정리하려 한다.
--
-- "ASM was not detected" 도 같은 이유다. ASM 디스크그룹은 공유 디스크에 있고
-- rac1 이 계속 쓰고 있으므로 건드리지 않는 것이 맞다.
