================================================================================
 제3장 실습 04 — 공유 디스크 인식 확인
 대상   : rac1 / rac2
 목표   : 두 노드가 같은 디스크를 보고 있는지 WWN 으로 대조하고,
          링크드 클론 환경에서 OS 디스크가 만드는 함정을 확인한다
 소요   : 약 30분
 선행   : 공유 디스크가 두 VM 에 연결되어 있을 것
 계정   : root
================================================================================

--------------------------------------------------------------------------------
 [1] 디스크 인식 확인
--------------------------------------------------------------------------------

[root@rac1 ~]# lsblk -d -o NAME,SIZE,TYPE | grep disk
sda   100G disk
sdb     4G disk
sdc     4G disk
sdd     4G disk
sde    20G disk
sdf    20G disk
sdg    20G disk
sdh    12G disk
sdi    12G disk

-- 계획한 구성과 맞는지 본다.
--   4GB x 3   +CRS 용 (Normal 이중화, 보팅 파일 3벌 확보용)
--   20GB x 3  +DATA 용
--   12GB x 2  +FRA 용
--   100GB x 1 OS 디스크 — 공유 대상이 아니다
--
-- 여기서 개수나 크기가 다르면 스토리지 매핑 또는 vmx 설정을 먼저 확인한다.

--------------------------------------------------------------------------------
 [2] WWN 을 뽑는다
--------------------------------------------------------------------------------

[root@rac1 ~]# for d in /dev/sd?; do
>   n=$(basename $d)
>   sz=$(lsblk -bdno SIZE $d); gb=$((sz/1073741824))
>   id=$(/lib/udev/scsi_id -g -u -d $d)
>   printf '  %-5s %3dGB  %s\n' $n $gb "${id:-없음}"
> done
  sda   100GB  36000c29d85490b5acfaa7629df3ef441
  sdb     4GB  36000c29895fb6f08c44ceae9d85133a2
  sdc     4GB  36000c29f6be51822332005c44b4352bd
  sdd     4GB  36000c29536db115b31143f96347fbe43
  sde    20GB  36000c29b21cbf101cf58984292a222c6
  sdf    20GB  36000c29fcedbd7520d3d9c110bb3413b
  sdg    20GB  36000c2976226c81c029d91038e5cfb24
  sdh    12GB  36000c294925c5c7b2df0386287d47f5d
  sdi    12GB  36000c299212006d4478b392df1de6e42

-- scsi_id 옵션의 뜻
--   -g  화이트리스트를 무시하고 모든 디바이스를 대상으로 한다
--   -u  결과를 UUID 형식으로 출력한다
--   -d  대상 디바이스를 지정한다

 [3] 두 노드 대조 — 이것이 핵심이다
--------------------------------------------------------------------------------

[root@rac2 ~]# (같은 명령)
  sda   100GB  36000c29d85490b5acfaa7629df3ef441
  sdb     4GB  36000c29895fb6f08c44ceae9d85133a2
  sdc     4GB  36000c29f6be51822332005c44b4352bd
  sdd     4GB  36000c29536db115b31143f96347fbe43
  sde    20GB  36000c29b21cbf101cf58984292a222c6
  sdf    20GB  36000c29fcedbd7520d3d9c110bb3413b
  sdg    20GB  36000c2976226c81c029d91038e5cfb24
  sdh    12GB  36000c294925c5c7b2df0386287d47f5d
  sdi    12GB  36000c299212006d4478b392df1de6e42

-- sdb 부터 sdi 까지 여덟 개의 WWN 이 양쪽에서 정확히 같다.
-- 같은 물리 디스크를 두 노드가 함께 보고 있다는 뜻이다.
-- 공유 스토리지가 제대로 붙었다는 증거다.
--
-- 자동으로 대조하려면 이렇게 한다.
[root@rac1 ~]# for h in rac1 rac2; do
>   ssh $h 'for d in /dev/sd?; do
>     printf "%s %s\n" $(basename $d) $(/lib/udev/scsi_id -g -u -d $d)
>   done' > /tmp/wwn.$h
> done
[root@rac1 ~]# diff /tmp/wwn.rac1 /tmp/wwn.rac2 && echo "두 노드 WWN 동일"
두 노드 WWN 동일

--------------------------------------------------------------------------------
 [4] 크기 비교만으로는 왜 부족한가
--------------------------------------------------------------------------------

-- 이 환경에는 같은 크기의 디스크가 여럿이다.
--   20GB : sde, sdf, sdg  (data1, data2, data3)
--   12GB : sdh, sdi       (fra1, fra2)
--    4GB : sdb, sdc, sdd  (crs1, crs2, crs3)
--
-- 같은 크기끼리 서로 바꿔 짚어도 크기만 봐서는 알 수 없다.
-- 실제로 LUN 매핑 순서가 어긋나 rac1 의 data1 과 rac2 의 data1 이 다른
-- 디스크를 가리킨 사이트가 있었다. GI 설치까지 통과했고, ASM 디스크그룹을
-- 만든 뒤에야 드러났다. 한 노드에서 마운트하면 다른 노드에서 디스크가
-- 보이지 않았고, 강제로 진행하자 헤더가 겹쳐 쓰여 다시 만들어야 했다.
--
-- WWN 대조 한 번이면 막을 수 있었다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — OS 디스크의 WWN 이 두 노드에서 같다
--------------------------------------------------------------------------------

-- 위 출력을 다시 본다. sda 의 WWN 도 양쪽이 같다.
--   rac1  sda  100GB  36000c29d85490b5acfaa7629df3ef441
--   rac2  sda  100GB  36000c29d85490b5acfaa7629df3ef441
--
-- sda 는 OS 디스크이며 노드마다 별개여야 한다. 그런데 WWN 이 같다.
-- WWN 만으로 공유 디스크를 판별하는 스크립트라면 sda 도 공유 디스크로
-- 분류해 udev 규칙에 넣게 된다.

-- 원인 파악
--   두 VM 이 같은 원본 이미지에서 만든 링크드 클론이기 때문이다.
--   클론은 원본의 디스크 식별자를 물려받는다.
[root@rac1 ~]# lsblk /dev/sda
NAME            MAJ:MIN RM  SIZE RO TYPE MOUNTPOINT
sda               8:0    0  100G  0 disk
├─sda1            8:1    0    1G  0 part /boot
└─sda2            8:2    0   99G  0 part
  ├─ol-root     252:0    0   84G  0 lvm  /
  └─ol-swap     252:1    0 15.6G  0 lvm  [SWAP]

--   sda 에는 /boot 와 루트 LVM 이 있다. 명백히 OS 디스크다.
--   실제 데이터는 각 노드가 따로 쓰지만 식별자만 같게 보인다.
--
--   여기에 ASM 헤더를 쓰면 OS 파티션이 깨져 노드가 부팅하지 못한다.

-- 조치 — 규칙에서 크기로 걸러 낸다.
[root@rac1 ~]# grep -o 'asmdisk/[a-z0-9]*' /etc/udev/rules.d/99-oracle-asmdisk.rules
asmdisk/crs1
asmdisk/crs2
asmdisk/crs3
asmdisk/data1
asmdisk/data2
asmdisk/data3
asmdisk/fra1
asmdisk/fra2

--   여덟 개뿐이다. sda 는 포함되지 않았다.
--   규칙 생성 스크립트가 크기로 용도를 가르기 때문이다.
--     case $gb in
--       4)  name="crs${ci}"  ;;
--       20) name="data${di}" ;;
--       12) name="fra${fi}"  ;;
--       *)  continue ;;      <- 100GB 는 여기서 걸러진다
--     esac

-- 재확인
[root@rac1 ~]# ls /dev/asmdisk/
crs1  crs2  crs3  data1  data2  data3  fra1  fra2

[root@rac1 ~]# ls -lL /dev/asmdisk/ | awk '{print $1, $3, $4, $NF}' | tail -8
brw-rw---- grid asmadmin crs1
brw-rw---- grid asmadmin crs2
brw-rw---- grid asmadmin crs3
brw-rw---- grid asmadmin data1
brw-rw---- grid asmadmin data2
brw-rw---- grid asmadmin data3
brw-rw---- grid asmadmin fra1
brw-rw---- grid asmadmin fra2

--   OS 디스크가 빠졌고 여덟 개만 grid:asmadmin 권한으로 준비되어 있다.

-- 배운 것
--   WWN 은 공유 여부를 판별하는 강력한 기준이지만 만능은 아니다.
--   가상화 환경에서는 OS 디스크의 식별자까지 같을 수 있다.
--   규칙에는 공유 대상이 아닌 디스크를 확실히 제외하는 장치가 있어야 한다.
--   크기, 멀티패스 이름, LUN 번호 중 환경에 맞는 것을 쓴다.

--------------------------------------------------------------------------------
 [6] 오류 재현 — cluvfy comp ssa 로 공유 여부를 확인하려 하면
--------------------------------------------------------------------------------

-- cluvfy 에 공유 스토리지 접근성을 검사하는 comp ssa 가 있다.
-- ASM 디스크에 써 보면 어떻게 되는지 확인한다.

[grid@rac1 ~]$ cluvfy comp ssa -n rac1,rac2 -s /dev/asmdisk/crs1

Verifying Shared Storage Accessibility:/dev/asmdisk/crs1 ...FAILED (PRVG-11502)

Verification of shared storage accessibility was unsuccessful on all the
specified nodes.

Failures were encountered during execution of CVU verification request
"shared storage accessibility".

Verifying Shared Storage Accessibility:/dev/asmdisk/crs1 ...FAILED
PRVG-11502 : Path "/dev/sdb" of type "Disk" is not suitable for usage as RAC
database file for release "19.0". Supported storage types are "NFS, ASM Disk
Group, OCFS2, VXFS, ACFS".

-- 원인 파악
--   메시지를 그대로 읽는다. /dev/sdb 는 Disk 유형이며 RAC 데이터베이스
--   파일용으로 적합하지 않다고 한다. 지원 유형으로 NFS, ASM Disk Group,
--   OCFS2, VXFS, ACFS 를 열거한다.
--
--   comp ssa 는 파일을 놓을 수 있는 공유 파일 시스템을 검사하는 명령이다.
--   원시 블록 디바이스는 그 대상이 아니다. ASM 디스크그룹이 만들어진 뒤에는
--   +DATA 같은 디스크그룹 이름을 넘겨야 의미가 있다.
--
--   즉 이 실패는 디스크가 공유되지 않았다는 뜻이 아니다. 검사 도구를
--   잘못 골랐다는 뜻이다.

-- 경로 자체가 없으면 또 다른 오류가 난다.
[grid@rac1 ~]$ cluvfy comp ssa -n rac1,rac2 -s /dev/asmdisk/nosuch

Verifying Shared Storage Accessibility:/dev/asmdisk/nosuch ...FAILED
PRVG-10130 : Unable to determine whether file path "/dev/asmdisk" is shared by
nodes "rac1,rac2"
PRKC-1025 : Failed to create a file under the filepath /dev/asmdisk because the
filepath is not executable or writable

--   PRKC-1025 가 근본 원인을 말해 준다. comp ssa 는 대상 경로에 시험용
--   파일을 만들어 두 노드에서 보이는지 확인하는 방식으로 동작한다.
--   /dev/asmdisk 는 udev 가 만든 디바이스 링크 디렉터리라 파일을 만들 수 없다.

-- 조치 — ASM 원시 디스크의 공유 여부는 WWN 대조로 확인한다.
[root@rac1 ~]# for h in rac1 rac2; do
>   ssh $h 'for d in /dev/sd?; do
>     printf "%s %s
" $(basename $d) $(/lib/udev/scsi_id -g -u -d $d)
>   done' > /tmp/wwn.$h
> done
[root@rac1 ~]# diff /tmp/wwn.rac1 /tmp/wwn.rac2 && echo "두 노드 WWN 동일"
두 노드 WWN 동일

-- 재확인
--   WWN 이 모두 같으므로 공유가 확인된다.
--   comp ssa 는 디스크그룹이 만들어진 뒤 그 이름으로 쓰거나, NFS·ACFS 같은
--   공유 파일 시스템을 검사할 때 쓴다.
--
-- 배운 것
--   FAILED 라고 모두 시스템 문제가 아니다. 검사 도구를 대상에 맞게 골라야 한다.
--   메시지의 "Supported storage types" 목록이 그 사실을 알려 주고 있었다.
--   오류 메시지를 끝까지 읽는 습관이 시간을 아낀다.

--------------------------------------------------------------------------------
 [7] scsi_id 가 빈 값을 돌려준다면
--------------------------------------------------------------------------------

-- 이 실습 환경에서는 모든 디스크가 WWN 을 돌려주지만, 그렇지 않은 경우가 있다.
--
--   VMware 환경 : vmx 에 disk.EnableUUID = "TRUE" 가 없으면 게스트가
--                 디스크 UUID 를 조회할 수 없다. VM 을 정지하고 추가한 뒤
--                 재기동한다. VM 이 켜진 상태에서 vmx 를 고치면 종료 시
--                 덮어써지므로 반드시 정지 후 편집한다.
--   물리 서버   : HBA 드라이버가 SCSI INQUIRY 의 Device Identification
--                 VPD 페이지를 노출하는지 확인한다. 멀티패스를 쓴다면
--                 multipath -ll 로 WWID 를 확인하고 /dev/mapper/mpathX 를
--                 대상으로 삼는다.
--
-- WWN 을 얻지 못하면 udev 규칙을 만들 수 없으므로 이 단계를 넘어갈 수 없다.
-- 크기나 디바이스 이름으로 임시 규칙을 만드는 것은 노드 간 순서가 달라질
-- 위험이 있어 권하지 않는다.

--------------------------------------------------------------------------------
 [8] 정리
--------------------------------------------------------------------------------

-- 확인한 것
--   1) 디스크 개수와 크기가 계획과 맞는지 먼저 본다
--   2) scsi_id -g -u -d 로 WWN 을 뽑는다
--   3) 두 노드의 출력을 diff 로 대조한다 — 여덟 개 모두 일치
--   4) 크기가 같은 디스크가 여럿이라 크기 비교는 무력하다
--   5) 링크드 클론에서는 OS 디스크의 WWN 도 같다
--   6) 규칙은 크기로 OS 디스크를 걸러 낸다
--   7) ls -lL 로 대상 파일의 권한을 확인한다
--
-- 이 실습은 조회만 했으므로 원상 복구할 것이 없다.
