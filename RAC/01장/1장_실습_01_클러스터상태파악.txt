================================================================================
 제1장 실습 01 — 클러스터 전체 상태 파악
 대상   : rac1(172.16.225.101) / rac2(172.16.225.102), 클러스터 rac-cluster
 목표   : crsctl status resource -t 한 화면에서 무엇을 읽어 낼 수 있는지 익히고
          Local Resources 와 Cluster Resources 의 구분, Target 과 State 의 의미를
          정리한다
 소요   : 약 25분
 선행   : 두 노드의 Clusterware 가 기동되어 있을 것
 계정   : grid(Clusterware 조회), oracle(데이터베이스 조회)
================================================================================

--------------------------------------------------------------------------------
 [1] 접속과 계정 확인
--------------------------------------------------------------------------------

[root@rac1 ~]# su - grid

-- Clusterware 조회는 grid 계정으로 한다. ORACLE_HOME 이 Grid 홈을 가리키는지
-- 먼저 확인한다. 이것을 건너뛰면 명령을 못 찾거나 엉뚱한 홈의 명령이 실행된다.
[grid@rac1 ~]$ echo $ORACLE_HOME
/u01/app/19.3.0/grid

[grid@rac1 ~]$ echo $ORACLE_SID
+ASM1

[grid@rac1 ~]$ which crsctl
/u01/app/19.3.0/grid/bin/crsctl

--------------------------------------------------------------------------------
 [2] 스택이 살아 있는지부터 본다
--------------------------------------------------------------------------------

-- 자원 목록을 보기 전에 Clusterware 자체가 정상인지 확인한다. 순서가 중요하다.
-- 스택이 내려가 있으면 자원 조회는 아무 의미가 없다.
[grid@rac1 ~]$ crsctl check crs
CRS-4638: Oracle High Availability Services is online
CRS-4537: Cluster Ready Services is online
CRS-4529: Cluster Synchronization Services is online
CRS-4533: Event Manager is online

-- -all 을 붙이면 클러스터의 모든 노드를 한 번에 본다. 한쪽 노드에서만 실행해도
-- 양쪽 결과가 나온다는 점이 RAC 조회의 편리한 부분이다.
[grid@rac1 ~]$ crsctl check cluster -all
**************************************************************
rac1:
CRS-4537: Cluster Ready Services is online
CRS-4529: Cluster Synchronization Services is online
CRS-4533: Event Manager is online
**************************************************************
rac2:
CRS-4537: Cluster Ready Services is online
CRS-4529: Cluster Synchronization Services is online
CRS-4533: Event Manager is online
**************************************************************

-- 노드 목록과 번호도 함께 확인해 둔다.
[grid@rac1 ~]$ olsnodes -n -i -s -t
rac1    1       <none>  Active  Unpinned
rac2    2       <none>  Active  Unpinned

-- 열의 뜻: 노드명 / 노드번호 / VIP / 상태 / 고정 여부
-- Active 는 클러스터 멤버로 참여 중이라는 뜻이다. Unpinned 는 노드 번호가
-- 고정되지 않았다는 뜻으로, 기본값이며 정상이다.

--------------------------------------------------------------------------------
 [4] 이 출력을 읽는 법
--------------------------------------------------------------------------------

-- (1) 두 덩어리로 나뉜다.
--     Local Resources   : 모든 노드에 하나씩 뜨는 자원. 노드 이름이 줄마다 나온다.
--                         리스너, ONS, 네트워크 자원이 여기 속한다.
--     Cluster Resources : 클러스터 전체에서 개수가 정해진 자원. 왼쪽에 인스턴스
--                         번호가 붙는다.
--
-- (2) Target 과 State 를 함께 읽는다.
--     Target = 이 자원이 어떤 상태여야 하는가 (관리자의 지시)
--     State  = 지금 실제로 어떤 상태인가
--     둘 다 ONLINE 이면 정상. Target 이 ONLINE 인데 State 가 OFFLINE 이면
--     뭔가 죽었고 Clusterware 가 되살리려는 중이다. 둘 다 OFFLINE 이면
--     누군가 의도적으로 내린 것이다.
--
-- (3) ora.asmgroup 의 3번은 왜 OFFLINE 인가.
--     19c 의 Flex ASM 은 기본적으로 ASM 인스턴스 카디널리티를 3 으로 잡는다.
--     노드가 둘뿐이므로 3번 자리는 채워질 노드가 없어 OFFLINE 으로 남는다.
--     정상이며 조치가 필요 없다. 노드를 하나 더 추가하면 3번이 채워진다.
--
-- (4) State details 열에 추가 정보가 붙는다.
--     데이터베이스는 Open 인지 Mounted 인지, 어느 홈에서 떴는지가 표시된다.
--     STABLE 은 상태가 안정되어 전이 중이 아니라는 뜻이다.

-- 특정 자원만 좁혀 보려면 이름을 지정한다.
[grid@rac1 ~]$ crsctl status resource ora.racdb.db -t
--------------------------------------------------------------------------------
Name           Target  State        Server                   State details
--------------------------------------------------------------------------------
Cluster Resources
--------------------------------------------------------------------------------
ora.racdb.db
      1        ONLINE  ONLINE       rac1                     Open,HOME=...,STABLE
      2        ONLINE  ONLINE       rac2                     Open,HOME=...,STABLE
--------------------------------------------------------------------------------

--------------------------------------------------------------------------------
 [6] 오류 재현 — 노드를 재부팅한 직후 조회하면
--------------------------------------------------------------------------------

-- 실습 환경을 재부팅한 뒤 곧바로 데이터베이스에 접속해 보면 이런 일이 생긴다.
-- 아래는 실제로 재부팅 직후에 발생한 상황을 그대로 옮긴 것이다.

[oracle@rac1 ~]$ sqlplus -s / as sysdba
SQL> select inst_id, thread#, group# from gv$log where inst_id=1;
select inst_id, thread#, group# from gv$log where inst_id=1
*
ERROR at line 1:
ORA-01034: ORACLE not available
Process ID: 0
Session ID: 0 Serial number: 0

-- 원인 파악
--   ORA-01034 는 접속하려는 인스턴스가 기동되어 있지 않다는 뜻이다.
--   RAC 에서는 Clusterware 는 정상인데 데이터베이스만 아직 안 올라온 상태가
--   흔하다. 계층을 아래에서 위로 순서대로 확인한다.

[oracle@rac1 ~]$ srvctl status database -d racdb
Instance racdb1 is not running on node rac1
Instance racdb2 is not running on node rac2

[oracle@rac1 ~]$ exit
[root@rac1 ~]# su - grid
[grid@rac1 ~]$ crsctl check crs
CRS-4638: Oracle High Availability Services is online
CRS-4537: Cluster Ready Services is online
CRS-4529: Cluster Synchronization Services is online
CRS-4533: Event Manager is online

--   Clusterware 는 이미 올라와 있는데 데이터베이스 인스턴스가 아직 없다.
--   Management policy 가 AUTOMATIC 이므로 곧 올라오겠지만, 재부팅 직후에는
--   ASM 디스크그룹 마운트와 리스너 등록이 끝날 때까지 시간이 걸린다.
--   기다리거나 직접 올린다.

-- 조치
[grid@rac1 ~]$ exit
[root@rac1 ~]# su - oracle
[oracle@rac1 ~]$ srvctl start database -d racdb

[oracle@rac1 ~]$ srvctl status database -d racdb
Instance racdb1 is running on node rac1
Instance racdb2 is running on node rac2

-- 재확인
[oracle@rac1 ~]$ sqlplus -s / as sysdba
SQL> select inst_id, instance_name, status from gv$instance order by 1;

   INST_ID INSTANCE_NAME    STATUS
---------- ---------------- ------------
         1 racdb1           OPEN
         2 racdb2           OPEN

2 rows selected.

-- 배운 것: RAC 에서 조회가 실패하면 계층을 아래에서 위로 확인한다.
--   (1) Clusterware 가 살아 있는가          crsctl check cluster -all
--   (2) ASM 디스크그룹이 마운트되었는가     crsctl status resource -t 의 ora.*.dg
--   (3) 데이터베이스 인스턴스가 떴는가      srvctl status database -d racdb
--   (4) 그제야 SQL 로 내려간다
-- 이 순서를 지키면 원인을 훨씬 빨리 찾는다. ORA-01034 를 보고 곧바로
-- startup 을 치는 것은 RAC 에서 잘못된 습관이다. Clusterware 가 관리하는
-- 자원을 SQL*Plus 로 직접 올리면 자원 상태와 실제가 어긋난다.

--------------------------------------------------------------------------------
 [7] 정리
--------------------------------------------------------------------------------

-- crsctl check cluster -all         : 스택 생존 확인, 가장 먼저
-- olsnodes -n -i -s -t              : 노드 목록과 상태
-- crsctl status resource -t         : 자원 전체 한 화면
-- crsctl status resource NAME -t    : 자원 하나만
-- srvctl status database -d racdb -v : 데이터베이스 관점의 상태
-- srvctl config database -d racdb   : 데이터베이스 구성 정보
--
-- 이 실습은 조회만 했으므로 원상 복구할 것이 없다.
