--------------------------------------------------------------------------------
 [2] 32자리 16진수의 정체
--------------------------------------------------------------------------------

SQL> select con_id, name, guid, open_mode from v$pdbs order by con_id;

    CON_ID NAME         GUID                               OPEN_MODE
---------- ------------ ---------------------------------- ----------
         2 PDB$SEED     5A566B559A25734BE06365E110AC68E0   READ ONLY
         3 RACPDB1      5A569448ED6C2E8DE06365E110ACB045   READ WRITE

-- 맞춰 보면 이렇다.
--   86B637B62FE07A65E053F706E80A27CA  -> v$pdbs 에 없다. PDB$SEED 의 이전 GUID
--                                        또는 CDB 생성 시 만들어진 것
--   5A566B559A25734BE06365E110AC68E0  -> PDB$SEED
--   5A569448ED6C2E8DE06365E110ACB045  -> RACPDB1
--   GUID 조각 없음                    -> CDB$ROOT
--
-- 왜 PDB 이름 대신 GUID 를 쓰는가.
-- PDB 는 이름을 바꿀 수 있고(alter pluggable database ... rename global_name to)
-- 다른 CDB 로 옮길 수도 있다(unplug / plug). 이름을 경로에 넣으면 그때마다
-- 파일을 옮겨야 한다. GUID 는 만들 때 부여되어 평생 바뀌지 않으므로 경로가
-- 안정적이다.

--------------------------------------------------------------------------------
 [3] 다섯 조각으로 나눈다
--------------------------------------------------------------------------------

-- +DATA/RACDB/5A569448ED6C2E8DE06365E110ACB045/DATAFILE/system.274.1242762677
--
--   조각 1  +DATA                              디스크그룹
--   조각 2  RACDB                              DB_UNIQUE_NAME
--   조각 3  5A569448ED6C2E8DE06365E110ACB045   PDB GUID (RACPDB1)
--   조각 4  DATAFILE                           파일 종류. ASM 이 만든 디렉터리
--   조각 5  system.274.1242762677              태그.파일번호.incarnation
--
-- 조각 5 를 더 나누면
--   system      태그. 데이터베이스가 붙인다. 보통 테이블스페이스 이름
--   274         ASM 파일 번호. v$asm_file 의 FILE_NUMBER
--   1242762677  incarnation
--
-- 이 규칙을 알면 경로만 보고 어느 데이터베이스의 어느 PDB 에 속한 무슨
-- 종류의 파일인지 즉시 판단할 수 있다.

--------------------------------------------------------------------------------
 [6] 소수점이 왜 붙는가
--------------------------------------------------------------------------------

-- 890.007813MB 처럼 끝이 딱 떨어지지 않는다. 계산해 본다.
--
--   113921 블록 x 8192 바이트 = 933,232,640 바이트
--   933,232,640 / 1024 / 1024 = 890.0078125 MB
--
-- 정확히 맞는다. 그런데 왜 113921 인가. SYSTEM 테이블스페이스를 890MB 로
-- 만들었다면 113920 블록이어야 한다.
--
--   890 x 1024 x 1024 / 8192 = 113920
--
-- 한 블록이 더 있다. 데이터파일 헤더 블록이다. 오라클 데이터파일은 맨 앞에
-- 파일 헤더 블록 하나를 두고 그 뒤에 실제 데이터 블록이 온다.
--
-- 그래서 데이터파일 크기는 항상 지정한 크기에 한 블록이 더 붙는다.
-- 8KB 블록이면 0.0078125MB 다. 실습 06 에서 50MB 테이블스페이스를 복사하면
-- 51MB 파일이 나오는 것도 같은 이유다.

 [7] 오류 재현 — 이름을 한 자리 틀리면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ sqlplus -s / as sysdba

SQL> alter database datafile '+DATA/RACDB/DATAFILE/users.260.1242761684' offline;
alter database datafile '+DATA/RACDB/DATAFILE/users.260.1242761684' offline
*
ERROR at line 1:
ORA-01516: nonexistent log file, data file, or temporary file
           "+DATA/RACDB/DATAFILE/users.260.1242761684" in the current container

-- 오류 : ORA-01516
-- 원인 파악 : incarnation 의 마지막 자리를 3에서 4로 바꿔 입력했다.
--             메시지는 nonexistent data file 이라고 하지만 파일은 존재한다.
--             이름만 틀린 것이다. 파일이 사라진 줄 알고 복구 절차를 밟기
--             쉬운 함정이다.
-- 조치     : v$datafile 의 NAME 을 그대로 복사해 쓴다. 더 안전한 방법은
--             파일 번호로 지정하는 것이다.
-- 재확인   : 파일이 멀쩡한지 확인한다.

SQL> select file#, name, status from v$datafile where file#=7;

     FILE# NAME                                          STATUS
---------- --------------------------------------------- -------
         7 +DATA/RACDB/DATAFILE/users.260.1242761683     ONLINE

-- 파일은 ONLINE 이다. 이름 끝이 ...683 이었다.
--
-- incarnation 이 이런 실수를 잡아 준다는 점이 오히려 중요하다.
-- 만약 이 값이 없어서 users.260 만으로 파일을 지정한다면, 그 번호가
-- 재사용된 뒤에는 전혀 다른 파일을 조용히 건드리게 된다.
-- 시끄럽게 실패하는 편이 조용히 잘못 동작하는 것보다 안전하다.

--------------------------------------------------------------------------------
 [9] 정리
--------------------------------------------------------------------------------

-- 재현한 오류
--   ORA-01516   incarnation 한 자리 오타
--
-- 확인한 것
--   1) 파일 이름은 다섯 조각. 디스크그룹·DB·컨테이너·종류·이름
--   2) 32자리 16진수는 PDB 의 GUID. v$pdbs 로 확인된다
--   3) CDB$ROOT 와 컨트롤파일·리두는 GUID 조각이 없다
--   4) 파일 번호는 디스크그룹 안에서만 유일하다
--   5) v$asm_file 에는 이름 컬럼이 없다. 번호로 관리한다
--   6) 크기 소수점은 데이터파일 헤더 블록 하나 때문이다
--   7) 이름을 손으로 입력하면 ORA-01516 을 만난다
--
-- 기억할 것
--   이름은 복사하거나 파일 번호로 지정한다.
--   alter database datafile 7 offline 처럼 쓰면 오타가 생기지 않는다.
