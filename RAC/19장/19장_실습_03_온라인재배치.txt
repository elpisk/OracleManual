 [1] 후보 서버에 rac2 를 넣는다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl modify database -db racdb -node rac1,rac2
PRKF-1125 : multiple values specified for the single value option "node": rac1,rac2

-- 0.6초. 거부됐다.
--
-- 원인 파악 : 이 버전의 srvctl 에서 modify database 의 -node 는 값 하나만 받는다.
--             후보 서버 목록을 통째로 넣는 옵션이 아니다.
--             목록을 넣으려는 생각은 Candidate servers 줄이 쉼표 목록이라
--             자연스럽게 나오지만, 명령은 그렇게 동작하지 않는다.
-- 조치     : 재배치 대상 노드를 relocate 의 -node 로 직접 준다.
--             관리자 관리형 RAC One Node 는 후보에 없는 노드로 재배치하면
--             그 노드를 후보에 추가한다. [2] 의 첫 줄에서 확인한다.

[oracle@rac1 ~]$ srvctl config database -db racdb | grep -E "Candidate|instances|Type"

Candidate servers: rac1
Database instances: racdb_1

 [3-1] 흔한 오류 — rac2 에서 sqlplus 로 붙으면
--------------------------------------------------------------------------------

[oracle@rac2 ~]$ sqlplus / as sysdba
SQL> select inst_id, instance_name, host_name, thread#, status from gv$instance;
select inst_id, instance_name, host_name, thread#, status from gv$instance
*
ERROR at line 1:
ORA-01034: ORACLE not available
Process ID: 0
Session ID: 0 Serial number: 0

-- 데이터베이스는 rac2 에서 Open 인데 ORA-01034 가 났다.
--
-- 원인 파악 : 접속한 인스턴스 이름이 다르다.

[oracle@rac2 ~]$ ps -ef | grep [o]ra_pmon
oracle   22653     1  0 18:13 ?        00:00:00 ora_pmon_racdb_2
[oracle@rac2 ~]$ echo $ORACLE_SID
racdb2

--             oracle 계정의 프로필은 ORACLE_SID=racdb2 로 되어 있다.
--             실제 인스턴스는 racdb_2 다. 로컬 접속(/ as sysdba)은 ORACLE_SID 로
--             인스턴스를 찾으므로 없는 인스턴스 racdb2 에 붙으려 했다.
--             RAC One Node 의 인스턴스 이름은 재배치할 때마다 접두어_1, 접두어_2 로
--             바뀐다. 고정된 ORACLE_SID 와 맞지 않는 것이 정상이다.
-- 조치     : 이번 접속에서만 SID 를 바꾼다.

[oracle@rac2 ~]$ export ORACLE_SID=racdb_2
[oracle@rac2 ~]$ sqlplus / as sysdba

SQL> select inst_id, instance_name, host_name, thread#, status from gv$instance;

   INST_ID INSTANCE_NAM HOST_NAME             THREAD# STATUS
---------- ------------ ------------------ ---------- ------------
         2 racdb_2      rac2.localdomain            2 OPEN

SQL> select value from v$parameter where name='undo_tablespace';

VALUE
------------
UNDOTBS2

SQL> select name, open_mode from v$pdbs where name='RACPDB1';

NAME             OPEN_MODE
---------------- ----------
RACPDB1          READ WRITE

-- 재확인 : racdb_2 는 스레드 2 와 UNDOTBS2 를 쓰고 있다.
--
-- SPFILE 을 다시 보면 racdb_2 전용 항목은 없다.

SQL> select sid, name, value from v$spparameter
  2   where name in ('instance_number','thread','undo_tablespace') and value is not null
  3   order by sid, name;

SID      NAME             VALUE
-------- ---------------- ------------
racdb1   instance_number  1
racdb1   thread           1
racdb1   undo_tablespace  UNDOTBS1
racdb2   instance_number  2
racdb2   thread           2
racdb2   undo_tablespace  UNDOTBS2

-- 인스턴스 racdb_2 에 지정된 thread, undo_tablespace 가 없는데도
-- 열려 있는 스레드 2 와 쓰이지 않던 UNDOTBS2 를 잡았다.
-- 실습 02 에서 srvctl remove instance 로 등록만 해제하고 스레드와 UNDO 를
-- 남겨 둔 것이 여기서 쓰였다.
--
-- 운영에서는 ORACLE_SID 에 기대지 말고 서비스(ronsvc)로 접속한다.
-- 어느 노드의 어느 인스턴스로 가든 서비스 이름은 그대로다.
