-- OCR 과 세 가지가 다르다.
--   1) ID 가 노드마다 다르다. rac1 은 1636561748, rac2 는 877385927
--   2) 위치가 ASM 이 아니라 로컬 파일시스템이다
--   3) Logical corruption check 줄이 없다. OLR 은 구조가 단순하다
--
-- ID 가 다른 것이 핵심이다. OCR 은 클러스터 하나에 하나이므로 어느 노드에서
-- 조회해도 665666554 가 나오지만, OLR 은 노드마다 별개의 저장소다.

--------------------------------------------------------------------------------
 [3] 부트스트랩 파일 두 개
--------------------------------------------------------------------------------

[root@rac1 ~]# cat /etc/oracle/ocr.loc
#Device/file +CRS getting replaced by device +CRS/rac-cluster/OCRFILE/registry.255.1242760249
ocrconfig_loc=+CRS/rac-cluster/OCRFILE/registry.255.1242760249
local_only=false

[root@rac1 ~]# cat /etc/oracle/olr.loc
olrconfig_loc=/u01/app/grid/crsdata/rac1/olr/rac1_19.olr
crs_home=/u01/app/19.3.0/grid

-- ocr.loc 은 ocrcheck -config 보다 해상도가 높다. 디스크그룹 이름만이 아니라
-- 그 안의 ASM 파일 번호 registry.255.1242760249 까지 적혀 있다.
--
-- 첫 줄 주석은 ocrconfig 명령이 자동으로 남긴 변경 이력이다. 누가 언제 OCR
-- 위치를 바꿨는지 추적하는 단서가 된다. 실습 04 에서 이 줄이 바뀌는 것을 본다.
--
-- local_only=false 는 클러스터 구성원이라는 뜻이다. Oracle Restart 단일 노드
-- 환경에서는 true 가 된다.

-- rac2 의 ocr.loc 도 확인한다. 두 노드가 일치해야 한다.
[root@rac2 ~]# cat /etc/oracle/ocr.loc
ocrconfig_loc=+CRS/rac-cluster/OCRFILE/registry.255.1242760249
local_only=FALSE

-- ocrconfig_loc 값이 rac1 과 같다. local_only 의 대소문자가 다른 것은
-- 설치 시점에 작성된 파일과 이후 명령이 갱신한 파일의 차이이며 동작에는
-- 영향이 없다. 값이 같은지만 보면 된다.

-- 3개인 이유는 +CRS 가 Normal 중복도이기 때문이다. 우리가 개수를 정한 것이
-- 아니라 디스크그룹의 중복도가 정한다.
--
--   External  1개   손실 허용 0
--   Normal    3개   손실 허용 1
--   High      5개   손실 허용 2
--
-- 항상 홀수인 이유는 과반수 규칙 때문이다. 노드는 Voting File 의 과반수에
-- 접근할 수 있어야 클러스터 구성원으로 남는다. 짝수면 반반으로 갈릴 때
-- 결론이 나지 않는다.
--
-- File Name 컬럼이 ASM 파일 경로가 아니라 원시 장치 경로라는 점을 눈여겨본다.
-- 이유는 [5] 에서 확인한다.

[root@rac2 ~]# crsctl query css votedisk

 [5] ASM 쪽에서 본 +CRS
--------------------------------------------------------------------------------

[grid@rac1 ~]$ sqlplus -s / as sysasm

SQL> select group_number gn, name, state, type, total_mb, free_mb, voting_files
  2    from v$asm_diskgroup order by group_number;

        GN NAME       STATE       TYPE     TOTAL_MB    FREE_MB V
---------- ---------- ----------- ------ ---------- ---------- -
         1 CRS        MOUNTED     NORMAL      12288      11372 Y
         2 DATA       MOUNTED     EXTERN      61440      56986 N
         3 FRA        MOUNTED     EXTERN      24576      23636 N

SQL> select d.disk_number dn, d.name, d.path, d.mount_status, d.header_status,
  2         d.mode_status, d.state, d.failgroup, d.total_mb, d.free_mb
  3    from v$asm_disk d, v$asm_diskgroup g
  4   where d.group_number=g.group_number and g.name='CRS' order by d.disk_number;

        DN NAME       PATH                   MOUNT_S HEADER_STATU MODE_ST STATE    FAILGROUP    TOTAL_MB    FREE_MB
---------- ---------- ---------------------- ------- ------------ ------- -------- ---------- ---------- ----------
         0 CRS_0000   /dev/asmdisk/crs1      CACHED  MEMBER       ONLINE  NORMAL   CRS_0000         4096       3792
         1 CRS_0001   /dev/asmdisk/crs2      CACHED  MEMBER       ONLINE  NORMAL   CRS_0001         4096       3788
         2 CRS_0002   /dev/asmdisk/crs3      CACHED  MEMBER       ONLINE  NORMAL   CRS_0002         4096       3792

-- VOTING_FILES 컬럼이 Y 인 디스크그룹은 CRS 하나다.
-- 디스크 3개가 각각 다른 장애 그룹(FAILGROUP)에 속해 있다. 그래서 하나가
-- 죽어도 나머지 둘이 과반수를 유지한다.

-- 여기서 디스크그룹 속성도 함께 확인해 둔다. 실습 06 에서 문제가 된다.
SQL> select name, value from v$asm_attribute
  2   where group_number=(select group_number from v$asm_diskgroup where name='CRS')
  3     and name in ('compatible.asm','compatible.rdbms','au_size','sector_size');

NAME                VALUE
------------------- ------------
sector_size         512
au_size             4194304
compatible.asm      19.0.0.0.0
compatible.rdbms    10.1.0.0.0

-- compatible.rdbms 가 10.1.0.0.0 이다. 19c 로 설치했는데 왜 10.1 인지,
-- 이 값이 무엇을 막는지는 실습 06 에서 다룬다. 지금은 값만 기록해 둔다.

-- 이제 Voting File 을 파일 목록에서 찾아본다.
[grid@rac1 ~]$ asmcmd ls +CRS/rac-cluster/
ASMPARAMETERFILE/
OCRBACKUP/
OCRFILE/

-- 없다. 이것이 정상이다.
-- Voting File 은 ASM 파일이 아니다. 디스크의 지정된 오프셋에 직접 기록된다.
-- 이유는 부팅 순서다. CSS 는 ASM 인스턴스가 뜨기 전에 Voting File 을 읽어야
-- 하는데, ASM 이 필요한 자리에 ASM 을 쓸 수는 없다.
-- 그래서 crsctl 출력의 File Name 이 원시 장치 경로였던 것이다.

(ocrcheck 정상 출력. root 로 실행한 것과 같다)

-- 그런데 변경 계열 명령은 다르다.
[grid@rac1 ~]$ ocrconfig -manualbackup
PROT-20: Insufficient permission to proceed. Require privileged user

[grid@rac1 ~]$ ocrconfig -export /tmp/g.dmp
PROT-20: Insufficient permission to proceed. Require privileged user

-- 오류 : PROT-20
-- 원인 파악 : Insufficient permission to proceed. Require privileged user.
--             OCR 은 Clusterware 의 구성 그 자체이므로 변경 권한을 root 로
--             좁혀 두었다. 읽기(ocrcheck, -showbackup)는 grid 도 되지만
--             쓰기(-manualbackup, -export, -add, -delete)는 root 만 된다.
-- 조치     : root 로 실행한다. 백업 스크립트를 cron 에 넣을 때도 grid 의
--             cron 이 아니라 root 의 cron 이거나 sudo 를 거쳐야 한다.
-- 재확인   : 실패한 두 명령이 아무것도 만들지 않았는지 확인한다.

[grid@rac1 ~]$ ls -l /tmp/g.dmp
ls: cannot access /tmp/g.dmp: No such file or directory

[root@rac1 ~]# ocrconfig -showbackup
PROT-24: Auto backups for the Oracle Cluster Registry are not available
PROT-25: Manual backups for the Oracle Cluster Registry are not available

-- 파일도 만들어지지 않았고 백업 목록도 그대로 비어 있다.

--------------------------------------------------------------------------------
 [8] 정리
--------------------------------------------------------------------------------

-- 기록한 대조 기준
--   OCR ID          665666554        (끝까지 바뀌면 안 된다)
--   OCR 위치        +CRS             (실습 04 에서 일시적으로 늘어난다)
--   OLR ID          rac1 1636561748 / rac2 877385927
--   Voting FUID     838fca04 / 4c38f09d / 370e2d1c
--   Voting 개수     3 (Normal 중복도)
--   클러스터명      rac-cluster
--   misscount 30 초, disktimeout 200 초, reboottime 3 초
--   compatible.asm 19.0.0.0.0 / compatible.rdbms 10.1.0.0.0
--
-- 확인한 사실
--   1) OCR 은 최대 5벌. 지금은 1벌이고 슬롯 4개가 비어 있다
--   2) OCR ID 는 클러스터당 하나, OLR ID 는 노드마다 다르다
--   3) ocrcheck 는 물리 검사와 논리 검사를 따로 한다
--   4) Voting File 개수는 중복도가 정한다. Normal 이라 3개
--   5) Voting File 은 ASM 파일이 아니라 디스크 오프셋에 직접 있다
--   6) 상태 조회는 grid 로도 되지만 변경은 root 만 된다(PROT-20)
--
-- FUID 는 실습 05 와 06 에서 바뀐다. 바뀌는 것이 정상이며, 무엇이 바뀌고
-- 무엇이 그대로인지를 구분하는 것이 이 기록의 목적이다.
