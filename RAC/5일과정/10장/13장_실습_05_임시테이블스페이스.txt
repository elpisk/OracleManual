================================================================================
 제13장 실습 05 — 임시 테이블스페이스, 공유되는 이유와 노드별 사용량
 대상   : rac1
 목표   : TEMP 가 하나뿐임을 확인하고, 그런데도 사용량이 인스턴스별로 잡히는
          구조를 이해한다. 임시 공간 부족 진단의 두 축을 익힌다
 소요   : 약 25분
 선행   : racdb 양 노드 OPEN
 계정   : oracle
================================================================================

--------------------------------------------------------------------------------
 [1] 임시 테이블스페이스는 하나다
--------------------------------------------------------------------------------

SQL> select tablespace_name, contents, status, extent_management
  2    from dba_tablespaces where contents='TEMPORARY';

TABLESPACE_NAME  CONTENTS              STATUS    EXTENT_MAN
---------------- --------------------- --------- ----------
TEMP             TEMPORARY             ONLINE    LOCAL

1 row selected.

-- 인스턴스가 둘인데 임시 테이블스페이스는 하나다.
-- 리두로그와 UNDO 는 인스턴스마다 있었다. 여기서 갈린다.

--------------------------------------------------------------------------------
 [2] 템프파일이 셋인 이유
--------------------------------------------------------------------------------

SQL> select con_id, name, bytes/1024/1024 mb from v$tempfile order by con_id;

    CON_ID NAME                                                        MB
---------- --------------------------------------------------- ----------
         1 +DATA/RACDB/TEMPFILE/temp.264.1242761781                    36
         2 +DATA/RACDB/5A566B559A25.../TEMPFILE/temp.268.1242762013    36
         3 +DATA/RACDB/5A569448ED6C.../TEMPFILE/temp.276.1242762707    36

3 rows selected.

-- 파일이 셋이라 인스턴스별로 나뉜 것처럼 오해하기 쉽다. 아니다.
-- CON_ID 를 함께 조회하면 이유가 드러난다.
--
--   CON_ID 1 = CDB$ROOT
--   CON_ID 2 = PDB$SEED
--   CON_ID 3 = RACPDB1
--
-- 컨테이너마다 자기 템프파일을 갖는 것이지 인스턴스와는 무관하다.
-- 9장에서 GUID 디렉터리를 추적할 때 본 그 구조다.
-- 인스턴스가 몇이든 임시 테이블스페이스 개수는 변하지 않는다.

--------------------------------------------------------------------------------
 [3] 왜 나누지 않는가
--------------------------------------------------------------------------------

-- 세 자원의 쓰기 방식을 나란히 놓으면 답이 나온다.
--
--   리두 : 로그 라이터가 파일 끝에 순차로 이어 붙인다.
--          두 인스턴스가 같은 파일에 붙일 수 없다.          → 나눈다
--
--   UNDO : 트랜잭션마다 롤백 세그먼트를 배타적으로 배정받는다.
--          같은 것을 쓰면 배정을 두고 경합한다.              → 나눈다
--
--   임시 : 세션이 필요한 만큼 익스텐트를 할당받아 쓰고 반납한다.
--          할당 단위가 처음부터 나뉘어 있다.                 → 공유한다
--
-- 기준은 하나다. 동시에 쓸 수 있는가.

--------------------------------------------------------------------------------
 [4] 그런데 사용량은 노드별로 잡힌다
--------------------------------------------------------------------------------

SQL> select inst_id, tablespace_name, total_blocks, used_blocks, free_blocks
  2    from gv$sort_segment order by inst_id, total_blocks desc;

   INST_ID TABLESPACE_NAME    TOTAL_BLOCKS USED_BLOCKS FREE_BLOCKS
---------- ------------------ ------------ ----------- -----------
         1 TEMP                       2688         256        2432
         1 TEMP                        512           0         512
         2 TEMP                       1280         128        1152
         2 TEMP                        384           0         384

4 rows selected.

-- 테이블스페이스는 하나인데 인스턴스별로 행이 나온다.
--
-- 각 인스턴스가 자기 몫의 익스텐트를 미리 확보해 두고
-- 자기 세션들에 나눠 주기 때문이다.
-- 확보한 익스텐트는 세션이 끝나도 바로 반납되지 않고 인스턴스가 계속 들고 있다.
--
-- 인스턴스 1 이 2688 + 512, 인스턴스 2 가 1280 + 384 를 확보했다.
-- 인스턴스 1 쪽이 많은 것은 그동안 더 많은 정렬 작업을 했기 때문이다.
--
-- 행이 인스턴스마다 둘씩인 것은 컨테이너 때문이다. [2] 와 같은 이유다.
--
-- 이것이 "구조는 공유, 사용은 분리" 다.
-- 테이블스페이스라는 구조는 하나이고 양 노드가 같은 것을 보지만
-- 실제 쓰는 공간은 인스턴스별로 나뉘어 있다.

--------------------------------------------------------------------------------
 [5] 흔한 오류 — 없는 뷰로 세그먼트를 세려 하면
--------------------------------------------------------------------------------

SQL> select inst_id, count(*) segs from gv$rollname group by inst_id order by 1;
select inst_id, count(*) segs from gv$rollname group by inst_id order by 1
                                   *
ERROR at line 1:
ORA-00942: table or view does not exist

-- 원인 파악 : V$ROLLNAME 은 있지만 GV$ROLLNAME 은 없다.
--             모든 V$ 뷰에 대응하는 GV$ 뷰가 있는 것은 아니다.
--             GV$ 는 클러스터 전체를 모아 보여 주는 뷰이므로
--             그럴 필요가 있는 것에만 만들어져 있다.
-- 조치     : 필요한 것이 세그먼트 개수라면 딕셔너리 뷰를 쓴다.
--             DBA_ROLLBACK_SEGS 는 테이블스페이스별로 세면 되므로
--             인스턴스 축이 필요 없다. UNDO 테이블스페이스가 이미
--             인스턴스에 대응하기 때문이다.
--             GV$ 뷰의 존재 여부는 다음으로 확인한다.
--               select view_name from v$fixed_view_definition
--                where view_name like 'GV$ROLL%';
-- 재확인   : 테이블스페이스별 개수가 나오는지 본다.

SQL> select tablespace_name, count(*) cnt from dba_rollback_segs
  2   where tablespace_name like 'UNDO%' group by tablespace_name;

TABLESPACE_NAM        CNT
-------------- ----------
UNDOTBS1               10
UNDOTBS2               10

2 rows selected.

--------------------------------------------------------------------------------
 [6] 임시 공간 부족을 진단하는 두 축
--------------------------------------------------------------------------------

-- ORA-01652 를 만났을 때 RAC 에서는 두 가지를 함께 봐야 한다.

-- 축 하나 : 전체 여유가 있는가

SQL> select file_id, tablespace_name, bytes/1024/1024 mb, autoextensible,
  2         maxbytes/1024/1024 max_mb from dba_temp_files;

   FILE_ID TABLESPACE_NAME            MB AUT     MAX_MB
---------- ------------------ ---------- --- ----------
         1 TEMP                       36 YES 32767.9844

1 row selected.

-- 자동 확장이 켜져 있고 상한이 32767.98MB 다. 여유가 크다.
-- 자동 확장이 꺼져 있거나 상한에 도달했다면 이쪽이 원인이다.

-- 축 둘 : 인스턴스별 확보량이 균형인가
--         → [4] 의 GV$SORT_SEGMENT 조회다.
--
-- 전체는 충분한데 한 인스턴스가 대부분을 확보하고 있으면
-- 다른 쪽 세션이 부족을 만난다. 단일 인스턴스에는 없는 상황이다.
--
-- 두 조회를 한 세트로 실행하는 습관을 들이면 진단이 빨라진다.
-- 첫 번째만 보고 "공간은 충분한데 왜 오류가 나지" 하며 헤매는 일이 사라진다.
