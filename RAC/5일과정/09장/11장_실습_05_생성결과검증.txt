--------------------------------------------------------------------------------
 [1] 접속 준비
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ export ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_rac
[oracle@rac1 ~]$ export ORACLE_SID=testdb1
[oracle@rac1 ~]$ $ORACLE_HOME/bin/sqlplus -s / as sysdba

-- ORACLE_SID 를 testdb1 로 바꿔야 한다. racdb1 인 채로 접속하면
-- 엉뚱한 데이터베이스를 보게 된다.
--
-- 조용한 함정 : sqlplus 를 PATH 없이 호출하면 command not found 가 난다.
--               $ORACLE_HOME/bin/ 을 붙이거나 PATH 에 추가한다.
--
-- 접속한 뒤 어느 데이터베이스인지 반드시 확인한다.

SQL> select name, db_unique_name from v$database;

NAME      DB_UNIQUE_NAME
--------- --------------
TESTDB    testdb

1 row selected.

SQL> select name from v$datafile where name not like '+%';

no rows selected

-- 두 번째 조회는 10장에서 배운 점검이다. 0 행이면 모든 데이터파일이
-- ASM 에 있다는 뜻이다.
--
-- 원인 파악 : "no rows selected" 를 조회 실패로 오해하기 쉽다.
--             여기서는 0 행이 곧 정상이라는 답이다.
--             파일이 하나라도 파일 시스템에 있으면 그 이름이 나온다.
-- 조치     : 조건을 고칠 것이 아니라 반대 조건으로 대조해 조회 자체가
--             동작하는지 확인한다.
-- 재확인   : like '+%' 로 바꿔 행이 나오는지 본다. 실습 05 [6] 에서
--             ASM 쪽으로도 교차 확인한다.

--------------------------------------------------------------------------------
 [2] 검증 1·2 — 인스턴스 둘, 리두 스레드 둘
--------------------------------------------------------------------------------

SQL> col host_name for a10
SQL> select inst_id, instance_name, host_name, thread#, status
  2    from gv$instance order by 1;

   INST_ID INSTANCE_NAME    HOST_NAME     THREAD# STATUS
---------- ---------------- ---------- ---------- ------------
         1 testdb1          rac1.local          1 OPEN
                            domain
         2 testdb2          rac2.local          2 OPEN
                            domain

2 rows selected.

-- 세 가지를 함께 확인한다.
--   인스턴스가 둘인가              → 2 행
--   서로 다른 노드에 있는가        → rac1, rac2
--   리두 스레드가 다른가           → 1, 2
--
-- 인스턴스가 둘이라는 것만으로는 부족하다. 셋이 모두 맞아야 한다.
-- THREAD# 가 같으면 두 인스턴스가 같은 리두로그에 동시에 쓰려 해 로그가 망가진다.

--------------------------------------------------------------------------------
 [3] 검증 3 — 리두로그가 스레드마다 있는가
--------------------------------------------------------------------------------

SQL> select group#, thread#, bytes/1024/1024 mb, status from v$log order by group#;

    GROUP#    THREAD#         MB STATUS
---------- ---------- ---------- ----------------
         1          1         20 ACTIVE
         2          1         20 CURRENT
         3          2         20 INACTIVE
         4          2         20 CURRENT

4 rows selected.

-- 4 개이고 스레드마다 2 개씩이다. 크기는 지정한 대로 20MB 다.
-- 스레드마다 CURRENT 가 하나씩 있다.
-- 각 인스턴스가 독립적으로 로그 스위치를 한다는 뜻이다.
--
-- racdb 와 비교하면 차이가 있다.
--   racdb  — 8 행 (4 그룹 × 2 멤버, +DATA 와 +FRA 에 다중화)
--   testdb — 4 행 (다중화하지 않음)
--
-- 이것은 생성 옵션의 차이이지 RAC 여부와 무관하다.
-- 운영에서는 리두로그를 다중화하는 것이 원칙이다.

--------------------------------------------------------------------------------
 [4] 검증 4 — UNDO 가 인스턴스 수만큼 있는가
--------------------------------------------------------------------------------

SQL> col tablespace_name for a16
SQL> select tablespace_name, status from dba_tablespaces
  2   where contents='UNDO' order by 1;

TABLESPACE_NAME  STATUS
---------------- ---------
UNDOTBS1         ONLINE
UNDOTBS2         ONLINE

2 rows selected.

SQL> select inst_id, value from gv$parameter where name='undo_tablespace' order by 1;

   INST_ID VALUE
---------- ----------
         1 UNDOTBS1
         2 UNDOTBS2

2 rows selected.

-- UNDO 테이블스페이스가 둘이고 인스턴스마다 하나씩 배정되었다.
--
-- 단일 인스턴스라면 하나뿐이다. 둘인 것 자체가 RAC 이라는 증거다.
-- 같은 것을 쓰면 두 인스턴스가 UNDO 세그먼트를 두고 다투게 된다.

--------------------------------------------------------------------------------
 [5] 검증 5 — SPFILE 의 인스턴스별 지정
--------------------------------------------------------------------------------

SQL> col name for a20
SQL> col value for a26
SQL> col sid for a10
SQL> select sid, name, value from v$spparameter
  2   where isspecified='TRUE' and sid<>'*' order by sid, name;

SID        NAME                 VALUE
---------- -------------------- --------------------------
testdb1    instance_number      1
testdb1    thread               1
testdb1    undo_tablespace      UNDOTBS1
testdb2    instance_number      2
testdb2    thread               2
testdb2    undo_tablespace      UNDOTBS2

6 rows selected.

SQL> select sid, count(*) cnt from v$spparameter where isspecified='TRUE'
  2   group by sid order by 1;

SID               CNT
---------- ----------
*                  23
testdb1             3
testdb2             3

3 rows selected.

-- 12장에서 다룬 구조가 그대로 나타난다.
-- dbca 가 세 파라미터를 인스턴스별로 지정했다.
--
-- 공통 항목이 23 개다. racdb 의 24 개와 다르다.
--
-- 원인 파악 : 두 데이터베이스의 생성 옵션이 달라서다.
--             리두 크기, 다중화, EM 구성 등 선택이 SPFILE 항목 수를 바꾼다.
-- 조치     : 이 숫자를 절대값으로 삼지 않는다.
--             각 데이터베이스의 기준선으로 따로 기록해 둔다.
-- 재확인   : 인스턴스별 3 개씩은 둘 다 같다.
--             이 셋은 RAC 의 구조적 요구이므로 어느 데이터베이스에서나 같다.
--
-- 검증에서 기준으로 삼을 것은 인스턴스별 항목 쪽이다.
-- 공통 항목이 23 이든 24 든 정상이지만, 인스턴스별 항목이 3 개가 아니거나
-- 그 셋이 아닌 다른 파라미터가 들어 있다면 문제다.

--------------------------------------------------------------------------------
 [6] 검증 6 — 데이터파일이 의도한 곳에 있는가
--------------------------------------------------------------------------------

[grid@rac1 ~]$ asmcmd ls +DATA
RACDB/
TESTDB/

[grid@rac1 ~]$ asmcmd du +DATA/TESTDB
Used_MB      Mirror_used_MB
   3525                3525

[grid@rac1 ~]$ asmcmd ls +FRA
RACDB/

-- +DATA 에 TESTDB 디렉터리가 생겼고 3525MB 를 쓰고 있다.
-- 데이터파일이 의도한 디스크그룹에 들어갔다는 확인이다.
--
-- 실습 02 [2] 에서 -diskGroupName 이 도움말에 없는 인자임을 확인했다.
-- 그럼에도 파일이 +DATA 에 들어갔는지는 이렇게 결과로 판정한다.
-- 인자가 어떻게 해석되었든 결과가 의도와 맞으면 된다.
--
-- 데이터베이스 쪽에서도 같은 확인이 가능하다. 10장에서 배운 조회다.
--   select name from v$datafile where name not like '+%';
-- 0 행이면 모든 데이터파일이 ASM 에 있다는 뜻이다.
--
-- +FRA 에는 아직 RACDB 만 있다. 아카이브를 켜지 않았고 백업도 받지 않아
-- testdb 가 FRA 를 쓸 일이 아직 없었기 때문이다.
-- 컨트롤파일 자동 백업이 일어나면 그때 TESTDB 디렉터리가 생긴다.

--------------------------------------------------------------------------------
 [7] 여섯 기준 정리
--------------------------------------------------------------------------------

--  검증 항목             기대값                        실측
--  -------------------  ----------------------------  ----------------------
--  1 CRS 자원            양 노드 ONLINE                ONLINE / ONLINE
--  2 인스턴스            2개, 서로 다른 노드           testdb1@rac1, testdb2@rac2
--  3 리두 스레드         2개                           THREAD# 1, 2
--  4 리두로그            스레드당 2개                  4개 (1·2=스레드1, 3·4=스레드2)
--  5 UNDO                2개                           UNDOTBS1, UNDOTBS2
--  6 SPFILE 인스턴스별   각 3항목                      3 / 3
--
-- 하나라도 어긋나면 단일 인스턴스가 만들어졌거나 생성이 완전하지 않은 것이다.
-- 특히 -databaseConfigType RAC 을 빠뜨리면 1·2·3·5·6 이 모두 어긋난다.

--------------------------------------------------------------------------------
 [8] racdb 와 나란히 놓고 보기
--------------------------------------------------------------------------------

--  항목                  racdb              testdb             성격
--  -------------------  -----------------  -----------------  ---------------
--  인스턴스 수           2                  2                  RAC 이면 공통
--  리두 스레드           1, 2               1, 2               RAC 이면 공통
--  UNDO                  UNDOTBS1/2         UNDOTBS1/2         RAC 이면 공통
--  SPFILE 인스턴스별     3항목              3항목              RAC 이면 공통
--  SPFILE 공통 항목      24                 23                 DB 마다 다름
--  리두로그 크기         200MB              20MB               생성 옵션
--  리두로그 다중화       +DATA·+FRA 2벌     +DATA 1벌          생성 옵션
--  PDB 저장 상태         2행                0행                사람이 해야 함
--
-- 위 넷은 RAC 데이터베이스라면 반드시 그래야 하는 것이다.
-- 아래 넷은 만들 때의 선택이나 그 뒤의 관리에 달린 것이다.
--
-- 이 구분이 검증에서 중요하다.
-- 위 넷 중 하나라도 어긋나면 생성이 잘못된 것이고,
-- 아래 넷이 다른 것은 문제가 아니다.
--
-- 마지막 줄이 다음 실습의 주제다.
-- 둘 다 dbca 로 만들었는데 PDB 저장 상태가 다르다.
-- racdb 쪽은 누군가 나중에 저장했기 때문이다.
