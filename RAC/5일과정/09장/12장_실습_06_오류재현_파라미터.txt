================================================================================
 제12장 실습 06 — 오류 재현 : 같아야 할 것을 한쪽만 바꾸면
 대상   : rac1
 목표   : 오라클이 막아 주는 경우와 막지 않는 경우를 직접 확인하고, 되돌리는
          올바른 방법을 익힌다
 소요   : 약 25분
 선행   : racdb 양 노드 OPEN, SPFILE 항목 수 24 / 3 / 3
 계정   : oracle
 주의   : SPFILE 을 실제로 수정한다. [6] 의 RESET 단계를 반드시 수행해야 하며,
          건너뛰면 다음 재기동에서 racdb1 이 올라오지 않는다
================================================================================

--------------------------------------------------------------------------------
 [1] 시작 전 항목 수 기록
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ sqlplus -s / as sysdba

SQL> select sid, count(*) cnt from v$spparameter where isspecified='TRUE'
  2   group by sid order by 1;

SID              CNT
-------- ----------
*                 24
racdb1             3
racdb2             3

3 rows selected.

-- 24 / 3 / 3 이 시작점이다. 실습이 끝나면 이 값으로 돌아와야 한다.
--
-- 이 실습의 모든 변경은 scope=spfile 로만 한다.
-- 지금 돌고 있는 인스턴스에는 영향이 없으므로 실습 도중 클러스터가
-- 무너지지 않는다. 대신 문제가 다음 재기동에서 드러난다.

--------------------------------------------------------------------------------
 [2] 오류 재현 1 — 오라클이 막아 주는 경우
--------------------------------------------------------------------------------

SQL> alter system set db_block_size=16384 sid='racdb1' scope=spfile;
alter system set db_block_size=16384 sid='racdb1' scope=spfile
                 *
ERROR at line 1:
ORA-32017: failure in updating SPFILE
ORA-32016: parameter "db_block_size" cannot be updated in SPFILE

SQL> alter system set db_name='xxx' sid='racdb1' scope=spfile;
alter system set db_name='xxx' sid='racdb1' scope=spfile
                 *
ERROR at line 1:
ORA-32017: failure in updating SPFILE
ORA-32016: parameter "db_name" cannot be updated in SPFILE

-- 오류 : ORA-32017 / ORA-32016
--
-- 원인 파악 : db_block_size 와 db_name 은 데이터베이스를 만들 때 확정되어
--             이후 어떤 방법으로도 바꿀 수 없는 값이다.
--             인스턴스별 지정 이전에 변경 자체가 막혀 있다.
--             sid= 를 빼고 실행해도 같은 오류가 난다.
-- 조치     : 조치할 것이 없다. 애초에 시도하면 안 되는 변경이다.
-- 재확인   : 다음 단계에서 SPFILE 에 아무것도 들어가지 않았는지 확인한다.
--
-- 여기까지 보면 "오라클이 알아서 막아 주는구나" 하고 안심하게 된다.
-- 같은 "반드시 같아야" 분류인 control_files 도 막힐 것 같은가.

--------------------------------------------------------------------------------
 [3] 오류 재현 2 — 그런데 막지 않는다
--------------------------------------------------------------------------------

SQL> alter system set cluster_database=false sid='racdb1' scope=spfile;

System altered.

SQL> alter system set control_files='+DATA/x' sid='racdb1' scope=spfile;

System altered.

-- 둘 다 통과했다. 경고 하나 없다.
-- 실제로 SPFILE 에 들어갔는지 확인한다.

SQL> col name for a20
SQL> col value for a34
SQL> select sid, name, value from v$spparameter
  2   where name in ('db_block_size','cluster_database','control_files','db_name')
  3     and isspecified='TRUE' order by name, sid;

SID      NAME                 VALUE
-------- -------------------- ----------------------------------
*        cluster_database     true
racdb1   cluster_database     FALSE
*        control_files        +DATA/RACDB/CONTROLFILE/current.26
                              1.1242761745
*        control_files        +FRA/RACDB/CONTROLFILE/current.256
                              .1242761745
racdb1   control_files        +DATA/x
*        db_block_size        8192
*        db_name              racdb

7 rows selected.

-- 들어갔다.
--   racdb1 전용 항목이 두 개 생겼고 `*` 값과 공존한다.
--   db_block_size 와 db_name 은 [2] 에서 거부되었으므로 `*` 만 있다.
--
-- 인스턴스별 값이 있으면 그것이 `*` 보다 우선한다.
-- 다음에 racdb1 을 기동하면 cluster_database=FALSE 와
-- control_files=+DATA/x 를 읽는다.

SQL> select sid, count(*) cnt from v$spparameter where isspecified='TRUE'
  2   group by sid order by 1;

SID              CNT
-------- ----------
*                 24
racdb1             5
racdb2             3

3 rows selected.

-- racdb1 항목이 3 에서 5 로 늘었다. 기준선이 깨졌다.
-- 항목 수만 세어도 무언가 들어간 것을 알 수 있다.

--------------------------------------------------------------------------------
 [4] 이 상태로 재기동하면 무슨 일이 생기는가
--------------------------------------------------------------------------------

-- 실제로 재기동하지는 않는다. 예상되는 결과를 정리한다.
--
--  control_files='+DATA/x'
--    존재하지 않는 경로다. racdb1 은 컨트롤파일을 읽지 못해 MOUNT 단계에서
--    실패한다. ORA-00205 계열로 기동이 중단된다.
--
--  cluster_database=FALSE
--    control_files 를 고쳐 기동이 된다 해도, racdb1 이 자기를 단일 인스턴스로
--    여기고 데이터베이스를 배타적으로 열려 한다.
--    racdb2 가 이미 열고 있으므로 ORA-01102 계열로 실패한다.
--
-- 두 경우 모두 CRS 가 RESTART_ATTEMPTS(2) 만큼 재시도한 뒤 포기한다.
-- 그 사이 몇 분이 지나간다.
--
-- 가장 무서운 점 : 명령을 내리는 순간에는 아무 일도 일어나지 않는다.
--                  System altered 만 보고 성공했다고 판단하기 쉽다.
--                  문제는 다음 재기동에서 드러나고, 그때는 이 변경을 한 사람이
--                  자리에 없을 수도 있다.

--------------------------------------------------------------------------------
 [5] 왜 오라클은 막지 않는가
--------------------------------------------------------------------------------

-- 막지 않는 데는 이유가 있다. 두 파라미터 모두 정당한 사용 사례가 있다.
--
--  cluster_database=FALSE
--    RAC 데이터베이스를 단일 인스턴스 모드로 열어야 하는 작업이 있다.
--    업그레이드 스크립트나 특정 복구 작업이 그렇다.
--    그때는 다른 인스턴스를 모두 내리고 한 인스턴스만 이 값으로 올린다.
--
--  control_files
--    10장 실습 05 에서 본 컨트롤파일 이동 절차가 이 파라미터를
--    scope=spfile 로 바꾸는 것에서 시작한다.
--
-- 즉 그 명령 자체가 틀린 것이 아니라 지금 이 상황에서 틀린 것이다.
-- 도구는 상황을 모른다. 오라클이 이것까지 막으면 정당한 작업이 불가능해진다.
--
-- 그러므로 안전장치는 사람 쪽에 있어야 한다.
-- sid= 를 붙이기 전에 그 파라미터가 세 분류 중 어디에 속하는지 확인한다.

--------------------------------------------------------------------------------
 [6] 되돌리기 — 반드시 수행한다
--------------------------------------------------------------------------------

SQL> alter system reset cluster_database sid='racdb1' scope=spfile;

System altered.

SQL> alter system reset control_files sid='racdb1' scope=spfile;

System altered.

SQL> col name for a20
SQL> col value for a40
SQL> select sid, name, value from v$spparameter
  2   where name in ('cluster_database','control_files') and isspecified='TRUE'
  3   order by name, sid;

SID      NAME                 VALUE
-------- -------------------- ----------------------------------------
*        cluster_database     true
*        control_files        +DATA/RACDB/CONTROLFILE/current.261.1242
                              761745
*        control_files        +FRA/RACDB/CONTROLFILE/current.256.12427
                              61745

3 rows selected.

-- racdb1 전용 항목이 사라지고 `*` 만 남았다.
--
-- RESET 은 값을 원래대로 되돌리는 것이 아니라 그 항목 자체를 SPFILE 에서
-- 지우는 것이다. 그러면 상위인 `*` 값이 다시 적용된다.
--
-- 흔한 실수 : 값을 다시 SET 하는 방식으로 되돌리는 것.
--             예를 들어 alter system set cluster_database=true sid='racdb1' 이다.
--             값은 맞아지지만 racdb1 전용 항목이 그대로 남는다.
--             나중에 누군가 `*` 값을 바꿔도 그 인스턴스만 옛 값을 유지한다.
--             변경이 한쪽에만 반영되지 않는 현상으로 나타나 원인을 찾기 어렵다.

SQL> select sid, count(*) cnt from v$spparameter where isspecified='TRUE'
  2   group by sid order by 1;

SID              CNT
-------- ----------
*                 24
racdb1             3
racdb2             3

3 rows selected.

-- 24 / 3 / 3 으로 복귀했다. 이 확인을 반드시 한다.

--------------------------------------------------------------------------------
 [7] 달라도 되는 파라미터는 통과한다 — 정상적인 용법
--------------------------------------------------------------------------------

SQL> alter system set open_cursors=350 sid='racdb1' scope=spfile;

System altered.

SQL> col value for a20
SQL> select sid, name, value from v$spparameter where name='open_cursors'
  2     and isspecified='TRUE' order by sid;

SID      NAME             VALUE
-------- ---------------- ------
*        open_cursors     300
racdb1   open_cursors     350

2 rows selected.

-- 같은 문법인데 이번에는 정상적인 용법이다.
-- 어느 쪽인지를 가르는 것은 문법이 아니라 그 파라미터가 어느 분류에
-- 속하는가다. open_cursors 는 "같은 것이 권장" 분류다.
--
-- 다르게 잡는 정당한 경우
--   노드의 하드웨어가 다르다
--   부하가 한쪽에 의도적으로 몰려 있다
--
-- 다만 권장은 같게 두는 것이다. 값이 다르면 어느 노드에 접속했느냐에 따라
-- 동작이 달라지고, 장애 조치로 세션이 넘어갔을 때 예상 밖의 결과가 나온다.

SQL> alter system reset open_cursors sid='racdb1' scope=spfile;

System altered.

SQL> select sid, name, value from v$spparameter where name='open_cursors'
  2     and isspecified='TRUE' order by sid;

SID      NAME             VALUE
-------- ---------------- ------
*        open_cursors     300

1 row selected.

-- 실습이므로 되돌린다. 실무에서 의도적으로 다르게 두는 것이라면 그대로 둔다.

--------------------------------------------------------------------------------
 [8] scope 를 고르는 기준
--------------------------------------------------------------------------------

--  scope=memory  지금 돌고 있는 인스턴스에만 적용, SPFILE 은 그대로.
--                재기동하면 사라진다. 임시 조정에 쓴다.
--  scope=spfile  SPFILE 만 바꾸고 지금은 적용하지 않는다.
--                정적 파라미터이거나 다음 재기동에 반영하려는 경우.
--  scope=both    둘 다. 기본값이며 동적 파라미터에 주로 쓴다.
--
-- 이 실습이 모두 scope=spfile 인 것은 의도적이다.
-- 잘못된 값이 지금 돌고 있는 인스턴스에 즉시 반영되면 실습 도중 클러스터가
-- 무너진다.
--
-- 실무 원칙 : 변경은 scope=spfile 로 하고, 즉시 반영이 필요하면
--             scope=memory 를 별도로 실행한다. 한 번에 both 로 하지 않는다.
