-- 새 6장과 기존 8장의 차이를 다섯 컬럼에서 볼 수 있다.
--
--   컬럼            CANDIDATE      MEMBER
--   ----            ---------      ------
--   NAME            비어 있음      TEST_0000 형식
--   HEADER_STATUS   CANDIDATE      MEMBER
--   MOUNT_STATUS    CLOSED         CACHED
--   GROUP_NUMBER    0              1, 2, 3
--   TOTAL_MB        0              실제 크기
--
-- MOUNT_STATUS 가 CLOSED 라는 것은 ASM 이 헤더만 읽고 장치를 열어 두지
-- 않았다는 뜻이다. CACHED 는 열어 두고 메타데이터를 캐시하고 있다는 뜻이다.

--------------------------------------------------------------------------------
 [3] TOTAL_MB 가 0인데 OS_MB 는 4096 이다
--------------------------------------------------------------------------------

SQL> select path, total_mb, os_mb, os_mb-total_mb diff
  2    from v$asm_disk where path like '%test%' order by path;

PATH                       TOTAL_MB      OS_MB       DIFF
------------------------ ---------- ---------- ----------
/dev/asmdisk/test1                0       4096       4096
/dev/asmdisk/test2                0       4096       4096
/dev/asmdisk/test3                0       4096       4096
/dev/asmdisk/test4                0       4096       4096
/dev/asmdisk/test5                0       4096       4096
/dev/asmdisk/test6                0       4096       4096

-- 이 대비를 놓치면 디스크가 인식되지 않았다고 오해한다.
--
--   TOTAL_MB  ASM 이 쓸 수 있는 크기. 디스크그룹 편입 전에는 0
--   OS_MB     운영체제가 보고한 크기. 장치가 정상이면 값이 나온다
--
-- OS_MB 가 정상적으로 나온다는 것은 장치와 udev 링크에 문제가 없다는 뜻이다.
-- ASM 이 아직 그 디스크를 쓰지 않기로 했을 뿐이다.
--
-- 반대로 OS_MB 가 0이거나 예상보다 작다면 장치 계층의 문제이므로 lsblk 와
-- udev 규칙부터 확인해야 한다.

--------------------------------------------------------------------------------
 [4] 상태별 개수 세기
--------------------------------------------------------------------------------

SQL> select header_status, count(*) from v$asm_disk group by header_status order by 1;

HEADER_STATU   COUNT(*)
------------ ----------
CANDIDATE             6
MEMBER                8

-- 점검 스크립트에 넣기 좋은 조회다. 기대값과 다르면 바로 알 수 있다.
--
-- HEADER_STATUS 여섯 가지를 사용 가능 여부로 나누면 이렇다.
--
--   그대로 넣을 수 있다
--     CANDIDATE     한 번도 쓰인 적 없는 새 디스크
--     FORMER        디스크그룹에서 정상적으로 빠져나온 디스크
--     PROVISIONED   관리자가 ASM 용으로 표시한 디스크
--
--   넣을 수 없거나 위험하다
--     MEMBER        현재 어느 디스크그룹의 구성원. FORCE 필요
--     FOREIGN       다른 제품이 사용 중
--     CONFLICT      헤더 정보가 모순됨

--------------------------------------------------------------------------------
 [5] 오류 재현 — MEMBER 디스크를 쓰려 하면
--------------------------------------------------------------------------------

SQL> create diskgroup TESTX external redundancy disk '/dev/asmdisk/data1';
create diskgroup TESTX external redundancy disk '/dev/asmdisk/data1'
*
ERROR at line 1:
ORA-15018: diskgroup cannot be created
ORA-15029: disk '/dev/asmdisk/data1' is already mounted by this instance

-- 오류 : ORA-15018, ORA-15029
-- 원인 파악 : ORA-15018 은 결과, ORA-15029 가 원인이다. data1 은 +DATA 의
--             DATA_0000 이며 데이터베이스 파일이 들어 있다.
--             HEADER_STATUS 가 MEMBER 인 디스크를 지정하면 이 오류가 난다.
-- 조치     : v$asm_disk 에서 GROUP_NUMBER 를 확인해 어느 디스크그룹에
--             속했는지 파악한다. 정말 회수해야 한다면 원래 디스크그룹에서
--             정상적으로 drop disk 한 뒤에 쓴다. FORCE 는 헤더를 덮어써
--             기존 데이터를 파괴하므로 쓰지 않는다.
-- 재확인   : 아무것도 만들어지지 않았는지 확인한다.

SQL> select name, state from v$asm_diskgroup order by name;

NAME       STATE
---------- -----------
CRS        MOUNTED
DATA       MOUNTED
FRA        MOUNTED

-- TESTX 는 생기지 않았다.

-- 여기서 개수가 다르면 디스크그룹을 만든 뒤 그 노드에서 마운트가 실패한다.

--------------------------------------------------------------------------------
 [7] 장치 이름은 노드마다 다를 수 있다
--------------------------------------------------------------------------------

[root@rac2 ~]# lsblk -d -o NAME,SIZE | head -4
NAME  SIZE
sda     4G
sdb   100G
sdc     4G

-- rac2 에서는 sda 가 4GB 공유 디스크이고 sdb 가 100GB OS 디스크다.
-- rac1 에서는 sda 가 OS 디스크였다. 순서가 뒤바뀌었다.

[root@rac2 ~]# ls -l /dev/asmdisk/crs1
lrwxrwxrwx. 1 root root 6 Sep  1 19:17 crs1 -> ../sda

[root@rac1 ~]# ls -l /dev/asmdisk/crs1
lrwxrwxrwx. 1 root root 6 Sep  1 19:17 crs1 -> ../sdb

-- rac1 의 crs1 은 sdb, rac2 의 crs1 은 sda 를 가리킨다.
-- 그런데 두 링크는 같은 물리 디스크다. WWN 으로 확인한다.

[root@rac1 ~]# /lib/udev/scsi_id -g -u -d /dev/sdb
36000c29895fb6f08c44ceae9d85133a2
[root@rac2 ~]# /lib/udev/scsi_id -g -u -d /dev/sda
36000c29895fb6f08c44ceae9d85133a2

-- WWN 이 같다. 3장에서 udev 규칙을 WWN 기준으로 만든 이유가 이것이다.
-- 장치 이름으로 ASM_DISKSTRING 을 지정했다면 이 클러스터는 이미 깨졌다.

--------------------------------------------------------------------------------
 [8] 정리
--------------------------------------------------------------------------------

-- 재현한 오류
--   ORA-15018 / ORA-15029   MEMBER 디스크로 디스크그룹 생성 시도
--
-- 확인한 것
--   1) 공유 디스크 14장 중 6장이 CANDIDATE, 8장이 MEMBER
--   2) CANDIDATE 는 NAME 없음, MOUNT_STATUS CLOSED, GROUP_NUMBER 0
--   3) TOTAL_MB 0 / OS_MB 4096 은 정상이다. 두 컬럼을 함께 본다
--   4) HEADER_STATUS 여섯 가지 중 셋은 그대로 쓸 수 있다
--   5) 양 노드에서 같은 6장이 같은 이름으로 보인다
--   6) 장치 이름은 노드마다 다르다. rac1 의 sdb 가 rac2 에서는 sda
--   7) WWN 이 같으므로 udev 링크가 같은 디스크를 가리킨다
--
-- 점검 요령
--   select header_status, count(*) from v$asm_disk group by header_status;
--   이 한 줄을 양 노드에서 실행해 결과가 같은지 비교한다.
