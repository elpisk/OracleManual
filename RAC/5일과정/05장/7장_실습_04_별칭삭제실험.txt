--------------------------------------------------------------------------------
 [2] 오류 재현 — delete alias 라고 쓰면
--------------------------------------------------------------------------------

[grid@rac1 ~]$ sqlplus -s / as sysasm

SQL> alter diskgroup TEST delete alias '+TEST/mydir/my_alias.dbf';
alter diskgroup TEST delete alias '+TEST/mydir/my_alias.dbf'
                     *
ERROR at line 1:
ORA-00905: missing keyword

-- 오류 : ORA-00905
-- 원인 파악 : 오류 번호대가 다르다는 점이 첫 단서다. ASM 관련 오류는
--             ORA-150xx 대에 몰려 있는데 00905 는 그 범위 밖이다.
--             SQL 파서가 문장을 해석하는 단계에서 delete 라는 키워드가
--             alter diskgroup 절에 올 수 없다고 판단해 거절한 것이다.
--             명령이 ASM 계층까지 도달하지도 못했다.
--             별표가 delete 를 가리키는 것도 이를 뒷받침한다.
-- 조치     : DROP 을 쓴다. ADD 의 반대가 DELETE 일 것 같지만 오라클 DDL
--             관례는 DROP 이다. drop diskgroup, drop directory, drop template
--             모두 일관된다.
-- 재확인   : DROP 으로 실행한다.

SQL> alter diskgroup TEST drop alias '+TEST/mydir/my_alias.dbf';

Diskgroup altered.

 [4] 데이터베이스는 아무 일도 겪지 않았다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ sqlplus -s / as sysdba

SQL> select file#, name, status, bytes/1024/1024 mb from v$datafile where name like '%TEST%';

     FILE# NAME                                              STATUS        MB
---------- ------------------------------------------------- ------- --------
        16 +TEST/RACDB/DATAFILE/ts_alias.256.1242847959      ONLINE       100

SQL> select tablespace_name, status from dba_tablespaces where tablespace_name='TS_ALIAS';

TABLESPACE_NAME                STATUS
------------------------------ ---------
TS_ALIAS                       ONLINE

-- 실제로 데이터를 넣어 본다.
SQL> create table ts_alias_test (id number, txt varchar2(100)) tablespace ts_alias;

Table created.

SQL> insert into ts_alias_test select level, 'row '||level from dual connect by level<=1000;

1000 rows created.

SQL> commit;

Commit complete.

SQL> select count(*) from ts_alias_test;

  COUNT(*)
----------
      1000

-- 정상 동작한다. 별칭 삭제가 데이터베이스에 아무 영향을 주지 않았다.
--
-- 이유는 명확하다. 데이터베이스가 참조하는 이름은 시스템이 만든 정규 이름
-- ts_alias.256.1242847959 이고, 우리가 지운 것은 사용자가 만든 별칭이다.
-- 데이터베이스는 그 별칭의 존재조차 몰랐다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — 시스템 이름을 지우려 하면
--------------------------------------------------------------------------------

[grid@rac1 ~]$ sqlplus -s / as sysasm

SQL> alter diskgroup TEST drop alias '+TEST/RACDB/DATAFILE/ts_alias.256.1242847959';
alter diskgroup TEST drop alias '+TEST/RACDB/DATAFILE/ts_alias.256.1242847959'
*
ERROR at line 1:
ORA-15032: not all alterations performed
ORA-15177: cannot operate on system aliases

-- 오류 : ORA-15032, ORA-15177
-- 원인 파악 : 시스템이 만든 이름은 지울 수 없다.
--             이 보호가 없다면 어떻게 되는지 생각해 보라. 파일 자체는
--             남는다. v$asm_file 에 FILE_NUMBER 256 이 계속 있고 익스텐트도
--             디스크에 그대로다. 그러나 그 파일을 가리키는 이름이 사라지므로
--             데이터베이스는 파일을 열 수 없다.
--             복구도 어렵다. 별칭을 다시 만들려면 정규 이름 형식을 정확히
--             알아야 하는데 그 이름이 이미 사라졌다. 사실상 파일을 잃는다.
-- 조치     : 시스템 이름은 건드리지 않는다. 파일을 없애려면 데이터베이스
--             쪽에서 drop tablespace 로 지운다.
-- 재확인   : 이름이 그대로인지 확인한다.

SQL> select name from v$asm_alias
  2   where group_number=(select group_number from v$asm_diskgroup where name='TEST')
  3     and system_created='Y' and file_number <> 4294967295;

NAME
--------------------------
TS_ALIAS.256.1242847959

--------------------------------------------------------------------------------
 [6] 디렉터리도 같은 원리다
--------------------------------------------------------------------------------

SQL> alter diskgroup TEST drop directory '+TEST/RACDB';
alter diskgroup TEST drop directory '+TEST/RACDB'
*
ERROR at line 1:
ORA-15032: not all alterations performed
ORA-15177: cannot operate on system aliases

-- RACDB 는 SYSTEM_CREATED 가 Y 였다. 실습 02 에서 asmcmd 의 Sys 컬럼이
-- N 으로 보였던 것과 다르다.
-- 두 값이 다른 이유는 관점의 차이다. asmcmd 의 Sys 는 +DATA 아래에서 볼 때의
-- 값이고, 여기서는 +TEST 안의 v$asm_alias 값이다. 실습 환경에서 +TEST/RACDB 는
-- 데이터베이스가 파일을 만들면서 ASM 이 생성한 것이므로 Y 다.
--
-- 어느 쪽이든 결론은 같다. 안에 파일이 있는 디렉터리는 지울 수 없다.

-- 사용자가 만든 빈 디렉터리는 지워진다. 먼저 남은 별칭을 지운다.
SQL> alter diskgroup TEST drop alias '+TEST/mydir/second.dbf';

Diskgroup altered.

SQL> alter diskgroup TEST drop directory '+TEST/mydir';

Diskgroup altered.

SQL> select name, alias_directory ad, system_created sc from v$asm_alias
  2   where group_number=(select group_number from v$asm_diskgroup where name='TEST')
  3   order by reference_index;

NAME                       A S
-------------------------- - -
RACDB                      Y Y
DATAFILE                   Y Y
TS_ALIAS.256.1242847959    N Y

-- 사용자가 만든 것이 모두 사라졌다. 시스템 항목만 남았다.

--------------------------------------------------------------------------------
 [7] 오류 재현 — 비어 있지 않은 디렉터리
--------------------------------------------------------------------------------

-- 순서를 바꿔 시도했다면 어떻게 되는지 확인한다.
-- 별칭이 남아 있는 상태에서 디렉터리를 지우려 하면 거절된다.
-- (위에서 별칭을 먼저 지웠으므로 재현을 위해 다시 만든다)

SQL> alter diskgroup TEST add directory '+TEST/mydir2';

Diskgroup altered.

SQL> alter diskgroup TEST add alias '+TEST/mydir2/x.dbf'
  2    for '+TEST/RACDB/DATAFILE/ts_alias.256.1242847959';

Diskgroup altered.

SQL> alter diskgroup TEST drop directory '+TEST/mydir2';
alter diskgroup TEST drop directory '+TEST/mydir2'
*
ERROR at line 1:
ORA-15032: not all alterations performed
ORA-15027: active use of diskgroup "TEST" precludes its dismount

-- 오류 : ORA-15032, ORA-15027
-- 원인 파악 : 메시지가 dismount 를 말하지만 실제 상황은 디렉터리가 비어
--             있지 않다는 것이다. ASM 은 참조가 남아 있는 항목을 지우지
--             않는다.
--             이런 식으로 메시지가 상황과 정확히 맞지 않는 경우가 있으므로
--             메시지만 믿지 말고 상태를 직접 조회해야 한다.
-- 조치     : 안의 별칭을 먼저 지운 뒤 디렉터리를 지운다.
-- 재확인   : 순서대로 실행한다.

SQL> alter diskgroup TEST drop alias '+TEST/mydir2/x.dbf';

Diskgroup altered.

SQL> alter diskgroup TEST drop directory '+TEST/mydir2';

Diskgroup altered.

 [8] 정리 — 데이터는 끝까지 무사했다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ sqlplus -s / as sysdba

SQL> select count(*) from ts_alias_test;

  COUNT(*)
----------
      1000

SQL> select file#, name, status from v$datafile where name like '%TEST%';

     FILE# NAME                                              STATUS
---------- ------------------------------------------------- -------
        16 +TEST/RACDB/DATAFILE/ts_alias.256.1242847959      ONLINE

-- 별칭을 세 번 만들고 세 번 지우고 디렉터리를 두 번 만들고 두 번 지웠다.
-- 그동안 데이터는 한 번도 영향을 받지 않았다.

--------------------------------------------------------------------------------
 [9] 정리
--------------------------------------------------------------------------------

-- 재현한 오류
--   ORA-00905              delete alias 라는 구문 오류 (SQL 파서 계층)
--   ORA-15032 / ORA-15177  시스템 별칭·디렉터리 삭제 시도
--   ORA-15032 / ORA-15027  비어 있지 않은 디렉터리 삭제 시도
--
-- 확인한 것
--   1) ADD 의 반대는 DELETE 가 아니라 DROP 이다
--   2) ORA-00905 는 ASM 오류가 아니다. 번호대로 계층을 판단한다
--   3) 별칭을 지워도 실제 파일과 데이터베이스는 무사하다
--   4) 데이터베이스는 정규 이름으로만 참조하므로 별칭을 모른다
--   5) 시스템 이름은 ORA-15177 로 보호된다
--   6) 이 보호가 없으면 파일은 남고 접근 경로만 사라진다
--   7) 디렉터리는 비운 뒤에 지운다
--
-- 실무 지침
--   사용자 별칭은 자유롭게 만들고 지워도 된다. 그러나 관리 대상이 늘어나므로
--   꼭 필요한 곳에만 만든다. v$asm_alias 에서 SYSTEM_CREATED='N' 조건으로
--   주기적으로 목록을 뽑아 문서화된 것만 남아 있는지 확인한다.
