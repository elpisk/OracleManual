================================================================================
 제1장 실습 02 — 두 인스턴스가 하나의 DB 를 공유한다는 것 확인
 대상   : rac1(racdb1) — 조회는 한 노드에서만 수행한다
 목표   : GV$ 와 V$ 의 차이를 눈으로 확인하고, INST_ID 열의 의미와
          공유되는 파일이 실제로 한 벌뿐임을 조회로 증명한다
 소요   : 약 20분
 선행   : racdb 두 인스턴스가 모두 OPEN 상태일 것
 계정   : oracle
================================================================================

--------------------------------------------------------------------------------
 [1] 접속과 환경 확인
--------------------------------------------------------------------------------

[root@rac1 ~]# su - oracle

-- RAC 실습에서 가장 흔한 실수는 조회 자체가 아니라 어느 인스턴스에 접속했는지를
-- 착각하는 것이다. ORACLE_SID 를 먼저 확인한다.
[oracle@rac1 ~]$ echo $ORACLE_SID
racdb1

[oracle@rac1 ~]$ echo $ORACLE_HOME
/u01/app/oracle/product/19.3.0/dbhome_rac

[oracle@rac1 ~]$ sqlplus / as sysdba

SQL*Plus: Release 19.0.0.0.0 - Production
Connected to:
Oracle Database 19c Enterprise Edition Release 19.0.0.0.0 - Production
Version 19.3.0.0.0

--------------------------------------------------------------------------------
 [2] GV$INSTANCE 와 V$INSTANCE — 같은 세션, 다른 결과
--------------------------------------------------------------------------------

SQL> SET LINES 160 PAGES 60
SQL> COL host_name FOR a18
SQL> COL instance_name FOR a10

SQL> SELECT inst_id, instance_number, instance_name, host_name, status, thread#
  2    FROM gv$instance ORDER BY inst_id;

   INST_ID INSTANCE_NUMBER INSTANCE_N HOST_NAME          STATUS      THREAD#
---------- --------------- ---------- ------------------ -------- ----------
         1               1 racdb1     rac1.localdomain   OPEN              1
         2               2 racdb2     rac2.localdomain   OPEN              2

2 rows selected.

SQL> SELECT instance_number, instance_name, host_name, thread# FROM v$instance;

INSTANCE_NUMBER INSTANCE_N HOST_NAME            THREAD#
--------------- ---------- ------------------ ----------
              1 racdb1     rac1.localdomain            1

1 row selected.

-- 같은 세션에서 연달아 조회했는데 GV$ 는 2행, V$ 는 1행이다.
-- V$ 는 접속한 인스턴스의 정보만 보여 준다. 다른 노드에서 벌어지는 일은
-- V$ 에 나타나지 않는다. RAC 진단은 반드시 GV$ 로 시작한다.

-- INST_ID 와 INSTANCE_NUMBER 의 차이도 확인해 둔다.
-- INST_ID  : GV$ 결과 행이 어느 인스턴스에서 왔는지
-- INSTANCE_NUMBER : 그 인스턴스에 부여된 고유 번호
-- 여기서는 두 값이 같지만 의미가 다르다. 인스턴스 번호를 1, 3 으로 부여한
-- 클러스터라면 INST_ID 도 1, 3 이 된다.

--------------------------------------------------------------------------------
 [3] 하나의 데이터베이스임을 확인
--------------------------------------------------------------------------------

SQL> SELECT name, dbid, open_mode, database_role, cdb, log_mode FROM v$database;

NAME            DBID OPEN_MODE            DATABASE_ROLE    CDB LOG_MODE
--------- ---------- -------------------- ---------------- --- ------------
RACDB     1240378513 READ WRITE           PRIMARY          YES NOARCHIVELOG

1 row selected.

-- V$DATABASE 는 GV$ 로 조회해도 두 행이 나오지만 내용은 같다. 데이터베이스가
-- 하나이기 때문이다. DBID 가 특히 그렇다 — 두 인스턴스가 같은 DBID 를 보고한다.

SQL> SELECT inst_id, name, dbid, open_mode FROM gv$database ORDER BY inst_id;

   INST_ID NAME            DBID OPEN_MODE
---------- --------- ---------- --------------------
         1 RACDB     1240378513 READ WRITE
         2 RACDB     1240378513 READ WRITE

2 rows selected.

-- 이것이 RAC 와 Data Guard 의 결정적 차이다. Data Guard 에서는 Primary 와
-- Standby 가 서로 다른 DB_UNIQUE_NAME 을 가지고 별개의 파일을 쓴다.
-- RAC 에서는 이름도 DBID 도 파일도 하나다.

--------------------------------------------------------------------------------
 [4] 공유 파일이 한 벌뿐임을 확인
--------------------------------------------------------------------------------

SQL> COL name FOR a62
SQL> SELECT name FROM v$controlfile;

NAME
--------------------------------------------------------------
+DATA/RACDB/CONTROLFILE/current.261.1242761745
+FRA/RACDB/CONTROLFILE/current.256.1242761745

2 rows selected.

-- 파일이 둘인 것은 다중화이지 인스턴스별 분리가 아니다. 두 파일은 같은 내용을
-- 담고 있고 모든 인스턴스가 두 파일을 모두 연다.

SQL> COL value FOR a62
SQL> SELECT value FROM v$parameter WHERE name = 'spfile';

VALUE
--------------------------------------------------------------
+DATA/RACDB/PARAMETERFILE/spfile.272.1242762487

-- 파라미터는 이 한 파일 안에서 sid 접두어로 구분한다.

SQL> SELECT con_id, name FROM v$datafile ORDER BY con_id, name;

    CON_ID NAME

         1 +DATA/RACDB/DATAFILE/system.257.1242761643

         2 +DATA/RACDB/86B637B62FE07A65E053F706E80A27CA/DATAFILE/sysaux...
         3 +DATA/RACDB/5A569448ED6C2E8DE06365E110ACB045/DATAFILE/system...

13 rows selected.

-- UNDOTBS1 과 UNDOTBS2 가 나란히 보인다. 인스턴스마다 UNDO 를 따로 쓰지만

-- CON_ID 1 은 CDB 루트, 2 는 PDB$SEED, 3 은 RACPDB1 이다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — V$ 뷰에 INST_ID 를 쓰면
--------------------------------------------------------------------------------

-- RAC 를 처음 다루는 사람이 가장 자주 저지르는 실수다. GV$ 에 익숙해진 뒤
-- V$ 에도 INST_ID 가 있으리라 생각하고 조건에 넣는다.

SQL> SELECT instance_name FROM v$instance WHERE inst_id = 2;
SELECT instance_name FROM v$instance WHERE inst_id = 2
                                           *
ERROR at line 1:
ORA-00904: "INST_ID": invalid identifier

SQL> SELECT COUNT(*) FROM v$undostat WHERE inst_id = 1;
SELECT COUNT(*) FROM v$undostat WHERE inst_id = 1
                                      *
ERROR at line 1:
ORA-00904: "INST_ID": invalid identifier

-- 원인 파악
--   V$ 뷰에는 INST_ID 열이 없다. 애초에 접속한 인스턴스의 정보만 담고 있으므로
--   구분할 필요가 없기 때문이다. INST_ID 는 GV$ 뷰에만 있다.
--   열 구성을 직접 확인해 본다.

SQL> DESC v$undostat
 Name                                Null?    Type
 ----------------------------------- -------- ------------------
 BEGIN_TIME                                   DATE
 END_TIME                                     DATE
 UNDOTSN                                      NUMBER
 UNDOBLKS                                     NUMBER
 TXNCOUNT                                     NUMBER
 ...
 CON_ID                                       NUMBER

--   INST_ID 가 없다. CON_ID 는 멀티테넌트 구분용이며 인스턴스와 무관하다.

-- 조치
--   특정 인스턴스만 보고 싶으면 GV$ 를 쓰고 INST_ID 로 거른다.

SQL> SELECT inst_id, COUNT(*) FROM gv$undostat GROUP BY inst_id ORDER BY 1;

   INST_ID   COUNT(*)
---------- ----------
         1          3
         2          2

2 rows selected.

-- 재확인
SQL> SELECT instance_name FROM gv$instance WHERE inst_id = 2;

INSTANCE_N
----------
racdb2

1 row selected.

-- 배운 것
--   V$  = 접속한 인스턴스만, INST_ID 열 없음
--   GV$ = 클러스터 전체, INST_ID 열 있음
--   특정 노드만 보려면 GV$ 에 INST_ID 조건을 건다.
--   반대로 GV$ 를 쓰면서 INST_ID 를 선택 목록에 넣지 않으면 어느 행이 어느
--   노드 것인지 알 수 없게 되므로 반드시 함께 조회한다.

--------------------------------------------------------------------------------
 [6] 정리
--------------------------------------------------------------------------------

-- 확인한 것
--   1) GV$INSTANCE 2행 / V$INSTANCE 1행 — 뷰의 범위가 다르다
--   2) 두 인스턴스가 같은 DBID(1240378513)를 보고한다 — 하나의 데이터베이스
--   3) 컨트롤 파일·SPFILE·데이터 파일이 공유 스토리지에 한 벌뿐이다
--   4) V$ 에는 INST_ID 가 없다
--
-- 이 실습은 조회만 했으므로 원상 복구할 것이 없다.
