================================================================================
 제16장 실습 08 — 장애와 계획 정지, FAN 이벤트가 어떻게 다른가
 대상   : rac1, rac2
 목표   : 인스턴스를 kill -9 로 죽였을 때와 srvctl 로 내렸을 때 콜아웃에 남는
          이벤트를 비교한다. reason 값, 이벤트가 생기는 노드, 서비스가 어디로
          가는지, 걸린 시간을 잰다
 소요   : 약 40분 (인스턴스 재기동 대기 포함)
 선행   : 실습 07 완료 (콜아웃 두 노드 755, ch16_fan 실행 중)
 계정   : root, oracle
 주의   : 인스턴스를 강제로 죽인다. 시험 랩에서만 한다
================================================================================

--------------------------------------------------------------------------------
 [1] 준비
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl relocate service -db racdb -service ch16_fan -oldinst racdb2 -newinst racdb1
[oracle@rac1 ~]$ srvctl status service -db racdb -service ch16_fan
Service ch16_fan is running on instance(s) racdb1

[root@rac1 ~]# rm -f /tmp/fan_*.log
[root@rac2 ~]# rm -f /tmp/fan_*.log

-- 서비스를 선호 인스턴스 racdb1 로 되돌리고 두 노드의 이벤트 로그를 비운다.

--------------------------------------------------------------------------------
 [2] 흔한 실수 — pgrep -f 로 죽일 프로세스를 고르면
--------------------------------------------------------------------------------

[root@rac1 ~]# ps -eo pid,user,cmd | grep [o]ra_pmon_racdb1; date '+%T.%3N'; \
               kill -9 $(pgrep -f ora_pmon_racdb1); date '+%T.%3N'
 5398 root     bash -c ps -eo pid,user,cmd | grep [o]ra_pmon_racdb1; date ... kill -9 $(pgrep -f ora_pmon_racdb1) ...
31677 oracle   ora_pmon_racdb1
14:11:38.828

[oracle@rac1 ~]$ srvctl status instance -db racdb -instance racdb1
Instance racdb1 is running on node rac1

-- 두 번째 date 가 찍히지 않았고 인스턴스는 2 분이 지나도 멀쩡했다. 이벤트 로그도 없다.
-- 원인 파악 : pgrep -f 는 명령행 전체에서 찾는다. 이 명령을 돌린 셸(PID 5398)의
--             명령행에도 ora_pmon_racdb1 이라는 글자가 있다. 셸이 자기 자신을 죽였다.
-- 조치     : 프로세스 이름으로 정확히 찾는다. pgrep -x ora_pmon_racdb1.
--             ps 에서 PID 를 확인하고 숫자로 kill 해도 된다.
-- 재확인   : 아래 [3] 은 pgrep -x 로 했다.

--------------------------------------------------------------------------------
 [3] 장애 — PMON 을 kill -9
--------------------------------------------------------------------------------

[root@rac1 ~]# date '+%T.%3N'; kill -9 $(pgrep -x ora_pmon_racdb1); date '+%T.%3N'
14:14:21.942
14:14:21.970

[oracle@rac1 ~]$ srvctl status service -db racdb -service ch16_fan; srvctl status instance -db racdb -instance racdb1
-- 10 초 남짓 간격으로 되풀이한 결과
14:14:38  Service ch16_fan is running on instance(s) racdb2   Instance racdb1 is not running on node rac1
14:17:45  Service ch16_fan is running on instance(s) racdb2   Instance racdb1 is not running on node rac1
14:18:00  Service ch16_fan is running on instance(s) racdb2   Instance racdb1 is running on node rac1
14:19:12  Service ch16_fan is running on instance(s) racdb2   Instance racdb1 is running on node rac1

[root@rac1 ~]# cat /tmp/fan_rac1.log
14:14:29.586 grid SERVICEMEMBER ... service=ch16_fan instance=racdb1 host=rac1 status=down reason=FAILURE timestamp=2026-11-10 14:14:29
14:14:29.553 grid SERVICE       ... service=ch16_fan instance=racdb1 host=rac1 status=down reason=FAILURE timestamp=2026-11-10 14:14:29
14:14:39.187 grid INSTANCE      ... service=racdb    instance=racdb1 host=rac1 status=down reason=FAILURE timestamp=2026-11-10 14:14:35
14:17:58.247 grid INSTANCE      ... service=racdb    instance=racdb1 host=rac1 status=up   reason=FAILURE timestamp=2026-11-10 14:17:57
[root@rac2 ~]# cat /tmp/fan_rac2.log
14:14:40.120 grid SERVICE       ... service=ch16_fan instance=racdb2 host=rac2 status=up reason=FAILURE timestamp=2026-11-10 14:14:39
14:14:40.178 grid SERVICEMEMBER ... service=ch16_fan instance=racdb2 host=rac2 status=up reason=FAILURE card=1 timestamp=2026-11-10 14:14:39
(VERSION · database · timezone · db_domain 은 줄였다)

-- 시간 순으로 보면 이렇다. 기준은 kill 시각 14:14:21.97.
--   +7.6 초   rac1  ch16_fan DOWN (SERVICE · SERVICEMEMBER)
--   +13 초    rac1  INSTANCE racdb1 DOWN   (timestamp 기준, 콜아웃 도착은 +17 초)
--   +18 초    rac2  ch16_fan UP, card=1
--   +3분 36초 rac1  INSTANCE racdb1 UP     (Clusterware 가 자동 재기동)
-- reason 이 모두 FAILURE 다. 풀은 이 값을 보면 그 인스턴스의 연결을 기다리지 않고 버린다.
-- 이 장애에서는 노드가 살아 있어 죽은 인스턴스의 이벤트도 rac1 에서 나왔다.
-- 노드가 통째로 죽으면 그 노드는 이벤트를 낼 수 없다. 살아 있는 쪽이 낸다(본문 4.1).
-- rac1 의 두 줄은 SERVICE 가 SERVICEMEMBER 보다 먼저 기록됐다. 이벤트 timestamp 는 같다.
-- 콜아웃 순서는 보장되지 않는다는 문서 설명이 그대로 보인다.
-- 인스턴스 이벤트의 service= 칸에는 서비스가 아니라 DB_UNIQUE_NAME(racdb)이 들어간다.

-- racdb1 이 돌아왔는데 서비스는 racdb2 에 남았다. 돌아오는 이벤트도 없다.
-- 서비스에 -failback 을 주지 않았으므로 자동으로 되돌아가지 않는다(15장 실습 09).
-- 풀은 INSTANCE UP 을 보고 racdb1 에도 연결을 만들지만, ch16_fan 으로 오는 작업은
-- 여전히 racdb2 로만 간다. 보통은 정비 창에 relocate 로 되돌린다.
-- 여기서는 되돌리지 않고 이 상태로 [4] 를 한다.

--------------------------------------------------------------------------------
 [4] 계획 정지 — 서비스가 도는 인스턴스를 srvctl 로 내리기
--------------------------------------------------------------------------------

-- 이 단계를 시작할 때 ch16_fan 은 racdb2 에 있었다([3] 의 자동 이동 결과).
-- racdb2 를 계획 정지해 본다.

[root@rac1 ~]# rm -f /tmp/fan_*.log
[root@rac2 ~]# rm -f /tmp/fan_*.log

[oracle@rac1 ~]$ date '+%T.%3N'; srvctl stop instance -db racdb -instance racdb2 -stopoption immediate; date '+%T.%3N'
14:19:35.727
PRCD-1131 : Failed to stop database racdb and its services on nodes rac2
PRCR-1133 : Failed to stop database racdb and its running services
PRCR-1132 : Failed to stop resources using a filter
CRS-2974: unable to act on resource 'ora.racdb.db' on server 'rac2' because that would
          require stopping or relocating resource 'ora.racdb.ch16_fan.svc' but the
          appropriate force flag was not specified
14:19:36.770

-- 원인 파악 : 그 인스턴스에서 사용자 서비스가 돌고 있다. 인스턴스를 내리면 서비스도
--             멈추거나 옮겨야 하는데 그 결정을 srvctl 이 혼자 내리지 않는다.
--             메시지가 CRS-2974 로 끝난다. 거부는 Clusterware 가 했다.
-- 조치     : 서비스를 먼저 relocate 하거나 -force 를 붙인다. 여기서는 -force 로 결과를 본다.
-- 이 거부 덕에 서비스가 도는 줄 모르고 인스턴스를 내리는 사고를 한 번 막는다.

[oracle@rac1 ~]$ date '+%T.%3N'; srvctl stop instance -db racdb -instance racdb2 -stopoption immediate -force; date '+%T.%3N'
14:19:36.830
14:24:45.264
[oracle@rac1 ~]$ srvctl status service -db racdb -service ch16_fan
Service ch16_fan is not running.
[oracle@rac1 ~]$ srvctl status database -db racdb
Instance racdb1 is running on node rac1
Instance racdb2 is not running on node rac2

[root@rac2 ~]# cat /tmp/fan_rac2.log
14:19:38.666 grid SERVICEMEMBER ... service=ch16_fan instance=racdb2 host=rac2 status=down reason=USER timestamp=2026-11-10 14:19:38
14:19:38.676 grid SERVICE       ... service=ch16_fan instance=racdb2 host=rac2 status=down reason=USER timestamp=2026-11-10 14:19:38
14:24:44.195 grid INSTANCE      ... service=racdb    instance=racdb2 host=rac2 status=down reason=USER timestamp=2026-11-10 14:24:43
[root@rac1 ~]# cat /tmp/fan_rac1.log
cat: /tmp/fan_rac1.log: No such file or directory

-- 서비스는 racdb1 로 옮겨 가지 않고 그냥 멈췄다. SERVICE DOWN 까지 나갔다.
-- 장애([3])에서는 가용 인스턴스로 갔는데 계획 정지에 -force 를 주면 서비스가 사라진다.
-- 선호 인스턴스 racdb1 이 멀쩡히 떠 있었는데도 그렇다.
-- 모든 이벤트가 reason=USER 다. 서비스 DOWN 은 정지 2 초 만에, 인스턴스 DOWN 은 5 분 뒤에 나왔다.
-- 5 분 넘게 걸린 것은 이 랩의 rac2 메모리 사정 때문이다(같은 시각 free 89MB).
-- 계획 정비 순서는 "relocate → 남은 세션 확인 → stop instance" 다. -force 로 바로 내리면
-- 그 시간 동안 ch16_fan 으로는 아무도 접속하지 못한다.

[oracle@rac1 ~]$ date '+%T.%3N'; srvctl start instance -db racdb -instance racdb2; date '+%T.%3N'
14:26:23.093
14:34:11.299
[oracle@rac1 ~]$ srvctl status service -db racdb -service ch16_fan
Service ch16_fan is running on instance(s) racdb2

[root@rac2 ~]# tail -3 /tmp/fan_rac2.log
14:34:03.015 grid INSTANCE      ... instance=racdb2 host=rac2 status=up reason=USER timestamp=2026-11-10 14:34:00
14:34:10.039 grid SERVICEMEMBER ... service=ch16_fan instance=racdb2 status=up reason=USER card=1 timestamp=2026-11-10 14:34:08
14:34:10.041 grid SERVICE       ... service=ch16_fan instance=racdb2 status=up reason=USER timestamp=2026-11-10 14:34:08

-- 인스턴스가 뜨자 서비스가 racdb2 에서 따라 올라왔다. 멈추기 직전에 돌던 자리로 돌아간
-- 것이다(서비스 자원의 AUTO_START=restore, 15장 실습 09 [5]).
-- 기동도 7 분 48 초가 걸렸다. 역시 메모리 탓이다. 실습 시간을 잡을 때 감안한다.

--------------------------------------------------------------------------------
 [5] 정리
--------------------------------------------------------------------------------

  구분            장애 (kill -9)                 계획 정지 (srvctl -force)
  --------------  -----------------------------  -----------------------------
  reason          FAILURE                        USER
  서비스          가용 인스턴스로 이동 (18 초)   멈춤. 다른 인스턴스로 가지 않음
  이벤트 노드     죽은 인스턴스 노드 + 받은 노드  내린 노드만
  인스턴스 복귀   Clusterware 가 자동 재기동     사람이 start
  복귀 후 서비스  가용 쪽에 남음                 원래 자리에서 다시 뜸

-- 진단 — "인스턴스를 내렸더니 서비스가 안 넘어갔다"
--   -force 로 계획 정지를 하면 서비스는 넘어가지 않는다는 것을 [4] 에서 확인했다.
--   이 서비스를 계속 쓰게 하려면 먼저 relocate 한다. 15장 실습 10 의 드레인 옵션을 같이 쓴다.
