 [6] 수집 중 새 오류가 났는지 확인
--------------------------------------------------------------------------------

SQL> select inst_id, to_char(originating_timestamp,'MM-DD HH24:MI') t, message_text
  2    from gv$diag_alert_ext
  3   where message_text like '%ORA-%'
  4     and originating_timestamp > systimestamp - interval '60' minute
  5   order by originating_timestamp;
  from gv$diag_alert_ext
       *
ERROR at line 2:
ORA-00942: table or view does not exist

-- 원인 파악: 모든 V$ 뷰에 GV$ 짝이 있는 것은 아니다.
--            딕셔너리를 확인해 보면 V_$DIAG_ALERT_EXT 만 존재하고
--            GV_$DIAG_ALERT_EXT 는 없다.
--
--   SQL> select object_name, object_type from dba_objects
--     2   where object_name in ('V_$DIAG_ALERT_EXT','GV_$DIAG_ALERT_EXT');
--
--   OBJECT_NAME          OBJECT_TYPE
--   -------------------- -----------
--   V_$DIAG_ALERT_EXT    VIEW
--
-- 조치     : V$ 로 바꾼다. 노드별로 봐야 하면 각 인스턴스에 접속해 실행한다.

SQL> select to_char(originating_timestamp,'MM-DD HH24:MI') t,
  2         substr(message_text,1,55) msg
  3    from v$diag_alert_ext
  4   where message_text like '%ORA-%'
  5     and originating_timestamp > systimestamp - interval '60' minute
  6   order by originating_timestamp fetch first 5 rows only;

no rows selected

-- 재확인: 최근 한 시간 동안 ORA- 오류가 없다. 수집 자체는 부하만 주고
--         새 오류를 만들지 않았다.
--
-- 여기서 얻는 교훈이 하나 더 있다. RAC 라고 해서 모든 진단 뷰를 GV$ 로
-- 볼 수 있는 것이 아니다. 없으면 노드마다 접속해 V$ 로 본다.

--------------------------------------------------------------------------------
 [7] 목적별 수집 방침
--------------------------------------------------------------------------------

-- 오라클 지원에 SR 을 열 때
--   tfactl diagcollect -srdc dbrac_evictions -node all
--   증상별로 필요한 파일 목록이 정의되어 있어 빠뜨림이 없다.
--   SR 왕복이 크게 줄어든다.
--
-- 스스로 분석할 때
--   컴포넌트를 좁히고 시간 범위를 준다.
--   클러스터 문제면 -crs, 데이터베이스 문제면 -database, ASM 이면 -asm.
--
-- 언제 모으는가
--   장애 직후가 원칙이다. 로그는 회전하므로 며칠 지나면 사라진다.
--   다만 장애가 진행 중이면 수집이 부하를 더하므로 급한 불을 끈 뒤에 모은다.
