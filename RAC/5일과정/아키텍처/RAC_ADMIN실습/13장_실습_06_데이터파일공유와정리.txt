================================================================================
 제13장 실습 06 — 데이터파일 추가, 양 노드 반영 확인과 정리
 대상   : rac1, rac2
 목표   : 데이터파일이 완전히 공유된다는 것을 양 노드에서 확인하고,
          인스턴스별 리두 쓰기량으로 부하 편중을 읽는 법을 익힌다
 소요   : 약 30분
 선행   : 실습 05 완료, 두 세션(rac1·rac2)을 함께 열어 둔다
 계정   : oracle, grid
================================================================================

--------------------------------------------------------------------------------
 [1] rac2 에서 테이블스페이스를 만든다
--------------------------------------------------------------------------------

[rac2] SQL> set timing on
[rac2] SQL> create tablespace ts_ch13 datafile '+DATA' size 100M;

Tablespace created.

Elapsed: 00:00:02.20

-- 2.2 초다. ASM 이 100MB 파일 하나를 만드는 시간이다.
-- 만든 노드는 rac2 다. 다음 단계에서 rac1 을 본다.

--------------------------------------------------------------------------------
 [2] rac1 에서 즉시 보이는가
--------------------------------------------------------------------------------

[rac1] SQL> select file#, name, bytes/1024/1024 mb from v$datafile
  2    where name like '%ts_ch13%';

[rac1] SQL> select tablespace_name, status from dba_tablespaces
  2    where tablespace_name='TS_CH13';

TABLESPACE_NAME                STATUS
------------------------------ ---------
TS_CH13                        ONLINE

1 row selected.

-- rac1 에서 아무것도 하지 않았는데 보인다.
-- 데이터파일 구성은 컨트롤파일에 기록되고 컨트롤파일은 공유 스토리지에 있다.
--
-- 파일 번호가 25 다. 앞 번호가 다 찼다는 뜻은 아니다.
-- 10장에서 확인했듯 파일 번호는 연속이라는 보장이 없다.
--
-- 경로가 +DATA 로 시작하는 것을 반드시 확인한다. 이것이 공유의 조건이다.
-- 로컬 파일시스템 경로였다면 다음 단계에서 문제가 생긴다.

--------------------------------------------------------------------------------
 [3] rac1 에서 데이터파일을 추가하고 rac2 에서 확인
--------------------------------------------------------------------------------

[rac1] SQL> alter tablespace ts_ch13 add datafile '+DATA' size 50M;

Tablespace altered.

[rac2] SQL> select file_id, file_name, bytes/1024/1024 mb from dba_data_files
  2    where tablespace_name='TS_CH13' order by file_id;

   FILE_ID FILE_NAME                                            MB
---------- --------------------------------------------- ----------
        25 +DATA/RACDB/DATAFILE/ts_ch13.295.1243078401         100
        26 +DATA/RACDB/DATAFILE/ts_ch13.284.1243078405          50

2 rows selected.

-- 방향을 바꿔도 같다. rac2 가 만든 것을 rac1 이 보고,
-- rac1 이 추가한 것을 rac2 가 본다. 어느 쪽이 원본이라는 개념이 없다.

--------------------------------------------------------------------------------
 [4] 양 노드에서 동시에 쓰기
--------------------------------------------------------------------------------

-- 조회되는 것과 실제로 쓸 수 있는 것은 다르므로 반드시 써 본다.
-- 10장에서 rac2 가 ORA-01157 을 냈던 것은 딕셔너리에 이름은 있는데
-- 파일이 공유 스토리지에 없었기 때문이다. 즉 [2] 는 통과하고 여기서 실패한다.

[rac1] SQL> create table sys.ch13_t (n varchar2(10), v number) tablespace ts_ch13;

Table created.

[rac1] SQL> insert into sys.ch13_t select 'rac1', level from dual connect by level<=1000;

1000 rows created.

[rac1] SQL> commit;

Commit complete.

[rac2] SQL> insert into sys.ch13_t select 'rac2', level from dual connect by level<=1000;

1000 rows created.

[rac2] SQL> commit;

Commit complete.

[rac1] SQL> select n, count(*) from sys.ch13_t group by n order by n;

N            COUNT(*)
---------- ----------
rac1             1000
rac2             1000

2 rows selected.

-- rac1 에서 조회했는데 rac2 가 넣은 1000 건도 보인다. rac2 에서도 같다.
--
-- 데이터는 하나다. 나뉜 것은 각 인스턴스가 변경을 기록하는 리두와 UNDO 이지
-- 데이터 자체가 아니다.
--
-- 두 인스턴스가 같은 블록을 다룰 수 있는 것은 캐시 퓨전 덕분이다.
-- 각 인스턴스가 자기 버퍼 캐시에 블록을 올려 두고, 다른 인스턴스가
-- 필요로 하면 디스크를 거치지 않고 인터커넥트로 직접 넘겨 준다.
-- 저장 계층이 할 일은 데이터파일을 공유해 두는 것뿐이다.

--------------------------------------------------------------------------------
 [5] 흔한 오류 — 노드마다 다른 con_id 에서 만들면
--------------------------------------------------------------------------------

-- rac2 에서 PDB 에 접속한 채로 테이블스페이스를 만들고
-- rac1 에서는 CDB$ROOT 에서 찾으면 이런 일이 생긴다.

[rac1] SQL> select count(*) from sys.ch13_t;
select count(*) from sys.ch13_t
                         *
ERROR at line 1:
ORA-00942: table or view does not exist

-- 원인 파악 : 객체가 없는 것이 아니라 다른 컨테이너에 있다.
--             현재 컨테이너는 다음으로 확인한다.
--               select sys_context('userenv','con_name') from dual;
--             RAC 의 공유 문제가 아니라 CDB 의 컨테이너 문제다.
--             데이터파일 공유는 정상인데 보는 위치가 다른 것이다.
-- 조치     : alter session set container=RACPDB1 로 컨테이너를 맞춘다.
--             양 노드에서 실습할 때는 두 세션의 컨테이너를 먼저 일치시킨다.
-- 재확인   : 양 세션에서 sys_context 로 같은 컨테이너인지 확인한 뒤
--             다시 조회한다.

--------------------------------------------------------------------------------
 [6] 인스턴스별 리두 쓰기량 — 부하 편중을 수치로
--------------------------------------------------------------------------------

[rac1] SQL> select inst_id, name, value from gv$sysstat
  2   where name in ('redo size','redo writes','redo log space requests')
  3   order by name, inst_id;

   INST_ID NAME                            VALUE
---------- ------------------------- ----------
         1 redo log space requests            0
         2 redo log space requests            0
         1 redo size                   16778836
         2 redo size                    9446300
         1 redo writes                     2556
         2 redo writes                     1457

6 rows selected.

-- 읽는 법.
--
--   redo size : rac1 이 16,778,836, rac2 가 9,446,300 이다. 약 1.8 배.
--   redo writes : 2556 대 1457 로 비슷한 비율이다.
--                 두 지표의 비율이 비슷하면 쓰기 패턴은 같고 양만 다르다.
--                 한쪽만 크게 다르면 트랜잭션 크기가 다르다는 뜻이다.
--   redo log space requests : 둘 다 0 이다.
--                 리두 버퍼가 가득 차 로그 스위치를 기다린 횟수이며
--                 0 이면 리두 크기가 충분하다는 뜻이다.
--                 계속 올라간다면 그룹을 늘리거나 크기를 키운다.
--
-- 주의할 점은 이 값이 인스턴스 기동 이후 누적이라는 것이다.
-- 한 노드를 재기동했다면 그 값이 작게 나오는 것은 당연하다.
-- gv$instance 의 startup_time 을 함께 보고 시간당 값으로 환산해 비교한다.
--
-- 실습 03 에서 UNDOTBS1 이 UNDOTBS2 보다 훨씬 컸던 것과 방향이 같다.
-- 두 지표가 같은 쪽을 가리키면 부하 편중이 확실하다.
-- 조치는 리두나 UNDO 크기 조절이 아니라 부하 분산이다. 15장의 서비스가 도구다.
