 대상   : rac1

--------------------------------------------------------------------------------
 [3] 이중화는 파일마다 다르다 — BYTES 와 SPACE
--------------------------------------------------------------------------------

[grid@rac1 ~]$ sqlplus / as sysasm
SQL> select g.name dg, f.type, f.redundancy red, count(*) cnt,
  2         round(sum(f.bytes)/1024/1024) mb, round(sum(f.space)/1024/1024) space_mb
  3    from v$asm_file f join v$asm_diskgroup g on g.group_number=f.group_number
  4   where g.name in ('DATA','FRA')
  5   group by g.name, f.type, f.redundancy order by 1,2;

DG    TYPE             RED           CNT         MB   SPACE_MB
----- ---------------- ------ ---------- ---------- ----------
DATA  CONTROLFILE      UNPROT          1         19         32
DATA  DATAFILE         UNPROT         13       4385       4448
DATA  ONLINELOG        UNPROT          4        800        816
DATA  PARAMETERFILE    UNPROT          1          0          4
DATA  PASSWORD         UNPROT          1          0          0
DATA  TEMPFILE         UNPROT          3        238        248
FRA   ARCHIVELOG       MIRROR         24        265        700
FRA   AUTOBACKUP       MIRROR          1         19         40
FRA   BACKUPSET        MIRROR          5       2999       6068
FRA   CONTROLFILE      HIGH            1         19         96
FRA   ONLINELOG        MIRROR          4        800       1680

-- BYTES 는 데이터베이스가 보는 크기, SPACE 는 ASM 이 실제로 잡은 공간이다.
--
--   DATA 는 EXTERNAL 이중화라 모두 UNPROT 다. SPACE 가 BYTES 보다 조금 큰 정도다.
--   FRA 는 NORMAL 이라 대부분 MIRROR 다. SPACE 가 BYTES 의 두 배를 조금 넘는다.
--     리두 4 개    800MB → 1680MB
--     백업셋 5 개 2999MB → 6068MB
--   FRA 의 컨트롤파일만 HIGH 다. 19MB 짜리가 96MB 를 쓴다. 세 벌이기 때문이다.
--
-- 같은 리두 그룹의 두 멤버가 실제로 쓰는 공간이 다르다는 점을 본다.
-- +DATA 멤버는 한 벌, +FRA 멤버는 두 벌이다. 리두 멤버 둘과 FRA 미러를
-- 합치면 한 그룹의 리두가 디스크에 세 벌 있다.

SQL> select g.name dg, t.name template, t.redundancy, t.stripe
  2    from v$asm_template t join v$asm_diskgroup g on g.group_number=t.group_number
  3   where t.name in ('CONTROLFILE','ONLINELOG','DATAFILE','ARCHIVELOG')
  4     and g.name in ('DATA','FRA') order by 1,2;

DG    TEMPLATE       REDUND STRIPE
----- -------------- ------ ------
DATA  ARCHIVELOG     UNPROT COARSE
DATA  CONTROLFILE    UNPROT FINE
DATA  DATAFILE       UNPROT COARSE
DATA  ONLINELOG      UNPROT COARSE
FRA   ARCHIVELOG     MIRROR COARSE
FRA   CONTROLFILE    HIGH   FINE
FRA   DATAFILE       MIRROR COARSE
FRA   ONLINELOG      MIRROR COARSE

-- 파일마다 다른 이중화는 템플릿에서 온다. NORMAL 디스크그룹의 CONTROLFILE
-- 템플릿은 기본이 HIGH 다. 컨트롤파일은 FINE 스트라이프도 쓴다.
-- 파일을 만들 때 템플릿이 정해지므로 나중에 템플릿을 바꿔도 기존 파일은 그대로다.

[grid@rac1 ~]$ asmcmd du +DATA/RACDB
Used_MB      Mirror_used_MB
   5548                5548
[grid@rac1 ~]$ asmcmd du +FRA/RACDB
Used_MB      Mirror_used_MB
   4276                8584

-- du 는 같은 차이를 디렉터리 단위로 보여 준다. FRA 는 4276MB 의 파일이
-- 8584MB 를 쓴다. v$asm_diskgroup 의 FREE_MB 가 이 Mirror_used_MB 기준으로 줄어든다.

--------------------------------------------------------------------------------
 [4] 어느 인스턴스가 어느 ASM 에 붙어 있는가
--------------------------------------------------------------------------------

SQL> select g.name dg, c.instance_name, c.db_name, c.status
  2    from v$asm_client c join v$asm_diskgroup g on g.group_number=c.group_number
  3   order by 1,2;

DG     INSTANCE_N DB_NAME  STATUS
------ ---------- -------- ----------
CRS    +ASM1      +ASM     CONNECTED
CRS    rac1.local _OCR     CONNECTED
       domain
DATA   +ASM1      +ASM     CONNECTED
DATA   racdb1     racdb    CONNECTED
FRA    racdb1     racdb    CONNECTED

-- +ASM1 에 붙은 데이터베이스 인스턴스는 racdb1 하나다. racdb2 가 없다.
-- _OCR 은 rac1 의 Clusterware 가 OCR 을 읽으려고 붙은 연결이다.
-- V$ 뷰는 접속한 ASM 인스턴스 것만 보여 준다.

SQL> select c.inst_id, g.name dg, c.instance_name, c.db_name, c.status
  2    from gv$asm_client c join v$asm_diskgroup g on g.group_number=c.group_number
  3   where c.db_name='racdb' order by 1,2;

   INST_ID DG    INSTANCE_N DB_NAME  STATUS
---------- ----- ---------- -------- ------------
         1 DATA  racdb1     racdb    CONNECTED
         1 FRA   racdb1     racdb    CONNECTED
         2 DATA  racdb2     racdb    CONNECTED
         2 FRA   racdb2     racdb    CONNECTED

-- GV$ 로 보면 racdb2 는 +ASM2(INST_ID 2)에 붙어 있다. 노드마다 자기 ASM 을 쓴다.
-- 데이터베이스 인스턴스는 CRS 디스크그룹에 붙지 않는다. 쓰는 파일이 없기 때문이다.
-- Flex ASM 이라 한 노드의 ASM 이 내려가면 그 노드의 racdb 가 다른 노드의
-- ASM 으로 옮겨 붙는다. 그 장면은 6장 실습 08(Flex ASM 장애 전환)에서 봤다.

--------------------------------------------------------------------------------
 [5] 클러스터 자원이 기억하는 위치
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl config database -d racdb | grep -iE 'spfile|password|disk groups|oracle home'
Oracle home: /u01/app/oracle/product/19.3.0/dbhome_rac
Spfile: +DATA/RACDB/PARAMETERFILE/spfile.272.1248276167
Password file: +DATA/RACDB/PASSWORD/pwdracdb.256.1248275447
Disk Groups: DATA,FRA

-- SPFILE 과 비밀번호 파일도 +DATA 에 있다. [3] 의 PARAMETERFILE · PASSWORD 행이다.
-- 인스턴스는 기동할 때 이 경로를 클러스터 자원에서 받아 SPFILE 을 연다.
-- Disk Groups 는 이 데이터베이스 자원이 의존하는 디스크그룹이다.
-- 데이터베이스 자원은 이 디스크그룹 자원이 올라온 뒤에 기동한다.
-- 이 랩에서는 실습용 디스크그룹 TEST 가 이 목록에 남은 채 디스크그룹만 지워져
-- srvctl config database 가 PRCD-1012 로 실패한 적이 있다.
-- srvctl modify database -db racdb -diskgroup "DATA,FRA" 로 목록을 고쳐 풀었다.
-- 디스크그룹을 지울 때는 이 목록에서 먼저 뺀다.
