================================================================================
 제16장 실습 07 — FAN 콜아웃으로 이벤트를 직접 받아 보기
 대상   : rac1, rac2
 목표   : Grid 홈 racg/usrco 에 콜아웃을 두고 서비스를 멈추고 올리고 옮기며
          FAN 이벤트가 어느 노드에 어떤 모양으로 오는지 기록한다.
          실행 권한이 없는 콜아웃이 어떻게 되는지도 본다
 소요   : 약 20분
 선행   : 16장 실습 06 완료 (사용자 서비스 없음)
 계정   : root, grid, oracle
 주의   : 콜아웃은 실습 11 끝에서 지운다. 그때까지 두 노드에 그대로 둔다
================================================================================

--------------------------------------------------------------------------------
 [2] 콜아웃 작성 — 일부러 실행 권한 없이
--------------------------------------------------------------------------------

[root@rac1 ~]# cat > /u01/app/19.3.0/grid/racg/usrco/callout.sh <<'X'
#!/bin/bash
# FAN callout — 이벤트 한 줄을 받은 시각과 함께 남긴다
echo "$(date '+%T.%3N') $(id -un) $*" >> /tmp/fan_$(hostname -s).log
X
[root@rac1 ~]# chown grid:oinstall /u01/app/19.3.0/grid/racg/usrco/callout.sh
[root@rac1 ~]# chmod 644 /u01/app/19.3.0/grid/racg/usrco/callout.sh

-- rac2 에도 같은 파일을 같은 소유자 · 권한으로 둔다.

[grid@rac1 ~]$ ls -l /u01/app/19.3.0/grid/racg/usrco/
-rw-r--r--. 1 grid oinstall 154 Nov 10 14:10 callout.sh

[oracle@rac1 ~]$ srvctl add service -db racdb -service ch16_fan -pdb racpdb1 \
                   -preferred racdb1 -available racdb2
[oracle@rac1 ~]$ srvctl start service -db racdb -service ch16_fan
[oracle@rac1 ~]$ srvctl status service -db racdb -service ch16_fan
Service ch16_fan is running on instance(s) racdb1

[root@rac1 ~]# ls -l /tmp/fan_*.log
ls: cannot access /tmp/fan_*.log: No such file or directory
[root@rac2 ~]# ls -l /tmp/fan_*.log
ls: cannot access /tmp/fan_*.log: No such file or directory

-- 서비스가 떴는데 로그가 없다. 오류 메시지도 어디에도 나오지 않았다.
-- 원인 파악 : 콜아웃에 실행 권한이 없다. Clusterware 는 usrco 의 파일을 실행 파일로 부른다.
--             실행할 수 없으면 조용히 건너뛴다. srvctl 은 성공으로 끝난다.
-- 조치     : 두 노드 모두 chmod 755.
-- 콜아웃은 "아무 일도 안 일어난다" 는 모습으로 실패한다. 배포한 뒤 서비스를 한 번
-- 멈췄다 올려 로그가 생기는지 꼭 확인한다.

--------------------------------------------------------------------------------
 [3] 실행 권한을 주고 멈췄다 올린다
--------------------------------------------------------------------------------

[root@rac1 ~]# chmod 755 /u01/app/19.3.0/grid/racg/usrco/callout.sh
[root@rac2 ~]# chmod 755 /u01/app/19.3.0/grid/racg/usrco/callout.sh

[oracle@rac1 ~]$ date +%T; srvctl stop service -db racdb -service ch16_fan; date +%T
14:10:52
14:10:53
[oracle@rac1 ~]$ date +%T; srvctl start service -db racdb -service ch16_fan; date +%T
14:10:59
14:10:59

[root@rac1 ~]# cat /tmp/fan_rac1.log
14:10:53.086 grid SERVICEMEMBER VERSION=1.0 service=ch16_fan database=racdb instance=racdb1
             host=rac1 status=down reason=USER timestamp=2026-11-10 14:10:52 timezone=+09:00 db_domain=
14:10:53.089 grid SERVICE VERSION=1.0 service=ch16_fan database=racdb instance=racdb1
             host=rac1 status=down reason=USER timestamp=2026-11-10 14:10:52 timezone=+09:00 db_domain=
14:10:59.823 grid SERVICEMEMBER VERSION=1.0 service=ch16_fan database=racdb instance=racdb1
             host=rac1 status=up reason=USER card=1 timestamp=2026-11-10 14:10:59 timezone=+09:00 db_domain=
14:10:59.825 grid SERVICE VERSION=1.0 service=ch16_fan database=racdb instance=racdb1
             host=rac1 status=up reason=USER timestamp=2026-11-10 14:10:59 timezone=+09:00 db_domain=
(한 줄로 찍힌 것을 읽기 좋게 나눴다)

-- 정지 하나에 이벤트가 둘이다. SERVICEMEMBER 는 "racdb1 에서의 ch16_fan" 이고
-- SERVICE 는 "ch16_fan 전체" 다. 인스턴스가 하나뿐인 서비스라 둘이 같이 내려갔다.
-- 콜아웃은 grid 로 실행됐다. 로그 파일을 쓸 디렉터리는 grid 가 쓸 수 있어야 한다.
-- reason=USER 는 srvctl 로 한 계획 작업이라는 뜻이다. 풀은 이 값을 보고 사용 중인
-- 연결이 반납되기를 기다렸다가 닫는다(본문 4.2.2).
-- UP 에만 card=1 이 붙는다. 지금 이 서비스를 제공하는 인스턴스 수다.
-- 이벤트 줄의 timestamp 는 초 단위이고, 콜아웃이 받은 시각은 그보다 1 초 안쪽으로 늦었다.
-- 서비스가 PDB(racpdb1)에 속해 있어도 이벤트에 PDB 이름은 없다. db_domain= 은 비어 있다.

[root@rac2 ~]# ls -l /tmp/fan_*.log
ls: cannot access /tmp/fan_*.log: No such file or directory

-- rac2 에는 아무것도 오지 않았다. 이벤트는 그 일이 생긴 노드의 콜아웃에만 간다.

--------------------------------------------------------------------------------
 [4] 서비스를 옮기면 두 노드에 하나씩
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ date +%T; srvctl relocate service -db racdb -service ch16_fan \
                   -oldinst racdb1 -newinst racdb2; date +%T
14:11:05
14:11:07
[oracle@rac1 ~]$ srvctl status service -db racdb -service ch16_fan
Service ch16_fan is running on instance(s) racdb2

[root@rac1 ~]# tail -1 /tmp/fan_rac1.log
14:11:06.984 grid SERVICEMEMBER VERSION=1.0 service=ch16_fan database=racdb instance=racdb1
             host=rac1 status=down reason=USER timestamp=2026-11-10 14:11:06 ...
[root@rac2 ~]# cat /tmp/fan_rac2.log
14:11:06.994 grid SERVICEMEMBER VERSION=1.0 service=ch16_fan database=racdb instance=racdb2
             host=rac2 status=up reason=USER card=2 timestamp=2026-11-10 14:11:06 ...

-- rac1 에는 racdb1 쪽 DOWN, rac2 에는 racdb2 쪽 UP 이 왔다. 각자 자기 일만 받는다.
-- SERVICE 이벤트는 없다. 서비스 전체로 보면 한 번도 멈추지 않았기 때문이다.
-- UP 의 card=2 가 눈에 띈다. 선호 인스턴스는 하나뿐인데 옮기는 순간에는 잠깐
-- 두 인스턴스에서 돈 것으로 셌다. 콜아웃 기록으로는 DOWN 이 10ms 먼저다.
-- 콜아웃은 비동기라 노드 사이의 기록 순서로 앞뒤를 따지지 않는다(본문 4.2.3).
-- 풀 입장에서 relocate 는 "새 쪽에 연결을 만들고, 옛 쪽 연결은 반납될 때 닫는다" 가 된다.

--------------------------------------------------------------------------------
 [5] 이벤트 한 줄 읽는 법
--------------------------------------------------------------------------------

  항목           [3]·[4] 의 값            쓰는 곳
  -------------  -----------------------  --------------------------------------
  첫 단어        SERVICEMEMBER · SERVICE   무엇에 대한 이벤트인가
  service        ch16_fan                  SYS_CONTEXT('USERENV','SERVICE_NAME')
  instance       racdb1 · racdb2           SYS_CONTEXT('USERENV','INSTANCE_NAME')
  host           rac1 · rac2               SYS_CONTEXT('USERENV','SERVER_HOST')
  status         up · down
  reason         USER                      계획 작업. 실습 08 에서 FAILURE 와 비교
  card           1 · 2                     UP 에만. 현재 제공 인스턴스 수

-- 콜아웃 안에서 status=down reason=FAILURE 일 때만 경보를 보내게 거르면
-- 정비 작업마다 경보가 울리는 일을 막는다.
-- 콜아웃은 빨리 끝나야 한다. 메일 · 티켓 같은 오래 걸리는 일은 뒤로 돌리고(&) 바로 끝낸다.
