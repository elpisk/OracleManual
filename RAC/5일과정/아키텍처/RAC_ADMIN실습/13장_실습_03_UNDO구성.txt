================================================================================
 제13장 실습 03 — UNDO 구성, 인스턴스별로 나뉜 이유
 대상   : rac1
 목표   : UNDO 테이블스페이스가 인스턴스마다 다르게 지정되어 있음을 확인하고,
          롤백 세그먼트 수준에서 왜 나누는지 이해한다
 소요   : 약 25분
 선행   : racdb 양 노드 OPEN
 계정   : oracle
================================================================================

--------------------------------------------------------------------------------
 [1] UNDO 테이블스페이스는 둘이다
--------------------------------------------------------------------------------

SQL> select tablespace_name, status, contents, retention
  2    from dba_tablespaces where contents='UNDO' order by tablespace_name;

TABLESPACE_NAME  STATUS    CONTENTS              RETENTION
---------------- --------- --------------------- -----------
UNDOTBS1         ONLINE    UNDO                  NOGUARANTEE
UNDOTBS2         ONLINE    UNDO                  NOGUARANTEE

2 rows selected.

-- 이름 끝의 숫자가 인스턴스 번호를 암시하지만 그것은 관례일 뿐이다.
-- 실제 대응은 파라미터가 정한다.

--------------------------------------------------------------------------------
 [2] 어느 인스턴스가 어느 것을 쓰는가
--------------------------------------------------------------------------------

SQL> select inst_id, value from gv$parameter
  2   where name='undo_tablespace' order by inst_id;

   INST_ID VALUE
---------- ------------
         1 UNDOTBS1
         2 UNDOTBS2

2 rows selected.

-- 12장에서 본 racdb1./racdb2. 접두어 항목이 바로 이것이다.
-- SPFILE 안에는 이렇게 들어 있다.
--
--   racdb1.undo_tablespace='UNDOTBS1'
--   racdb2.undo_tablespace='UNDOTBS2'
--
-- 12장의 세 분류로 보면 "반드시 달라야" 에 속하는 파라미터다.

--------------------------------------------------------------------------------
 [3] 흔한 오류 — 데이터파일 크기를 보려는데 행이 배수로 나온다
--------------------------------------------------------------------------------

SQL> select ts.name, df.name file_name, df.bytes/1024/1024 mb
  2    from v$tablespace ts, v$datafile df
  3   where ts.ts#=df.ts# and ts.name like 'UNDO%' order by ts.name;

NAME       FILE_NAME                                                       MB
---------- ------------------------------------------------------- ----------
UNDOTBS1   +DATA/RACDB/DATAFILE/undotbs1.259.1242761683                    350
UNDOTBS1   +DATA/RACDB/DATAFILE/undotbs1.259.1242761683                    350
UNDOTBS1   +DATA/RACDB/DATAFILE/undotbs1.259.1242761683                    350
UNDOTBS1   +DATA/RACDB/86B637.../DATAFILE/undotbs1.267.1242761985          100
UNDOTBS1   +DATA/RACDB/5A5694.../DATAFILE/undotbs1.273.1242762677          100
...
UNDOTBS2   +DATA/RACDB/DATAFILE/undotbs2.269.1242762281                     25
UNDO_2     +DATA/RACDB/5A5694.../DATAFILE/undo_2.277.1242762729            100

13 rows selected.

-- 원인 파악 : CON_ID 조건이 빠졌다. CDB 에서는 V$TABLESPACE 와 V$DATAFILE 이
--             모두 컨테이너별로 행을 가지므로 TS# 만으로 조인하면
--             컨테이너 간 교차 조인이 일어난다. 컨테이너가 셋이라
--             같은 파일이 세 번씩 나왔다.
--             이름이 다른 UNDO_2 까지 섞여 나온 것도 같은 이유다.
--             PDB 안의 UNDO 가 함께 걸려든 것이다.
-- 조치     : 조인 조건에 컨테이너를 함께 넣는다.
--             ts.con_id=df.con_id 가 교차 조인을 막고,
--             df.con_id=1 이 CDB$ROOT 로 범위를 좁힌다.
--             UNDO 는 CDB 레벨 자원이므로 CON_ID 1 만 보면 된다.
-- 재확인   : 행 수가 UNDO 테이블스페이스 수와 맞는지 본다.

SQL> select df.con_id, ts.name ts_name, df.name file_name, df.bytes/1024/1024 mb
  2    from v$tablespace ts, v$datafile df
  3   where ts.ts#=df.ts# and ts.con_id=df.con_id and df.con_id=1
  4     and ts.name like 'UNDO%' order by ts.name;

    CON_ID TS_NAME    FILE_NAME                                            MB
---------- ---------- --------------------------------------------- ----------
         1 UNDOTBS1   +DATA/RACDB/DATAFILE/undotbs1.259.1242761683         350
         1 UNDOTBS2   +DATA/RACDB/DATAFILE/undotbs2.269.1242762281          25

2 rows selected.

-- 이제 두 행이다.
--
-- 행 수가 예상보다 배수로 많으면 조인 조건에 CON_ID 가 빠졌는지
-- 먼저 의심하는 습관을 들인다. CDB 를 다룰 때 자주 만나는 함정이다.

--------------------------------------------------------------------------------
 [4] 크기가 다른 것은 문제인가
--------------------------------------------------------------------------------

-- UNDOTBS1 이 350MB, UNDOTBS2 가 25MB 다. 열네 배 차이다.
--
-- 크기 차이 자체는 문제가 아니다. 만들어진 시점과 사용량이 다르기 때문이다.
--   UNDOTBS1 : dbca 가 처음 만들었고 그동안 자동 확장되었다.
--   UNDOTBS2 : 두 번째 인스턴스를 추가할 때 만들어져 거의 쓰이지 않았다.
--
-- 다만 운영 환경에서 두 노드에 비슷한 부하가 걸린다면 결국 비슷해져야 한다.
-- 한쪽만 계속 커진다면 부하가 그 노드에 몰려 있다는 신호다.
-- 조치는 UNDO 를 키우는 것이 아니라 부하를 나누는 것이다. 15장의 서비스가 도구다.

--------------------------------------------------------------------------------
 [5] 롤백 세그먼트 — 나누는 이유는 여기에 있다
--------------------------------------------------------------------------------

SQL> select tablespace_name, count(*) cnt, max(status) status
  2    from dba_rollback_segs where tablespace_name like 'UNDO%'
  3   group by tablespace_name order by tablespace_name;

TABLESPACE_NAM        CNT STATUS
-------------- ---------- ----------------
UNDOTBS1               10 ONLINE
UNDOTBS2               10 ONLINE

2 rows selected.

-- 각각 10 개씩이다.
--
-- 트랜잭션이 시작되면 이 세그먼트 중 하나를 배정받아 이전 이미지를 쓴다.
-- 여러 개를 두는 이유는 동시 트랜잭션 간 경합을 줄이기 위해서다.
--
-- 두 인스턴스가 같은 UNDO 테이블스페이스를 쓰면 이 배정을 두고 다투게 된다.
-- 그래서 나눈다. 이것이 UNDO 를 인스턴스별로 두는 근본 이유다.
--
-- 자동 UNDO 관리에서는 오라클이 부하에 따라 세그먼트 개수를 조절하므로
-- 관리자가 개수를 신경 쓸 일은 없다. 정하는 것은 어느 테이블스페이스를 쓸지뿐이다.

--------------------------------------------------------------------------------
 [6] 보존 설정은 나뉘지 않는다
--------------------------------------------------------------------------------

SQL> select inst_id, name, value from gv$parameter
  2   where name in ('undo_management','undo_retention') order by name, inst_id;

   INST_ID NAME                 VALUE
---------- -------------------- ------
         1 undo_management      AUTO
         2 undo_management      AUTO
         1 undo_retention       900
         2 undo_retention       900

4 rows selected.

-- 나뉘는 것은 undo_tablespace 하나뿐이다.
-- 방식(AUTO)과 보존 기간(900초)은 두 인스턴스가 같은 값을 쓴다.
--
-- 다르게 잡을 수는 있지만 그러면 어느 노드에 접속했느냐에 따라
-- ORA-01555 가 나는 시점이 달라진다. 같은 쿼리가 한 노드에서는 성공하고
-- 다른 노드에서는 실패하게 되어 진단이 어려워진다.

--------------------------------------------------------------------------------
 [7] UNDO 통계는 반드시 GV$ 로
--------------------------------------------------------------------------------

SQL> select inst_id, undotsn, undoblks, txncount, maxquerylen
  2    from gv$undostat order by inst_id, begin_time desc;

   INST_ID    UNDOTSN   UNDOBLKS   TXNCOUNT MAXQUERYLEN
---------- ---------- ---------- ---------- -----------
         1          2          1         37         975
         1          2          1         28         971
         1          2          1         18        1561
         1          2          1        200         365
         1          2        310        207           0
         1          2          1         24         969

6 rows selected.

-- 10 분 단위로 쌓이는 통계다. 읽는 법.
--
--   UNDOBLKS 가 대부분 1 인데 한 구간만 310 이다.
--   그 시간대에 대량 변경이 있었다는 뜻이다. TXNCOUNT 도 207 로 높다.
--
--   MAXQUERYLEN 이 1561 인 구간이 있다. 그 구간에서 가장 오래 돈 쿼리가
--   1561 초짜리였다는 뜻이다. undo_retention 900 을 크게 넘었으므로
--   그 쿼리가 필요로 하는 이전 이미지가 덮였을 수 있었다.
--   즉 ORA-01555 가 날 수 있었던 구간이다.
--
-- RAC 에서는 반드시 GV$ 로 본다. UNDO 가 인스턴스별로 나뉘어 있으므로
-- V$UNDOSTAT 만 보면 접속한 인스턴스의 절반짜리 그림만 얻는다.
