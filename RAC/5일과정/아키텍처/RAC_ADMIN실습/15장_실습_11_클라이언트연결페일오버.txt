================================================================================
 제15장 실습 11 — 클라이언트 쪽 연결 페일오버, 주소 목록과 시간 제한
 대상   : rac1 (클라이언트 역할)
 목표   : 접속 기술자의 주소 목록, FAILOVER · LOAD_BALANCE,
          TRANSPORT_CONNECT_TIMEOUT · RETRY_COUNT 가 접속 시간과 결과를
          어떻게 바꾸는지 실측한다. 접속 오류 번호로 틀린 곳을 가린다
 소요   : 약 25분
 선행   : 실습 10 완료 (ch15_pdb 가 racdb1 에서 실행 중)
 계정   : oracle
 주의   : 실제 tnsnames.ora 는 건드리지 않는다. TNS_ADMIN 을 /tmp/tns15 로 돌려 쓴다
================================================================================

--------------------------------------------------------------------------------
 [1] 준비 — 시험용 tnsnames.ora
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ mkdir /tmp/tns15; vi /tmp/tns15/tnsnames.ora

DEAD_FIRST =
  (DESCRIPTION =
    (ADDRESS_LIST = (LOAD_BALANCE=OFF)(FAILOVER=ON)
      (ADDRESS = (PROTOCOL = TCP)(HOST = 10.255.255.1)(PORT = 1521))
      (ADDRESS = (PROTOCOL = TCP)(HOST = rac-scan)(PORT = 1521)))
    (CONNECT_DATA = (SERVICE_NAME = ch15_pdb)))

DEAD_FIRST_TO =
  (DESCRIPTION = (TRANSPORT_CONNECT_TIMEOUT=3)(CONNECT_TIMEOUT=60)
    (ADDRESS_LIST = (LOAD_BALANCE=OFF)(FAILOVER=ON)
      ... DEAD_FIRST 와 같은 두 주소 ...
    (CONNECT_DATA = (SERVICE_NAME = ch15_pdb)))

DEAD_FIRST_NOFO =
  (DESCRIPTION = (TRANSPORT_CONNECT_TIMEOUT=3)
    (ADDRESS_LIST = (LOAD_BALANCE=OFF)(FAILOVER=OFF)
      ... DEAD_FIRST 와 같은 두 주소 ...
    (CONNECT_DATA = (SERVICE_NAME = ch15_pdb)))

ALL_DEAD_RETRY =
  (DESCRIPTION = (TRANSPORT_CONNECT_TIMEOUT=3)(RETRY_COUNT=2)(RETRY_DELAY=2)
    (ADDRESS_LIST = (FAILOVER=ON)
      (ADDRESS = (PROTOCOL = TCP)(HOST = 10.255.255.1)(PORT = 1521))
      (ADDRESS = (PROTOCOL = TCP)(HOST = 10.255.255.2)(PORT = 1521)))
    (CONNECT_DATA = (SERVICE_NAME = ch15_pdb)))

VIP_LB / VIP_NOLB =
  (DESCRIPTION =
    (ADDRESS_LIST = (LOAD_BALANCE=ON 또는 OFF)(FAILOVER=ON)
      (ADDRESS = (PROTOCOL = TCP)(HOST = rac1-vip)(PORT = 1521))
      (ADDRESS = (PROTOCOL = TCP)(HOST = rac2-vip)(PORT = 1521)))
    (CONNECT_DATA = (SERVICE_NAME = racpdb1)))

VIP_PDBSVC =
  (DESCRIPTION =
    (ADDRESS_LIST = (LOAD_BALANCE=OFF)(FAILOVER=ON)
      (ADDRESS = (PROTOCOL = TCP)(HOST = rac2-vip)(PORT = 1521))
      (ADDRESS = (PROTOCOL = TCP)(HOST = rac1-vip)(PORT = 1521)))
    (CONNECT_DATA = (SERVICE_NAME = ch15_pdb)))

[oracle@rac1 ~]$ cat /tmp/tns15/t.sh            (요지)
#!/bin/bash
export TNS_ADMIN=/tmp/tns15
# sqlplus -L system/<비밀번호>@$1 로 한 번 붙어 인스턴스 · 컨테이너를 찍고, 걸린 시간을 잰다

[oracle@rac1 ~]$ ping -c1 -W2 10.255.255.1 | tail -2
1 packets transmitted, 0 received, 100% packet loss, time 0ms

[oracle@rac1 ~]$ grep -E 'vip|scan' /etc/hosts
172.16.225.111  rac1-vip.localdomain  rac1-vip
172.16.225.112  rac2-vip.localdomain  rac2-vip
172.16.225.120  rac-scan.localdomain  rac-scan

-- 살아 있는 주소는 이 셋이다. 기술자에서는 이름(rac-scan, rac1-vip, rac2-vip)으로 쓴다.
-- 10.255.255.1 · 10.255.255.2 는 랩에 없는 가짜 주소다. 답하지 않는 주소가 필요해서
-- 일부러 넣었고, 꺼진 노드나 끊긴 망을 대신한다.
-- 같은 서브넷 주소는 ARP 단계에서 실패 모양이 달라지므로 일부러 게이트웨이 너머를 골랐다.

--------------------------------------------------------------------------------
 [2] 죽은 주소가 맨 앞 — 시간 제한이 없으면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ /tmp/tns15/t.sh DEAD_FIRST
DEAD_FIRST         22.1s  racdb1 RACPDB1

-- 접속은 됐다. 첫 주소에서 막히자 두 번째 주소(SCAN)로 넘어갔다. 연결 페일오버다.
-- 그런데 22 초가 걸렸다. 사용자에게는 응답 없는 화면이다.

[root@rac1 ~]# time bash -c 'echo > /dev/tcp/10.255.255.1/1521'
bash: connect: Connection refused
real    0m21.0s

[root@rac1 ~]# sysctl net.ipv4.tcp_syn_retries
net.ipv4.tcp_syn_retries = 6

-- 22 초 가운데 21 초는 TCP 연결 시도다. 이 랩은 VMware NAT 게이트웨이가 21 초 뒤에
-- 거절 응답을 돌려줘서 이 정도로 끝났다.
-- 패킷이 그냥 버려지는 실제 망이라면 더 길다. 리눅스 커널은 SYN 을 6 번 재전송한다.
-- 문서상 sqlnet.ora 의 TCP.CONNECT_TIMEOUT 기본값은 60 초다.

--------------------------------------------------------------------------------
 [3] TRANSPORT_CONNECT_TIMEOUT=3
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ /tmp/tns15/t.sh DEAD_FIRST_TO
DEAD_FIRST_TO       4.1s  racdb1 RACPDB1

-- 같은 주소 목록에 시간 제한 3 초만 더했다. 22 초가 4 초가 됐다.
-- 3 초 만에 첫 주소를 포기하고 SCAN 으로 넘어가 1 초 안에 붙었다.
-- 문서가 접속 문자열에 시간 제한을 넣으라고 하는 이유다.
-- TRANSPORT_CONNECT_TIMEOUT 은 TCP 연결까지, CONNECT_TIMEOUT 은 연결 전체의 시간이다.

--------------------------------------------------------------------------------
 [4] FAILOVER=OFF 이면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ /tmp/tns15/t.sh DEAD_FIRST_NOFO
DEAD_FIRST_NOFO     3.0s  ERROR: ORA-12170: TNS:Connect timeout occurred

-- 원인 파악 : 주소 목록이 있어도 FAILOVER=OFF 면 첫 주소만 시도한다.
--             3 초 시간 제한에 걸려 ORA-12170 으로 끝났다. SCAN 은 시도하지 않았다.
--             주소 목록의 FAILOVER 기본값은 ON 이다. OFF 는 누가 일부러 적은 것이다.
-- 조치     : FAILOVER=OFF 를 지우거나 ON 으로 바꾼다.
-- 재확인   : DEAD_FIRST_TO 처럼 두 번째 주소로 붙는지 본다.

--------------------------------------------------------------------------------
 [5] 모두 죽었을 때 — RETRY_COUNT 가 시간을 늘린다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ /tmp/tns15/t.sh ALL_DEAD_RETRY
ALL_DEAD_RETRY     22.0s  ERROR: ORA-12170: TNS:Connect timeout occurred

-- 22 초가 어디서 나왔는지 계산이 맞는다.
--   주소 2 개 × 시도 3 회(처음 1 + 재시도 2) × 3 초 = 18 초
--   목록을 다시 돌기 전 RETRY_DELAY 2 초 × 2 번      =  4 초
-- RETRY_COUNT 는 목록 전체를 다시 도는 횟수다. 노드 재기동처럼 잠깐 사라지는 장애를
-- 넘기는 데 쓴다. 대신 정말 다 죽었을 때 오류를 받기까지의 시간도 그만큼 늘어난다.

--------------------------------------------------------------------------------
 [6] ORA-12514 도 다음 주소로 넘어간다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ /tmp/tns15/t.sh rac2-vip:1521/ch15_pdb
rac2-vip:1521/ch15_pdb    0.0s  ERROR: ORA-12514: TNS:listener does not currently know of service requested in connect

[oracle@rac1 ~]$ /tmp/tns15/t.sh VIP_PDBSVC
VIP_PDBSVC          1.1s  racdb1 RACPDB1

-- ch15_pdb 는 racdb1 에서만 돈다. rac2 의 노드 리스너는 이 서비스를 모른다.
-- rac2-vip 하나로 붙으면 ORA-12514 다. 주소 목록 맨 앞에 rac2-vip 를 두면
-- 그 오류를 받고 다음 주소 rac1-vip 로 넘어가 1.1 초에 붙었다.
-- 연결 페일오버는 TCP 실패뿐 아니라 리스너의 거절에도 일어난다.
-- 다만 노드 VIP 목록으로 이걸 해결하는 것보다 SCAN 하나를 쓰는 편이 낫다.
-- SCAN 리스너는 어느 인스턴스에 서비스가 있는지 알고 바로 보낸다.

--------------------------------------------------------------------------------
 [7] LOAD_BALANCE — 노드 VIP 를 나열하면 주소가 곧 인스턴스다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ /tmp/tns15/loop.sh VIP_LB 20          (20 번 붙어 인스턴스별로 센다)
     13 racdb1
      7 racdb2

[oracle@rac1 ~]$ /tmp/tns15/loop.sh VIP_NOLB 20
     20 racdb1

-- racpdb1 은 두 인스턴스에서 도는데도 LOAD_BALANCE=OFF 면 20 번 모두 racdb1 이다.
-- 노드 리스너는 자기 노드의 인스턴스만 등록받는다(3.3.1). 첫 주소 rac1-vip 가 살아 있으면
-- 늘 racdb1 로 간다.
-- LOAD_BALANCE=ON 이면 클라이언트가 주소를 무작위로 고른다. 13 대 7 로 나뉘었다.
-- 번갈아 고르는 것이 아니라 무작위라 고르게 나뉘지 않는다.
-- 이것이 클라이언트 쪽 부하 분산이다. 서버 쪽(실습 07 의 CLB goal)과는 별개이고,
-- SCAN 을 쓰면 클라이언트가 할 일이 거의 없다. 옛 클라이언트용 방식이다.

--------------------------------------------------------------------------------
 [8] 오류 번호로 틀린 곳을 가린다
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ for a in rac-scan:1522/ch15_pdb rac-scn:1521/ch15_pdb \
                          rac-scan:1521/ch15_pdbx NO_SUCH_ALIAS; do /tmp/tns15/t.sh $a; done
rac-scan:1522/ch15_pdb    0.0s  ERROR: ORA-12541: TNS:no listener
rac-scn:1521/ch15_pdb     0.0s  ERROR: ORA-12154: TNS:could not resolve the connect identifier specified
rac-scan:1521/ch15_pdbx   0.0s  ERROR: ORA-12514: TNS:listener does not currently know of service requested in connect
NO_SUCH_ALIAS             0.0s  ERROR: ORA-12154: TNS:could not resolve the connect identifier specified

  틀린 곳                       오류        시간
  ---------------------------   ---------   -------------------------
  별칭이 tnsnames.ora 에 없다   ORA-12154   즉시
  호스트 이름 철자              ORA-12154   즉시 (이름 풀이 실패)
  포트 (그 포트에 리스너 없음)  ORA-12541   즉시 (상대가 거절)
  서비스 이름 · 서비스 정지     ORA-12514   즉시 (리스너가 거절)
  주소가 응답하지 않음          ORA-12170   시간 제한만큼 기다린 뒤

-- 앞의 넷은 바로 돌아온다. 오래 걸리고 ORA-12170 이면 망이나 노드 쪽을 본다.
-- EZConnect 에서 호스트 이름을 틀려도 별칭이 없을 때와 같은 ORA-12154 가 난다.
-- 번호만 보고 tnsnames.ora 를 뒤지지 말고, 접속 문자열에 무엇을 썼는지부터 본다.

[oracle@rac1 ~]$ rm -rf /tmp/tns15
