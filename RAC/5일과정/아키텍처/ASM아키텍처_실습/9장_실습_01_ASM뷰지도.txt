 [4] TOTAL_MB 가 0 인데 디스크는 4GB — OS_MB 와 구별하기
--------------------------------------------------------------------------------

SQL> col path for a22
SQL> select path, total_mb, os_mb, header_status
  2    from v$asm_disk where group_number=0 order by path;

PATH                     TOTAL_MB      OS_MB HEADER_STATU
---------------------- ---------- ---------- ------------
/dev/asmdisk/test1              0       4096 FORMER
/dev/asmdisk/test2              0       4096 FORMER
/dev/asmdisk/test3              0       4096 FORMER
/dev/asmdisk/test4              0       4096 FORMER
/dev/asmdisk/test5              0       4096 FORMER
/dev/asmdisk/test6              0       4096 FORMER

6 rows selected.

-- TOTAL_MB = "ASM 이 쓸 수 있는 공간"  → 편입 전이므로 0
-- OS_MB    = "디스크의 물리 크기"       → 4096 (4GB)
--
-- 실무 함정 : 신규 디스크를 붙이고 TOTAL_MB 가 0 이라 udev 가 잘못된 줄
--             아는 경우가 많다. OS_MB 가 정확히 읽힌다는 것 자체가
--             장치 파일과 권한이 정상이라는 증거다.
--             권한이 없으면 HEADER_STATUS 가 UNKNOWN 이 되거나
--             행 자체가 조회되지 않는다.

         2 ONLINELOG                     4        800
         2 PARAMETERFILE                 1          0
         2 PASSWORD                      1          0
         2 TEMPFILE                      3        108
         3 CONTROLFILE                   1         19
         3 ONLINELOG                     4        800

12 rows selected.

-- 그룹 1 (+CRS)  : OCR 158MB, OCR 백업 3, ASM 자신의 SPFILE·패스워드.
--                  데이터베이스 파일은 하나도 없다.
-- 그룹 2 (+DATA) : 데이터파일 13 개 3445MB. 데이터베이스의 본체.
-- 그룹 3 (+FRA)  : 컨트롤파일 1, 리두로그 4. 다중화된 사본만 있다.
--
-- 컨트롤파일과 리두로그가 +DATA 와 +FRA 양쪽에 있는 것은 dbca 가 기본으로
-- 구성한 다중화다. 디스크그룹 하나를 통째로 잃어도 둘 다 남는다.
-- 반면 데이터파일은 다중화 대상이 아니라 백업 대상이다.

--------------------------------------------------------------------------------
 [6] V$ASM_OPERATION — 지금 무슨 작업이 도는가
--------------------------------------------------------------------------------

SQL> select group_number, operation, state, power, sofar, est_work, est_minutes
  2    from v$asm_operation;

no rows selected

-- 여기서 "조회가 안 된다" 고 판단하기 쉽다. 다른 네 뷰는 모두 행이 있었다.
--
-- 원인 파악 : V$ASM_OPERATION 은 "지금 진행 중인 작업" 만 담는다.
--             리밸런스가 돌고 있지 않으면 0 행이 정상이다.
--             8장에서는 디스크를 추가하는 동안 이 뷰에 행이 있었다.
--             시점에 종속적인 뷰라는 점이 다른 넷과 다르다.
-- 조치     : 조치할 것이 없다. 0 행이 곧 "지금 아무 작업도 없다" 는 답이다.
--             권한이나 접속을 의심할 필요가 없다.
-- 재확인   : 같은 세션에서 행이 있는 뷰를 하나 조회해 접속이 정상임을 확인한다.

SQL> select count(*) from v$asm_diskgroup;

  COUNT(*)
----------
         3

1 row selected.

-- 접속과 권한은 정상이다. V$ASM_OPERATION 이 비어 있던 것은 상태를 반영한
-- 결과였을 뿐이다.
--
-- 교훈 : 뷰가 비어 있는 것과 조회가 실패한 것은 다르다.
--        0 행도 하나의 답이다.

--------------------------------------------------------------------------------
 [7] 정리 — 다섯 뷰가 답하는 질문
--------------------------------------------------------------------------------

--  V$ASM_DISKGROUP : 공간이 얼마나 남았나, 마운트되어 있나   (디스크그룹 1개)
--  V$ASM_DISK      : 어떤 디스크가 어디 붙어 있나, 상태는     (디스크 1장)
--  V$ASM_FILE      : 무엇이 얼마나 들어 있나                  (ASM 파일 1개)
--  V$ASM_CLIENT    : 누가 이 ASM 을 쓰고 있나                 (인스턴스 1개)
--  V$ASM_OPERATION : 지금 무슨 작업이 돌고 있나               (작업 1개)
--
-- 위에서부터 단위가 크고, 아래로 갈수록 시점에 종속적이다.
-- 뷰 이름을 외우지 말고 "지금 알고 싶은 것이 무엇인가" 에서 출발한다.

SQL> exit
