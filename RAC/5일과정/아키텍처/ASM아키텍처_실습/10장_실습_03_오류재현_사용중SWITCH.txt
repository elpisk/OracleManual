-- 실제로 해 보자.

--------------------------------------------------------------------------------
 [3] 오류 재현 — ONLINE 상태에서 SWITCH
--------------------------------------------------------------------------------

RMAN> switch datafile 23 to copy;

using target database control file instead of recovery catalog
RMAN-00571: ===========================================================
RMAN-00569: =============== ERROR MESSAGE STACK FOLLOWS ===============
RMAN-00571: ===========================================================
RMAN-03002: failure of switch to copy command at 09/03/2026 19:34:01
RMAN-06572: database is open and datafile 23 is not offline

-- 오류 : RMAN-06572
--
-- 원인 파악 : 메시지가 조건을 그대로 말해 준다.
--             "데이터베이스가 열려 있는데 데이터파일 23 이 오프라인이 아니다"
--
--             실습 02 의 ORA-19602 와 요구 조건이 겹쳐 헷갈리기 쉬우므로
--             정리해 둔다.
--
--             단계    명령                요구 조건
--             ----   ------------------  --------------------------------
--             복사    BACKUP AS COPY      NOARCHIVELOG 면 OFFLINE 필요
--                                         (ARCHIVELOG 면 ONLINE 가능)
--             전환    SWITCH ... TO COPY  DB 가 OPEN 이면 항상 OFFLINE 필요
--                                         (아카이브 모드와 무관)
--
--             즉 SWITCH 의 OFFLINE 요구는 아카이브 모드와 상관없다.
--             ARCHIVELOG 라도 SWITCH 순간에는 파일이 오프라인이어야 한다.
--
--             이유는 두 작업의 성격이 다르기 때문이다.
--             복사는 파일을 읽는 작업이라 일관된 시점만 확보되면 된다.
--             SWITCH 는 그 파일에 대한 데이터베이스의 인식을 통째로 바꾸는
--             작업이라 누구도 쓰고 있지 않아야 한다.
-- 조치     : 다시 OFFLINE 으로 내리고 SWITCH 한다.
-- 재확인   : [4] 에서 수행한다.

--------------------------------------------------------------------------------
 [4] OFFLINE 후 SWITCH — 그런데 이번엔 다른 문제가
--------------------------------------------------------------------------------

SQL> alter tablespace ts_mig offline;

Tablespace altered.

RMAN> switch datafile 23 to copy;

datafile 23 switched to datafile copy "+FRA/RACDB/DATAFILE/ts_mig.262.1243020835"
                                                               (2.2초 소요)

-- SWITCH 는 성공했다. 이제 ONLINE 으로 올리면 될 것 같다.

SQL> alter tablespace ts_mig online;
alter tablespace ts_mig online
*
ERROR at line 1:
ORA-01113: file 23 needs media recovery
ORA-01110: data file 23: '+FRA/RACDB/DATAFILE/ts_mig.262.1243020835'

-- 오류 : ORA-01113
--
-- 실습 02 에서는 복구 없이 바로 ONLINE 이 되었다. 무엇이 달라졌는가.

--------------------------------------------------------------------------------

--------------------------------------------------------------------------------

CURRENT_SCN
-----------
    3994002

-- 원인 파악 : 사본의 체크포인트 SCN 은 3993926 인데
--             데이터베이스의 현재 SCN 은 3994002 다. 76 만큼 벌어져 있다.
--
--             원인은 [2] 의 실수다. 사본을 만든 뒤 SWITCH 전에
--             테이블스페이스를 ONLINE 으로 올렸고, 그 사이에 데이터베이스가
--             앞으로 나아갔다. 사본은 그 이전 시점의 것이라 뒤처졌다.
--
--             뒤처진 파일을 그대로 열면 데이터가 어긋나므로 오라클이 막는다.

SQL> select count(*) from sys.mig_t;
select count(*) from sys.mig_t
*
ERROR at line 1:
ORA-00376: file 23 cannot be read at this time
ORA-01110: data file 23: '+FRA/RACDB/DATAFILE/ts_mig.262.1243020835'

-- 읽기도 안 된다. ORA-00376 은 OFFLINE 상태인 파일에 접근할 때 나온다.
-- ONLINE 이 실패했으니 파일은 여전히 OFFLINE 이다.

--------------------------------------------------------------------------------
 [7] 반드시 짚어야 할 질문 — NOARCHIVELOG 인데 왜 됐는가
--------------------------------------------------------------------------------

-- 아카이브 로그가 없는데 미디어 복구가 어떻게 성공했는가.
--
-- 미디어 복구는 아카이브 로그에서만 리두를 읽는 것이 아니다.
-- 필요한 리두가 아직 온라인 리두로그에 남아 있으면 거기서 읽어 적용한다.
-- 벌어진 SCN 이 76 뿐이라 그 사이의 변경이 아직 덮어써지지 않았다.
--
-- 그러나 이것은 보장되지 않는다.
-- 온라인 리두로그는 순환한다. 로그 스위치가 여러 번 일어나 필요한 리두가
-- 밀려나면 복구할 방법이 없고, 그 데이터파일은 되살릴 수 없다.
--
-- NOARCHIVELOG 에서 마이그레이션이 위험한 진짜 이유가 이것이다.
-- 실패했을 때 되돌릴 수 있느냐가 시간에 달려 있다.
--
-- 예방은 간단하다.
--   복사부터 SWITCH 까지 OFFLINE 을 유지하면 SCN 이 벌어지지 않는다.
--   실습 06 에서 이 방식을 써서 복구가 0 초로 끝나는 것을 확인한다.
