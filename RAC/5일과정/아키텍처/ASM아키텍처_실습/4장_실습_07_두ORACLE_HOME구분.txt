================================================================================
 제4장 실습 07 — 두 ORACLE_HOME 구분
 대상   : rac1
 목표   : 계정별 환경 변수를 확인하고, 인벤토리와 oratab 으로 이 서버에 설치된
          Oracle 홈을 파악하며, 진단 로그 위치가 어떻게 갈리는지 확인한다
 소요   : 약 25분
 선행   : GI 와 DB 소프트웨어가 설치되어 있을 것
 계정   : grid, oracle
================================================================================

--------------------------------------------------------------------------------
 [1] 계정별 환경 변수
--------------------------------------------------------------------------------

[root@rac1 ~]# su - grid -c 'echo "GRID: $ORACLE_HOME / SID=$ORACLE_SID / BASE=$ORACLE_BASE"'
GRID: /u01/app/19.3.0/grid / SID=+ASM1 / BASE=/u01/app/grid

[root@rac1 ~]# su - oracle -c 'echo "DB  : $ORACLE_HOME / SID=$ORACLE_SID / BASE=$ORACLE_BASE"'
DB  : /u01/app/oracle/product/19.3.0/dbhome_rac / SID=racdb1 / BASE=/u01/app/oracle

-- 세 가지가 모두 다르다.
--   ORACLE_HOME  소프트웨어가 설치된 위치
--   ORACLE_SID   기본으로 접속할 인스턴스
--   ORACLE_BASE  진단 정보와 관리 파일의 기준 경로
--
-- 특히 ORACLE_BASE 가 다르다는 점이 중요하다. 로그 위치가 여기서 갈린다.

-- 눈여겨볼 점 하나. Grid 홈이 ORACLE_BASE 의 하위가 아니다.
--   ORACLE_BASE = /u01/app/grid
--   ORACLE_HOME = /u01/app/19.3.0/grid
-- 처음 보면 오타처럼 보이지만 19c 의 표준 구성이다.
-- 데이터베이스 쪽은 하위 관계다.
--   ORACLE_BASE = /u01/app/oracle
--   ORACLE_HOME = /u01/app/oracle/product/19.3.0/dbhome_rac

[root@rac1 ~]# su - grid -c 'cat ~/.bash_profile'
[ -f /etc/bashrc ] && . /etc/bashrc
umask 022
export ORACLE_BASE=/u01/app/grid
export ORACLE_HOME=/u01/app/19.3.0/grid
export ORACLE_SID=+ASM1
export PATH=$ORACLE_HOME/bin:$PATH
export LD_LIBRARY_PATH=$ORACLE_HOME/lib

-- 두 번째 노드에서는 ORACLE_SID 가 +ASM2, racdb2 다.
-- 이 값을 잘못 두면 rac2 에서 접속했는데 racdb1 을 찾아 ORA-01034 가 난다.

--------------------------------------------------------------------------------
 [2] 인벤토리로 설치된 홈 전체 파악
--------------------------------------------------------------------------------

[grid@rac1 ~]$ grep -oE 'NAME="[^"]+"|LOC="[^"]+"' /u01/app/oraInventory/ContentsXML/inventory.xml
NAME="OraDB19Home1"
LOC="/u01/app/oracle/product/19.3.0/dbhome_1"
NAME="OraGI19Home1"
LOC="/u01/app/19.3.0/grid"
NAME="OraDB19Home2"
LOC="/u01/app/oracle/product/19.3.0/dbhome_rac"

-- 세 개가 등록되어 있다.
--   OraDB19Home1  원본 이미지의 단일 인스턴스용 홈. 이 교재에서는 쓰지 않는다
--   OraGI19Home1  Grid 홈
--   OraDB19Home2  이 교재에서 새로 만든 RAC 용 데이터베이스 홈
--
-- 인벤토리가 가장 확실한 근거다. 환경 변수는 지금 쓰는 것만 알려 주고
-- oratab 은 단일 인스턴스만 담는다.

--------------------------------------------------------------------------------
 [3] oratab 은 무엇을 담는가
--------------------------------------------------------------------------------

[grid@rac1 ~]$ grep -vE '^#|^$' /etc/oratab
orcl:/u01/app/oracle/product/19.3.0/dbhome_1:N
PROD:/u01/app/oracle/product/19.3.0/dbhome_1:N
KNH:/u01/app/oracle/product/19.3.0/dbhome_1:N

-- 원본 이미지의 단일 인스턴스 데이터베이스 셋만 있다. racdb 는 없다.
--
-- 형식은 SID:ORACLE_HOME:자동기동여부 다.
-- 세 번째 필드가 N 이므로 dbstart 로 자동 기동하지 않는다.
--
-- racdb 가 없는 이유는 RAC 데이터베이스가 Clusterware 자원으로 관리되기
-- 때문이다. oratab 의 자동 기동 표시에 의존하지 않는다.

[oracle@rac1 ~]$ srvctl config database -d racdb | grep -E 'Management policy|Type'
Management policy: AUTOMATIC
Type: RAC

-- 자동 기동은 이 Management policy 가 결정한다. AUTOMATIC 이면 노드가
-- 부팅할 때 Clusterware 가 올린다. 12장에서 다룬다.
--
-- 따라서 oratab 만 보고 이 서버의 데이터베이스를 판단하면 안 된다.
-- RAC 데이터베이스는 srvctl config database 로 확인한다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — oracle 계정으로 Clusterware 진단을 찾으면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ adrci exec="show homes"
ADR Homes:
diag/rdbms/orcl/orcl
diag/rdbms/knh/KNH
diag/rdbms/prod/PROD
diag/rdbms/racdb/racdb1
diag/clients/user_oracle/host_3192166451_110
diag/clients/user_oracle/host_203307297_110
diag/tnslsnr/oel7v9/listener
diag/tnslsnr/oel7v9/lsnr
diag/tnslsnr/oel7v9/lsnrprod

-- diag/crs/rac1/crs 가 없다. Clusterware 홈이 목록에 나오지 않는다.
-- 여기서 "Clusterware 진단 정보가 없다"고 판단하면 잘못이다.

-- 원인 파악
--   adrci 는 그 계정의 ORACLE_BASE 아래 진단 홈만 보여 준다.
[oracle@rac1 ~]$ echo $ORACLE_BASE
/u01/app/oracle
[root@rac1 ~]# su - grid -c 'echo $ORACLE_BASE'
/u01/app/grid

--   Clusterware 홈은 /u01/app/grid 아래에 있으므로 oracle 계정의 목록에
--   나올 수 없다. 계층마다 계정이 다르듯 ADR base 도 다르다.
--
--   목록에 orcl, knh, prod 처럼 낯선 데이터베이스가 보이는 것도 눈여겨본다.
--   원본 이미지에 있던 단일 인스턴스 데이터베이스들의 진단 홈이 그대로 남아
--   있다. tnslsnr/oel7v9 도 원본 호스트명 시절의 흔적이다.
--   실제로 쓰지 않는 항목이며 클론 환경 특유의 잔재다.

-- 조치 — grid 계정으로 실행한다.
[oracle@rac1 ~]$ exit
[root@rac1 ~]# su - grid
[grid@rac1 ~]$ adrci exec="show homes" | grep crs
diag/crs/rac1/crs

-- 재확인
--   grid 계정에서는 Clusterware 홈이 정상적으로 나온다.

-- 다만 파일 읽기 자체는 oracle 로도 된다. 권한을 확인해 본다.
[root@rac1 ~]# ls -l /u01/app/grid/diag/crs/rac1/crs/trace/alert.log
-rw-rw----. 1 grid oinstall 35072 alert.log

[root@rac1 ~]# id oracle
uid=1000(oracle) gid=54321(oinstall) groups=54321(oinstall),54322(dba),...

[oracle@rac1 ~]$ tail -2 /u01/app/grid/diag/crs/rac1/crs/trace/alert.log
2026-09-01 13:05:23.135 [CRSD(4611)]CRS-2772: Server 'rac2' has been assigned to pool 'Generic'.
2026-09-01 13:05:23.136 [CRSD(4611)]CRS-2772: Server 'rac2' has been assigned to pool 'ora.racdb'.

--   읽힌다. 파일이 grid:oinstall 소유에 그룹 읽기 권한이 있고
--   oracle 의 기본 그룹도 oinstall 이기 때문이다.
--   3장에서 두 계정의 기본 그룹을 모두 oinstall 로 맞춘 결과가 여기서 나타난다.
--
-- 배운 것
--   경로를 알면 oracle 로도 읽을 수 있다. 문제는 adrci 로 경로를 찾을 때다.
--   Clusterware 진단은 grid 계정으로 하는 것이 원칙이며 그래야 adrci 가
--   올바른 홈 목록을 보여 준다.
--   "로그가 없다"고 판단하기 전에 계정부터 확인한다.

--------------------------------------------------------------------------------
 [6] 홈이 여럿일 때의 혼란
--------------------------------------------------------------------------------

[grid@rac1 ~]$ which sqlplus
/u01/app/19.3.0/grid/bin/sqlplus

[oracle@rac1 ~]$ which sqlplus
/u01/app/oracle/product/19.3.0/dbhome_rac/bin/sqlplus

-- 같은 명령인데 계정에 따라 다른 실행 파일이 잡힌다.
-- 원본 이미지의 dbhome_1 에도 sqlplus 가 있으므로 이 서버에는 셋이 있다.
--
-- root 로 작업할 때는 PATH 에 어느 홈도 없다.
[root@rac1 ~]# which crsctl
/usr/bin/which: no crsctl in (/usr/local/sbin:/usr/local/bin:/sbin:/bin:...)

-- 그래서 root 로 crsctl 을 쓸 때는 전체 경로를 써야 한다.
[root@rac1 ~]# /u01/app/19.3.0/grid/bin/crsctl config crs
CRS-4622: Oracle High Availability Services autostart is enabled.

-- 작업을 시작할 때 세 가지를 먼저 출력해 보는 습관을 들인다.
[grid@rac1 ~]$ whoami; echo $ORACLE_HOME; echo $ORACLE_SID
grid
/u01/app/19.3.0/grid
+ASM1

--------------------------------------------------------------------------------
 [7] 정리
--------------------------------------------------------------------------------

-- 확인한 것
--   1) 계정마다 ORACLE_HOME, ORACLE_SID, ORACLE_BASE 가 모두 다르다
--   2) Grid 홈은 ORACLE_BASE 의 하위가 아니다. 19c 의 표준이다
--   3) 인벤토리에 세 홈이 등록되어 있다
--   4) oratab 에는 단일 인스턴스만 있고 racdb 는 없다
--   5) RAC DB 의 자동 기동은 Management policy 가 결정한다
--   6) adrci 는 그 계정의 ORACLE_BASE 아래 홈만 보여 준다
--   7) 파일 읽기는 oinstall 그룹이면 oracle 로도 된다
--   8) root 는 PATH 에 홈이 없으므로 전체 경로를 쓴다
--
-- 이 실습은 조회만 했으므로 원상 복구할 것이 없다.
