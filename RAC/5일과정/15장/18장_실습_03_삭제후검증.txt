[root@rac1 ~]# ocrcheck | grep ' ID '
         ID                       :  665666554

--   항목            삭제 전 (실습 01)    삭제 후
--   ------------    -----------------    -----------------
--   노드            rac1, rac2           rac1
--   CRS 자원        17                   16
--   -init 자원      13                   13
--   보팅 디스크     3 ONLINE             3 ONLINE
--   OCR ID          665666554            665666554
--
-- 자원 하나가 줄었다. rac2-vip 다.
-- 나머지 노드별 자원(LISTENER, ons, net1 등)은 자원 정의가 하나이고
-- 노드마다 행이 붙는 구조라 개수가 줄지 않는다. 행만 줄었다.
--
-- -init 자원 13 은 rac1 의 하위 스택(ohasd 가 관리하는 cssd, crsd, ctssd 등)이라
-- 다른 노드를 지워도 변하지 않는다.
--
-- 보팅 디스크와 OCR ID 가 그대로인 것이 중요하다.
-- 노드 삭제는 OCR 안의 항목을 지운 것이지 OCR 을 새로 만든 것이 아니다.

[root@rac2 ~]# ls -d /home/grid/.ssh
/home/grid/.ssh

-- 남은 것을 정리하면 이렇다.
--
--   /etc/oracle            빈 디렉터리 두 개. olr.loc, ocr.loc 은 없다
--   그리드 홈              빈 디렉터리
--   /u01/app/grid          261MB (CHM 데이터, 진단 로그)
--   OS 계정·그룹           그대로
--   SSH 키                 그대로
--   udev 규칙·공유 디스크  그대로 (실습 04 의 cluvfy 가 rac2 에서 공유 접근을 확인한다)
--
-- deinstall 은 오라클이 설치한 것만 지운다. OS 수준에서 준비한 것
-- (계정, 그룹, 커널 파라미터, SSH 등가성, udev)은 남는다.
-- 그래서 다시 넣을 때 OS 준비를 처음부터 하지 않아도 된다.
--
-- 반대로 /etc/oracle 에 olr.loc 이나 ocr.loc 이 남아 있었다면
-- rac2 는 "이미 클러스터 구성이 있는 노드" 로 보여 재추가가 꼬인다.
-- 이 랩에서는 rootcrs.sh -deconfig 가 깨끗이 지웠다.

--------------------------------------------------------------------------------
 [5-1] 흔한 오류 — 삭제한 노드에서 Clusterware 명령을 치면
--------------------------------------------------------------------------------

[root@rac2 ~]# ls /u01/app/19.3.0/grid/bin/crsctl
ls: cannot access /u01/app/19.3.0/grid/bin/crsctl: No such file or directory

-- 원인 파악 : 그리드 홈을 지웠으므로 crsctl 을 치려 해도 실행 파일이 없다.
--             rootcrs.sh -deconfig 직후, deinstall 전에 쳤다면
--             CRS-4047: No Oracle Clusterware components configured. 가 나온다.
--             (실습 02 [5] 에서 실제로 본 메시지다)
--             두 메시지는 삭제가 어디까지 진행됐는지 알려 준다.
--               CRS-4047          구성만 해제, 소프트웨어는 남음
--               No such file      소프트웨어까지 삭제
-- 조치     : 상태 확인은 남은 노드(rac1)에서 한다.
-- 재확인   : [grid@rac1 ~]$ olsnodes -s -t 에서 rac2 가 보이지 않는다.
