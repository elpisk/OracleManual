================================================================================
 제19장 실습 05 — RAC 로 다시 변환하고 인스턴스 이름 원복  [주의]
 대상   : rac1, rac2
 목표   : RAC One Node 를 RAC 로 되돌리고, 접두어 방식으로 바뀐 인스턴스 이름을
          racdb1, racdb2 로 원복해 18장 끝의 기준선으로 돌아온다
 소요   : 약 25분 (DB 정지 47.7초, 기동 30.6초)
 선행   : 실습 04 완료 — racdb_1 이 rac1 에서 Open
 계정   : oracle, grid
 주의   : 이름을 바꾸려면 데이터베이스를 한 번 완전히 내린다
================================================================================

--------------------------------------------------------------------------------
 [1] RAC 로 변환
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl convert database -db racdb -dbtype RAC -node rac1

-- 3.3초. 출력 없이 끝났다. 데이터베이스는 멈추지 않았다.

[oracle@rac1 ~]$ srvctl config database -db racdb | grep -E "Type|instances|nodes|Candidate|prefix|timeout"
Type: RAC
Database instances: racdb_1
Configured nodes: rac1

[oracle@rac1 ~]$ srvctl status database -db racdb -v
Instance racdb_1 is running on node rac1 with online services ronsvc. Instance status: Open.

-- Type 이 RAC 로 돌아왔다. 접두어, 후보 서버, 재배치 시간 제한 줄은 사라졌다.
-- 인스턴스는 racdb_1 이름 그대로다. 사용법에 적힌 대로
-- "RAC database having one instance" 가 됐다.

--------------------------------------------------------------------------------
 [2] 바로 racdb2 를 붙이면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl add instance -db racdb -instance racdb2 -node rac2
[oracle@rac1 ~]$ srvctl config database -db racdb | grep -E "instances|nodes"
Database instances: racdb2,racdb_1
Configured nodes: rac2,rac1

-- 받아들여졌다. 그러나 이름이 섞였다. racdb2 와 racdb_1.
--
-- 이대로 써도 데이터베이스는 돈다. racdb_1 이 무엇을 쓰고 있는지 본다.

[oracle@rac1 ~]$ export ORACLE_SID=racdb_1; sqlplus / as sysdba
SQL> select inst_id, instance_name, host_name, thread#, status from gv$instance order by 1;

   INST_ID INSTANCE_NAM HOST_NAME             THREAD# STATUS
---------- ------------ ------------------ ---------- ------------
         1 racdb_1      rac1.localdomain            1 OPEN

SQL> select inst_id, value from gv$parameter where name='undo_tablespace' order by 1;

   INST_ID VALUE
---------- ------------
         1 UNDOTBS1

-- 스레드 1, UNDOTBS1 이다. 지금은 우연히 맞다.
-- 그러나 SPFILE 의 racdb1.thread=1, racdb1.undo_tablespace=UNDOTBS1 은
-- racdb1 이라는 이름에 묶인 항목이라 racdb_1 에는 적용되지 않는다.
-- racdb_1 은 비어 있는 스레드와 UNDO 를 그때그때 잡는다.
-- 기동 순서가 바뀌면 rac1 이 스레드 2 를 잡을 수도 있다.
-- 12·13장에서 인스턴스별로 고정한 설계가 이름 하나로 무력해진다.
--
-- oracle 계정의 ORACLE_SID=racdb1 프로필과도 맞지 않는다(실습 03 [3-1]).
-- racdb_1 을 racdb1 로 되돌린다.

--------------------------------------------------------------------------------
 [3] 실행 중인 인스턴스의 등록을 해제하면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl config service -db racdb -service ronsvc | grep -iE "preferred|available"
Preferred instances: racdb_1
Available instances:

-- 서비스의 선호 인스턴스도 racdb_1 로 바뀌어 있다.
-- 실습 02 에서 racdb1 로 만들었는데 재배치를 거치며 따라 바뀌었다.

[oracle@rac1 ~]$ srvctl remove instance -db racdb -instance racdb_1 -noprompt
PRKO-3147 : Instance racdb_1 cannot be removed because it is the only preferred instance for service(s) ronsvc for database racdb

-- 1.0초 만에 거부됐다.
--
-- 원인 파악 : ronsvc 의 유일한 선호 인스턴스가 racdb_1 이다.
--             인스턴스를 지우면 서비스가 갈 곳이 없어진다.
--             ronsvc 는 실습 02 에서 변환 조건(PRCD-1242)을 맞추려고 만든 서비스다.
--             RAC 로 돌아왔으므로 더는 필요 없다.
-- 조치     : 서비스를 먼저 지운다. 그다음 데이터베이스를 내리고 이름을 바꾼다.

--------------------------------------------------------------------------------
 [4] 서비스 제거와 이름 원복
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ srvctl stop service -db racdb -service ronsvc
[oracle@rac1 ~]$ srvctl remove service -db racdb -service ronsvc

[oracle@rac1 ~]$ srvctl stop database -db racdb

-- 47.7초. 인스턴스 이름은 실행 중에 바꿀 수 없으므로 한 번 완전히 내린다.

[oracle@rac1 ~]$ srvctl remove instance -db racdb -instance racdb_1 -noprompt
[oracle@rac1 ~]$ srvctl add instance -db racdb -instance racdb1 -node rac1
[oracle@rac1 ~]$ srvctl config database -db racdb | grep -E "Type|instances|nodes|Services"
Services:
Type: RAC
Database instances: racdb1,racdb2
Configured nodes: rac1,rac2

-- 해제 5.2초, 등록 1.4초. 구성이 18장 끝과 같아졌다.

[oracle@rac1 ~]$ srvctl start database -db racdb
[oracle@rac1 ~]$ srvctl status database -db racdb -v
Instance racdb1 is running on node rac1. Instance status: Open.
Instance racdb2 is running on node rac2. Instance status: Open.

-- 기동 30.6초. srvctl 한 번으로 두 인스턴스가 함께 올라왔다.

SQL> select inst_id, instance_name, host_name, thread#, status from gv$instance order by 1;

   INST_ID INSTANCE_NAM HOST_NAME             THREAD# STATUS
---------- ------------ ------------------ ---------- ------------
         1 racdb1       rac1.localdomain            1 OPEN
         2 racdb2       rac2.localdomain            2 OPEN

SQL> select inst_id, value from gv$parameter where name='undo_tablespace' order by 1;

   INST_ID VALUE
---------- ------------
         1 UNDOTBS1
         2 UNDOTBS2

-- 재확인 : 이름이 racdb1, racdb2 로 돌아왔고 SPFILE 의 인스턴스별 항목이
--          다시 적용된다. 스레드와 UNDO 가 이름에 고정됐다.

--------------------------------------------------------------------------------
 [5] 기준선과 잔재
--------------------------------------------------------------------------------

SQL> select con_name, instance_name, state from dba_pdb_saved_states order by 2;

CON_NAME   INSTANCE_NAM STATE
---------- ------------ --------------
RACPDB1    racdb1       OPEN
RACPDB1    racdb2       OPEN

SQL> select sid, name, value from v$spparameter where sid like 'racdb%' and value is not null order by sid, name;

SID      NAME                 VALUE
-------- -------------------- --------------
racdb1   instance_number      1
racdb1   thread               1
racdb1   undo_tablespace      UNDOTBS1
racdb2   instance_number      2
racdb2   thread               2
racdb2   undo_tablespace      UNDOTBS2

SQL> select count(*) services from gv$active_services;     →  10

[grid@rac1 ~]$ crsctl stat res | grep -c ^NAME=; crsctl stat res -init | grep -c ^NAME=
17
13

-- PDB 저장 상태, SPFILE, 서비스 10행, CRS 자원 17/13 모두 18장 끝과 같다.
-- racdb_1, racdb_2 용 SPFILE 항목이나 저장 상태는 생기지 않았다.

[oracle@rac1 ~]$ ls /u01/app/oracle/diag/rdbms/racdb/
i_1.mif  racdb1  racdb_1
[oracle@rac2 ~]$ ls /u01/app/oracle/diag/rdbms/racdb/
i_1.mif  racdb2  racdb_2

-- 남은 것은 진단 디렉터리 racdb_1, racdb_2 다. 그 이름으로 돌았던 동안의
-- 알럿 로그가 들어 있다. 재배치 기록을 볼 때 이쪽을 연다. 지우지 않아도 된다.
--
-- 이 장에서 바꾼 OCR 의 데이터베이스 자원은 모두 원래대로 돌아왔다.
-- 실습 01 의 OCR 수동 백업은 쓰지 않았다.
