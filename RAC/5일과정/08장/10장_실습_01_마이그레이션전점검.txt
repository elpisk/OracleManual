 [2] 옮길 대상을 찾는 한 줄
--------------------------------------------------------------------------------

SQL> select name from v$datafile where name not like '+%';

no rows selected

-- 이 조회가 이 장 전체에서 가장 자주 쓰이는 한 줄이다.
-- ASM 파일 이름은 항상 + 로 시작하므로, 0 행이면 모든 데이터파일이
-- 공유 스토리지에 있다는 뜻이다.
--
-- 여기서 "0 행" 을 조회 실패로 오해하기 쉽다. 실제로 마이그레이션 대상을
-- 찾는 작업에서 결과가 비어 있으면 조건이 잘못된 줄 알고 다시 짜기 쉽다.
--
-- 원인 파악 : 0 행은 실패가 아니라 답이다. "옮길 것이 없다" 를 뜻한다.
--             이 랩은 dbca 가 모든 파일을 +DATA 에 만들었으므로 정상이다.
-- 조치     : 조건을 고칠 것이 아니라, 반대 조건으로 대조해 조회 자체가
--             동작하는지 확인한다.
-- 재확인   : like '+%' 로 바꿔 행이 나오는지 본다.

SQL> select count(*) from v$datafile where name like '+%';

  COUNT(*)
----------
        13

1 row selected.

-- 13 행이 나온다. 조회는 정상이고 모든 데이터파일이 ASM 에 있다는 것이
-- 확정되었다.
--
-- 이 습관은 실무에서 중요하다. 점검 스크립트가 0 행을 반환할 때
-- "이상 없음" 인지 "조회가 잘못됨" 인지 구분해야 하기 때문이다.
-- 반대 조건으로 한 번 대조하면 그 구분이 확실해진다.
--
-- 단일 인스턴스를 RAC 으로 전환하는 작업의 완료 판정 기준으로 쓴다.
-- 실습 02 에서 파일을 만들면 이 조회에 행이 나타난다.

SQL> select round(sum(bytes)/1024/1024) datafile_mb from v$datafile;

DATAFILE_MB
-----------
       3455

SQL> select round(sum(bytes)/1024/1024) tempfile_mb from v$tempfile;

TEMPFILE_MB
-----------
        108

SQL> select round(sum(bytes)/1024/1024) redo_mb from v$log;

   REDO_MB
----------
        800

-- 합쳐 약 4.3GB 다. 그런데 필요한 여유는 이 값이 아니다.
--
-- 이유 두 가지.
--  1) BACKUP AS COPY 는 원본을 그대로 둔 채 사본을 만든다.
--     옮기는 동안 두 벌이 동시에 존재하므로 원본 크기만큼의 여유가 필요하다.
--  2) 중복도를 곱해야 한다. Normal 이라면 3455MB 에 원시 공간 6910MB 가 든다.
--
-- 판단 기준은 FREE_MB 가 아니라 USABLE_FILE_MB 다. 9장에서 확인했듯
-- FREE_MB 가 넉넉해 보여도 USABLE_FILE_MB 는 이미 음수일 수 있다.
-- 실무 공식은 옮길 총량의 1.2 배 이상이다.

SQL> select log_mode, name, open_mode from v$database;

LOG_MODE     NAME      OPEN_MODE
------------ --------- --------------------
NOARCHIVELOG RACDB     READ WRITE

-- 이 한 줄이 이 장의 절차 전체를 결정한다.
--
-- ARCHIVELOG 라면
--   복사 중 변경이 아카이브 로그에 남으므로 온라인 복사가 가능하다.
--   복사를 온라인으로 끝내고 SWITCH 순간에만 짧게 OFFLINE 하면 된다.
--   서비스 중단이 수 초로 줄어든다.
--
-- NOARCHIVELOG 라면
--   복사본이 어느 시점의 것인지 맞출 방법이 없어 온라인 복사가 금지된다.
--   대상을 OFFLINE 으로 내려야 하고 복사 시간 전체가 서비스 중단이다.
--
-- 실습 02 의 첫 오류가 바로 이것 때문에 나온다.
