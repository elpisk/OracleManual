================================================================================
 제10장 실습 04 — 디스크그룹 간 이동
 대상   : rac1 (검증은 rac2 포함)
 목표   : +DATA 와 +FRA 사이를 왕복하며 절차가 동일하다는 것과, OFFLINE 을
          유지하면 복구가 필요 없다는 것을 확인한다
 소요   : 약 25분
 선행   : racdb 양 노드 OPEN, +DATA·+FRA 여유 충분
 계정   : oracle
 주의   : 여기서 만든 ts_dg 는 이 실습 마지막에 삭제한다
================================================================================

--------------------------------------------------------------------------------
 [1] 대상 만들기 — 이번에는 처음부터 ASM 에
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ sqlplus -s / as sysdba

SQL> create tablespace ts_dg datafile '+DATA' size 200M autoextend off;

Tablespace created.                                            (4.5초 소요)

SQL> col name for a52
SQL> select file#, name, status from v$datafile where name like '%ts_dg%';

     FILE# NAME                                       STATUS
---------- ------------------------------------------ -------
        24 +DATA/RACDB/DATAFILE/ts_dg.279.1243022085  ONLINE

1 row selected.

-- 실습 02 와 달리 경로를 '+DATA' 로만 줬다.
-- 파일 이름은 ASM 이 OMF 규칙으로 지었다. 279 가 파일 번호다.
-- 파일 번호 24 를 받았다. 실습 02·03 에서 쓴 23 번 다음이다.

SQL> create table sys.dg_t (a number, b varchar2(100)) tablespace ts_dg;

Table created.

SQL> insert into sys.dg_t select level, lpad('d',100,'d')
  2    from dual connect by level<=30000;

30000 rows created.

SQL> commit;

Commit complete.

SQL> select count(*) from sys.dg_t;

  COUNT(*)
----------
     30000

1 row selected.

-- 3 만 건을 넣었다. 실습 02 보다 데이터가 많아 복사 시간을 비교하기 좋다.

--------------------------------------------------------------------------------
 [2] +DATA → +FRA — 이번에는 순서를 지킨다
--------------------------------------------------------------------------------

-- 실습 03 의 교훈을 적용한다.
-- 복사부터 SWITCH 까지 OFFLINE 을 계속 유지하면 SCN 이 벌어지지 않는다.
-- 세 명령을 한 번에 실행해 중간에 ONLINE 으로 올릴 틈을 두지 않는다.

SQL> alter tablespace ts_dg offline;

Tablespace altered.

[oracle@rac1 ~]$ rman target /

RMAN> backup as copy datafile 24 format '+FRA';
RMAN> switch datafile 24 to copy;
RMAN> recover datafile 24;

-- 세 명령 전체가 13.7 초다.
--
-- 복사 자체는 7 초로, 실습 02 의 100MB 와 비슷하다. 200MB 인데도 비슷한 것은
-- 원본이 이미 ASM 에 있어 읽기가 빠르기 때문이다.
--
-- 주목할 것은 마지막 줄이다.
--   media recovery complete, elapsed time: 00:00:00
--
-- 복구가 0 초다. 실습 03 에서는 2 초가 걸렸고 그 전에 ORA-01113 도 났다.
-- 차이는 하나뿐이다. 이번에는 중간에 ONLINE 으로 올리지 않았다.
-- SCN 이 벌어지지 않았으므로 적용할 리두가 없었던 것이다.

SQL> alter tablespace ts_dg online;

Tablespace altered.

SQL> col name for a52
SQL> select file#, name, status from v$datafile where file#=24;

     FILE# NAME                                       STATUS
---------- ------------------------------------------ -------
        24 +FRA/RACDB/DATAFILE/ts_dg.264.1243022097   ONLINE

1 row selected.

SQL> select count(*) from sys.dg_t;

  COUNT(*)
----------
     30000

1 row selected.

-- ORA-01113 없이 한 번에 ONLINE 이 되었다.
-- 실습 03 과 대비하면 순서를 지키는 것의 효과가 분명하다.

--------------------------------------------------------------------------------
 [3] rac2 확인
--------------------------------------------------------------------------------

[oracle@rac2 ~]$ sqlplus -s / as sysdba

SQL> select count(*) from sys.dg_t;

  COUNT(*)
----------
     30000

1 row selected.

-- 정상이다. +DATA 에서 +FRA 로 옮겼지만 rac2 는 아무 영향을 받지 않는다.
-- 둘 다 공유 스토리지 위의 디스크그룹이기 때문이다.
--
-- 실습 02 에서 로컬 파일 시스템일 때 ORA-01157 이 났던 것과 대비된다.
-- 차이를 만드는 것은 디스크그룹 이름이 아니라 공유 여부다.

-- ASM 파일 번호는 재사용되지 않는다.

[oracle@rac2 ~]$ sqlplus -s / as sysdba

     30000

1 row selected.

-- rac2 도 계속 정상이다. 왕복하는 동안 한 번도 문제가 없었다.

 [6] 정리
--------------------------------------------------------------------------------

RMAN> delete noprompt backup device type disk;

deleted backup piece

-- DROP TABLESPACE 로 지워진 것은 현재 사용 중이던 280 하나뿐이다.
-- 나머지 두 사본은 RMAN 저장소에만 남아 있어 DELETE COPY 로 지웠다.
-- 자동 백업 조각도 둘 생겼다. SWITCH 를 두 번 했기 때문이다.

SQL> col name for a52
SQL> select file#, name from v$datafile where name like '%ts_dg%';

no rows selected

RMAN> list copy;

using target database control file instead of recovery catalog
specification does not match any datafile copy in the repository
specification does not match any control file copy in the repository
specification does not match any archived log in the repository

-- 두 조회 모두 비어 있다. 여기서 판단이 갈린다.
--
-- 원인 파악 : "specification does not match any datafile copy" 는 오류가
--             아니라 정상 결과다. RMAN 저장소에 남은 사본이 하나도 없다는
--             뜻이며, 정리가 끝났다는 확인이다.
--             DELETE COPY 가 실패했다면 여기에 사본이 그대로 나왔을 것이다.
-- 조치     : 조치할 것이 없다. 다만 이 메시지만으로는 "정말 지워졌는지"
--             와 "애초에 조회가 잘못됐는지" 를 구분할 수 없으므로
--             다른 각도에서 한 번 더 확인한다.
-- 재확인   : 데이터파일 개수와 디스크그룹 여유를 본다.
