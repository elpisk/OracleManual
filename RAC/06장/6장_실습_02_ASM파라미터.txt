 제6장 실습 02 — ASM 파라미터
 대상   : rac1
 목표   : ASM 주요 파라미터를 확인하고 변경한 뒤 되돌린다. 파라미터가
          인스턴스 종류와 값 범위로 이중 보호되는 것을 확인한다
 소요   : 약 25분
 선행   : 실습 01 완료
 계정   : grid, oracle

--------------------------------------------------------------------------------
 [1] 주요 파라미터 한눈에
--------------------------------------------------------------------------------

[grid@rac1 ~]$ sqlplus / as sysasm

SQL> select name, value, isdefault, isinstance_modifiable, ismodified
  2    from v$parameter
  3   where name in ('instance_type','asm_diskstring','asm_power_limit',
  4                  'asm_diskgroups','cluster_database','memory_target',
  5                  'memory_max_target','large_pool_size','processes')
  6   order by name;

NAME                   VALUE              ISDEFAULT ISINSTANCE_MODIFIABLE ISMODIFIED
---------------------- ------------------ --------- --------------------- ----------
asm_diskgroups         DATA, FRA          FALSE     TRUE                  FALSE
asm_diskstring         /dev/asmdisk/*     FALSE     TRUE                  FALSE
asm_power_limit        1                  FALSE     TRUE                  FALSE
cluster_database       TRUE               TRUE      FALSE                 FALSE
instance_type          ASM                TRUE      FALSE                 FALSE
large_pool_size        12582912           FALSE     TRUE                  FALSE
memory_max_target      1128267776         TRUE      FALSE                 FALSE
memory_target          1128267776         TRUE      TRUE                  FALSE
processes              1080               TRUE      FALSE                 FALSE

-- ISDEFAULT 가 FALSE 인 네 개가 설치 시 명시적으로 설정된 값이다.
-- ISINSTANCE_MODIFIABLE 은 alter system 으로 바꿀 수 있는지를 뜻한다.
-- instance_type 과 cluster_database 는 FALSE 다. 바꿀 수 없다.
--
-- memory_target 1128267776 은 약 1.05GB 다. ASM 인스턴스는 이 정도면 충분하다.
-- 같은 노드의 데이터베이스와 비교하면 훨씬 작다.

--------------------------------------------------------------------------------
 [2] ASM_DISKGROUPS 에 +CRS 가 없다
--------------------------------------------------------------------------------

SQL> show parameter asm_diskgroups

NAME              TYPE     VALUE
----------------- -------- -----------
asm_diskgroups    string   DATA, FRA

-- 실제로 마운트된 디스크그룹은 셋인데 목록에는 둘뿐이다.
SQL> select name, state from v$asm_diskgroup order by name;

NAME       STATE
---------- -----------
CRS        MOUNTED
DATA       MOUNTED
FRA        MOUNTED

-- 설정 누락이 아니다. 부트스트랩 순서 때문이다.
--
--   ASM_DISKGROUPS 는 SPFILE 에 있는 파라미터다
--   그런데 ASM 의 SPFILE 자체가 +CRS 안에 있다
--   따라서 이 파라미터를 읽으려면 이미 +CRS 가 마운트되어 있어야 한다
--
-- 순환이다. 이것을 Clusterware 가 GPnP 프로파일로 푼다.

SQL> show parameter spfile

NAME       TYPE     VALUE
---------- -------- --------------------------------------------------
spfile     string   +CRS/rac-cluster/ASMPARAMETERFILE/registry.253.1242760241

[grid@rac1 ~]$ asmcmd spget
+CRS/rac-cluster/ASMPARAMETERFILE/registry.253.1242760241

-- 5장의 OLR·OCR 관계와 같은 설계다. OCR 이 ASM 안에 있어 ASM 기동 정보를
-- 담을 수 없으므로 OLR 을 로컬에 두었듯, SPFILE 이 +CRS 안에 있어 +CRS 를
-- 여는 정보를 담을 수 없으므로 GPnP 프로파일을 따로 둔다.

--------------------------------------------------------------------------------
 [3] asmcmd dsget 은 두 줄을 보여 준다
--------------------------------------------------------------------------------

[grid@rac1 ~]$ asmcmd dsget
parameter:/dev/asmdisk/*
profile:/dev/asmdisk/*

-- parameter 는 현재 인스턴스의 ASM_DISKSTRING 값이다.
-- profile 은 GPnP 프로파일에 저장된 값이다.
--
-- 두 값이 같아야 한다. 다르면 인스턴스를 재기동했을 때 프로파일 값으로
-- 디스크를 찾으므로 디스크그룹을 마운트하지 못할 수 있다.
-- 점검 스크립트에 이 명령을 넣어 두 줄이 같은지 확인하는 것이 좋다.
--
-- 값을 바꿀 때는 asmcmd dsset 을 쓴다. alter system 만 쓰면 프로파일이
-- 갱신되지 않는다.

--------------------------------------------------------------------------------
 [4] asm_power_limit 을 변경하고 되돌린다
--------------------------------------------------------------------------------

SQL> show parameter asm_power_limit

NAME               TYPE      VALUE
------------------ --------- ------
asm_power_limit    integer   1

SQL> alter system set asm_power_limit=4 scope=memory;

System altered.

SQL> show parameter asm_power_limit

NAME               TYPE      VALUE
------------------ --------- ------
asm_power_limit    integer   4

SQL> alter system set asm_power_limit=1 scope=memory;

System altered.

SQL> show parameter asm_power_limit

NAME               TYPE      VALUE
------------------ --------- ------
asm_power_limit    integer   1

-- scope=memory 를 썼으므로 SPFILE 은 바뀌지 않는다. 재기동하면 원래 값으로
-- 돌아온다. 실습에서는 이 방식이 안전하다.
-- 영구 변경이 필요하면 scope=both 또는 scope=spfile 을 쓴다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — 값 범위를 벗어나면
--------------------------------------------------------------------------------

SQL> alter system set asm_power_limit=2000 scope=memory;
alter system set asm_power_limit=2000 scope=memory
*
ERROR at line 1:
ORA-00068: invalid value 2000 for parameter asm_power_limit, must be between 0 and 1024

-- 오류 : ORA-00068
-- 원인 파악 : 값 범위 검증에 걸렸다. asm_power_limit 의 유효 범위는 0에서
--             1024 다. 11g 까지는 상한이 11 이었고 12c 부터 1024 로 늘었다.
--             오래된 자료를 보고 11 이 상한이라고 알고 있으면 이 메시지가
--             혼란스러울 수 있다. 메시지가 정확한 범위를 알려 준다.
--             0은 리밸런스를 하지 않는다는 뜻이다. 계획 정지 시간에만
--             리밸런스하려는 경우 일시적으로 쓴다.
-- 조치     : 범위 안의 값을 쓴다. 실제로 의미 있는 값은 디스크 수와 I/O
--             대역폭이 정한다. 8장에서 1과 8의 차이를 실측한다.
-- 재확인   : 값이 바뀌지 않았는지 확인한다.

SQL> show parameter asm_power_limit

NAME               TYPE      VALUE
------------------ --------- ------
asm_power_limit    integer   1

--------------------------------------------------------------------------------
 [6] 오류 재현 — 인스턴스 종류를 벗어나면
--------------------------------------------------------------------------------

[oracle@rac1 ~]$ sqlplus / as sysdba

SQL> alter system set asm_power_limit=4 scope=memory;
alter system set asm_power_limit=4 scope=memory
*
ERROR at line 1:
ORA-15021: parameter "asm_power_limit" is not valid in RDBMS instance

-- 오류 : ORA-15021
-- 원인 파악 : ASM 파라미터는 ASM 인스턴스에서만 유효하다. 데이터베이스
--             인스턴스에는 그런 파라미터 자체가 없다.
--             실무에서 흔한 원인은 ORACLE_SID 를 잘못 설정한 채 접속한
--             것이다. grid 계정에서 작업하려다 oracle 계정 환경으로
--             들어간 경우다.
-- 조치     : 실행 전에 echo $ORACLE_SID 로 확인하는 습관을 들인다.
--             ASM 작업은 grid 계정, ORACLE_SID=+ASM1, sqlplus / as sysasm 이다.
-- 재확인   : 올바른 계정에서 실행하면 동작한다.

[grid@rac1 ~]$ echo $ORACLE_SID
+ASM1
[grid@rac1 ~]$ sqlplus -s / as sysasm <<< "alter system set asm_power_limit=1 scope=memory;"

System altered.

--------------------------------------------------------------------------------
 [7] 파라미터 점검 스크립트
--------------------------------------------------------------------------------

[grid@rac1 ~]$ cat > /tmp/asm_param_check.sh <<'SH'
#!/bin/bash
export ORACLE_HOME=/u01/app/19.3.0/grid
export ORACLE_SID=+ASM1
export PATH=$ORACLE_HOME/bin:$PATH
echo "=== dsget (두 줄이 같아야 한다) ==="
asmcmd dsget
echo "=== spget (공유 디스크그룹이어야 한다) ==="
asmcmd spget
echo "=== 파라미터 ==="
sqlplus -s / as sysasm <<'SQL'
set lines 120 pages 50
col name for a22
col value for a44
select name, value from v$parameter
 where name in ('instance_type','asm_diskstring','asm_power_limit','asm_diskgroups')
 order by name;
exit
SQL
SH
[grid@rac1 ~]$ chmod +x /tmp/asm_param_check.sh

--------------------------------------------------------------------------------
 [8] 정리
--------------------------------------------------------------------------------

-- 재현한 오류
--   ORA-00068   값 범위 초과 (0~1024)
--   ORA-15021   RDBMS 인스턴스에서 ASM 파라미터 설정
--
-- 확인한 것
--   1) ISINSTANCE_MODIFIABLE 로 변경 가능 여부를 알 수 있다
--   2) instance_type 과 cluster_database 는 변경 불가
--   3) ASM_DISKGROUPS 에 +CRS 가 없는 것은 부트스트랩 순서 때문
--   4) ASM 의 SPFILE 은 +CRS 안에 있다
--   5) asmcmd dsget 의 parameter 와 profile 두 줄이 같아야 한다
--   6) scope=memory 로 바꾸면 재기동 시 원복된다
--   7) asm_power_limit 상한은 1024 (11g 는 11)
--
-- 실무 지침
--   ASM 작업 전에 echo $ORACLE_SID 로 확인한다. ORA-15021 의 대부분은
--   계정과 SID 를 잘못 잡은 결과다.
--   ASM_DISKSTRING 을 바꿀 때는 alter system 이 아니라 asmcmd dsset 을 쓴다.
