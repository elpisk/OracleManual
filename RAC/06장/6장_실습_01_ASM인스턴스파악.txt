================================================================================
 제6장 실습 01 — ASM 인스턴스 파악
 대상   : rac1, rac2
 목표   : ASM 인스턴스와 데이터베이스 인스턴스를 나란히 조회해 차이를 확인하고,
          ASM 고유 백그라운드 프로세스의 역할을 안다
 소요   : 약 25분
 선행   : 클러스터와 racdb 가 정상 기동되어 있을 것
 계정   : grid, oracle
================================================================================

--------------------------------------------------------------------------------
 [1] 접속부터 다르다
--------------------------------------------------------------------------------

[grid@rac1 ~]$ echo $ORACLE_SID
+ASM1
[grid@rac1 ~]$ echo $ORACLE_HOME
/u01/app/19.3.0/grid

[grid@rac1 ~]$ sqlplus / as sysasm

SQL*Plus: Release 19.0.0.0.0 - Production
Connected to:
Oracle Database 19c Enterprise Edition Release 19.0.0.0.0 - Production

-- 접속 권한이 SYSDBA 가 아니라 SYSASM 이다. 11g R2 부터 ASM 관리 권한이
-- 분리되었다. OS 그룹은 asmadmin 이고, 3장에서 grid 계정을 만들 때 이
-- 그룹에 넣었다.

--------------------------------------------------------------------------------
 [2] v$instance 를 나란히 놓는다
--------------------------------------------------------------------------------

SQL> select instance_name, host_name, version, status, database_status,
  2         instance_role, active_state from v$instance;

INSTANCE_N HOST_NAME          VERSION     STATUS    DATABASE_STATUS INSTANCE_ROLE  ACTIVE_ST
---------- ------------------ ----------- --------- --------------- -------------- ---------
+ASM1      rac1.localdomain   19.0.0.0.0  STARTED   ACTIVE          UNKNOWN        NORMAL

-- 이제 데이터베이스 인스턴스에서 같은 조회를 한다.

[oracle@rac1 ~]$ sqlplus / as sysdba

SQL> select instance_name, host_name, version, status, database_status,
  2         instance_role, active_state from v$instance;

INSTANCE_NAME    HOST_NAME          VERSION     STATUS  DATABASE_STATUS INSTANCE_ROLE      ACTIVE_ST
---------------- ------------------ ----------- ------- --------------- ------------------ ---------
racdb1           rac1.localdomain   19.0.0.0.0  OPEN    ACTIVE          PRIMARY_INSTANCE   NORMAL

-- 두 가지가 다르다.
--   STATUS         STARTED  vs  OPEN
--   INSTANCE_ROLE  UNKNOWN  vs  PRIMARY_INSTANCE
--
-- ASM 인스턴스는 영원히 STARTED 에 머문다. 마운트할 데이터베이스가 없기
-- 때문이다. STARTED 는 실패 상태가 아니라 ASM 의 정상 상태다.

--------------------------------------------------------------------------------
 [3] 오류 재현 — ASM 에서 데이터파일을 조회한다
--------------------------------------------------------------------------------

SQL> -- +ASM1 에서
SQL> select count(*) from v$datafile;
select count(*) from v$datafile
                     *
ERROR at line 1:
ORA-01507: database not mounted

-- 오류 : ORA-01507
-- 원인 파악 : database not mounted. ASM 인스턴스는 데이터베이스를 마운트하지
--             않으므로 데이터파일 관련 뷰를 읽을 수 없다. 고장이 아니라
--             구조상 당연한 결과다.
--             ASM 이 관리하는 것은 디스크그룹의 메타데이터이지 데이터베이스의
--             파일 목록이 아니다.
-- 조치     : ASM 에서는 v$asm_ 으로 시작하는 뷰를 쓴다.
--             v$asm_diskgroup, v$asm_disk, v$asm_file, v$asm_alias,
--             v$asm_template, v$asm_operation, v$asm_attribute 다.
-- 재확인   : 데이터베이스 인스턴스에서는 정상 조회된다.

SQL> -- racdb1 에서
SQL> select count(*) datafiles from v$datafile;

 DATAFILES
----------
        13

--------------------------------------------------------------------------------
 [4] INSTANCE_TYPE 이 구분 기준이다
--------------------------------------------------------------------------------

SQL> -- +ASM1 에서
SQL> select name, value from v$parameter where name='instance_type';

NAME             VALUE
---------------- ------
instance_type    ASM

SQL> -- racdb1 에서
SQL> select name, value from v$parameter where name='instance_type';

NAME             VALUE
---------------- ------
instance_type    RDBMS

-- 이 파라미터가 인스턴스의 성격을 정한다. 변경할 수 없다.

--------------------------------------------------------------------------------
 [5] GV$ 는 ASM 에서도 동작한다
--------------------------------------------------------------------------------

SQL> select inst_id, instance_name, host_name, status from gv$instance order by inst_id;

   INST_ID INSTANCE_N HOST_NAME              STATUS
---------- ---------- ---------------------- ------------
         1 +ASM1      rac1.localdomain       STARTED
         2 +ASM2      rac2.localdomain       STARTED

-- 1장에서 배운 GV$ 의 원리가 그대로 적용된다. ASM 인스턴스도 클러스터를
-- 이루므로 서로의 정보를 주고받을 수 있다.
-- SID 규칙도 데이터베이스와 같다. 노드 번호가 뒤에 붙는다.
--   데이터베이스   racdb1 / racdb2
--   ASM            +ASM1  / +ASM2

--------------------------------------------------------------------------------
 [7] ASM 고유 프로세스 세 가지
--------------------------------------------------------------------------------

-- RBAL — 리밸런스 조정
--   실제 데이터 이동은 ARB0, ARB1 같은 슬레이브가 한다. 8장에서 디스크를
--   추가하면 ARB 프로세스가 뜨는 것을 보게 된다.
[root@rac1 ~]# ps -u grid -o cmd | grep 'asm_arb' | grep -v grep
(평상시에는 없다. 리밸런스 중에만 뜬다)

-- GMON — 디스크그룹 멤버십과 PST 관리
--   5장에서 디스크를 OFFLINE 했을 때 alert.log 에 이렇게 남았다.
[grid@rac1 ~]$ grep -i 'GMON' /u01/app/grid/diag/asm/+asm/+ASM1/trace/alert_+ASM1.log | tail -3
GMON updating disk modes for group 1 at 32 for pid 47, osid 4368
GMON updating disk modes for group 1 at 33 for pid 47, osid 4368
GMON querying group 1 at 44 for pid 26, osid 6314

-- ASMB — 데이터베이스와 ASM 을 잇는 채널
--   흥미롭게도 양쪽 모두에 존재한다.
[root@rac1 ~]# ps -ef | grep -E 'asm_asmb|ora_asmb' | grep -v grep
grid      9090     1  0 19:00 ?  00:00:01 asm_asmb_+ASM1
oracle   15705     1  0 19:26 ?  00:00:02 ora_asmb_racdb1

-- 데이터베이스 쪽 ora_asmb 가 ASM 쪽 asm_asmb 에 붙어 익스텐트 위치 정보를
-- 받아 온다. 실제 데이터 I/O 는 데이터베이스가 직접 디스크에 수행한다.
-- ASM 을 거치지 않는다. 그래서 ASM 인스턴스가 잠시 멈춰도 이미 열린 파일의
-- 읽기·쓰기는 계속된다.

--------------------------------------------------------------------------------
 [8] RAC 공통 프로세스도 있다
--------------------------------------------------------------------------------

[root@rac1 ~]# ps -u grid -o cmd | grep -E 'asm_(lmon|lmd|lms|lck)' | grep -v grep
asm_lmon_+ASM1
asm_lmd0_+ASM1
asm_lms0_+ASM1
asm_lck0_+ASM1
asm_lck1_+ASM1

-- 1장에서 캐시 퓨전을 담당하던 프로세스들이 ASM 에도 있다.
-- 이유는 같다. +ASM1 과 +ASM2 가 같은 디스크그룹의 메타데이터를 공유하므로,
-- 어느 인스턴스가 어떤 메타데이터 블록을 갱신하는지 조정해야 한다.
-- 데이터베이스가 데이터 블록을 조정하듯 ASM 은 메타데이터 블록을 조정한다.

--------------------------------------------------------------------------------
 [9] 정리
--------------------------------------------------------------------------------

-- 확인한 것
--   1) 접속 권한이 SYSASM 이다 (DB 는 SYSDBA)
--   2) STATUS 가 STARTED 다. OPEN 이 아니며 그것이 정상이다
--   3) v$datafile 조회는 ORA-01507 이다. ASM 은 v$asm_ 뷰를 쓴다
--   4) INSTANCE_TYPE 이 ASM / RDBMS 로 갈린다
--   5) GV$ 는 ASM 에서도 동작한다. +ASM1 과 +ASM2 가 보인다
--   6) 프로세스는 37개 대 115개
--   7) RBAL·GMON·ASMB 가 ASM 고유
--   8) LMON·LMD0·LMS0 도 있다. ASM 도 클러스터이기 때문
--
-- 기억할 대조 기준
--   +ASM1 / +ASM2, STATUS STARTED, INSTANCE_TYPE ASM
--   프로세스 37개
