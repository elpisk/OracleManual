 제12장 실습 04 — 이벤트 핸들러
 대상    : oel7v9 orcl
 목표    : 감사 조건이 충족되면 알림 파일에 한 줄을 쓰는 핸들러를 만들고,
          핸들러가 실패하면 사용자 문장이 어떻게 되는지 확인한다
 소요    : 약 20분
 변경    : 디렉터리 객체 FGA_ALERT_DIR, OS 디렉터리 /tmp/sec12_fga (05 에서 삭제)

--------------------------------------------------------------------------------
 [1] 디렉터리 객체와 권한
--------------------------------------------------------------------------------

-- 디렉터리 객체는 OS 경로를 가리키기만 한다. 일부러 OS 디렉터리를 아직 만들지 않았다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE DIRECTORY fga_alert_dir AS '/tmp/sec12_fga';

Directory created.

SYS@orcl> GRANT READ, WRITE ON DIRECTORY fga_alert_dir TO secadm;

Grant succeeded.

SYS@orcl> GRANT EXECUTE ON UTL_FILE TO secadm;

Grant succeeded.

--------------------------------------------------------------------------------
 [2] 핸들러와 정책
--------------------------------------------------------------------------------

SQL> CONNECT secadm/oracle
Connected.
SECADM@orcl> CREATE OR REPLACE PROCEDURE rt_alert (
  2    p_schema IN VARCHAR2,
  3    p_object IN VARCHAR2,
  4    p_policy IN VARCHAR2)
  5  AS
  6    f UTL_FILE.FILE_TYPE;
  7  BEGIN
  8    f := UTL_FILE.FOPEN('FGA_ALERT_DIR', 'fga_alert.log', 'a');
  9    UTL_FILE.PUT_LINE(f, TO_CHAR(SYSDATE, 'YYYY-MM-DD HH24:MI:SS') || ' '
 10                         || SYS_CONTEXT('USERENV', 'SESSION_USER') || ' '
 11                         || p_schema || '.' || p_object || ' ' || p_policy);
 12    UTL_FILE.FCLOSE(f);
 13  END;
 14  /

Procedure created.

SECADM@orcl> BEGIN
  2    DBMS_FGA.DROP_POLICY('HR', 'EMPLOYEES', 'EMP_SAL_DML');
  3    DBMS_FGA.ADD_POLICY(
  4      object_schema   => 'HR',
  5      object_name     => 'EMPLOYEES',
  6      policy_name     => 'EMP_HIGH_SAL',
  7      audit_condition => 'SALARY > 9999',
  8      audit_column    => 'SALARY',
  9      handler_schema  => 'SECADM',
 10      handler_module  => 'RT_ALERT');
 11  END;
 12  /

PL/SQL procedure successfully completed.

-- ★ 핸들러는 (스키마, 객체, 정책) 세 인자를 받는 프로시저다. 소유자 SECADM 권한으로 실행된다.
--    COMMIT 이 없는 UTL_FILE 기록이라 사용자 트랜잭션에 영향을 주지 않는다.
