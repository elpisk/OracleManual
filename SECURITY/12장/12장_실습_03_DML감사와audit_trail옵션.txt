 대상    : oel7v9 orcl

          XML+EXTENDED 기록 파일과 V$XML_AUDIT_TRAIL 을 확인한다
 소요    : 약 25분
 변경    : 정책 교체(EMP_SAL_DML). adump 에 XML 감사 파일 1개 생성(05 에서 삭제). DML 은 모두 ROLLBACK

-- ★ MERGE 를 넣자 ORA-28106: input value for argument #9 is not valid.
--    9번째 인자가 statement_types 다. 지정할 수 있는 값은 SELECT, INSERT, UPDATE, DELETE 뿐이다.
--    MERGE 를 빼고 INSERT, UPDATE 로 다시 등록해 성공했다.

--------------------------------------------------------------------------------
 [2] 바인드 변수로 급여를 올렸다가 ROLLBACK
--------------------------------------------------------------------------------

SQL> CONNECT hr/hr
Connected.
HR@orcl> VARIABLE b NUMBER
HR@orcl> EXEC :b := 105

PL/SQL procedure successfully completed.

HR@orcl> UPDATE employees SET salary = salary * 1.1 WHERE employee_id = :b;

1 row updated.

HR@orcl> ROLLBACK;

Rollback complete.

HR@orcl> SELECT salary FROM employees WHERE employee_id = 105;

    SALARY
----------
      4800

1 row selected.

-- ★ 급여는 ROLLBACK 으로 4800 그대로다.

--------------------------------------------------------------------------------
 [3] EXTENDED 기록
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL db_user FOR a8
SYS@orcl> COL stmt FOR a6
SYS@orcl> COL sql_text FOR a60
SYS@orcl> COL sql_bind FOR a20
SYS@orcl> SELECT db_user, statement_type stmt, sql_text, sql_bind
  2    FROM dba_fga_audit_trail WHERE policy_name = 'EMP_SAL_DML';

DB_USER  STMT   SQL_TEXT                                                     SQL_BIND
-------- ------ ------------------------------------------------------------ --------------------
HR       UPDATE UPDATE employees SET salary = salary * 1.1 WHERE employee_id  #1(3):105
                 = :b


1 row selected.

-- ★ 취소된 UPDATE 도 기록에 남았다. 감사 기록은 사용자 트랜잭션과 독립적이다.
--    SQL_TEXT 에는 바인드 변수 :b 가, SQL_BIND 에는 #1(3):105 형식으로 실제 값이 남았다.
--    (#번호(길이):값) 형식이다. 바인드 값이 주민번호였다면 그대로 기록됐을 것이다.

 [4] XML+EXTENDED 로 바꾸기
--------------------------------------------------------------------------------

-- 정의를 바꾸는 API 가 없으므로 한 블록에서 DROP 후 ADD 한다.

Connected.
SECADM@orcl> BEGIN
  2    DBMS_FGA.DROP_POLICY('HR', 'EMPLOYEES', 'EMP_SAL_DML');
  3    DBMS_FGA.ADD_POLICY(
  4      object_schema   => 'HR',
  5      object_name     => 'EMPLOYEES',
  6      policy_name     => 'EMP_SAL_DML',
  7      audit_column    => 'SALARY',
  8      statement_types => 'INSERT, UPDATE',
  9      audit_trail     => DBMS_FGA.XML + DBMS_FGA.EXTENDED);
 10  END;
 11  /

PL/SQL procedure successfully completed.

SECADM@orcl> CONNECT hr/hr
Connected.
HR@orcl> UPDATE employees SET salary = 9999 WHERE employee_id = 106;

1 row updated.

HR@orcl> ROLLBACK;

Rollback complete.

--------------------------------------------------------------------------------
 [6] V$XML_AUDIT_TRAIL
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL db_user FOR a8
SYS@orcl> COL policy_name FOR a12
SYS@orcl> COL sql_text FOR a50
SYS@orcl> SELECT db_user, policy_name, sql_text FROM v$xml_audit_trail WHERE policy_name = 'EMP_SAL_DML';

DB_USER  POLICY_NAME  SQL_TEXT
-------- ------------ --------------------------------------------------
HR       EMP_SAL_DML  UPDATE employees SET salary = 9999 WHERE employee_
                      id = 106


1 row selected.

SYS@orcl> SELECT COUNT(*) FROM dba_fga_audit_trail WHERE policy_name = 'EMP_SAL_DML';

  COUNT(*)
----------
         1

1 row selected.

-- ★ 같은 기록을 데이터베이스 안에서 V$XML_AUDIT_TRAIL 로 읽을 수 있다.
--    DBA_FGA_AUDIT_TRAIL 의 EMP_SAL_DML 1건은 앞 단계 DB 기록이다. XML 로 바꾼 뒤의 UPDATE 는 FGA_LOG$ 에 들어가지 않았다.

================================================================================
 정리
================================================================================

-- - statement_types 는 SELECT·INSERT·UPDATE·DELETE 만(MERGE → ORA-28106).
-- - ROLLBACK 한 DML 도 감사 기록은 남는다.
-- - EXTENDED 는 SQL_TEXT 와 SQL_BIND(#n(len):value)를 남긴다. 민감 정보가 평문으로 기록된다.
-- - XML 은 AUDIT_FILE_DEST 의 파일 + V$XML_AUDIT_TRAIL. 파일 시각은 UTC.
