================================================================================
 제8장 실습 04 — 리댁션의 한계와 면제
 대상    : oel7v9 orcl
 목표    : 리댁션이 막는 것(값을 다른 객체로 옮기기)과 막지 못하는 것(조건절 추론)을 구분하고,
          EXEMPT REDACTION POLICY 의 효과를 확인한다
 소요    : 약 20분
 변경    : sking 에 CREATE TABLE 권한, 뷰 HR.EMP_SAL_V (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] 조건절은 원래 값으로 평가된다
--------------------------------------------------------------------------------

SQL> CONNECT sking/sking
Connected.
SKING@orcl> SELECT employee_id, salary FROM hr.employees WHERE salary > 15000 ORDER BY employee_id;

EMPLOYEE_ID     SALARY
----------- ----------
        100      99000
        101      99000
        102      99000

3 rows selected.

SKING@orcl> SELECT COUNT(*) FROM hr.employees WHERE salary BETWEEN 23999 AND 24001;

  COUNT(*)
----------
         1

1 row selected.

SKING@orcl> SELECT employee_id FROM hr.employees WHERE employee_id < 110 ORDER BY salary DESC FETCH FIRST 3 ROWS ONLY;

EMPLOYEE_ID
-----------
        100
        101
        102

3 rows selected.

-- ★ sking 에게 급여는 모두 99000 처럼 보이지만 WHERE salary > 15000 은 원래 값으로 걸러 3명만 돌려준다.
--    BETWEEN 23999 AND 24001 로 1건이 나오면 100번 사원의 급여가 24000 이라는 것을 알아낸 셈이다.
--    ORDER BY 도 원래 값으로 정렬된다. 리댁션은 화면 노출을 줄이는 기능이지
--    SQL 을 직접 쓸 수 있는 사용자의 추론까지 막는 기능은 아니다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — CTAS 와 INSERT SELECT 로 빼내기
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> GRANT CREATE TABLE, UNLIMITED TABLESPACE TO sking;

Grant succeeded.

SYS@orcl> CONNECT sking/sking
Connected.
SKING@orcl> CREATE TABLE emp_copy AS SELECT employee_id, salary FROM hr.employees;
CREATE TABLE emp_copy AS SELECT employee_id, salary FROM hr.employees
                                             *
ERROR at line 1:
ORA-28081: Insufficient privileges - the command references a redacted object.


SKING@orcl> CREATE TABLE emp_copy AS SELECT employee_id, first_name FROM hr.employees;

Table created.

SKING@orcl> SELECT COUNT(*) FROM emp_copy;

  COUNT(*)
----------
       107

1 row selected.

SKING@orcl> CREATE TABLE emp_sal (employee_id NUMBER, salary NUMBER);

Table created.

SKING@orcl> INSERT INTO emp_sal SELECT employee_id, salary FROM hr.employees WHERE employee_id = 100;
INSERT INTO emp_sal SELECT employee_id, salary FROM hr.employees WHERE employee_id = 100
                                        *
ERROR at line 1:
ORA-28081: Insufficient privileges - the command references a redacted object.


SKING@orcl> SELECT * FROM emp_sal;

no rows selected

SKING@orcl> ROLLBACK;

Rollback complete.

-- ★ 리댁션된 컬럼을 CREATE TABLE AS SELECT, INSERT ... SELECT 로 옮기려 하면 ORA-28081.
--    리댁션 대상이 아닌 FIRST_NAME 만 옮기는 CTAS 는 성공했다(107건).
--    원래 값을 다른 테이블에 복사해 정책 밖으로 꺼내는 경로를 막는 장치다.

--------------------------------------------------------------------------------
 [3] 오류 재현 — 리댁션 컬럼을 연산하는 뷰
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE OR REPLACE VIEW hr.emp_sal_v AS SELECT employee_id, salary * 12 AS annual_sal FROM hr.employees;

View created.

SYS@orcl> GRANT SELECT ON hr.emp_sal_v TO sking;

Grant succeeded.

SYS@orcl> CONNECT sking/sking
Connected.
SKING@orcl> SELECT * FROM hr.emp_sal_v WHERE employee_id IN (100, 101);
SELECT * FROM hr.emp_sal_v WHERE employee_id IN (100, 101)
                 *
ERROR at line 1:
ORA-28094: SQL construct not supported by data redaction

-- ★ 뷰 정의 안에서 salary * 12 처럼 리댁션 컬럼을 가공하면 조회 시 ORA-28094:
--    SQL construct not supported by data redaction (19.3).
--    뷰 컬럼이 어떤 값으로 리댁션돼야 하는지 정할 수 없는 구문이다.

--------------------------------------------------------------------------------
 [4] 조치 — 컬럼은 그대로 노출하고 연산은 조회에서
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE OR REPLACE VIEW hr.emp_sal_v AS SELECT employee_id, salary FROM hr.employees;

View created.

SYS@orcl> CONNECT sking/sking
Connected.
SKING@orcl> SELECT * FROM hr.emp_sal_v WHERE employee_id IN (100, 101);

EMPLOYEE_ID     SALARY
----------- ----------
        100      99000
        101      99000

2 rows selected.

SKING@orcl> SELECT employee_id, salary * 12 AS annual_sal FROM hr.emp_sal_v WHERE employee_id IN (100, 101);

EMPLOYEE_ID ANNUAL_SAL
----------- ----------
        100    1188000
        101    1188000

2 rows selected.

-- ★ 뷰가 SALARY 를 그대로 내보내면 리댁션된 값(99000)이 나온다.
--    연산을 바깥 조회에서 하면 리댁션된 값 기준으로 계산된다(99000 × 12 = 1188000).

--------------------------------------------------------------------------------
 [5] EXEMPT REDACTION POLICY
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> SELECT employee_id, salary FROM hr.employees WHERE employee_id = 100;

EMPLOYEE_ID     SALARY
----------- ----------
        100      24000

1 row selected.

SYS@orcl> GRANT EXEMPT REDACTION POLICY TO sking;

Grant succeeded.

SYS@orcl> CONNECT sking/sking
Connected.
SKING@orcl> SELECT employee_id, salary FROM hr.employees WHERE employee_id = 100;

EMPLOYEE_ID     SALARY
----------- ----------
        100      24000

1 row selected.

SKING@orcl> CREATE TABLE emp_copy2 AS SELECT employee_id, salary FROM hr.employees WHERE employee_id = 100;

Table created.

SKING@orcl> SELECT * FROM emp_copy2;

EMPLOYEE_ID     SALARY
----------- ----------
        100      24000

1 row selected.

SKING@orcl> CONNECT / AS SYSDBA
Connected.
SYS@orcl> REVOKE EXEMPT REDACTION POLICY FROM sking;

Revoke succeeded.

-- ★ SYS 는 원래 값을 본다. EXEMPT REDACTION POLICY 를 받은 sking 도 원래 값을 보고,
--    CTAS 로 원래 급여를 복사할 수도 있다. 이 권한은 DB 의 모든 리댁션 정책을 건너뛴다.
--    Data Pump 로 리댁션 테이블을 내보내는 계정 정도에만 주고 바로 회수했다.

================================================================================
 정리
================================================================================

-- - WHERE·ORDER BY 는 원래 값으로 평가된다. 임의 SQL 을 쓰는 사용자에게는 추론 공격이 가능하다.
-- - CTAS·INSERT SELECT 로 리댁션 컬럼을 옮기면 ORA-28081.
-- - 뷰 안에서 리댁션 컬럼을 연산하면 ORA-28094(19.3).
-- - SYS 와 EXEMPT REDACTION POLICY 보유자는 모든 정책을 우회한다.
