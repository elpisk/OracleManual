 [2] 이름 붙은 정책 식, 오류 재현 — 연결된 식 삭제
--------------------------------------------------------------------------------

-- 정책 식에 이름을 붙여 따로 만들고 컬럼에 연결한다(12.2 이후).
-- 연결된 컬럼에서는 이 식이 정책의 기본 식보다 우선한다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> BEGIN
  2    DBMS_REDACT.CREATE_POLICY_EXPRESSION(
  3      policy_expression_name        => 'SEC08_NOT_HR_EXPR',
  4      expression                    => 'SYS_CONTEXT(''USERENV'', ''SESSION_USER'') != ''HR''',
  5      policy_expression_description => 'HR 외 사용자에게 리댁션');
  6    DBMS_REDACT.APPLY_POLICY_EXPR_TO_COL(
  7      object_schema          => 'HR',
  8      object_name            => 'EMPLOYEES',
  9      column_name            => 'SALARY',
 10      policy_expression_name => 'SEC08_NOT_HR_EXPR');
 11  END;
 12  /

PL/SQL procedure successfully completed.

SYS@orcl> COL policy_expression_name FOR a18

-- - 이름 붙은 정책 식은 컬럼 단위로 정책 기본 식을 덮어쓴다. 연결을 끊어야 지울 수 있다(ORA-28092).
-- - 없는 정책을 지우면 ORA-28068.
