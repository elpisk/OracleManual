================================================================================
 제2장 실습 01 — 프록시 계정과 대상 사용자
 대상    : oel7v9 orcl (non-CDB, 19.3)
 목표    : 중간 계층용 프록시 계정 prx_app 과 최종 사용자 prx_user1 을 만들고,
          GRANT CONNECT THROUGH 전후의 접속 결과와 프록시 세션의 USERENV 값을 비교한다
 소요    : 약 20분
 변경    : 역할 prx_role, 사용자 prx_app·prx_user1·prx_user2 생성 (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] 사용자와 역할 만들기
--------------------------------------------------------------------------------

-- 프록시 계정 prx_app 에는 CREATE SESSION 만 준다. 프록시 계정 자체로 할 수 있는 일은 접속뿐이어야 한다.
-- prx_user1 은 CONNECT·RESOURCE·prx_role, prx_user2 는 CONNECT·prx_role 을 받는다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE ROLE prx_role;

Role created.

SYS@orcl> GRANT CREATE TABLE TO prx_role;

Grant succeeded.

SYS@orcl> CREATE USER prx_app IDENTIFIED BY prx_app;

User created.

SYS@orcl> GRANT CREATE SESSION TO prx_app;

Grant succeeded.

SYS@orcl> CREATE USER prx_user1 IDENTIFIED BY prx_user1 QUOTA 10M ON users;

User created.

SYS@orcl> CREATE USER prx_user2 IDENTIFIED BY prx_user2;

User created.

SYS@orcl> GRANT CONNECT, RESOURCE, prx_role TO prx_user1;

Grant succeeded.

SYS@orcl> GRANT CONNECT, prx_role TO prx_user2;

Grant succeeded.

-- ★ 아직 어떤 사용자도 prx_app 을 통한 접속을 허용하지 않았다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — 허용 없이 대신 접속하기
--------------------------------------------------------------------------------

-- SQL*Plus 의 프록시 접속 형식은 프록시계정[대상사용자]/프록시계정비밀번호 다.

SQL> CONNECT prx_app[prx_user1]/prx_app
ERROR:
ORA-01017: invalid username/password; logon denied

-- ★ ORA-01017: invalid username/password. 프록시 허용이 없을 때도 비밀번호 오류와 같은 메시지가 나온다.
--    접속 시도자에게 "이 사용자는 존재하지만 프록시가 막혀 있다"는 정보를 주지 않는다.
--    비밀번호가 맞는데 ORA-01017 이 나면 DBA_PROXIES 부터 본다.

--------------------------------------------------------------------------------
 [3] 조치 — GRANT CONNECT THROUGH
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> ALTER USER prx_user1 GRANT CONNECT THROUGH prx_app;

User altered.

SYS@orcl> COL proxy FOR a10
SYS@orcl> COL client FOR a10
SYS@orcl> COL role FOR a10
SYS@orcl> SELECT proxy, client, authentication, authorization_constraint, role FROM dba_proxies WHERE proxy = 'PRX_APP';

PROXY      CLIENT     AUT AUTHORIZATION_CONSTRAINT            ROLE
---------- ---------- --- ----------------------------------- ----------
PRX_APP    PRX_USER1  NO  PROXY MAY ACTIVATE ALL CLIENT ROLES

1 row selected.

-- ★ 허용은 대상 사용자 쪽에 건다(ALTER USER 대상 GRANT CONNECT THROUGH 프록시).
--    역할 제한이 없으므로 AUTHORIZATION_CONSTRAINT 가 PROXY MAY ACTIVATE ALL CLIENT ROLES 다.
--    AUT=NO 는 대상 사용자의 비밀번호를 요구하지 않는다는 뜻이다(03 에서 바꿔 본다).

--------------------------------------------------------------------------------
 [4] 재확인 — 프록시로 접속
--------------------------------------------------------------------------------

SQL> CONNECT prx_app[prx_user1]/prx_app
Connected.
PRX_USER1@orcl> SHOW USER
USER is "PRX_USER1"
PRX_USER1@orcl> COL session_user FOR a14
PRX_USER1@orcl> COL proxy_user FOR a12
PRX_USER1@orcl> COL auth_identity FOR a14
PRX_USER1@orcl> COL auth_method FOR a12
PRX_USER1@orcl> SELECT SYS_CONTEXT('USERENV', 'SESSION_USER')           AS session_user,
  2         SYS_CONTEXT('USERENV', 'PROXY_USER')             AS proxy_user,
  3         SYS_CONTEXT('USERENV', 'AUTHENTICATED_IDENTITY') AS auth_identity,
  4         SYS_CONTEXT('USERENV', 'AUTHENTICATION_METHOD')  AS auth_method
  5    FROM dual;

SESSION_USER   PROXY_USER   AUTH_IDENTITY  AUTH_METHOD
-------------- ------------ -------------- ------------
PRX_USER1      PRX_APP      PRX_USER1      NONE

1 row selected.

PRX_USER1@orcl> SELECT role FROM session_roles ORDER BY role;

ROLE
------------------------------------------------------------------------------------------------------------------------
CONNECT
PRX_ROLE
RESOURCE
SODA_APP

4 rows selected.

-- ★ 세션 사용자는 PRX_USER1 이고 PROXY_USER 에 PRX_APP 이 남는다. 권한은 PRX_USER1 의 것이다.
--    AUTHENTICATION_METHOD 가 NONE 이다. PRX_USER1 은 이 세션에서 자기 비밀번호로 인증받지 않았다.
--    PRX_USER1 의 기본 역할 네 개(SODA_APP 은 RESOURCE 에 딸려 온 역할)가 모두 켜졌다.

--------------------------------------------------------------------------------
 [5] 프록시 계정 자신으로 접속하면
--------------------------------------------------------------------------------

SQL> CONNECT prx_app/prx_app
Connected.
PRX_APP@orcl> SHOW USER
USER is "PRX_APP"
PRX_APP@orcl> COL session_user FOR a14
PRX_APP@orcl> COL proxy_user FOR a12
PRX_APP@orcl> COL auth_identity FOR a14
PRX_APP@orcl> COL auth_method FOR a12
PRX_APP@orcl> SELECT SYS_CONTEXT('USERENV', 'SESSION_USER')           AS session_user,
  2         SYS_CONTEXT('USERENV', 'PROXY_USER')             AS proxy_user,
  3         SYS_CONTEXT('USERENV', 'AUTHENTICATED_IDENTITY') AS auth_identity,
  4         SYS_CONTEXT('USERENV', 'AUTHENTICATION_METHOD')  AS auth_method
  5    FROM dual;

SESSION_USER   PROXY_USER   AUTH_IDENTITY  AUTH_METHOD
-------------- ------------ -------------- ------------
PRX_APP                     PRX_APP        PASSWORD

1 row selected.

-- ★ 일반 접속에서는 PROXY_USER 가 비어 있고 AUTHENTICATION_METHOD 가 PASSWORD 다.
--    감사·VPD 정책 함수가 PROXY_USER 값으로 "애플리케이션 서버를 거친 접속인가"를 구분할 수 있다.

================================================================================
 정리
================================================================================

-- - 프록시 허용은 대상 사용자에 GRANT CONNECT THROUGH 로 건다. 허용이 없으면 ORA-01017.
-- - 프록시 세션의 사용자·권한은 대상 사용자 것이고, PROXY_USER 에 프록시 계정이 남는다.
-- - 프록시 계정에는 CREATE SESSION 외의 권한을 주지 않는다.
