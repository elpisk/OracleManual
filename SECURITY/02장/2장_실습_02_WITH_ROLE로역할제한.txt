================================================================================
 제2장 실습 02 — WITH ROLE 로 역할 제한
 대상    : oel7v9 orcl
 목표    : WITH ROLE, WITH NO ROLES 로 프록시 세션에서 켤 수 있는 역할을 제한하고,
          역할을 제한했을 때 CREATE SESSION 이 어디서 오는지 확인한다
 소요    : 약 20분
 변경    : prx_user2 의 프록시 허용 조건 변경, prx_user2 에 CREATE SESSION 직접 부여
================================================================================

--------------------------------------------------------------------------------
 [1] WITH ROLE prx_role 로 허용
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> ALTER USER prx_user2 GRANT CONNECT THROUGH prx_app WITH ROLE prx_role;

User altered.

SYS@orcl> COL proxy FOR a10
SYS@orcl> COL client FOR a10
SYS@orcl> COL role FOR a10
SYS@orcl> SELECT proxy, client, authorization_constraint, role FROM dba_proxies WHERE proxy = 'PRX_APP' ORDER BY client;

PROXY      CLIENT     AUTHORIZATION_CONSTRAINT            ROLE
---------- ---------- ----------------------------------- ----------
PRX_APP    PRX_USER1  PROXY MAY ACTIVATE ALL CLIENT ROLES
PRX_APP    PRX_USER2  PROXY MAY ACTIVATE ROLE             PRX_ROLE

2 rows selected.

-- ★ prx_user2 행은 PROXY MAY ACTIVATE ROLE / PRX_ROLE 이다. 허용한 역할마다 한 행씩 생긴다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — 접속 시도
--------------------------------------------------------------------------------

SQL> CONNECT prx_app[prx_user2]/prx_app
ERROR:
ORA-01045: user PRX_USER2 lacks CREATE SESSION privilege; logon denied

-- ★ ORA-01045: user PRX_USER2 lacks CREATE SESSION privilege.
--    프록시 허용은 됐지만 로그인 권한 검사에서 막혔다.

--------------------------------------------------------------------------------
 [3] 원인 확인
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL grantee FOR a10
SYS@orcl> COL privilege FOR a14
SYS@orcl> SELECT grantee, privilege FROM dba_sys_privs WHERE grantee IN ('PRX_USER2', 'PRX_ROLE', 'CONNECT') ORDER BY grantee;

GRANTEE    PRIVILEGE
---------- --------------
CONNECT    SET CONTAINER
CONNECT    CREATE SESSION
PRX_ROLE   CREATE TABLE

3 rows selected.

SYS@orcl> SELECT granted_role, default_role FROM dba_role_privs WHERE grantee = 'PRX_USER2' ORDER BY 1;

GRANTED_ROLE
------------------------------------------------------------------------------------------------------------------------
DEF
---
CONNECT
YES

PRX_ROLE
YES


2 rows selected.

-- ★ prx_user2 의 CREATE SESSION 은 CONNECT 역할을 통해서만 들어온다.
--    WITH ROLE prx_role 은 CONNECT 를 켜지 못하게 하므로 로그인 권한도 사라졌다.

--------------------------------------------------------------------------------
 [4] 조치 — CONNECT 까지 허용
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> ALTER USER prx_user2 GRANT CONNECT THROUGH prx_app WITH ROLE prx_role, connect;

User altered.

SYS@orcl> COL proxy FOR a10
SYS@orcl> COL client FOR a10
SYS@orcl> COL role FOR a10
SYS@orcl> SELECT proxy, client, authorization_constraint, role FROM dba_proxies WHERE proxy = 'PRX_APP' ORDER BY client, role;

PROXY      CLIENT     AUTHORIZATION_CONSTRAINT            ROLE
---------- ---------- ----------------------------------- ----------
PRX_APP    PRX_USER1  PROXY MAY ACTIVATE ALL CLIENT ROLES
PRX_APP    PRX_USER2  PROXY MAY ACTIVATE ROLE             CONNECT
PRX_APP    PRX_USER2  PROXY MAY ACTIVATE ROLE             PRX_ROLE

3 rows selected.

SYS@orcl> CONNECT prx_app[prx_user2]/prx_app
Connected.
PRX_USER2@orcl> SELECT role FROM session_roles ORDER BY role;

ROLE
----------
CONNECT
PRX_ROLE

2 rows selected.

PRX_USER2@orcl> CREATE TABLE prx_t (id NUMBER);

Table created.

PRX_USER2@orcl> SET ROLE ALL;

Role set.

PRX_USER2@orcl> SELECT role FROM session_roles ORDER BY role;

ROLE
----------
CONNECT
PRX_ROLE

2 rows selected.

-- ★ 허용 역할이 CONNECT, PRX_ROLE 두 행이 되고 접속에 성공했다. 세션 역할도 그 둘뿐이다.
--    SET ROLE ALL 을 실행해도 세션 역할은 그대로 두 개다. 프록시 세션의 ALL 은 허용된 역할 안에서의 ALL 이다.

--------------------------------------------------------------------------------
 [5] WITH NO ROLES
--------------------------------------------------------------------------------

-- 역할 없이 접속하게 하려면 CREATE SESSION 을 역할이 아닌 직접 권한으로 줘야 한다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> GRANT CREATE SESSION TO prx_user2;

Grant succeeded.

SYS@orcl> ALTER USER prx_user2 GRANT CONNECT THROUGH prx_app WITH NO ROLES;

User altered.

SYS@orcl> COL proxy FOR a10
SYS@orcl> COL client FOR a10
SYS@orcl> COL role FOR a10
SYS@orcl> SELECT proxy, client, authorization_constraint, role FROM dba_proxies WHERE client = 'PRX_USER2';

PROXY      CLIENT     AUTHORIZATION_CONSTRAINT            ROLE
---------- ---------- ----------------------------------- ----------
PRX_APP    PRX_USER2  NO CLIENT ROLES MAY BE ACTIVATED

1 row selected.

SYS@orcl> CONNECT prx_app[prx_user2]/prx_app
Connected.
PRX_USER2@orcl> SELECT role FROM session_roles;

no rows selected

PRX_USER2@orcl> SET ROLE prx_role;
SET ROLE prx_role
*
ERROR at line 1:
ORA-01924: role 'PRX_ROLE' not granted or does not exist


PRX_USER2@orcl> SELECT role FROM session_roles;

no rows selected

-- ★ 세션 역할이 하나도 없고, SET ROLE prx_role 은 ORA-01924 로 거부됐다.
--    prx_user2 가 직접 접속했다면 켤 수 있는 역할이지만, 프록시 세션에서는 부여되지 않은 역할처럼 취급된다.
--    애플리케이션 서버가 화면마다 필요한 역할만 켜도록 설계할 때 WITH ROLE 목록이 그 상한이 된다.

================================================================================
 정리
================================================================================

-- - WITH ROLE 목록 = 프록시 세션에서 켤 수 있는 역할의 상한. SET ROLE ALL 도 이 안에서만 동작한다.
-- - CREATE SESSION 이 역할로만 들어오는 사용자는 그 역할을 빼면 ORA-01045 로 로그인이 막힌다.
-- - WITH NO ROLES 에서는 어떤 역할도 켤 수 없다(ORA-01924). 직접 권한만 남는다.
