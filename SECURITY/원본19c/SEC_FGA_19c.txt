SYS@orcl> create or replace procedure secadm.rt_alert (
  2    object_schema varchar2, object_name varchar2, policy_name varchar2)
  3  as
  4    pragma autonomous_transaction;
  5  begin
  6    insert into secadm.fga_alert (db_user, pol, sql_text)
  7    values (sys_context('USERENV','SESSION_USER'), policy_name,
  8            substr(sys_context('USERENV','CURRENT_SQL'), 1, 200));
  9    commit;
 10  end;
 11  /
