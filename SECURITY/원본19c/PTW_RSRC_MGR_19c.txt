-- 소비자 그룹 생성
SYS@orcl> EXEC DBMS_RESOURCE_MANAGER.CREATE_CONSUMER_GROUP( -
>    CONSUMER_GROUP => 'OLTP_GROUP', -
>    COMMENT        => 'OLTP 온라인 사용자 그룹');

PL/SQL procedure successfully completed.

-- BATCH_GROUP 은 19c 가 기본 제공하는 소비자 그룹이다(DSS_PLAN, ETL_CRITICAL_PLAN, MIXED_WORKLOAD_PLAN 이 사용).
-- 이름이 겹쳐 ORA-29357 이 난다. 오류가 나도 Pending Area 는 그대로 남아 다음 호출을 이어 갈 수 있다.

-- ORA-29388(plan/consumer_group BATCH_GROUP is part of more than one top-plan)로 실패하고 플랜이 지워지지 않는다

SYS@orcl> EXEC DBMS_RESOURCE_MANAGER.CREATE_CONSUMER_GROUP( -
>    CONSUMER_GROUP => 'BATCH_GROUP', -
>    COMMENT        => '배치 작업 그룹');
