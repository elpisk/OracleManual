================================================================================
 제3장 실습 01 — 권한과 역할 조회
 대상    : oel7v9 orcl (non-CDB, 19.3)
 목표    : 사용자에게 직접 부여된 시스템·객체 권한과 역할을 딕셔너리 뷰로 확인한다.
          세션에 실제로 켜진 역할과 권한을 보고, 부여받지 않은 역할을 켤 때의 오류를 확인한다
 소요    : 약 20분
 변경    : 시험 사용자 sar_dev 생성 (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] DBA 역할을 가진 사용자
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL grantee FOR a10
SYS@orcl> COL granted_role FOR a12
SYS@orcl> SELECT grantee, granted_role, admin_option, default_role FROM dba_role_privs WHERE granted_role = 'DBA';

GRANTEE    GRANTED_ROLE ADM DEF
---------- ------------ --- ---
SARAH      DBA          NO  YES
SYS        DBA          YES YES
SYSTEM     DBA          NO  YES

3 rows selected.

-- ★ SYS, SYSTEM 외에 SARAH 에게도 DBA 가 있다. 이전 과정 실습에서 부여된 계정이다.
--    DBA 보유자 목록은 정기 점검의 첫 줄이다. 예상하지 못한 이름이 보이면 부여 경위를 확인한다.

--------------------------------------------------------------------------------
 [2] 시험 사용자 SAR_DEV 의 권한 구성
--------------------------------------------------------------------------------

-- 다른 실습이 HR 에 권한을 주고받으므로 이 장 전용 사용자를 만들어 조회한다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE USER sar_dev IDENTIFIED BY oracle;

User created.

SYS@orcl> GRANT CREATE SESSION, CREATE VIEW, CREATE SYNONYM TO sar_dev;

Grant succeeded.

SYS@orcl> GRANT RESOURCE TO sar_dev;

Grant succeeded.

SYS@orcl> GRANT UNLIMITED TABLESPACE TO sar_dev;

Grant succeeded.

SYS@orcl> GRANT SELECT ON scott.dept TO sar_dev;

Grant succeeded.

SYS@orcl> COL privilege FOR a25
SYS@orcl> SELECT grantee, privilege, admin_option FROM dba_sys_privs WHERE grantee = 'SAR_DEV' ORDER BY privilege;

GRANTEE PRIVILEGE                 ADM
------- ------------------------- ---
SAR_DEV CREATE SESSION            NO
SAR_DEV CREATE SYNONYM            NO
SAR_DEV CREATE VIEW               NO
SAR_DEV UNLIMITED TABLESPACE      NO

4 rows selected.

SYS@orcl> COL owner FOR a6
SYS@orcl> COL table_name FOR a10
SYS@orcl> SELECT owner, table_name, privilege, grantable FROM dba_tab_privs WHERE grantee = 'SAR_DEV';

OWNER  TABLE_NAME PRIVILEGE                 GRA
------ ---------- ------------------------- ---
SCOTT  DEPT       SELECT                    NO

1 row selected.

SYS@orcl> COL granted_role FOR a12
SYS@orcl> SELECT granted_role, admin_option, default_role FROM dba_role_privs WHERE grantee = 'SAR_DEV';

GRANTED_ROLE ADM DEF
------------ --- ---
RESOURCE     NO  YES

1 row selected.

SYS@orcl> COL grantee FOR a10
SYS@orcl> SELECT grantee, privilege FROM dba_sys_privs WHERE privilege = 'UNLIMITED TABLESPACE' AND grantee IN ('HR', 'SCOTT', 'SAR_DEV') ORDER BY 1;

GRANTEE    PRIVILEGE
---------- -------------------------
HR         UNLIMITED TABLESPACE
SAR_DEV    UNLIMITED TABLESPACE

2 rows selected.

-- ★ 시스템 권한(DBA_SYS_PRIVS), 객체 권한(DBA_TAB_PRIVS), 역할(DBA_ROLE_PRIVS)이 뷰 세 개에 나뉘어 있다.
--    RESOURCE 를 받은 SCOTT 에게는 UNLIMITED TABLESPACE 가 없다. 12c 부터 RESOURCE 를 줘도
--    UNLIMITED TABLESPACE 가 함께 부여되지 않는다. 필요하면 SAR_DEV 처럼 직접 준다.

--------------------------------------------------------------------------------
 [3] 역할 안의 권한
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL grantee FOR a10
SYS@orcl> COL privilege FOR a25
SYS@orcl> SELECT grantee, privilege FROM dba_sys_privs WHERE grantee IN ('CONNECT', 'RESOURCE') ORDER BY 1, 2;

GRANTEE    PRIVILEGE
---------- -------------------------
CONNECT    CREATE SESSION
CONNECT    SET CONTAINER
RESOURCE   CREATE CLUSTER
RESOURCE   CREATE INDEXTYPE
RESOURCE   CREATE OPERATOR
RESOURCE   CREATE PROCEDURE
RESOURCE   CREATE SEQUENCE
RESOURCE   CREATE TABLE
RESOURCE   CREATE TRIGGER
RESOURCE   CREATE TYPE

10 rows selected.

SYS@orcl> SELECT COUNT(*) FROM role_role_privs WHERE role = 'DBA';

  COUNT(*)
----------
        18

1 row selected.

SYS@orcl> SELECT COUNT(*) FROM dba_sys_privs WHERE grantee = 'DBA';

  COUNT(*)
----------
       235

1 row selected.

-- ★ 19c 의 CONNECT 는 CREATE SESSION, SET CONTAINER 두 개뿐이다.
--    DBA 역할은 시스템 권한 235개를 직접 갖고, 역할 18개를 다시 포함한다.
--    역할 안의 역할까지 펼쳐 봐야 실제 권한 범위가 보인다(ROLE_ROLE_PRIVS).

--------------------------------------------------------------------------------
 [4] 오류 재현 — 일반 사용자가 DBA_ 뷰를 조회
--------------------------------------------------------------------------------

SQL> CONNECT sar_dev/oracle
Connected.
SAR_DEV@orcl> SELECT privilege FROM dba_sys_privs WHERE grantee = 'SAR_DEV';
SELECT privilege FROM dba_sys_privs WHERE grantee = 'SAR_DEV'
                      *
ERROR at line 1:
ORA-00942: table or view does not exist


SAR_DEV@orcl> SELECT privilege FROM user_sys_privs ORDER BY 1;

PRIVILEGE
----------------------------------------
CREATE SESSION
CREATE SYNONYM
CREATE VIEW
UNLIMITED TABLESPACE

4 rows selected.

SAR_DEV@orcl> COL role FOR a20
SAR_DEV@orcl> SELECT role FROM session_roles;

ROLE
--------------------
RESOURCE
SODA_APP

2 rows selected.

SAR_DEV@orcl> SELECT COUNT(*) FROM session_privs;

  COUNT(*)
----------
        12

1 row selected.

-- ★ ORA-00942. DBA_ 뷰는 SELECT ANY DICTIONARY 나 SELECT_CATALOG_ROLE 이 있어야 보인다.
--    일반 사용자는 USER_SYS_PRIVS 로 자기 권한을, SESSION_ROLES·SESSION_PRIVS 로 지금 켜진 것을 본다.
--    SESSION_ROLES 에 SODA_APP 이 보인다. 19c 의 RESOURCE 역할 안에 들어 있는 역할이다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — 부여받지 않은 역할 켜기
--------------------------------------------------------------------------------

SQL> CONNECT sar_dev/oracle
Connected.
SAR_DEV@orcl> SET ROLE dba;
SET ROLE dba
*
ERROR at line 1:
ORA-01924: role 'DBA' not granted or does not exist


SAR_DEV@orcl> SET ROLE resource;

Role set.

SAR_DEV@orcl> SELECT role FROM session_roles;

ROLE
------------------------------------------------------------------------------------------------------------------------
RESOURCE
SODA_APP

2 rows selected.

-- ★ ORA-01924. SET ROLE 은 이미 부여받은 역할을 켜고 끄는 명령이다. 권한을 얻는 수단이 아니다.
--    받은 역할(RESOURCE)은 SET ROLE 로 켜진다. 이 구조 위에 "조건을 만족할 때만 켜지는 역할"을
--    올린 것이 Secure Application Role 이다.

================================================================================
 정리
================================================================================

-- - DBA_SYS_PRIVS / DBA_TAB_PRIVS / DBA_ROLE_PRIVS / ROLE_ROLE_PRIVS 로 부여 상태를,
--   SESSION_ROLES / SESSION_PRIVS 로 현재 세션에 켜진 상태를 본다.
-- - 19c 의 CONNECT 는 CREATE SESSION·SET CONTAINER, RESOURCE 에는 UNLIMITED TABLESPACE 가 없다.
-- - 부여받지 않은 역할은 SET ROLE 로 켤 수 없다(ORA-01924).
