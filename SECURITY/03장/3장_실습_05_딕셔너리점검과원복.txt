================================================================================
 제3장 실습 05 — 딕셔너리 점검과 원복
 대상    : oel7v9 orcl
 목표    : DBA_APPLICATION_ROLES 로 역할과 활성화 프로시저의 연결을 확인하고,
          프로시저가 사라졌을 때의 상태를 본 뒤 이 장에서 만든 것을 모두 철거한다
 소요    : 약 15분
 변경    : 이 장에서 만든 사용자·역할·뷰·권한 전부 철거
================================================================================

--------------------------------------------------------------------------------
 [1] DBA_APPLICATION_ROLES 와 관련 객체
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL role FOR a14
SYS@orcl> COL schema FOR a10
SYS@orcl> COL package FOR a22
SYS@orcl> SELECT role, schema, package FROM dba_application_roles WHERE role = 'MANAGER_ROLE';

ROLE           SCHEMA     PACKAGE
-------------- ---------- ----------------------
MANAGER_ROLE   SAR_ADMIN  ACCESS_CONTROL_POLICY

1 row selected.

SYS@orcl> COL owner FOR a10
SYS@orcl> COL object_name FOR a22
SYS@orcl> SELECT owner, object_name, object_type, status FROM dba_objects WHERE owner = 'SAR_ADMIN';

OWNER      OBJECT_NAME            OBJECT_TYPE             STATUS
---------- ---------------------- ----------------------- -------
SAR_ADMIN  ACCESS_CONTROL_POLICY  PROCEDURE               VALID

1 row selected.

SYS@orcl> COL grantee FOR a10
SYS@orcl> SELECT grantee, privilege FROM dba_tab_privs WHERE owner = 'SAR_ADMIN' AND table_name = 'ACCESS_CONTROL_POLICY' ORDER BY 1;

GRANTEE    PRIVILEGE
---------- ----------------------------------------
BLAKE      EXECUTE
JAMES      EXECUTE

2 rows selected.

-- ★ 역할 → 스키마 → 프로시저의 연결이 한 행으로 나온다. 점검 순서는
--    ① 연결된 프로시저가 VALID 인가 ② 그 프로시저 실행 권한을 누가 가졌나(DBA_TAB_PRIVS) 이다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — 활성화 프로시저가 없어지면
--------------------------------------------------------------------------------

SQL> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> DROP PROCEDURE access_control_policy;

Procedure dropped.

SAR_ADMIN@orcl> CONNECT blake/blake
Connected.
BLAKE@orcl> EXEC sar_admin.access_control_policy
BEGIN sar_admin.access_control_policy; END;

      *
ERROR at line 1:
ORA-06550: line 1, column 7:
PLS-00201: identifier 'SAR_ADMIN.ACCESS_CONTROL_POLICY' must be declared
ORA-06550: line 1, column 7:
PL/SQL: Statement ignored


BLAKE@orcl> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL role FOR a14
SYS@orcl> COL schema FOR a10
SYS@orcl> COL package FOR a22
SYS@orcl> SELECT role, schema, package FROM dba_application_roles WHERE role = 'MANAGER_ROLE';

ROLE           SCHEMA     PACKAGE
-------------- ---------- ----------------------
MANAGER_ROLE   SAR_ADMIN  ACCESS_CONTROL_POLICY

1 row selected.

-- ★ blake 의 호출은 PLS-00201 로 실패하고 역할을 켤 방법이 없어진다.
--    DBA_APPLICATION_ROLES 에는 연결이 그대로 남는다. 프로시저를 같은 이름으로 다시 만들면 복구된다.
--    딕셔너리에 남아 있다는 것만으로 동작을 판단하지 않는다.

--------------------------------------------------------------------------------
 [3] 원복
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> DROP ROLE manager_role;

Role dropped.

SYS@orcl> DROP ROLE manager_role;
DROP ROLE manager_role
          *
ERROR at line 1:
ORA-01919: role 'MANAGER_ROLE' does not exist


SYS@orcl> DROP USER blake CASCADE;

User dropped.

SYS@orcl> DROP USER james CASCADE;

User dropped.

SYS@orcl> DROP USER sar_admin CASCADE;

User dropped.

SYS@orcl> DROP USER sar_dev CASCADE;

User dropped.

SYS@orcl> DROP VIEW scott.emp_job;

View dropped.

SYS@orcl> REVOKE CREATE VIEW FROM scott;

Revoke succeeded.

SYS@orcl> SELECT COUNT(*) FROM dba_users WHERE username IN ('BLAKE', 'JAMES', 'SAR_ADMIN', 'SAR_DEV');

  COUNT(*)
----------
         0

1 row selected.

SYS@orcl> SELECT COUNT(*) FROM dba_roles WHERE role = 'MANAGER_ROLE';

  COUNT(*)
----------
         0

1 row selected.

SYS@orcl> SELECT COUNT(*) FROM dba_objects WHERE owner = 'SCOTT' AND object_name = 'EMP_JOB';

  COUNT(*)
----------
         0

1 row selected.

SYS@orcl> SELECT COUNT(*) FROM dba_sys_privs WHERE grantee = 'SCOTT' AND privilege = 'CREATE VIEW';

  COUNT(*)
----------
         0

1 row selected.

-- ★ 역할을 두 번 지워 ORA-01919 를 확인했다. 사용자 4, 역할, 뷰, SCOTT 의 CREATE VIEW 권한이 모두 0 이다.

================================================================================
 정리
================================================================================

-- - DBA_APPLICATION_ROLES: 역할·스키마·프로시저(PACKAGE 열) 연결.
-- - 프로시저가 없거나 INVALID 면 역할을 켤 수 없다. 딕셔너리 연결은 남아 있다.
-- - 이 장에서 만든 sar_admin·sar_dev·blake·james·manager_role·scott.emp_job 을 모두 지웠다.
