================================================================================
 제3장 실습 04 — 검사 조건과 활성화 범위
 대상    : oel7v9 orcl
 목표    : 모듈 이름·IP 주소를 검사 조건에 넣고 접속 방식에 따라 결과가 달라지는 것을 본다.
          기본 역할·LOGON 트리거로는 켜지지 않는 것, SET ROLE NONE 으로 꺼지는 것을 확인한다
 소요    : 약 25분
 변경    : access_control_policy 교체. blake 에 트리거를 만들었다 지움
================================================================================

--------------------------------------------------------------------------------
 [1] 모듈 이름과 IP 를 함께 검사하도록 교체
--------------------------------------------------------------------------------

SQL> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> CREATE OR REPLACE PROCEDURE access_control_policy
  2  AUTHID CURRENT_USER
  3  AS
  4    v_job VARCHAR2(9);
  5  BEGIN
  6    IF SYS_CONTEXT('USERENV', 'MODULE') <> 'HRAPP'
  7       OR SYS_CONTEXT('USERENV', 'IP_ADDRESS') NOT LIKE '172.16.%' THEN
  8      RETURN;
  9    END IF;
 10    SELECT job INTO v_job
 11      FROM scott.emp_job
 12     WHERE ename = SYS_CONTEXT('USERENV', 'SESSION_USER');
 13    IF v_job = 'MANAGER' THEN
 14      DBMS_SESSION.SET_ROLE('manager_role');
 15    END IF;
 16  EXCEPTION
 17    WHEN NO_DATA_FOUND THEN NULL;
 18  END;
 19  /

Procedure created.

-- ★ MODULE 이 HRAPP 이고 클라이언트 IP 가 172.16. 대역일 때만 직무 검사로 넘어간다.
--    매뉴얼 예제의 IP·시간대 검사와 같은 방식이다.

--------------------------------------------------------------------------------
 [2] 로컬(bequeath) 접속, 모듈 미설정
--------------------------------------------------------------------------------

SQL> CONNECT blake/blake
Connected.
BLAKE@orcl> COL module FOR a20
BLAKE@orcl> COL ip FOR a15
BLAKE@orcl> COL role FOR a14
BLAKE@orcl> SELECT SYS_CONTEXT('USERENV', 'MODULE') module, SYS_CONTEXT('USERENV', 'IP_ADDRESS') ip FROM dual;

MODULE               IP
-------------------- ---------------
SQL*Plus

1 row selected.

BLAKE@orcl> EXEC sar_admin.access_control_policy

PL/SQL procedure successfully completed.

BLAKE@orcl> SELECT role FROM session_roles;

no rows selected

-- ★ 리스너를 거치지 않은 로컬 접속이라 IP_ADDRESS 가 비어 있고 MODULE 은 SQL*Plus 다.
--    프로시저는 조용히 끝나고 역할은 켜지지 않았다.

--------------------------------------------------------------------------------
 [3] 리스너 경유 + 모듈 설정
--------------------------------------------------------------------------------

SQL> CONNECT blake/blake@orcl
Connected.
BLAKE@orcl> COL module FOR a20
BLAKE@orcl> COL ip FOR a15
BLAKE@orcl> COL role FOR a14
BLAKE@orcl> EXEC DBMS_APPLICATION_INFO.SET_MODULE('HRAPP', NULL)

PL/SQL procedure successfully completed.

BLAKE@orcl> SELECT SYS_CONTEXT('USERENV', 'MODULE') module, SYS_CONTEXT('USERENV', 'IP_ADDRESS') ip FROM dual;

MODULE               IP
-------------------- ---------------
HRAPP                172.16.225.50

1 row selected.

BLAKE@orcl> EXEC sar_admin.access_control_policy

PL/SQL procedure successfully completed.

BLAKE@orcl> SELECT role FROM session_roles;

ROLE
--------------
MANAGER_ROLE

1 row selected.

BLAKE@orcl> SELECT MAX(sal) FROM scott.emp;

  MAX(SAL)
----------
      5000

1 row selected.

BLAKE@orcl> SET ROLE NONE;

Role set.

BLAKE@orcl> SELECT role FROM session_roles;

no rows selected

BLAKE@orcl> SELECT MAX(sal) FROM scott.emp;
SELECT MAX(sal) FROM scott.emp
                           *
ERROR at line 1:
ORA-00942: table or view does not exist

-- ★ @orcl 로 리스너를 거치면 IP 가 채워지고, SET_MODULE 로 HRAPP 을 넣자 역할이 켜졌다.
--    MODULE 은 클라이언트가 마음대로 정하는 값이다. SQL*Plus 에서도 한 줄로 바꿀 수 있으므로
--    단독 근거로 쓰지 않는다. SET ROLE NONE 은 이 역할도 끈다. 끄는 데에는 제약이 없다.

--------------------------------------------------------------------------------
 [4] 기본 역할로 지정해 보기
--------------------------------------------------------------------------------

-- 역할을 blake 에 부여하고 기본 역할로 지정해 새 세션에서 자동으로 켜지는지 본다.

SQL> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> GRANT manager_role TO blake;

Grant succeeded.

SAR_ADMIN@orcl> CONNECT / AS SYSDBA
Connected.
SYS@orcl> ALTER USER blake DEFAULT ROLE manager_role;

User altered.

SYS@orcl> COL granted_role FOR a14
SYS@orcl> SELECT granted_role, default_role FROM dba_role_privs WHERE grantee = 'BLAKE';

GRANTED_ROLE   DEF
-------------- ---
MANAGER_ROLE   NO

1 row selected.

SYS@orcl> CONNECT blake/blake@orcl
Connected.
BLAKE@orcl> COL role FOR a14
BLAKE@orcl> SELECT role FROM session_roles;

no rows selected

BLAKE@orcl> SELECT MAX(sal) FROM scott.emp;
SELECT MAX(sal) FROM scott.emp
                           *
ERROR at line 1:
ORA-00942: table or view does not exist


BLAKE@orcl> CONNECT / AS SYSDBA
Connected.
SYS@orcl> ALTER USER blake DEFAULT ROLE NONE;

User altered.

SYS@orcl> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> REVOKE manager_role FROM blake;

Revoke succeeded.

-- ★ ALTER USER ... DEFAULT ROLE manager_role 은 "User altered." 로 끝났지만 DEFAULT_ROLE 은 NO 그대로다.
--    새 세션에서도 역할이 없고 EMP 는 ORA-00942. 오류 없이 무시된다(19.3 동작).
--    매뉴얼도 이 유형의 역할은 기본 역할이 될 수 없다고 적고 있다. 시험 후 원래대로 되돌리고 회수했다.

--------------------------------------------------------------------------------
 [5] LOGON 트리거에서 켜기
--------------------------------------------------------------------------------

-- blake 스키마 한정 LOGON 트리거에서 모듈을 HRAPP 으로 바꾸고 활성화 프로시저를 부른다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> GRANT CREATE TRIGGER TO blake;

Grant succeeded.

SYS@orcl> GRANT EXECUTE ON sar_admin.access_control_policy TO blake;

Grant succeeded.

SYS@orcl> CREATE OR REPLACE TRIGGER blake.sar_logon_trg AFTER LOGON ON blake.SCHEMA
  2  BEGIN
  3    DBMS_APPLICATION_INFO.SET_MODULE('HRAPP', NULL);
  4    sar_admin.access_control_policy;
  5  END;
  6  /

Trigger created.

SYS@orcl> CONNECT blake/blake@orcl
Connected.
BLAKE@orcl> COL role FOR a14
BLAKE@orcl> SELECT role FROM session_roles;

no rows selected

BLAKE@orcl> CONNECT / AS SYSDBA
Connected.
SYS@orcl> DROP TRIGGER blake.sar_logon_trg;

Trigger dropped.

SYS@orcl> REVOKE CREATE TRIGGER FROM blake;

Revoke succeeded.

-- ★ 로그인은 정상이지만 SESSION_ROLES 가 비어 있다. 트리거 안에서는 역할이 켜지지 않았다.
--    매뉴얼은 LOGON 트리거로 이 역할을 켤 수 없으니, 로그인 직후 애플리케이션이 직접
--    프로시저를 호출하라고 한다. 트리거와 권한은 바로 지웠다.

================================================================================
 정리
================================================================================

-- - 검사 조건에는 서버가 채우는 값(SESSION_USER, IP_ADDRESS)을 중심으로 쓴다. MODULE 은 위조 가능.
-- - 로컬 접속은 IP_ADDRESS 가 비어 있다.
-- - 기본 역할 지정은 오류 없이 무시되고, LOGON 트리거로도 켜지지 않는다.
--   애플리케이션이 로그인 직후 프로시저를 직접 호출해야 한다.
