================================================================================
 제3장 실습 02 — Secure Application Role 만들기
 대상    : oel7v9 orcl
 목표    : IDENTIFIED USING 으로 역할을 만들고, 지정한 프로시저 밖에서는
          SET ROLE·DBMS_SESSION.SET_ROLE 로 켤 수 없음을 확인한다
 소요    : 약 20분
 변경    : sar_admin·blake·james 사용자, 역할 manager_role, 뷰 scott.emp_job (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] 보안 관리자와 시험 사용자
--------------------------------------------------------------------------------

-- 원본 자료는 SYSTEM 스키마에 프로시저를 만들었다. 여기서는 보안 관리 전용 계정 sar_admin 을 쓴다.
-- blake(직무 MANAGER), james(직무 CLERK) 는 EMP 의 사원 이름과 같게 만들었다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE USER sar_admin IDENTIFIED BY oracle;

User created.

SYS@orcl> GRANT CREATE SESSION, CREATE PROCEDURE, CREATE ROLE TO sar_admin;

Grant succeeded.

SYS@orcl> CREATE USER blake IDENTIFIED BY blake;

User created.

SYS@orcl> CREATE USER james IDENTIFIED BY james;

User created.

SYS@orcl> GRANT CREATE SESSION TO blake, james;

Grant succeeded.

SYS@orcl> GRANT CREATE VIEW TO scott;

Grant succeeded.

-- ★ sar_admin 에게 CREATE ROLE 과 CREATE PROCEDURE 만 줬다. 역할과 활성화 프로시저를 같은 계정이 소유한다.

--------------------------------------------------------------------------------
 [2] 업무용 뷰와 직접 권한
--------------------------------------------------------------------------------

SQL> CONNECT scott/tiger
Connected.
SCOTT@orcl> CREATE VIEW emp_job AS SELECT ename, job FROM emp;

View created.

SCOTT@orcl> GRANT SELECT ON emp_job TO blake, james;

Grant succeeded.

SCOTT@orcl> GRANT SELECT ON emp_job TO sar_admin;

Grant succeeded.

SCOTT@orcl> CONNECT blake/blake
Connected.
BLAKE@orcl> SELECT COUNT(*) FROM scott.emp_job;

  COUNT(*)
----------
        14

1 row selected.

BLAKE@orcl> SELECT MAX(sal) FROM scott.emp;
SELECT MAX(sal) FROM scott.emp
                           *
ERROR at line 1:
ORA-00942: table or view does not exist

-- ★ blake 는 이름·직무만 담은 EMP_JOB 뷰는 읽지만 급여가 있는 EMP 는 ORA-00942 로 읽지 못한다.
--    EMP 조회 권한은 사용자에게 주지 않고 역할에만 준다.

--------------------------------------------------------------------------------
 [3] IDENTIFIED USING 역할 만들기
--------------------------------------------------------------------------------

-- CREATE ROLE ... IDENTIFIED USING 스키마.프로시저. 이 시점에 프로시저는 아직 없다.

SQL> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> CREATE ROLE manager_role IDENTIFIED USING sar_admin.access_control_policy;

Role created.

SAR_ADMIN@orcl> CONNECT / AS SYSDBA
Connected.
SYS@orcl> GRANT SELECT ON scott.emp TO manager_role;

Grant succeeded.

SYS@orcl> COL role FOR a14
SYS@orcl> COL authentication_type FOR a19
SYS@orcl> SELECT role, password_required, authentication_type FROM dba_roles WHERE role = 'MANAGER_ROLE';

ROLE           PASSWORD AUTHENTICATION_TYPE
-------------- -------- -------------------
MANAGER_ROLE   YES      APPLICATION

1 row selected.

SYS@orcl> COL grantee FOR a10
SYS@orcl> COL granted_role FOR a14
SYS@orcl> SELECT grantee, granted_role, admin_option, default_role FROM dba_role_privs WHERE granted_role = 'MANAGER_ROLE';

GRANTEE    GRANTED_ROLE   ADM DEF
---------- -------------- --- ---
SAR_ADMIN  MANAGER_ROLE   YES NO

1 row selected.

-- ★ AUTHENTICATION_TYPE 이 APPLICATION 이다. PASSWORD_REQUIRED 는 YES 로 보이지만 비밀번호 역할은 아니다.
--    역할을 만든 sar_admin 에게 ADMIN OPTION 과 함께 부여됐고 DEFAULT_ROLE 은 NO 다.

--------------------------------------------------------------------------------
 [4] 오류 재현 — SET ROLE 로 직접 켜기
--------------------------------------------------------------------------------

SQL> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> SET ROLE manager_role;
SET ROLE manager_role
*
ERROR at line 1:
ORA-28201: invalid command to enable secure application role 'MANAGER_ROLE'


SAR_ADMIN@orcl> EXEC DBMS_SESSION.SET_ROLE('manager_role')
BEGIN DBMS_SESSION.SET_ROLE('manager_role'); END;

*
ERROR at line 1:
ORA-28201: invalid command to enable secure application role 'MANAGER_ROLE'
ORA-06512: at "SYS.DBMS_SESSION", line 183
ORA-06512: at line 1

-- ★ 역할을 가진 sar_admin 이라도 SET ROLE, DBMS_SESSION.SET_ROLE 모두 ORA-28201 이다.
--    Oracle 은 역할을 켜는 호출 스택에 USING 으로 지정한 프로시저가 있는지 확인한다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — 사용자에게 직접 부여하면
--------------------------------------------------------------------------------

SQL> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> GRANT manager_role TO blake;

Grant succeeded.

SAR_ADMIN@orcl> CONNECT blake/blake
Connected.
BLAKE@orcl> SELECT role FROM session_roles;

no rows selected

BLAKE@orcl> SET ROLE manager_role;
SET ROLE manager_role
*
ERROR at line 1:
ORA-28201: invalid command to enable secure application role 'MANAGER_ROLE'


BLAKE@orcl> CONNECT sar_admin/oracle
Connected.
SAR_ADMIN@orcl> REVOKE manager_role FROM blake;

Revoke succeeded.

-- ★ GRANT 자체는 성공한다. 그러나 로그인 직후 SESSION_ROLES 에 나타나지 않고 SET ROLE 도 ORA-28201 이다.
--    부여해도 얻는 것이 없으므로 매뉴얼은 역할을 사용자에게 직접 주지 말라고 한다. 바로 회수했다.

================================================================================
 정리
================================================================================

-- - CREATE ROLE ... IDENTIFIED USING 으로 만든 역할은 DBA_ROLES 에 AUTHENTICATION_TYPE=APPLICATION 으로 보인다.
-- - 지정한 프로시저 밖에서 켜려 하면 SET ROLE 이든 DBMS_SESSION.SET_ROLE 이든 ORA-28201.
-- - 권한은 역할에만 주고, 역할은 사용자에게 주지 않는다.
