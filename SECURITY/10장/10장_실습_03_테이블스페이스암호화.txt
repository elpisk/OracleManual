================================================================================
 제10장 실습 03 — 테이블스페이스 암호화
 대상    : oel7v9 tdedb
 목표    : 암호화 테이블스페이스와 일반 테이블스페이스에 같은 데이터를 넣고 데이터 파일을 직접 읽어 비교한다.
          기존 테이블스페이스를 온라인으로 암호화한다
 소요    : 약 15분
 변경    : 테이블스페이스 TDE_TS·TDE_PLAIN, 사용자 TDE_LAB
================================================================================

--------------------------------------------------------------------------------
 [1] 암호화 테이블스페이스와 일반 테이블스페이스
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> CREATE TABLESPACE tde_ts DATAFILE '/u01/app/oracle/oradata/TDEDB/tde_ts01.dbf' SIZE 20M
  2    ENCRYPTION USING 'AES256' ENCRYPT;

Tablespace created.

SYS@tdedb> CREATE TABLESPACE tde_plain DATAFILE '/u01/app/oracle/oradata/TDEDB/tde_plain01.dbf' SIZE 20M;

Tablespace created.

SYS@tdedb> CREATE USER tde_lab IDENTIFIED BY "TdeLab#1" QUOTA UNLIMITED ON tde_ts QUOTA UNLIMITED ON tde_plain QUOTA UNLIMITED ON users;

User created.

SYS@tdedb> GRANT CREATE SESSION TO tde_lab;

Grant succeeded.

SYS@tdedb> CREATE TABLE tde_lab.card_enc (id NUMBER, card_no VARCHAR2(20)) TABLESPACE tde_ts;

Table created.

SYS@tdedb> CREATE TABLE tde_lab.card_plain (id NUMBER, card_no VARCHAR2(20)) TABLESPACE tde_plain;

Table created.

SYS@tdedb> INSERT INTO tde_lab.card_enc VALUES (1, 'CARD-4321-SECRET');

1 row created.

SYS@tdedb> INSERT INTO tde_lab.card_plain VALUES (1, 'CARD-4321-SECRET');

1 row created.

SYS@tdedb> COMMIT;

Commit complete.

SYS@tdedb> ALTER SYSTEM CHECKPOINT;

System altered.

SYS@tdedb> COL name FOR a12
SYS@tdedb> SELECT t.name, e.encryptedts, e.encryptionalg
  2    FROM v$encrypted_tablespaces e JOIN v$tablespace t ON e.ts# = t.ts#;

NAME         ENC ENCRYPT
------------ --- -------
TDE_TS       YES AES256

1 row selected.

SYS@tdedb> SELECT tablespace_name, encrypted FROM dba_tablespaces WHERE tablespace_name LIKE 'TDE%';

TABLESPACE_NAME                ENC
------------------------------ ---
TDE_PLAIN                      NO
TDE_TS                         YES

2 rows selected.

-- ★ TDE_TS 는 AES256 으로 암호화, TDE_PLAIN 은 일반. 같은 카드 번호를 한 행씩 넣고 체크포인트로 블록을 디스크에 썼다.
--    DBA_TABLESPACES 의 ENCRYPTED 와 V$ENCRYPTED_TABLESPACES 로 확인한다.

--------------------------------------------------------------------------------
 [2] 데이터 파일을 OS 에서 직접 읽으면
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ strings /u01/app/oracle/oradata/TDEDB/tde_plain01.dbf | grep -c 'CARD-4321-SECRET'
1
[oracle@oel7v9 ~]$ strings /u01/app/oracle/oradata/TDEDB/tde_ts01.dbf | grep -c 'CARD-4321-SECRET'
0

-- ★ 일반 테이블스페이스 파일에서는 strings 만으로 카드 번호가 1건 나온다. 암호화 테이블스페이스 파일에서는 0건.
--    SQL 로 조회하면 두 테이블 모두 평문이 나온다. TDE 는 파일을 보호하고 SQL 사용자를 막지 않는다.

--------------------------------------------------------------------------------
 [3] 기존 테이블스페이스 온라인 변환
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> SHOW PARAMETER db_create_file_dest

NAME                                 TYPE        VALUE
------------------------------------ ----------- ------------------------------
db_create_file_dest                  string
SYS@tdedb> ALTER TABLESPACE tde_plain ENCRYPTION ONLINE USING 'AES256' ENCRYPT;

Tablespace altered.

SYS@tdedb> COL file_name FOR a50
SYS@tdedb> SELECT file_name FROM dba_data_files WHERE tablespace_name = 'TDE_PLAIN';

FILE_NAME
--------------------------------------------------
/u01/app/oracle/oradata/TDEDB/tde_plain01.dbf

1 row selected.

SYS@tdedb> SELECT tablespace_name, encrypted FROM dba_tablespaces WHERE tablespace_name LIKE 'TDE%';

TABLESPACE_NAME                ENC
------------------------------ ---
TDE_PLAIN                      YES
TDE_TS                         YES

2 rows selected.

SYS@tdedb> ALTER SYSTEM CHECKPOINT;

System altered.

SYS@tdedb> SELECT * FROM tde_lab.card_plain;

        ID CARD_NO
---------- --------------------
         1 CARD-4321-SECRET

1 row selected.

-- ★ 열린 상태로 TDE_PLAIN 을 암호화했다. OMF 가 아닌데 FILE_NAME_CONVERT 를 주지 않았지만 19.3 에서 성공했고
--    파일 이름도 tde_plain01.dbf 그대로다. 변환 중에도 테이블 조회는 평소대로 된다.

--------------------------------------------------------------------------------
 [4] 변환된 파일을 직접 읽으면
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ ls /u01/app/oracle/oradata/TDEDB | grep tde_
tde_plain01.dbf
tde_ts01.dbf
[oracle@oel7v9 ~]$ strings /u01/app/oracle/oradata/TDEDB/tde_plain01.dbf | grep -c 'CARD-4321-SECRET'
0

-- ★ 같은 이름의 파일에서 카드 번호가 0건. 블록이 암호화된 새 파일로 바뀌었다.

--------------------------------------------------------------------------------
 [5] 오류 재현 — 이미 암호화된 테이블스페이스를 다시 변환
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> ALTER TABLESPACE tde_plain ENCRYPTION ONLINE USING 'AES256' ENCRYPT;
ALTER TABLESPACE tde_plain ENCRYPTION ONLINE USING 'AES256' ENCRYPT
*
ERROR at line 1:
ORA-28370: ENCRYPT, DECRYPT or REKEY option not allowed


SYS@tdedb> COL name FOR a12
SYS@tdedb> SELECT t.name, e.encryptedts, e.encryptionalg, e.status
  2    FROM v$encrypted_tablespaces e JOIN v$tablespace t ON e.ts# = t.ts#;

NAME         ENC ENCRYPT STATUS
------------ --- ------- ----------
TDE_TS       YES AES256  NORMAL
TDE_PLAIN    YES AES256  NORMAL

2 rows selected.

-- ★ ORA-28370: ENCRYPT, DECRYPT or REKEY option not allowed. 이미 암호화된 테이블스페이스에 ENCRYPT 를 다시 줬다.
--    V$ENCRYPTED_TABLESPACES 로 두 테이블스페이스 모두 AES256, NORMAL 상태임을 확인한다.
--    알고리즘이나 키를 바꾸려면 ENCRYPT 대신 REKEY 를 쓴다.

================================================================================
 정리
================================================================================

-- - 테이블스페이스 암호화는 디스크의 블록을 암호화한다. strings 로 파일을 읽어도 평문이 나오지 않는다.
-- - ALTER TABLESPACE ... ENCRYPTION ONLINE ... ENCRYPT 로 열린 채 변환. 19.3 에서는 FILE_NAME_CONVERT 없이도 같은 이름으로 변환됐다.
-- - 이미 암호화된 테이블스페이스에 ENCRYPT 는 ORA-28370.
