================================================================================
 제10장 실습 04 — 컬럼 암호화
 대상    : oel7v9 tdedb
 목표    : ENCRYPT 컬럼의 기본 알고리즘·salt 를 확인하고 인덱스, 알고리즘 변경의 제약을 재현한다
 소요    : 약 15분
 변경    : TDE_LAB.CUST 테이블
================================================================================

--------------------------------------------------------------------------------
 [1] 암호화 컬럼 만들기
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> CREATE TABLE tde_lab.cust (
  2    cust_id  NUMBER,
  3    name     VARCHAR2(30),
  4    card_no  VARCHAR2(20) ENCRYPT) TABLESPACE users;

Table created.

SYS@tdedb> INSERT INTO tde_lab.cust VALUES (1, 'KIM', '1111-2222-3333-4444');

1 row created.

SYS@tdedb> COMMIT;

Commit complete.

SYS@tdedb> COL table_name FOR a8
SYS@tdedb> COL column_name FOR a10
SYS@tdedb> COL encryption_alg FOR a18
SYS@tdedb> COL salt FOR a4
SYS@tdedb> COL integrity_alg FOR a10
SYS@tdedb> SELECT table_name, column_name, encryption_alg, salt, integrity_alg FROM dba_encrypted_columns WHERE owner = 'TDE_LAB';

TABLE_NA COLUMN_NAM ENCRYPTION_ALG     SALT INTEGRITY_
-------- ---------- ------------------ ---- ----------
CUST     CARD_NO    AES 192 bits key   YES  SHA-1

1 row selected.

-- ★ 알고리즘을 지정하지 않은 ENCRYPT 컬럼은 AES 192, SALT YES, 무결성 SHA-1.
--    USERS 는 암호화되지 않은 테이블스페이스라 이 테이블은 컬럼 암호화만 받는다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — salt 컬럼에 인덱스
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> CREATE INDEX tde_lab.cust_card_ix ON tde_lab.cust (card_no);
CREATE INDEX tde_lab.cust_card_ix ON tde_lab.cust (card_no)
                                                   *
ERROR at line 1:
ORA-28338: Column(s) cannot be both indexed and encrypted with salt

-- ★ ORA-28338: Column(s) cannot be both indexed and encrypted with salt.

--------------------------------------------------------------------------------
 [3] 조치 — NO SALT, 그리고 다른 알고리즘 시도
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> ALTER TABLE tde_lab.cust MODIFY (card_no ENCRYPT NO SALT);

Table altered.

SYS@tdedb> CREATE INDEX tde_lab.cust_card_ix ON tde_lab.cust (card_no);

Index created.

SYS@tdedb> ALTER TABLE tde_lab.cust MODIFY (name ENCRYPT USING 'AES256');
ALTER TABLE tde_lab.cust MODIFY (name ENCRYPT USING 'AES256')
                                 *
ERROR at line 1:
ORA-28340: A different encryption algorithm has been chosen for the table or tablespace.


SYS@tdedb> COL table_name FOR a8
SYS@tdedb> COL column_name FOR a10
SYS@tdedb> COL encryption_alg FOR a18
SYS@tdedb> SELECT table_name, column_name, encryption_alg, salt FROM dba_encrypted_columns WHERE owner = 'TDE_LAB';

TABLE_NA COLUMN_NAM ENCRYPTION_ALG     SAL
-------- ---------- ------------------ ---
CUST     CARD_NO    AES 192 bits key   NO

1 row selected.

-- ★ NO SALT 로 바꾸자 인덱스가 만들어졌다.
--    이어서 NAME 컬럼만 AES256 으로 암호화하려 하자 ORA-28340: 테이블의 기존 암호화 알고리즘(AES192)과 다르다.
--    한 테이블의 암호화 컬럼은 같은 테이블 키·같은 알고리즘을 쓴다.

--------------------------------------------------------------------------------
 [4] 조치 — 테이블 단위 REKEY
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@tdedb> ALTER TABLE tde_lab.cust REKEY USING 'AES256';

Table altered.

SYS@tdedb> ALTER TABLE tde_lab.cust MODIFY (name ENCRYPT);

Table altered.

SYS@tdedb> COL table_name FOR a8
SYS@tdedb> COL column_name FOR a10
SYS@tdedb> COL encryption_alg FOR a18
SYS@tdedb> SELECT table_name, column_name, encryption_alg, salt FROM dba_encrypted_columns WHERE owner = 'TDE_LAB';

TABLE_NA COLUMN_NAM ENCRYPTION_ALG     SAL
-------- ---------- ------------------ ---
CUST     NAME       AES 256 bits key   YES
CUST     CARD_NO    AES 256 bits key   NO

2 rows selected.

SYS@tdedb> SELECT * FROM tde_lab.cust;

   CUST_ID NAME                           CARD_NO
---------- ------------------------------ --------------------
         1 KIM                            1111-2222-3333-4444

1 row selected.

-- ★ REKEY USING 'AES256' 로 테이블 키를 바꾼 뒤 NAME 을 알고리즘 지정 없이 암호화했다.
--    두 컬럼 모두 AES 256 이다. SALT 는 컬럼마다 따로라 NAME 은 YES, CARD_NO 는 NO.
--    SQL 결과는 평문이다.

================================================================================
 정리
================================================================================

-- - 컬럼 암호화 기본: AES192, salt, SHA-1.
-- - 인덱스 컬럼은 NO SALT(아니면 ORA-28338).
-- - 알고리즘은 테이블 단위. 다른 알고리즘은 ORA-28340, 바꾸려면 ALTER TABLE ... REKEY.
