================================================================================
 제13장 실습 04 — syslog 전송 설정
 대상    : oel7v9 orcl, root 셸
 목표    : rsyslog 에 local1 규칙을 추가하고 AUDIT_SYSLOG_LEVEL 과 UNIFIED_AUDIT_SYSTEMLOG 를 설정해
          SYS 감사 기록과 통합 감사 기록이 syslog 로 가는 것을 확인한다
 소요    : 약 30분 (재기동 포함)
 변경    : /etc/rsyslog.conf 수정(백업 후), orcl SPFILE 파라미터 2개, orcl 재기동 1회 (05 에서 원복)
================================================================================

--------------------------------------------------------------------------------
 [1] rsyslog 설정 (root)
--------------------------------------------------------------------------------

[root@oel7v9 ~]# cp -p /etc/rsyslog.conf /etc/rsyslog.conf.bak_sec13
[root@oel7v9 ~]# grep -n '^local' /etc/rsyslog.conf
73:local7.*                                                /var/log/boot.log
[root@oel7v9 ~]# echo 'local1.*                                                /var/log/oracle_db.log' >> /etc/rsyslog.conf
[root@oel7v9 ~]# grep -n '^local' /etc/rsyslog.conf
73:local7.*                                                /var/log/boot.log
92:local1.*                                                /var/log/oracle_db.log
[root@oel7v9 ~]# systemctl restart rsyslog
[root@oel7v9 ~]# systemctl is-active rsyslog
active

-- ★ 원본 파일을 백업한 뒤 local1 의 모든 메시지를 /var/log/oracle_db.log 로 보내는 규칙을 추가했다.
--    기존 규칙은 local7(boot.log) 하나라서 local1 과 겹치지 않는다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — 정적 파라미터를 바로 바꾸기
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE PFILE = '/tmp/sec13_init_before.ora' FROM SPFILE;

File created.

SYS@orcl> ALTER SYSTEM SET audit_syslog_level = 'local1.warning';
ALTER SYSTEM SET audit_syslog_level = 'local1.warning'
                 *
ERROR at line 1:
ORA-02095: specified initialization parameter cannot be modified


SYS@orcl> ALTER SYSTEM SET audit_syslog_level = 'local1.warning' SCOPE = SPFILE;

System altered.

SYS@orcl> ALTER SYSTEM SET unified_audit_systemlog = 'local1.notice' SCOPE = SPFILE;

System altered.

SYS@orcl> SHOW PARAMETER syslog

NAME                                 TYPE        VALUE
------------------------------------ ----------- ------------------------------
audit_syslog_level                   string

-- ★ AUDIT_SYSLOG_LEVEL 은 정적 파라미터라 SCOPE 없이 바꾸면 ORA-02095.
--    SCOPE=SPFILE 로 넣으니 성공했다. 아직 메모리 값은 비어 있다(SHOW PARAMETER 결과).
--    통합 감사 기록 쪽 UNIFIED_AUDIT_SYSTEMLOG 도 같은 방식으로 넣었다. 재기동 전에 SPFILE 을 PFILE 로 백업했다.

================================================================================
 정리
================================================================================

-- - AUDIT_SYSLOG_LEVEL 은 정적(ORA-02095). SCOPE=SPFILE + 재기동.
-- - AUDIT_SYS_OPERATIONS = TRUE 면 SYS 작업이 .aud 대신 syslog 로 간다('Oracle Audit').
-- - UNIFIED_AUDIT_SYSTEMLOG 는 통합 감사 기록의 핵심 필드를 보낸다('Oracle Unified Audit'). 19.3 혼합 모드에서도 동작.
-- - 긴 기록은 여러 syslog 항목으로 나뉜다.
