================================================================================
 제6장 실습 03 — LOGON 트리거로 자동 설정
 대상    : oel7v9 orcl
 목표    : 로그인 사용자와 사원 매핑으로 부서를 찾아 LOGON 트리거가 컨텍스트를 채우게 하고,
          트리거 안의 예외가 로그인 자체를 막는 상황과 그 처리 방법을 확인한다
 소요    : 약 25분
 변경    : c6_king·c6_hunold·c6_nomap 사용자, 매핑 테이블, C6_EMP_CTX, 스키마 LOGON 트리거 3개 (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] 사용자와 매핑 테이블
--------------------------------------------------------------------------------

-- 원본은 HR.EMPLOYEES 의 EMAIL 이 로그인 이름과 같다고 가정했다(SKING, AHUNOLD).
-- 여기서는 로그인 이름과 사원 EMAIL 을 잇는 매핑 테이블을 따로 둔다. c6_nomap 은 매핑이 없는 사용자다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE USER c6_king IDENTIFIED BY oracle;

User created.

SYS@orcl> CREATE USER c6_hunold IDENTIFIED BY oracle;

User created.

SYS@orcl> CREATE USER c6_nomap IDENTIFIED BY oracle;

User created.

SYS@orcl> GRANT CREATE SESSION TO c6_king, c6_hunold, c6_nomap;

Grant succeeded.

SYS@orcl> CONNECT ctxadm/oracle
Connected.
CTXADM@orcl> CREATE TABLE c6_user_map (username VARCHAR2(128) PRIMARY KEY, email VARCHAR2(25));

Table created.

CTXADM@orcl> INSERT INTO c6_user_map VALUES ('C6_KING', 'SKING');

1 row created.

CTXADM@orcl> INSERT INTO c6_user_map VALUES ('C6_HUNOLD', 'AHUNOLD');

1 row created.

CTXADM@orcl> COMMIT;

Commit complete.

-- ★ 매핑 테이블을 고칠 수 있는 사람이 곧 사용자의 부서를 정하는 사람이다. 소유자는 ctxadm.

--------------------------------------------------------------------------------
 [2] 컨텍스트, 예외 처리 없는 패키지, 보안 뷰
--------------------------------------------------------------------------------

SQL> CONNECT ctxadm/oracle
Connected.
CTXADM@orcl> CREATE CONTEXT c6_emp_ctx USING ctxadm.c6_emp_pkg;

Context created.

CTXADM@orcl> CREATE OR REPLACE PACKAGE c6_emp_pkg AS
  2    PROCEDURE set_department;
  3  END;
  4  /

Package created.

CTXADM@orcl> CREATE OR REPLACE PACKAGE BODY c6_emp_pkg AS
  2    PROCEDURE set_department IS
  3      v_dept hr.employees.department_id%TYPE;
  4    BEGIN
  5      SELECT e.department_id INTO v_dept
  6        FROM hr.employees e JOIN c6_user_map m ON e.email = m.email
  7       WHERE m.username = SYS_CONTEXT('USERENV', 'SESSION_USER');
  8      DBMS_SESSION.SET_CONTEXT('c6_emp_ctx', 'dept_id', v_dept);
  9    END;
 10  END;
 11  /

Package body created.

CTXADM@orcl> CREATE OR REPLACE VIEW c6_secure_emp AS
  2    SELECT employee_id, first_name, department_id
  3      FROM hr.employees
  4     WHERE department_id = SYS_CONTEXT('c6_emp_ctx', 'dept_id');

View created.

CTXADM@orcl> GRANT SELECT ON c6_secure_emp TO c6_king, c6_hunold, c6_nomap;

Grant succeeded.

-- ★ 패키지는 SELECT INTO 로 부서를 찾는다. 원본과 달리 예외 처리를 일부러 뺐다.

--------------------------------------------------------------------------------
 [3] 사용자별 LOGON 트리거
--------------------------------------------------------------------------------

-- 원본은 AFTER LOGON ON DATABASE 였다. 다른 실습과 같은 DB 를 쓰므로 사용자별 스키마 트리거로 범위를 좁힌다.
-- 다른 스키마에 트리거를 거는 것이라 SYS 가 만든다(CREATE ANY TRIGGER 필요).

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE OR REPLACE TRIGGER ctxadm.c6_king_logon AFTER LOGON ON c6_king.SCHEMA
  2  BEGIN
  3    ctxadm.c6_emp_pkg.set_department;
  4  END;
  5  /

Trigger created.

SYS@orcl> CREATE OR REPLACE TRIGGER ctxadm.c6_hunold_logon AFTER LOGON ON c6_hunold.SCHEMA
  2  BEGIN
  3    ctxadm.c6_emp_pkg.set_department;
  4  END;
  5  /

Trigger created.

SYS@orcl> CREATE OR REPLACE TRIGGER ctxadm.c6_nomap_logon AFTER LOGON ON c6_nomap.SCHEMA
  2  BEGIN
  3    ctxadm.c6_emp_pkg.set_department;
  4  END;
  5  /

Trigger created.

--------------------------------------------------------------------------------
 [5] 오류 재현 — 매핑에 없는 사용자의 로그인
--------------------------------------------------------------------------------

SQL> CONNECT c6_nomap/oracle
ERROR:
ORA-04088: error during execution of trigger 'CTXADM.C6_NOMAP_LOGON'
ORA-01403: no data found
ORA-06512: at "CTXADM.C6_EMP_PKG", line 5
ORA-06512: at line 2


SQL> SELECT USER FROM dual;
SP2-0640: Not connected

-- ★ ORA-04088 + ORA-01403. 트리거 안의 NO_DATA_FOUND 가 그대로 올라와 로그인이 거부됐다.
--    ON DATABASE 트리거였다면 매핑에 없는 모든 사용자가 같은 일을 겪는다.
--    ADMINISTER DATABASE TRIGGER 권한이 있는 DBA 계정은 LOGON 트리거 오류가 나도 로그인된다.

--------------------------------------------------------------------------------
 [6] 조치 — 예외 처리 추가 후 재확인
--------------------------------------------------------------------------------

SQL> CONNECT ctxadm/oracle
Connected.
CTXADM@orcl> CREATE OR REPLACE PACKAGE BODY c6_emp_pkg AS
  2    PROCEDURE set_department IS
  3      v_dept hr.employees.department_id%TYPE;
  4    BEGIN
  5      SELECT e.department_id INTO v_dept
  6        FROM hr.employees e JOIN c6_user_map m ON e.email = m.email
  7       WHERE m.username = SYS_CONTEXT('USERENV', 'SESSION_USER');
  8      DBMS_SESSION.SET_CONTEXT('c6_emp_ctx', 'dept_id', v_dept);
  9    EXCEPTION
 10      WHEN NO_DATA_FOUND THEN
 11        NULL;
 12    END;
 13  END;
 14  /

Package body created.

CTXADM@orcl> CONNECT c6_nomap/oracle
Connected.
C6_NOMAP@orcl> COL dept FOR a6
C6_NOMAP@orcl> SELECT NVL(SYS_CONTEXT('c6_emp_ctx', 'dept_id'), '(null)') AS dept FROM dual;

DEPT
------
(null)

1 row selected.

C6_NOMAP@orcl> SELECT COUNT(*) FROM ctxadm.c6_secure_emp;

  COUNT(*)
----------
         0

1 row selected.

-- ★ NO_DATA_FOUND 를 처리하자 c6_nomap 도 로그인된다. 컨텍스트는 비어 있고 보안 뷰는 0건이다.
--    값이 없으면 아무것도 보이지 않는 쪽으로 닫히는 구조다.

================================================================================
 정리
================================================================================

-- - LOGON 트리거 + 신뢰 패키지로 사용자가 값을 정할 기회 없이 컨텍스트를 채운다.
-- - 트리거 안의 처리되지 않은 예외는 ORA-04088 로 로그인을 막는다. 신뢰 패키지에 예외 처리는 필수.
-- - 범위가 필요한 사용자로 한정된다면 ON DATABASE 대신 ON <schema>.SCHEMA 트리거를 쓴다.
