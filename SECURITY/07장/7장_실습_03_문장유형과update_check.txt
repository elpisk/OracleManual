================================================================================
 제7장 실습 03 — 문장 유형과 update_check
 대상    : oel7v9 orcl
 목표    : statement_types 로 정책이 적용될 문장을 고르고,
          INSERT 와 update_check 의 관계, update_check 가 막는 것을 확인한다
 소요    : 약 25분
 변경    : EMP_DEPT_POLICY 재등록. DML 은 모두 ROLLBACK
================================================================================

--------------------------------------------------------------------------------
 [1] 오류 재현 — INSERT 까지 넣어 다시 등록
--------------------------------------------------------------------------------

-- statement_types 는 ALTER_POLICY 로 바꿀 수 없다(ALTER_POLICY 는 컨텍스트 속성 연결만 바꾼다).
-- 지우고 다시 등록한다. INSERT 를 포함해 네 문장 유형을 모두 준다.

  6      function_schema => 'SYSADMIN_VPD',

-- ★ ORA-28104: input value for statement_types is not valid.
--    ARPLS DBMS_RLS 에 적힌 규칙대로, INSERT 를 쓰려면 update_check => TRUE 가 함께 있어야 한다.
--    INSERT 를 빼고 SELECT, UPDATE, DELETE 로 등록하니 성공했다.
--    첫 줄에서 이미 DROP 했으므로 두 번째 ADD 가 성공하기 전까지 EMP 는 정책 없이 열려 있었다.

--------------------------------------------------------------------------------
 [2] update_check 없이 DML 해 보기 (컨텍스트 20)
--------------------------------------------------------------------------------

SQL> CONNECT scott/tiger
Connected.
SCOTT@orcl> EXEC sysadmin_vpd.hr_ctx_pkg.set_dept(20)

PL/SQL procedure successfully completed.

SCOTT@orcl> INSERT INTO emp VALUES (5556, 'RON1', 'DEV', 7902, DATE '2002-11-02', 2000, 100, 10);

1 row created.

SCOTT@orcl> SELECT empno, ename, deptno FROM emp WHERE empno = 5556;

no rows selected

SCOTT@orcl> UPDATE emp SET sal = sal + 1 WHERE empno = 5556;

0 rows updated.

SCOTT@orcl> DELETE FROM emp WHERE empno = 7839;

0 rows deleted.

SCOTT@orcl> UPDATE emp SET deptno = 30 WHERE empno = 7369;

1 row updated.

SCOTT@orcl> SELECT empno, ename, deptno FROM emp WHERE empno = 7369;

no rows selected

SCOTT@orcl> ROLLBACK;

Rollback complete.

-- ★ INSERT 는 정책 대상이 아니라 10번 행이 그대로 들어갔다. 그런데 넣은 사람은 그 행을 볼 수 없다.
--    UPDATE·DELETE 는 술어가 붙어 다른 부서 행(5556, 7839)을 0건 처리했다.
--    7369 의 부서를 30 으로 바꾸는 UPDATE 는 성공했고, 바꾼 순간 그 행은 내 시야에서 사라졌다.
--    볼 수 없는 데이터를 만들거나 내 데이터를 남의 부서로 넘기는 일을 막지 못한다.

--------------------------------------------------------------------------------
 [3] update_check 를 켜서 다시 등록
--------------------------------------------------------------------------------

SQL> CONNECT sysadmin_vpd/oracle
Connected.
SYSADMIN_VPD@orcl> BEGIN
  2    DBMS_RLS.ADD_POLICY(
  3      object_schema   => 'SCOTT',
  4      object_name     => 'EMP',
  5      policy_name     => 'EMP_DEPT_POLICY',
  6      function_schema => 'SYSADMIN_VPD',
  7      policy_function => 'RLS_DEPT',
  8      statement_types => 'SELECT, INSERT, UPDATE, DELETE',
  9      update_check    => TRUE);
 10  END;
 11  /
BEGIN
*
ERROR at line 1:
ORA-28101: policy already exists
ORA-06512: at "SYS.DBMS_RLS_INT", line 23
ORA-06512: at "SYS.DBMS_RLS", line 54
ORA-06512: at line 2


SYSADMIN_VPD@orcl> BEGIN
  2    DBMS_RLS.DROP_POLICY('SCOTT', 'EMP', 'EMP_DEPT_POLICY');
  3    DBMS_RLS.ADD_POLICY(
  4      object_schema   => 'SCOTT',
  5      object_name     => 'EMP',
  6      policy_name     => 'EMP_DEPT_POLICY',
  7      function_schema => 'SYSADMIN_VPD',
  8      policy_function => 'RLS_DEPT',
  9      statement_types => 'SELECT, INSERT, UPDATE, DELETE',
 10      update_check    => TRUE);
 11  END;
 12  /

PL/SQL procedure successfully completed.

SYSADMIN_VPD@orcl> CONNECT / AS SYSDBA

SYS@orcl> COL policy_name FOR a16

POLICY_NAME      SEL INS UPD DEL CHK
---------------- --- --- --- --- ---
EMP_DEPT_POLICY  YES YES YES YES YES

1 row selected.

-- ★ 정책이 남아 있는 상태에서 같은 이름으로 ADD 하면 ORA-28101: policy already exists.
--    DROP 후 ADD 로 바꾸니 INS=YES, CHK=YES 로 등록됐다.

--------------------------------------------------------------------------------
 [4] update_check 이후의 DML
--------------------------------------------------------------------------------

SQL> CONNECT scott/tiger
Connected.
SCOTT@orcl> EXEC sysadmin_vpd.hr_ctx_pkg.set_dept(20)

PL/SQL procedure successfully completed.

SCOTT@orcl> INSERT INTO emp VALUES (5557, 'RON1', 'DEV', 7902, DATE '2002-11-02', 2000, 100, 10);
INSERT INTO emp VALUES (5557, 'RON1', 'DEV', 7902, DATE '2002-11-02', 2000, 100, 10)
            *
ERROR at line 1:
ORA-28115: policy with check option violation


SCOTT@orcl> INSERT INTO emp VALUES (5557, 'RON1', 'DEV', 7902, DATE '2002-11-02', 2000, 100, 20);

1 row created.

SCOTT@orcl> UPDATE emp SET deptno = 30 WHERE empno = 5557;
UPDATE emp SET deptno = 30 WHERE empno = 5557
       *
ERROR at line 1:
ORA-28115: policy with check option violation


SCOTT@orcl> UPDATE emp SET sal = 2100 WHERE empno = 5557;

1 row updated.

SCOTT@orcl> ROLLBACK;

Rollback complete.

-- ★ 10번 부서 행 INSERT, 30번으로 옮기는 UPDATE 가 ORA-28115 로 막혔다.
--    변경 후의 행이 정책 술어를 만족하는지 다시 검사한다. 뷰의 WITH CHECK OPTION 과 같은 동작이다.
--    20번 안에서의 INSERT·급여 UPDATE 는 통과한다.

================================================================================
 정리
================================================================================

-- - statement_types 기본값은 SELECT, UPDATE, DELETE 다(INSERT·INDEX 제외).
-- - INSERT 를 넣으려면 update_check => TRUE 가 필수다(아니면 ORA-28104).
-- - update_check 는 변경 후 행이 술어를 벗어나는 INSERT/UPDATE 를 ORA-28115 로 막는다.
-- - statement_types·update_check 를 바꾸려면 DROP + ADD 다. 그 사이에는 보호가 없으므로 한 블록에서 연달아 처리한다.
