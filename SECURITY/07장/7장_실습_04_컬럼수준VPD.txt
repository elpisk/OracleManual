================================================================================
 제7장 실습 04 — 컬럼 수준 VPD
 대상    : oel7v9 orcl
 목표    : sec_relevant_cols 로 민감 컬럼을 부를 때만 정책이 걸리게 하고,
          ALL_ROWS 옵션으로 행은 보이고 값만 NULL 이 되는 동작을 확인한다
 소요    : 약 20분
 변경    : EMP_DEPT_POLICY 대신 EMP_SAL_POLICY
================================================================================

--------------------------------------------------------------------------------
 [1] SAL·COMM 에만 걸리는 정책
--------------------------------------------------------------------------------

SQL> CONNECT sysadmin_vpd/oracle
Connected.
SYSADMIN_VPD@orcl> BEGIN
  2    DBMS_RLS.DROP_POLICY('SCOTT', 'EMP', 'EMP_DEPT_POLICY');
  3    DBMS_RLS.ADD_POLICY(
  4      object_schema     => 'SCOTT',
  5      object_name       => 'EMP',
  6      policy_name       => 'EMP_SAL_POLICY',
  7      function_schema   => 'SYSADMIN_VPD',
  8      policy_function   => 'RLS_DEPT',
  9      statement_types   => 'SELECT',
 10      sec_relevant_cols => 'SAL,COMM');
 11  END;
 12  /

PL/SQL procedure successfully completed.

-- ★ 정책 함수는 그대로 rls_dept 다. 달라진 것은 언제 술어를 붙일지에 대한 조건뿐이다.

-- ★ ENAME·DEPTNO 만 부른 조회에는 정책이 붙지 않아 10번 부서 사원도 보인다.
--    SAL 을 SELECT 목록에 넣거나 WHERE 에서 SAL·COMM 을 참조하면 술어가 붙어 5건이 된다.
--    COUNT(*) 는 컬럼을 적지 않았는데도 5건이다. 19.3 에서는 COUNT(*) 에도 정책이 붙었다.
--    행 수가 필요한 화면은 정책을 고려해 설계한다.

--------------------------------------------------------------------------------
 [5] DBA_SEC_RELEVANT_COLS 로 확인
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL policy_name FOR a16
SYS@orcl> COL sec_rel_col FOR a10
SYS@orcl> COL sec_rel_column FOR a14
SYS@orcl> COL column_option FOR a13
SYS@orcl> SELECT policy_name, sec_rel_column, column_option FROM dba_sec_relevant_cols WHERE object_owner = 'SCOTT';

POLICY_NAME      SEC_REL_COLUMN COLUMN_OPTION
---------------- -------------- -------------
EMP_SAL_POLICY   SAL            ALL_ROWS
EMP_SAL_POLICY   COMM           ALL_ROWS

2 rows selected.

-- ★ 민감 컬럼과 옵션(ALL_ROWS)이 정책별로 남는다. 기본 방식이면 COLUMN_OPTION 이 NONE 이다.

================================================================================
 정리
================================================================================

-- - sec_relevant_cols: 지정한 컬럼을 참조하는 문장에만 정책 술어가 붙는다.
-- - sec_relevant_cols_opt => DBMS_RLS.ALL_ROWS: 모든 행을 보여 주고 민감 컬럼만 NULL. SELECT 전용(ORA-28104).
-- - NULL 로 가려진 값은 집계 결과를 바꾼다.
