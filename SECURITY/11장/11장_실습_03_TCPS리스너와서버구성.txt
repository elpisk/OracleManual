================================================================================
 제11장 실습 03 — TCPS 리스너와 서버 구성
 대상    : oel7v9 orcl, oracle OS 사용자
 목표    : 별도 TNS_ADMIN 으로 TCPS 리스너(2484)와 서버 sqlnet.ora 를 구성하고,
          인증서 DN 으로 로그인할 DB 사용자를 만든다
 소요    : 약 15분
 변경    : 리스너 LISTENER_TLS 기동, DB 사용자 TLS_CLIENT (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] 서버 쪽 listener.ora 와 sqlnet.ora
--------------------------------------------------------------------------------

-- 기존 리스너와 공용 sqlnet.ora 를 바꾸지 않으려고 /home/oracle/tls/server_tns 에 따로 만든다.
-- 정적 등록의 ENVS 로 TNS_ADMIN 을 넘겨, 이 리스너가 띄우는 서버 프로세스만 이 sqlnet.ora 를 읽게 한다.

[oracle@oel7v9 ~]$ cd /home/oracle/tls/server_tns
[oracle@oel7v9 server_tns]$ printf '%s\n' 'LISTENER_TLS =' '  (DESCRIPTION_LIST =' '    (DESCRIPTION =' '      (ADDRESS = (PROTOCOL = TCPS)(HOST = oel7v9)(PORT = 2484))))' 'SID_LIST_LISTENER_TLS =' '  (SID_LIST =' '    (SID_DESC =' '      (GLOBAL_DBNAME = orcl)' '      (ORACLE_HOME = /u01/app/oracle/product/19.3.0/dbhome_1)' '      (SID_NAME = orcl)' '      (ENVS = "TNS_ADMIN=/home/oracle/tls/server_tns")))' 'WALLET_LOCATION = (SOURCE = (METHOD = FILE)(METHOD_DATA = (DIRECTORY = /home/oracle/tls/server_wallet)))' 'SSL_CLIENT_AUTHENTICATION = FALSE' > listener.ora
[oracle@oel7v9 server_tns]$ printf '%s\n' 'WALLET_LOCATION = (SOURCE = (METHOD = FILE)(METHOD_DATA = (DIRECTORY = /home/oracle/tls/server_wallet)))' 'SQLNET.AUTHENTICATION_SERVICES = (BEQ, TCPS)' 'SSL_CLIENT_AUTHENTICATION = TRUE' > sqlnet.ora
[oracle@oel7v9 server_tns]$ cat listener.ora sqlnet.ora
LISTENER_TLS =
  (DESCRIPTION_LIST =
    (DESCRIPTION =
      (ADDRESS = (PROTOCOL = TCPS)(HOST = oel7v9)(PORT = 2484))))
SID_LIST_LISTENER_TLS =
  (SID_LIST =
    (SID_DESC =
      (GLOBAL_DBNAME = orcl)
      (ORACLE_HOME = /u01/app/oracle/product/19.3.0/dbhome_1)
      (SID_NAME = orcl)
      (ENVS = "TNS_ADMIN=/home/oracle/tls/server_tns")))
WALLET_LOCATION = (SOURCE = (METHOD = FILE)(METHOD_DATA = (DIRECTORY = /home/oracle/tls/server_wallet)))
SSL_CLIENT_AUTHENTICATION = FALSE
WALLET_LOCATION = (SOURCE = (METHOD = FILE)(METHOD_DATA = (DIRECTORY = /home/oracle/tls/server_wallet)))
SQLNET.AUTHENTICATION_SERVICES = (BEQ, TCPS)
SSL_CLIENT_AUTHENTICATION = TRUE

-- ★ 리스너: TCPS 2484, 서버 지갑 위치, 리스너 단계 클라이언트 인증은 FALSE.
--    서버 sqlnet.ora: 서버 지갑, 인증 서비스 (BEQ, TCPS), SSL_CLIENT_AUTHENTICATION = TRUE(상호 TLS).

--------------------------------------------------------------------------------
 [3] 인증서 사용자 만들기
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE USER tls_client IDENTIFIED EXTERNALLY AS 'CN=tls_client,O=itwill,C=KR';

User created.

SYS@orcl> GRANT CREATE SESSION TO tls_client;

Grant succeeded.

SYS@orcl> COL username FOR a12
SYS@orcl> COL external_name FOR a32
SYS@orcl> SELECT username, authentication_type, external_name FROM dba_users WHERE username = 'TLS_CLIENT';

USERNAME     AUTHENTI EXTERNAL_NAME
------------ -------- --------------------------------
TLS_CLIENT   EXTERNAL cn=tls_client,o=itwill,c=KR

1 row selected.

SYS@orcl> SHOW PARAMETER os_authent_prefix

NAME                                 TYPE        VALUE
------------------------------------ ----------- ------------------------------
os_authent_prefix                    string      ops$
SYS@orcl> SHOW PARAMETER remote_os_authent

NAME                                 TYPE        VALUE
------------------------------------ ----------- ------------------------------
remote_os_authent                    boolean     FALSE

-- ★ AUTHENTICATION_TYPE 은 EXTERNAL. EXTERNAL_NAME 은 속성 이름이 소문자(cn=, o=, c=)로 저장됐다.
--    원본 자료는 이 단계에서 OS_AUTHENT_PREFIX 를 비우고 재기동하지만, 여기서는 ops$ 그대로 두었다.
--    04 에서 이 상태 그대로 인증서 로그인이 되는지 확인한다.

-- - TCPS 리스너도 지갑 위치가 필요하다. 정적 등록 ENVS 로 서버 프로세스의 TNS_ADMIN 을 분리했다.
-- - IDENTIFIED EXTERNALLY AS 'DN' 사용자는 비밀번호가 없다.
