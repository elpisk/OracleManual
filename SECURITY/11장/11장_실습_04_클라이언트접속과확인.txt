================================================================================
 제11장 실습 04 — 클라이언트 접속과 확인
 대상    : oel7v9 orcl, oracle OS 사용자 (클라이언트 TNS_ADMIN)
 목표    : 클라이언트 sqlnet.ora·tnsnames.ora 를 구성해 인증서로 로그인하고,
          접속이 TCPS 인지, 인증 방식이 무엇인지 세션 안과 서버 쪽에서 확인한다
 소요    : 약 20분
 변경    : /home/oracle/tls/client_tns 설정
================================================================================

--------------------------------------------------------------------------------
 [1] 오류 재현 — 클라이언트 지갑 위치 없이 접속
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ cd /home/oracle/tls/client_tns
[oracle@oel7v9 client_tns]$ printf '%s\n' 'SQLNET.AUTHENTICATION_SERVICES = (TCPS)' 'SSL_CLIENT_AUTHENTICATION = TRUE' > sqlnet.ora
[oracle@oel7v9 client_tns]$ printf '%s\n' 'ORCL_TLS =' '  (DESCRIPTION =' '    (ADDRESS = (PROTOCOL = TCPS)(HOST = oel7v9)(PORT = 2484))' '    (CONNECT_DATA = (SERVER = DEDICATED)(SERVICE_NAME = orcl)))' > tnsnames.ora
[oracle@oel7v9 client_tns]$ cat sqlnet.ora
SQLNET.AUTHENTICATION_SERVICES = (TCPS)
SSL_CLIENT_AUTHENTICATION = TRUE
[oracle@oel7v9 client_tns]$ export TNS_ADMIN=/home/oracle/tls/client_tns
[oracle@oel7v9 client_tns]$ echo "SELECT SUBSTR(SYS_CONTEXT('USERENV','SESSION_USER'),1,12) usr, SUBSTR(SYS_CONTEXT('USERENV','NETWORK_PROTOCOL'),1,6) proto, SUBSTR(SYS_CONTEXT('USERENV','AUTHENTICATION_METHOD'),1,10) auth FROM dual;" | sqlplus -s -L /@orcl_tls
ERROR:
ORA-28759: failure to open file


SP2-0751: Unable to connect to Oracle.  Exiting SQL*Plus

-- ★ ORA-28759: failure to open file. TCPS 접속에는 클라이언트 지갑이 필요한데 sqlnet.ora 에 WALLET_LOCATION 이 없다.
--    클라이언트는 기본 위치에서 지갑을 찾다가 열지 못했다.

--------------------------------------------------------------------------------
 [2] 조치 — WALLET_LOCATION 추가
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ cd /home/oracle/tls/client_tns
[oracle@oel7v9 client_tns]$ printf '%s\n' 'WALLET_LOCATION = (SOURCE = (METHOD = FILE)(METHOD_DATA = (DIRECTORY = /home/oracle/tls/client_wallet)))' >> sqlnet.ora
[oracle@oel7v9 client_tns]$ cat sqlnet.ora
SQLNET.AUTHENTICATION_SERVICES = (TCPS)
SSL_CLIENT_AUTHENTICATION = TRUE
WALLET_LOCATION = (SOURCE = (METHOD = FILE)(METHOD_DATA = (DIRECTORY = /home/oracle/tls/client_wallet)))
[oracle@oel7v9 client_tns]$ export TNS_ADMIN=/home/oracle/tls/client_tns
[oracle@oel7v9 client_tns]$ echo "SELECT SUBSTR(SYS_CONTEXT('USERENV','SESSION_USER'),1,12) usr, SUBSTR(SYS_CONTEXT('USERENV','NETWORK_PROTOCOL'),1,6) proto, SUBSTR(SYS_CONTEXT('USERENV','AUTHENTICATION_METHOD'),1,10) auth FROM dual;" | sqlplus -s -L /@orcl_tls

USR						 PROTO
------------------------------------------------ ------------------------
AUTH
----------------------------------------
TLS_CLIENT					 tcps
SSL

-- ★ 사용자 이름·비밀번호 없이 /@orcl_tls 로 TLS_CLIENT 가 로그인했다.
--    NETWORK_PROTOCOL = tcps, AUTHENTICATION_METHOD = SSL.
--    원본 자료의 OS_AUTHENT_PREFIX 변경과 재기동 없이 인증서 로그인이 된다.

--------------------------------------------------------------------------------
 [3] 비밀번호 사용자로 TCPS 와 TCP 비교
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ export TNS_ADMIN=/home/oracle/tls/client_tns
[oracle@oel7v9 ~]$ echo "SELECT SUBSTR(SYS_CONTEXT('USERENV','SESSION_USER'),1,12) usr, SUBSTR(SYS_CONTEXT('USERENV','NETWORK_PROTOCOL'),1,6) proto, SUBSTR(SYS_CONTEXT('USERENV','AUTHENTICATION_METHOD'),1,10) auth FROM dual;" | sqlplus -s -L scott/tiger@orcl_tls

USR						 PROTO
------------------------------------------------ ------------------------
AUTH
----------------------------------------
SCOTT						 tcps
PASSWORD


[oracle@oel7v9 ~]$ echo "SELECT SUBSTR(SYS_CONTEXT('USERENV','SESSION_USER'),1,12) usr, SUBSTR(SYS_CONTEXT('USERENV','NETWORK_PROTOCOL'),1,6) proto, SUBSTR(SYS_CONTEXT('USERENV','AUTHENTICATION_METHOD'),1,10) auth FROM dual;" | sqlplus -s -L scott/tiger@oel7v9:1521/orcl

USR						 PROTO
------------------------------------------------ ------------------------
AUTH
----------------------------------------
SCOTT						 tcp
PASSWORD

-- ★ 같은 SCOTT 이 TCPS 별칭으로 들어오면 tcps/PASSWORD, Easy Connect(1521)로 들어오면 tcp/PASSWORD.
--    TLS 는 인증 방식과 별개로 통신 구간을 암호화한다. 비밀번호 사용자도 TCPS 로 붙으면 암호화된다.

--------------------------------------------------------------------------------
 [4] 인증서 세션 하나를 10초 유지
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ export TNS_ADMIN=/home/oracle/tls/client_tns
[oracle@oel7v9 ~]$ nohup sh -c "echo 'EXEC DBMS_SESSION.SLEEP(10)' | sqlplus -s -L /@orcl_tls" >/dev/null 2>&1 &
[oracle@oel7v9 ~]$ sleep 3

-- ★ 서버 쪽에서 볼 세션을 백그라운드로 잠시 붙여 둔다.

--------------------------------------------------------------------------------
 [5] 서버에서 본 세션
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> COL username FOR a10
SYS@orcl> COL authentication_type FOR a8
SYS@orcl> COL banner FOR a62
SYS@orcl> SELECT s.username, i.authentication_type, SUBSTR(i.network_service_banner, 1, 62) banner
  2    FROM v$session s JOIN v$session_connect_info i ON s.sid = i.sid AND s.serial# = i.serial#
  3   WHERE s.username = 'TLS_CLIENT';

USERNAME   AUTHENTI BANNER

TLS_CLIENT GLOBAL

TLS_CLIENT GLOBAL   Encryption service for Linux: Version 19.0.0.0.0 - Production
TLS_CLIENT GLOBAL   Crypto-checksumming service for Linux: Version 19.0.0.0.0 - Pr

-- ★ V$SESSION_CONNECT_INFO 의 AUTHENTICATION_TYPE 이 GLOBAL 로 나왔다.
--    배너에는 인증·암호화·체크섬 서비스가 보이지만 "TLS" 나 "TCPS" 라는 글자는 없다.
--    다른 세션의 TLS 여부는 배너 문구보다 AUTHENTICATION_TYPE 과 감사 기록으로 판단한다.

--------------------------------------------------------------------------------
 [6] 인증서 사용자의 DN 확인
--------------------------------------------------------------------------------

[oracle@oel7v9 ~]$ export TNS_ADMIN=/home/oracle/tls/client_tns
[oracle@oel7v9 ~]$ echo "SELECT SUBSTR(SYS_CONTEXT('USERENV','AUTHENTICATED_IDENTITY'),1,30) idn, SUBSTR(SYS_CONTEXT('USERENV','ENTERPRISE_IDENTITY'),1,30) eid FROM dual;" | sqlplus -s -L /@orcl_tls

IDN
--------------------------------------------------------------------------------
EID
--------------------------------------------------------------------------------
CN=tls_client,O=itwill,C=KR
cn=tls_client,o=itwill,c=KR

-- ★ AUTHENTICATED_IDENTITY 는 인증서에 적힌 그대로의 DN, ENTERPRISE_IDENTITY 는 DB 에 저장된 형식(소문자 속성 이름)이다.

-- - 클라이언트 지갑 위치가 없으면 ORA-28759.
-- - 확인은 NETWORK_PROTOCOL(tcps), AUTHENTICATION_METHOD(SSL), AUTHENTICATED_IDENTITY(DN).
-- - 서버 쪽에서는 V$SESSION_CONNECT_INFO.AUTHENTICATION_TYPE 이 인증서 로그인에서 GLOBAL 이었다.
