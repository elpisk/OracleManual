--------------------------------------------------------------------------------
 [3] 조치 — 철거
--------------------------------------------------------------------------------

-- 구버전 ACL 은 DROP_ACL 로 먼저 지우고, 남은 ACE 는 DBA_HOST_ACES 를 읽어 그대로 REMOVE_HOST_ACE 한다.

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> EXEC DBMS_NETWORK_ACL_ADMIN.DROP_ACL('acl_legacy.xml')

PL/SQL procedure successfully completed.

SYS@orcl> BEGIN
  2    FOR r IN (SELECT host, lower_port, upper_port, principal, privilege, grant_type
  3                FROM dba_host_aces WHERE principal LIKE 'ACL%') LOOP
  4      DBMS_NETWORK_ACL_ADMIN.REMOVE_HOST_ACE(
  5        host             => r.host,
  6        lower_port       => r.lower_port,
  7        upper_port       => r.upper_port,
  8        ace              => xs$ace_type(privilege_list => xs$name_list(r.privilege),
  9                                        principal_name => r.principal,
 10                                        principal_type => xs_acl.ptype_db,
 11                                        granted        => (r.grant_type = 'GRANT')),
 12        remove_empty_acl => TRUE);
 13    END LOOP;
 14  END;
 15  /

PL/SQL procedure successfully completed.

SYS@orcl> DROP USER acl_u1 CASCADE;

User dropped.

SYS@orcl> DROP USER acl_u2 CASCADE;

User dropped.

SYS@orcl> DROP USER acl_u3 CASCADE;

User dropped.

SYS@orcl> DROP ROLE acl_web_role;

Role dropped.

-- ★ remove_empty_acl => TRUE 라 ACE 가 빠진 ACL 도 함께 지워진다. 사용자·역할도 삭제했다.
