================================================================================
 제9장 실습 03 — AES256 CBC PKCS5 와 RAW 저장
 대상    : oel7v9 orcl
 목표    : AES256 + CBC + PKCS5 조합으로 암호화하고, 키 길이·패딩·IV 가 결과에 주는 영향을 본다.
          암호문을 RAW 컬럼에 저장하고 키가 다르면 어떻게 실패하는지 확인한다
 소요    : 약 25분
 변경    : cr_emp 에 sal_enc, sal_iv 컬럼, 임시 키 테이블 cr_tmp_key
================================================================================

--------------------------------------------------------------------------------
 [1] 오류 재현 — 키 길이가 맞지 않으면
--------------------------------------------------------------------------------

SQL> CONNECT cr_owner/oracle

  2    v_key RAW(32) := DBMS_CRYPTO.RANDOMBYTES(16);
  3    v_enc RAW(2000);
  4  BEGIN
  5    v_enc := DBMS_CRYPTO.ENCRYPT(
  6               src => UTL_I18N.STRING_TO_RAW('King', 'AL32UTF8'),
  7               typ => DBMS_CRYPTO.ENCRYPT_AES256 + DBMS_CRYPTO.CHAIN_CBC + DBMS_CRYPTO.PAD_PKCS5,
  8               key => v_key);

 10  END;

-- ★ AES256 에 16바이트 키를 넣어 ORA-28234: key length too short 가 났다.
--    AES 키는 128, 192, 256 비트(16, 24, 32 바이트) 중 알고리즘에 맞는 길이여야 한다.

--------------------------------------------------------------------------------
 [2] 조치 — 32바이트 키로 다시, 출력 길이
--------------------------------------------------------------------------------

SQL> CONNECT cr_owner/oracle
Connected.
CR_OWNER@orcl> SET SERVEROUTPUT ON
CR_OWNER@orcl> DECLARE
  2    v_typ PLS_INTEGER := DBMS_CRYPTO.ENCRYPT_AES256 + DBMS_CRYPTO.CHAIN_CBC + DBMS_CRYPTO.PAD_PKCS5;
  3    v_key RAW(32) := DBMS_CRYPTO.RANDOMBYTES(32);
  4    v_enc RAW(2000);
  5  BEGIN
  6    FOR r IN (SELECT 'King' s FROM dual UNION ALL SELECT 'Abcdefghijklmno' FROM dual
  7              UNION ALL SELECT 'Abcdefghijklmnop' FROM dual) LOOP
  8      v_enc := DBMS_CRYPTO.ENCRYPT(UTL_I18N.STRING_TO_RAW(r.s, 'AL32UTF8'), v_typ, v_key);
  9      DBMS_OUTPUT.PUT_LINE(RPAD(r.s, 17) || LENGTH(r.s) || ' bytes -> ' || UTL_RAW.LENGTH(v_enc) || ' bytes');
 10    END LOOP;
 11  END;
 12  /
King             4 bytes -> 16 bytes
Abcdefghijklmno  15 bytes -> 16 bytes
Abcdefghijklmnop 16 bytes -> 32 bytes

PL/SQL procedure successfully completed.

-- ★ 4바이트와 15바이트 평문은 16바이트, 16바이트 평문은 32바이트 암호문이 됐다.
--    PKCS5 패딩은 블록(AES 16바이트)을 채우고, 딱 맞으면 한 블록을 더 붙인다.
--    저장 컬럼 크기는 (평문 바이트 / 16 의 몫 + 1) x 16 으로 잡는다. IV 를 함께 두면 16바이트를 더한다.

--------------------------------------------------------------------------------
 [3] 같은 평문, 같은 키, IV 유무
--------------------------------------------------------------------------------

SQL> CONNECT cr_owner/oracle
Connected.
CR_OWNER@orcl> SET SERVEROUTPUT ON
CR_OWNER@orcl> DECLARE
  2    v_typ PLS_INTEGER := DBMS_CRYPTO.ENCRYPT_AES256 + DBMS_CRYPTO.CHAIN_CBC + DBMS_CRYPTO.PAD_PKCS5;
  3    v_key RAW(32) := DBMS_CRYPTO.RANDOMBYTES(32);
  4    v_src RAW(100) := UTL_I18N.STRING_TO_RAW('24000', 'AL32UTF8');

-- ★ IV 를 주지 않으면 같은 평문은 항상 같은 암호문이다. 급여가 같은 사원끼리 암호문이 같아
--    값을 몰라도 "같다"는 사실이 드러난다. IV 를 매번 RANDOMBYTES(16) 로 만들면 결과가 달라진다.

--------------------------------------------------------------------------------
 [4] RAW 컬럼에 저장하고 되돌리기
--------------------------------------------------------------------------------

-- 키를 임시 테이블에 둔 것은 이 실습을 단순하게 하려는 것이다. 04 에서 계정을 분리한다.

SQL> CONNECT cr_owner/oracle
Connected.
CR_OWNER@orcl> ALTER TABLE cr_emp ADD (sal_enc RAW(64), sal_iv RAW(16));

CR_OWNER@orcl> CREATE TABLE cr_tmp_key (k RAW(32));

Table created.

CR_OWNER@orcl> INSERT INTO cr_tmp_key VALUES (DBMS_CRYPTO.RANDOMBYTES(32));

1 row created.

CR_OWNER@orcl> COMMIT;

Commit complete.

CR_OWNER@orcl> DECLARE
  2    v_typ PLS_INTEGER := DBMS_CRYPTO.ENCRYPT_AES256 + DBMS_CRYPTO.CHAIN_CBC + DBMS_CRYPTO.PAD_PKCS5;
  3    v_key RAW(32);
  4    v_iv  RAW(16);
  5  BEGIN
  6    SELECT k INTO v_key FROM cr_tmp_key;
  7    FOR r IN (SELECT employee_id, salary FROM cr_emp) LOOP
  8      v_iv := DBMS_CRYPTO.RANDOMBYTES(16);
  9      UPDATE cr_emp
 10         SET sal_iv  = v_iv,
 11             sal_enc = DBMS_CRYPTO.ENCRYPT(UTL_I18N.STRING_TO_RAW(TO_CHAR(r.salary), 'AL32UTF8'), v_typ, v_key, v_iv)
 12       WHERE employee_id = r.employee_id;
 13    END LOOP;
 14    COMMIT;
 15  END;
 16  /

 정리

-- - 권장 조합: ENCRYPT_AES256 + CHAIN_CBC + PAD_PKCS5, 32바이트 키, 값마다 새 16바이트 IV.
-- - 키 길이가 틀리면 ORA-28234, 키·IV 가 틀리면 대개 ORA-28817.
-- - 암호문 길이는 블록 단위로 늘어난다. 컬럼 크기를 미리 계산한다.
