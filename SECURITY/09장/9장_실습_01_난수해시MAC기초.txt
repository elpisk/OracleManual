 제9장 실습 01 — 난수·해시·MAC 기초
 대상    : oel7v9 orcl (non-CDB, 19.3, AL32UTF8)
 목표    : DBMS_CRYPTO 실행 권한을 주고 RANDOMBYTES, HASH, MAC 의 결과 형태를 확인한다.
          VARCHAR2 를 바로 넘기거나 SQL 에서 패키지 상수를 쓸 때 나는 오류를 구분한다
 소요    : 약 20분
 변경    : cr_owner 사용자 생성 (05 에서 철거)
================================================================================

--------------------------------------------------------------------------------
 [1] 실습 계정과 DBMS_CRYPTO 권한 확인
--------------------------------------------------------------------------------

SQL> CONNECT / AS SYSDBA
Connected.
SYS@orcl> CREATE USER cr_owner IDENTIFIED BY oracle QUOTA 10M ON users;

User created.

SYS@orcl> GRANT CREATE SESSION, CREATE TABLE, CREATE PROCEDURE TO cr_owner;

Grant succeeded.

SYS@orcl> COL grantee FOR a12
SYS@orcl> SELECT grantee, privilege FROM dba_tab_privs WHERE table_name = 'DBMS_CRYPTO';

no rows selected

-- ★ DBMS_CRYPTO 에 대한 실행 권한은 PUBLIC 을 포함해 누구에게도 부여돼 있지 않다.
--    쓰려는 계정마다 SYS 가 직접 EXECUTE 를 줘야 한다.

--------------------------------------------------------------------------------
 [2] 오류 재현 — 권한 없이 호출
--------------------------------------------------------------------------------

SQL> CONNECT cr_owner/oracle
Connected.
CR_OWNER@orcl> SELECT DBMS_CRYPTO.RANDOMBYTES(16) FROM dual;
SELECT DBMS_CRYPTO.RANDOMBYTES(16) FROM dual
       *
ERROR at line 1:
ORA-00904: "DBMS_CRYPTO"."RANDOMBYTES": invalid identifier

-- ★ SQL 문 안에서 권한 없는 패키지 함수를 부르면 ORA-00904: invalid identifier 로 나온다.
--    PL/SQL 블록이었다면 PLS-00201 이다. 둘 다 "없다"가 아니라 "보이지 않는다"는 뜻이다.

--------------------------------------------------------------------------------

--    32바이트 난수는 64글자가 된다. 키와 IV 는 이 함수로 만든다.

--------------------------------------------------------------------------------
 [4] 오류 재현 — 문자열을 그대로 해시
--------------------------------------------------------------------------------

SQL> CONNECT cr_owner/oracle
Connected.
CR_OWNER@orcl> SELECT DBMS_CRYPTO.HASH('oracle', 4) FROM dual;
SELECT DBMS_CRYPTO.HASH('oracle', 4) FROM dual
                        *
ERROR at line 1:
ORA-01465: invalid hex number


CR_OWNER@orcl> SELECT DBMS_CRYPTO.HASH(UTL_I18N.STRING_TO_RAW('oracle', 'AL32UTF8'), DBMS_CRYPTO.HASH_SH256) FROM dual;
SELECT DBMS_CRYPTO.HASH(UTL_I18N.STRING_TO_RAW('oracle', 'AL32UTF8'), DBMS_CRYPTO.HASH_SH256) FROM dual
                                                                      *
ERROR at line 1:
ORA-06553: PLS-221: 'HASH_SH256' is not a procedure or is undefined

-- ★ 'oracle' 을 바로 넘기면 RAW 인자로 암묵 변환(HEXTORAW)을 시도하다 ORA-01465 가 난다.
--    두 번째는 SQL 문에서 패키지 상수 HASH_SH256 을 썼기 때문에 PLS-221 이다.
--    패키지 상수는 PL/SQL 안에서만 쓸 수 있다. SQL 에서는 숫자 값(HASH_SH256 = 4)을 쓴다.

--------------------------------------------------------------------------------
 [6] PL/SQL 안에서 상수 쓰기와 MAC
--------------------------------------------------------------------------------

SQL> CONNECT cr_owner/oracle
Connected.
CR_OWNER@orcl> SET SERVEROUTPUT ON
CR_OWNER@orcl> DECLARE
  2    v_src RAW(100) := UTL_I18N.STRING_TO_RAW('salary=24000', 'AL32UTF8');
  3    v_k1  RAW(32)  := UTL_I18N.STRING_TO_RAW('mac-key-one', 'AL32UTF8');
  4    v_k2  RAW(32)  := UTL_I18N.STRING_TO_RAW('mac-key-two', 'AL32UTF8');
  5  BEGIN
  6    DBMS_OUTPUT.PUT_LINE('HASH  : ' || RAWTOHEX(DBMS_CRYPTO.HASH(v_src, DBMS_CRYPTO.HASH_SH256)));
  7    DBMS_OUTPUT.PUT_LINE('MAC k1: ' || RAWTOHEX(DBMS_CRYPTO.MAC(v_src, DBMS_CRYPTO.HMAC_SH256, v_k1)));
  8    DBMS_OUTPUT.PUT_LINE('MAC k1: ' || RAWTOHEX(DBMS_CRYPTO.MAC(v_src, DBMS_CRYPTO.HMAC_SH256, v_k1)));
  9    DBMS_OUTPUT.PUT_LINE('MAC k2: ' || RAWTOHEX(DBMS_CRYPTO.MAC(v_src, DBMS_CRYPTO.HMAC_SH256, v_k2)));
 10  END;
 11  /
HASH  : 01723D210A02A50FC12147DCEE1E6CAA0231011C8E81903C680377C5AB277BFA
MAC k1: 79262ADF56E9A290C49F019828567312D7C521607DF02EE338EEAB3B74D4026C
MAC k1: 79262ADF56E9A290C49F019828567312D7C521607DF02EE338EEAB3B74D4026C
MAC k2: C33A7F99979885DDE80D91B6730DE6116AFA8F719A40E618B5D761520133C1A6

PL/SQL procedure successfully completed.

-- ★ PL/SQL 블록 안에서는 DBMS_CRYPTO.HASH_SH256, HMAC_SH256 상수를 그대로 쓸 수 있다.
--    같은 키로 만든 MAC 은 같고, 키가 바뀌면 완전히 다르다.
--    해시는 누구나 다시 계산할 수 있지만 MAC 은 키를 가진 쪽만 만들고 검증할 수 있다.

================================================================================
 정리
================================================================================

-- - DBMS_CRYPTO 는 기본 권한이 없다. 필요한 계정에만 EXECUTE 를 준다.
-- - 입력은 RAW 다. 문자열은 UTL_I18N.STRING_TO_RAW(s, 'AL32UTF8') 로 바꾼다(아니면 ORA-01465).
-- - 패키지 상수는 PL/SQL 전용이다. SQL 문에서는 PLS-221.
-- - 해시는 무결성 확인, MAC 은 키를 건 무결성 확인과 검색 키에 쓴다.
