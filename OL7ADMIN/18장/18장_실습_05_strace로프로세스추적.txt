================================================================================
 제18장 실습 05 — strace 로 프로세스가 하는 일과 기다리는 것 보기
 대상   : ol7a
 목표   : strace 로 명령 하나의 시스템 호출을 처음부터 따라가고(-o, -f, -e,
          -c, -tt, -T), 이미 돌고 있는 프로세스에 -p 로 붙어 무엇을 기다리는지
          확인한다. 설정 파일을 못 찾는 스크립트, 잠금을 기다리는 작업, 응답
          없는 주소에 연결하는 명령을 각각 시스템 호출로 진단한다
 배경   : 프로그램이 파일·네트워크·프로세스를 다루려면 반드시 커널에 시스템
          호출을 한다. strace 는 ptrace 로 대상에 붙어 그 호출과 반환값을 찍는다.
          오류 메시지가 뭉뚱그려져 있어도 호출 단위로 보면 어떤 경로에서 ENOENT
          가 났는지, 어느 호출에서 멈춰 있는지가 드러난다. 붙는 동안 대상은
          느려지므로 운영 서버에서는 짧게, 필터를 걸어 쓴다(본문 18.7)
 소요   : 약 25분
 선행   : ol7-base 스냅샷
 계정   : root, student
 주의   : 없음. 만든 스크립트·파일은 끝에서 지운다
 되돌림 : 시험 프로세스 0개, /usr/local/bin/appctl·/etc/appctl·/tmp 시험 파일 삭제.
          확인은 pgrep -c flock · ls /usr/local/bin /etc/appctl

44 /tmp/s1.txt

 [3] -e trace=file — 설정 파일을 못 찾는 스크립트
--------------------------------------------------------------------------------
-- 오류 메시지가 어느 파일인지 말해 주지 않는 스크립트를 만든다.
[root@ol7a ~]# cat > /usr/local/bin/appctl <<'EOF'
#!/bin/bash
for f in /etc/appctl/appctl.conf /opt/appctl/etc/appctl.conf; do
    [ -r "$f" ] && { . "$f"; break; }
done
[ -z "$PORT" ] && { echo "appctl: configuration error"; exit 1; }
echo "appctl: listening on port $PORT"
EOF
[root@ol7a ~]# chmod 755 /usr/local/bin/appctl; appctl; echo "exit=$?"
appctl: configuration error
exit=1
-- 오류 메시지는 configuration error 한 줄뿐. 어느 파일을 찾았는지
-- 말해 주지 않는다
-- trace=file : 경로 이름을 인자로 받는
-- 호출만(open·stat·access·execve 등).
[root@ol7a ~]# strace -e trace=file -o /tmp/f.txt appctl; grep -v -E '\.so|locale|/dev/tty|nsswitch|ld.so' /tmp/f.txt

-- 2초마다 파일에 시각을 적는 작업을 띄운다.
[root@ol7a ~]# nohup sh -c 'while :; do date >> /tmp/tick.log; sleep 2; done' >/dev/null 2>&1 </dev/null &

-- 그 뜻이다

 [6] 멈춘 작업 — 잠금을 기다린다
--------------------------------------------------------------------------------
-- 배치 작업 둘이 같은 잠금 파일을 쓴다. 먼저 잡은 쪽이 오래
-- 걸리면 뒤쪽은 그냥 멈춰 보인다.
[root@ol7a ~]# nohup flock /tmp/app.lock sleep 300 >/dev/null 2>&1 </dev/null &
[root@ol7a ~]# nohup flock /tmp/app.lock echo got-lock >/tmp/waiter.out 2>&1 </dev/null &
[root@ol7a ~]# sleep 1; ps -o pid,stat,etime,wchan:20,cmd -C flock
  PID STAT     ELAPSED WCHAN                CMD
 2327 S          00:01 do_wait              flock /tmp/app.lock sleep 300
 2346 S          00:01 locks_lock_inode_wai flock /tmp/app.lock echo got-lock
-- 2327 은 잠금을 잡고 자식 sleep 300 을 기다린다(do_wait)
-- 2346 은 같은 잠금을 기다린다. WCHAN locks_lock_inode_wait(열 폭
-- 때문에 잘림). STAT 은 S, CPU 는 쓰지 않는다
-- 뒤쪽 flock 에 3초 붙는다. -y : 파일 디스크립터 번호 옆에 경로를
-- 보여 준다.
[root@ol7a ~]# W=$(pgrep -f 'flock /tmp/app.lock echo'); timeout 3 strace -y -p $W
strace: Process 2346 attached
flock(3</tmp/app.lock>, LOCK_EXstrace: Process 2346 detached
 <detached ...>
-- 증상   : 배치 작업(2346)이 끝나지 않고 CPU 도 쓰지 않는다
-- 원인   : strace 에 flock(3</tmp/app.lock>, LOCK_EX 까지만
--          찍히고 반환값이 없다. -y 덕분에
--          fd 3 이 /tmp/app.lock 인 것도 보인다. 다른 프로세스가 그 잠금을 들고 있다
-- 조치   : 잠금 주인을 찾아 끝나게 한다(아래 /proc/locks,
--          /proc/PID/fd)
-- 재확인 : 주인이 끝나자 waiter.out 에 got-lock 이 찍혔다
-- 3초 뒤 timeout 으로 strace 가 떨어지며 detached 가 호출 중간에
-- 끼어들었다
-- 같은 잠금을 누가 들고 있는지: /proc/locks 와 /proc/PID/fd.
-- (fuser 를 주는 psmisc 는 기준 상태에 없다)

/proc/2346/fd/3 -> /tmp/app.lock
-- /proc/locks 의 7번 줄: FLOCK ADVISORY WRITE, PID 2327 이 inode
-- 67256443 을 잡고 있다
-- 7: -> 로 시작하는 줄이 그 잠금을 기다리는 2346 이다
-- 장치 fc:00 은 dm-0(ol-root), 마지막 숫자가 inode 다. ls -i
-- /tmp/app.lock 으로 맞춰 볼 수 있다
-- /proc/PID/fd 쪽: 2327(flock), 2328(그 자식 sleep 300),
-- 2346(대기) 셋이 app.lock 을 열고 있다
-- 2328 이 함께 보이는 것이 다음 단계의 함정이다
-- strace 없이도 기다리는 곳을 본다. wchan 은 잠든 커널 함수,
-- stack 은 커널 호출 경로(root 만)
[root@ol7a ~]# W=$(pgrep -f 'flock /tmp/app.lock echo'); cat /proc/$W/wchan; echo; cat /proc/$W/stack | head -6
locks_lock_inode_wait
[<0>] locks_lock_inode_wait+0x85/0x16e
[<0>] __x64_sys_flock+0xfc/0x105
[<0>] x64_sys_call+0x22f5/0x237b
[<0>] do_syscall_64+0x58/0x1d9
[<0>] entry_SYSCALL_64_after_hwframe+0x17b/0x0
-- wchan 은 locks_lock_inode_wait. stack 의 아래에서 위로 읽으면
-- 시스템 호출 → __x64_sys_flock → 잠금 대기
-- strace 처럼 붙지 않으므로 대상에 영향이 없다. 운영 중인 DB
-- 프로세스에도 안전하게 쓸 수 있다
-- 앞의 flock 을 끝낸다. 기다리던 쪽이 바로 잠금을 얻는지 본다.
[root@ol7a ~]# pkill -f 'flock /tmp/app.lock sleep'; sleep 1; ps -o pid,ppid,stat,cmd -C flock,sleep; cat /tmp/waiter.out
  PID  PPID STAT CMD
 2328     1 S    sleep 300
 2346     1 S    flock /tmp/app.lock echo got-lock
 2453  2241 S    sleep 2
-- flock(2327)을 죽였는데 waiter.out 은 비어 있고 2346 은 계속
-- 기다린다
-- sleep 300(2328)이 부모를 잃고 PPID 1 로 남았다. flock 이 연
-- 잠금 fd 를 자식이 물려받아 여전히 들고 있다
-- flock(1) 은 -o 를 주지 않으면 잠금 fd 를 실행할 명령에 넘긴다
[root@ol7a ~]# pkill -f '^sleep 300'; sleep 1; cat /tmp/waiter.out; pgrep -c flock
got-lock
0
-- sleep 300 까지 끝내자 got-lock 이 찍히고 flock 은 0개
-- 잠금으로 멈춘 작업을 풀 때는 /proc/PID/fd 로 잠금 파일을 연
-- 프로세스 전부를 확인한다

--          ptrace(PTRACE_SEIZE, 1):
--          Operation not permitted
-- 원인   : PID 1(systemd)은 root 소유다. 일반 사용자는 다른
--          사용자의 프로세스에 ptrace 로 붙을 수 없다
-- 조치   : root(또는 sudo)로 추적한다. 일반 사용자에게는 자기
--          프로세스만 허용된다
-- 재확인 : 아래에서 student 가 자기 sleep 에 붙자 attached 가
--          나왔다
-- student 가 자기 프로세스에 붙는다. 자기가 띄운 프로세스라도
-- 자식이 아니면 ptrace_scope 에 따라 다르다.
[student@ol7a ~]$ nohup sleep 120 >/dev/null 2>&1 </dev/null &
[student@ol7a ~]$ P=$(pgrep -u student -x sleep); timeout 2 strace -p $P; echo "exit=$?"

restart_syscall(<... resuming interrupted read ...>strace: Process 2571 detached
 <detached ...>
exit=124
-- 같은 student 의 sleep 에는 붙었다. ptrace_scope 0 이라 자식이
-- 아니어도 된다
-- restart_syscall : 시그널로 끊긴 sleep(nanosleep) 을 커널이
-- 이어서 하는 호출. 2초 뒤 timeout 으로 떨어짐
-- exit=124 는 timeout 이 명령을 끝냈다는 코드다

 [9] 출력 폭주 — 필터와 -o
--------------------------------------------------------------------------------
-- 아무 필터 없이 -f 로 find 를 추적하면 몇 줄이 나오는지 센다.
-- 화면으로 보냈다면 이 줄이 다 찍힌다.
[root@ol7a ~]# strace -f find /usr/lib64 -name '*.so' 2>&1 >/dev/null | wc -l
5411
-- 필터 없는 -f 추적은 5411 줄. find 의 출력은 >/dev/null 로
-- 버리고 추적(표준 오류)만 셌다
-- 증상   : strace 출력이 화면을 수천 줄 넘겨 원하는 줄을 찾을 수
--          없다(5411 줄)
-- 원인   : 모든 호출을 찍었다. fcntl·close·fstat 같은 일상 호출이
--          대부분이다
-- 조치   : -e trace=로 호출을 고르고 -o 로 파일에 쓴다. 집계만
--          필요하면 -c
-- 재확인 : 아래 open·openat 만 고르자 299 줄, -c 는 표 하나로
--          줄었다
-- 알고 싶은 것이 "어떤 디렉터리를 여는가" 뿐이면 호출을 고르고
-- 파일로 보낸다.
[root@ol7a ~]# strace -f -e trace=openat,open -o /tmp/o.txt find /usr/lib64 -name '*.so' >/dev/null; wc -l /tmp/o.txt; head -3 /tmp/o.txt
299 /tmp/o.txt
2651  open("/etc/ld.so.cache", O_RDONLY|O_CLOEXEC) = 3
2651  open("/lib64/libselinux.so.1", O_RDONLY|O_CLOEXEC) = 3
2651  open("/lib64/libm.so.6", O_RDONLY|O_CLOEXEC) = 3
-- 299 줄 중 앞은 라이브러리, 뒤가 find 가 연 디렉터리(openat)다
-- 시간을 어디에 쓰는지만 알면 되면 -c 로 줄인다.
[root@ol7a ~]# strace -f -c -o /tmp/oc.txt find /usr/lib64 -name '*.so' >/dev/null; head -8 /tmp/oc.txt
% time     seconds  usecs/call     calls    errors syscall

 43.73    0.019637           7      2494           fcntl
 19.30    0.008668           7      1108           close
 11.78    0.005290           9       558           newfstatat
 11.45    0.005143           9       558           getdents
  6.56    0.002945          10       281           openat
  5.51    0.002473           8       292           fstat
-- 같은 find 의 집계. fcntl 2494 번이 시간의 43.73% 를 차지한다
