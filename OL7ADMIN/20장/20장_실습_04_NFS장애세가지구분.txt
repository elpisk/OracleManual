================================================================================
 제20장 실습 04 — NFS 장애 세 가지 구분
 대상   : ol7a (NFS 서버), ol7b (NFS 클라이언트)
 목표   : 서버 서비스 정지·내보내기 권한 제거·방화벽 차단을 하나씩 걸고, 클라이언트에서
          보이는 증상과 확인 명령의 결과로 셋을 구분한다. hard 마운트가 걸린 채로
          umount 가 안 되는 상황을 -f·-l 로 푼다
 배경   : 클라이언트에서 NFS 가 "안 된다"는 증상은 셋 다 비슷해 보인다. 확인은 바깥에서
          안으로 한다. ping(네트워크) → TCP 2049 연결(방화벽·nfsd) → rpcinfo(RPC 등록)
          → showmount(mountd·exports) → 실제 접근. 어느 층에서 처음 실패하는지가
          원인을 가른다. hard 마운트는 서버가 돌아올 때까지 재시도하므로 프로세스가
          D 상태로 멈춘다 (본문 20.6, 14장)
 소요   : 약 40분
 선행   : ol7-base, ol7b-base 스냅샷
 계정   : root
 주의   : ol7a 의 exports·방화벽(런타임)을 바꿨다가 되돌린다. 영구 설정은 건드리지 않는다
 되돌림 : ol7b 마운트 해제, ol7a 의 exports 비움·nfs-server 정지·방화벽 런타임 원복
          → findmnt -t nfs4 (빈 출력), firewall-cmd --list-services (dhcpv6-client ssh)

 [1] 정상 상태 만들기 — 서버 내보내기와 방화벽
--------------------------------------------------------------------------------

-- 14장 실습 01 과 같은 구성을 짧게 만든다.
-- 방화벽은 --permanent 없이 런타임에만 연다.
-- reload 나 재부팅이면 원래대로 돌아간다.
[root@ol7a ~]# mkdir -p /srv/nfs/share; echo "hello from ol7a" > /srv/nfs/share/readme.txt
[root@ol7a ~]# echo '/srv/nfs/share  ol7b.example.com(rw,sync)' > /etc/exports; systemctl start nfs-server; exportfs -v
/srv/nfs/share  ol7b.example.com(sync,wdelay,hide,no_subtree_check,sec=sys,rw,secure,root_squash,no_all_squash)
-- exportfs -v 는 커널이 실제로 쓰는 옵션을 모두 펼쳐 보인다.
-- 적은 것은 rw,sync 뿐인데 root_squash·secure·wdelay 등 기본값이 붙었다.
[root@ol7a ~]# firewall-cmd --add-service=nfs --add-service=mountd --add-service=rpc-bind; firewall-cmd --list-services
success
dhcpv6-client mountd nfs rpc-bind ssh
-- nfs(2049), mountd(20048),
-- rpc-bind(111) 세 서비스가 런타임에 열렸다.

 [2] 클라이언트 마운트와 점검 스크립트
--------------------------------------------------------------------------------

-- hard 는 기본값이라 옵션에 적지 않아도 된다.
-- timeo=100 은 응답 대기 10초(단위 0.1초),
-- retrans=2 는 그 뒤 재전송 두 번.
-- 기본값(TCP timeo=600)이면 "not responding" 이 몇 분 뒤에 나온다.
-- 실습 시간에 맞춰 줄였다.
[root@ol7b ~]# mkdir -p /mnt/share; mount -t nfs -o timeo=100,retrans=2 ol7a:/srv/nfs/share /mnt/share; nfsstat -m
/mnt/share from ol7a:/srv/nfs/share
 Flags: rw,relatime,vers=4.1,rsize=524288,wsize=524288,namlen=255,hard,proto=tcp,timeo=100,retrans=2,sec=sys,clientaddr=172.16.225.62,local_lock=none,addr=172.16.225.61
-- vers=4.1 : 7.9 클라이언트가 고른 최신 버전.
-- hard, timeo=100, retrans=2 : 지정한 대로다.
-- proto=tcp, addr=172.16.225.61 : 서버 주소,
-- clientaddr 는 ol7b 자신.

-- 장애마다 같은 순서로 확인하려고 점검 스크립트를 만든다.
-- 바깥 층부터 안쪽 층 순서다.
-- ping     : 네트워크(ICMP)
-- tcp2049 : bash 의 /dev/tcp 로 연결만 시도. 방화벽과 nfsd 리슨 여부를 본다
-- rpcinfo : -T tcp 로 nfs 프로그램 버전 4 에 NULL 호출. rpcbind(111)를 거친다
-- showmount: mountd 와 exports 목록(TCP 111·20048)
-- ls       : 실제 마운트 접근. 15초 뒤 KILL 로 끊는다(rc=137)
[root@ol7b ~]# cat > /root/nfschk.sh <<'EOF'
#!/bin/bash
echo "ping     : $(ping -c1 -W2 ol7a >/dev/null 2>&1 && echo ok || echo FAIL)"
r=$( (timeout 5 bash -c '</dev/tcp/ol7a/2049' && echo open) 2>&1 | tail -1); echo "tcp2049  : ${r:-no answer in 5s}"
echo "rpcinfo  : $(rpcinfo -T tcp ol7a nfs 4 2>&1 | head -1)"
echo "showmount: $(timeout 10 showmount -e ol7a 2>&1 | tr '\n' ' ')"
out=$(timeout -s KILL 15 ls /mnt/share 2>&1); rc=$?; echo "ls       : $(echo $out) rc=$rc"
EOF
[root@ol7b ~]# chmod +x /root/nfschk.sh
-- r=$( … ) 부분은 연결 결과의 마지막 줄만 남긴다.
-- 아무 응답이 없으면 no answer in 5s.

-- 정상 상태의 기준값.
-- 장애마다 이 다섯 줄과 비교한다.
[root@ol7b ~]# time /root/nfschk.sh
ping     : ok
tcp2049  : open
rpcinfo  : program 100003 version 4 ready and waiting
showmount: Export list for ol7a: /srv/nfs/share ol7b.example.com 
ls       : readme.txt rc=0
-- 다섯 층 모두 정상.
-- rpcinfo 의 100003 은 NFS 프로그램 번호다.

real    0m0.123s
user    0m0.003s
sys     0m0.011s
-- 전체가 0.1초 남짓에 끝났다.
-- 장애 때의 시간과 비교한다.

--------------------------------------------------------------------------------
 [4] 오류 흐름 — hard 마운트가 걸린 채 umount 가 안 됨
--------------------------------------------------------------------------------

-- 서버가 멈춘 동안 누군가 마운트 지점에 접근했다.
-- 백그라운드로 ls 를 띄운다.
[root@ol7b ~]# nohup ls -l /mnt/share >/tmp/ls_bg.log 2>&1 </dev/null &
-- STAT 의 D 는 인터럽트 불가 대기.
-- WCHAN 은 커널 안에서 멈춘 함수다.
[root@ol7b ~]# ps -o pid,stat,wchan:22,etime,cmd -C ls
  PID STAT WCHAN                      ELAPSED CMD
10263 D    rpc_wait_bit_killable        00:05 ls -l /mnt/share
-- 5초째 D 상태.
-- 함수 이름 rpc_wait_bit_killable 은 RPC 응답을 기다리는 중이며
-- SIGKILL 로는 끊을 수 있다는 뜻이다.
-- load average 에도 D 상태가 더해진다.

-- 이 상태에서 마운트를 떼려고 한다.
[root@ol7b ~]# timeout 30 umount /mnt/share; echo "rc=$?"
rc=124
-- 30초 동안 끝나지 않아 timeout 이 끊었다(rc=124).
-- umount 자체가 서버를 기다렸다.
-- 누가 붙잡고 있는지 본다.
-- fuser 는 마운트 지점을 쓰는 프로세스를 PID 로 보여 준다.
[root@ol7b ~]# timeout 20 fuser -vm /mnt/share; echo "rc=$?"
timeout: failed to run command ‘fuser’: No such file or directory
rc=127
-- fuser 가 없다(rc=127).
-- fuser 는 psmisc 패키지에 있다.
-- 최소 설치였던 ol7b 에는 없을 수 있다.
[root@ol7b ~]# rpm -q psmisc; yum -y -q install psmisc; rpm -q psmisc
package psmisc is not installed
psmisc-22.20-17.el7.x86_64
-- 서버가 아니라 ol7b 가 인터넷 저장소에서 받는다.
-- 진단 도구 설치에도 네트워크가 필요하다.
[root@ol7b ~]# timeout 20 fuser -vm /mnt/share; echo "rc=$?"
rc=124
-- fuser 도 20초 안에 끝나지 않았다.
-- 마운트 지점을 stat 하다 같이 멈췄다.
-- 멈춘 NFS 에서는 lsof·df 도 멈출 수 있다.
-- 진단 명령에 timeout 을 붙이는 이유다.

-- -f 는 진행 중인 RPC 를 끊고 떼려고 한다.
[root@ol7b ~]# timeout 30 umount -f /mnt/share; echo "rc=$?"; findmnt /mnt/share | tail -1
umount.nfs4: /mnt/share: device is busy
rc=16
/mnt/share ol7a:/srv/nfs/share nfs4   rw,relatime,vers=4.1,rsize=524288,wsize=524288,namlen=255,hard,proto=tcp,timeo=100,retrans=2,sec=sys,clientaddr=172.16.225.62,local_lock=none,addr=172.16.225.61
-- device is busy(rc=16).
-- ls 가 붙잡고 있어 떼지 못했다.
-- 마운트는 그대로 남았다.
-- -l(lazy)은 이름 공간에서 먼저 떼고,
-- 쓰는 프로세스가 사라질 때 정리한다.
[root@ol7b ~]# umount -l /mnt/share; echo "rc=$?"; findmnt /mnt/share; echo "findmnt rc=$?"
rc=0
findmnt rc=1
-- rc=0, findmnt 에서 사라졌다.
[root@ol7b ~]# ps -o pid,stat,wchan:22,etime,cmd -C ls; cat /tmp/ls_bg.log
  PID STAT WCHAN                      ELAPSED CMD
ls: cannot access /mnt/share: Unknown error 512
-- ls 는 이미 끝났다.
-- -f 가 끊은 RPC 가 ls 에 오류로 돌아갔다.
-- Unknown error 512 는 커널 내부 값(ERESTARTSYS)이 그대로 새어 나온 것이다.
-- 증상   : "umount" 가 30초 넘게 멈추고, "umount.nfs4: /mnt/share: device is busy" (rc=16)
-- 원인   : hard 마운트에서 서버 응답을 기다리는 ls 가 D 상태로 마운트를 붙잡고 있다
-- 조치   : umount -f 로 RPC 를 끊고, 그래도 busy 면 umount -l 로 이름 공간에서 뗀다
-- 재확인 : findmnt /mnt/share 출력 없음(rc=1), ls 프로세스 사라짐
-- 서버가 멈춘 지 1분 남짓 지났다.
-- "not responding" 이 찍혔는지 본다.
[root@ol7b ~]# dmesg -T | grep -iE 'not responding|still trying|OK$' | tail -3
[Mon Nov 16 21:01:46 2026] nfs: server ol7a not responding, still trying
-- 장애 A 시작 뒤 약 50초.
-- still trying 은 hard 마운트라서 붙는 말이다.
-- soft 마운트였다면 timed out 이 찍힌다.

 [6] 장애 B — 내보내기에서 클라이언트 제거

 [8] 장애 C — 방화벽에서 nfs 서비스 차단
--------------------------------------------------------------------------------

-- 장애 C. nfs(2049) 만 닫는다.
-- rpc-bind·mountd 는 열어 둔다.
[root@ol7a ~]# firewall-cmd --remove-service=nfs; firewall-cmd --list-services
success
dhcpv6-client mountd rpc-bind ssh
-- 런타임 목록에서 nfs 만 빠졌다.

-- 먼저 이미 붙어 있는 마운트로 점검한다.
[root@ol7b ~]# time /root/nfschk.sh
ping     : ok
tcp2049  : bash: /dev/tcp/ol7a/2049: No route to host
rpcinfo  : rpcinfo: RPC: Remote system error - No route to host
showmount: Export list for ol7a: /srv/nfs/share ol7b.example.com 
ls       : readme.txt rc=0
-- tcp2049 : No route to host. ping 은 되는데 이 말이 나오면 방화벽의 REJECT 를 의심한다.
-- firewalld 는 icmp-host-prohibited 로 거절해 클라이언트는 "경로 없음"으로 받는다.
-- rpcinfo 도 TCP 2049 로 가는 호출이라 같은 오류.
-- showmount 는 111·20048 로 가므로 정상이다.
-- ls 는 정상! 이미 맺어진 연결은 방화벽의 상태 추적(ESTABLISHED)으로 통과한다.

real    0m0.011s
user    0m0.007s
sys     0m0.002s
-- 이 연결의 TCP 상태.
[root@ol7b ~]# ss -tn dst 172.16.225.61 | grep -v Recv
ESTAB 0      0      172.16.225.62:911  172.16.225.61:2049       
-- ESTAB. ol7b 의 911 번 포트에서 ol7a 2049 로 연결이 살아 있다.

-- 다른 마운트 지점에 새로 마운트를 시도한다.
[root@ol7b ~]# time timeout 40 mount -t nfs ol7a:/srv/nfs/share /mnt/t2; echo "rc=$?"

real    0m0.010s
user    0m0.001s
sys     0m0.002s
rc=0
-- 새 마운트도 0.01초에 성공했다.
-- 같은 서버로 가는 마운트는 클라이언트가 기존 연결을 나눠 쓴다.
[root@ol7b ~]# ss -tn dst 172.16.225.61 | grep -v Recv | wc -l
1
-- 연결은 여전히 하나뿐이다.
-- 마운트를 모두 떼고,
-- 연결이 없는 상태에서 다시 마운트한다.
[root@ol7b ~]# umount /mnt/t2 /mnt/share; sleep 2; ss -tn dst 172.16.225.61 | grep -v Recv | wc -l
0
-- 연결 0개.
[root@ol7b ~]# time timeout 40 mount -t nfs ol7a:/srv/nfs/share /mnt/share; echo "rc=$?"

real    0m40.003s
user    0m0.001s
sys     0m0.001s
rc=124
-- 이번에는 40초 동안 끝나지 않아 timeout 이 끊었다.
-- 마운트는 거절을 받고도 재시도를 계속한다.
-- 오류 문구 없이 멈춰만 있다.

-- 서버가 막은 패킷을 기록하게 한다.
-- 7.9 의 firewalld 0.6 은 LogDenied 기본값이 off.
-- --set-log-denied 는 영구 설정을 바꾸고 다시 읽는다.
-- [9] 에서 off 로 돌린다.
-- 다시 읽으면 런타임에만 연 서비스가 사라지므로 mountd·rpc-bind 를 다시 연다.
[root@ol7a ~]# firewall-cmd --get-log-denied; firewall-cmd --set-log-denied=unicast; firewall-cmd --add-service=mountd --add-service=rpc-bind; firewall-cmd --list-services
off
success
success
dhcpv6-client mountd rpc-bind ssh
[root@ol7b ~]# timeout 15 bash -c '</dev/tcp/ol7a/2049'; echo "rc=$?"
bash: connect: No route to host
bash: /dev/tcp/ol7a/2049: No route to host
rc=1
-- 클라이언트는 No route to host 두 줄.
[root@ol7a ~]# dmesg -T | grep REJECT | tail -2 | cut -c1-200
[Mon Nov 16 21:03:20 2026] FINAL_REJECT: IN=ens32 OUT= MAC=00:0c:29:ac:5e:a0:00:0c:29:7c:e4:df:08:00 SRC=172.16.225.62 DST=172.16.225.61 LEN=60 TOS=0x00 PREC=0x00 TTL=64 ID=42497 DF PROTO=TCP SPT=4033
-- 서버 커널 로그에 FINAL_REJECT.
-- 영역의 마지막 규칙에서 거절됐다는 뜻이다.
-- IN=ens32 : 들어온 NIC,
-- SRC=172.16.225.62 : ol7b, PROTO=TCP.
-- 200자에서 잘라 DPT(목적지 포트 2049)는 보이지 않는다.

--------------------------------------------------------------------------------
 [9] 장애 C 복구
--------------------------------------------------------------------------------

[root@ol7a ~]# firewall-cmd --set-log-denied=off; firewall-cmd --add-service=nfs --add-service=mountd --add-service=rpc-bind; firewall-cmd --list-services
success
success
dhcpv6-client mountd nfs rpc-bind ssh
-- 기록을 끄고 세 서비스를 다시 열었다.
[root@ol7b ~]# mount -t nfs -o timeo=100,retrans=2 ol7a:/srv/nfs/share /mnt/share; findmnt -n -o SOURCE /mnt/share
ol7a:/srv/nfs/share
[root@ol7b ~]# /root/nfschk.sh
ping     : ok
tcp2049  : open
rpcinfo  : program 100003 version 4 ready and waiting
showmount: Export list for ol7a: /srv/nfs/share ol7b.example.com 
ls       : readme.txt rc=0
-- 다시 기준값과 같다.

--------------------------------------------------------------------------------
 [10] 오류 흐름 — 세 장애가 비슷해 보인다
--------------------------------------------------------------------------------

-- 사용자 신고는 셋 다 "NFS 가 안 된다"였다.
-- 점검 결과를 층별로 모아 본다.
--
--  층          장애 A(서비스 정지)        장애 B(exports)          장애 C(방화벽)
--  ping        ok                         ok                       ok
--  tcp2049     Connection refused         open                     No route to host
--  rpcinfo     Program not registered     ready                    No route to host
--  showmount   Program not registered     목록에 ol7b 없음         정상
--  기존 마운트 15초 멈춤(rc=137)          즉시 Permission denied   정상(기존 연결)
--  새 마운트   (해 보지 않음)             Operation not permitted  40초 멈춤
--  서버 증거   ss 에 2049 없음            mountd: unmatched host   FINAL_REJECT(log-denied)
--
-- 기존 마운트로만 보면 A 와 C 는 "멈춤",
-- B 는 "권한"으로 보인다.
-- 처음 실패한 층을 보면 셋이 갈린다.
-- A 는 2049 거절, B 는 목록, C 는 경로 없음.
-- 증상   : 세 장애 모두 사용자에게는 "NFS 가 안 된다"(멈춤 또는 Permission denied)
-- 원인   : 서비스 정지·exports·방화벽은 서로 다른 층에서 실패하지만 ls 결과만 보면 구분되지 않는다
-- 조치   : ping → TCP 2049 → rpcinfo → showmount → 접근 순서로 처음 실패하는 층을 찾는다
-- 재확인 : 점검 스크립트 다섯 줄이 [2] 의 기준값과 같다

--------------------------------------------------------------------------------
 [변형] soft 마운트로 같은 장애 A
--------------------------------------------------------------------------------

-- soft 는 재시도 횟수를 다 쓰면 호출을 오류로 돌려준다.
-- 프로세스가 D 에 갇히지 않는다.
-- timeo=30(3초), retrans=1 로 짧게 잡았다.
[root@ol7b ~]# umount /mnt/share; mount -t nfs -o soft,timeo=30,retrans=1 ol7a:/srv/nfs/share /mnt/share; findmnt -n -o OPTIONS /mnt/share | tr ',' '\n' | grep -E '^(soft|hard|timeo|retrans|vers)'
vers=4.1
soft
timeo=30
retrans=1
-- 옵션에 soft 가 보인다. hard 는 없다.
[root@ol7a ~]# systemctl stop nfs-server
[root@ol7b ~]# time ls /mnt/share; echo "rc=$?"
ls: cannot open directory /mnt/share: Stale file handle

real    0m13.118s
user    0m0.001s
sys     0m0.001s
rc=2
-- 13초 뒤 오류로 돌아왔다.
-- 기다리던 프로세스가 스스로 끝난다.
-- 오류 문구는 Stale file handle.
-- 서버가 없다는 뜻으로 읽히지 않는다.
-- soft 는 쓰기 도중 오류가 나면 데이터가 조용히 깨질 수 있다.
-- DB 파일을 두는 NFS 는 hard 로 마운트한다(17장, DB 설치 가이드).
[root@ol7a ~]# systemctl start nfs-server
[root@ol7b ~]# ls /mnt/share; echo "rc=$?"
readme.txt
rc=0
-- 서버가 돌아오자 바로 읽힌다.
