-- 1. 아래 스크립트의 장애를 하나 바꿔 짝에게 진단을 맡긴다.
--    짝의 보고서가 [정리] 항목을 다 채웠는지 본다.
-- 2. sosreport 를 전체 플러그인으로 돌려 8초·1.67MiB 와 비교한다.
-- 3. report-gen 에 CPUQuota=50% 를 걸고 vmstat 의 id 가 어떻게 변하는지 본다.

-- 이 실습의 장애는 아래 스크립트로 심었다.
-- 다른 장애로 바꿔 짝과 서로 진단해 본다.
[root@ol7a ~]# cat /root/plant.sh
#!/bin/bash
# 장애 심기 (실습 06 준비) — /app 은 UUID 가 맞지 않게, report-gen 은 CPU 두 개를 다 쓰게
vgcreate -q vgapp /dev/sdb
lvcreate -q -L 1G -n lvapp vgapp; lvcreate -q -L 1G -n lvlog vgapp
mkfs.xfs -q /dev/vgapp/lvapp; mkfs.xfs -q /dev/vgapp/lvlog
mkdir -p /app /applog
echo "UUID=$(blkid -s UUID -o value /dev/vgapp/lvapp) /app xfs defaults,nofail 0 0" >> /etc/fstab
echo "/dev/mapper/vgapp-lvlog /applog xfs defaults 0 0" >> /etc/fstab
mkfs.xfs -q -f /dev/vgapp/lvapp      # 누군가 다시 만들어 UUID 가 바뀜
printf '#!/bin/bash\nsha256sum /dev/zero &\nsha256sum /dev/zero &\nwait\n' > /usr/local/bin/report-gen.sh
chmod 755 /usr/local/bin/report-gen.sh
printf '[Unit]\nDescription=Nightly report generator\n[Service]\nExecStart=/usr/local/bin/report-gen.sh\nRestart=always\n[Install]\nWantedBy=multi-user.target\n' > /etc/systemd/system/report-gen.service
systemctl daemon-reload; systemctl enable report-gen.service
-- 장애 심기 스크립트.
