================================================================================
 제3장 실습 02 — 기본 커널 변경과 복귀
 대상   : ol7a
 목표   : GRUB_DEFAULT=saved 환경에서 기본 부팅 커널을 grub2-set-default 와
          grubby --set-default 로 바꾸고, RHCK 로 부팅했다가 UEK 로 되돌린다.
          grub2-reboot 로 한 번만 다른 커널로 부팅하는 방법도 확인한다
 배경   : /etc/default/grub 의 GRUB_DEFAULT=saved 이면 GRUB 은 grubenv 파일의
          saved_entry 값으로 기본 항목을 고른다. 이 값은 메뉴 항목 번호(0부터)나
          제목 문자열이다. grub.cfg 는 grub2-mkconfig 가 만드는 파일이라 직접 고친
          내용은 다음 재생성 때 사라진다 (본문 3.2, 3.3)
 소요   : 약 35분 (재부팅 네 번)
 선행   : ol7-base. 설치된 커널 네 개(UEK 2개, RHCK 2개)
 계정   : root
 주의   : 커널을 바꿔 부팅한다. 이 랩은 마지막에 UEK 로 되돌리고 스냅샷으로도
          되돌릴 수 있다
 되돌림 : 기본 커널 5.4.17-2136.338.4.2.el7uek.x86_64.
          확인은 grubby --default-kernel; uname -r
================================================================================

--------------------------------------------------------------------------------
 [0] 시작 전 상태 기록
--------------------------------------------------------------------------------

-- 지금 커널, 기본 커널, grubenv 내용을 남긴다.
-- grubby --default-kernel : 기본 항목의 커널 파일 경로
-- grubby --default-index  : 기본 항목의 번호 (0부터)
-- grub2-editenv list      : /boot/grub2/grubenv 의 변수 목록
-- 실행 커널.
[root@ol7a ~]# uname -r
5.4.17-2136.338.4.2.el7uek.x86_64
-- uek 가 붙은 릴리스 = UEK R6. RHCK 에는 uek 가 없다.
[root@ol7a ~]# grubby --default-kernel; grubby --default-index
/boot/vmlinuz-5.4.17-2136.338.4.2.el7uek.x86_64
0
-- 기본 항목 번호는 0. 메뉴 맨 위 항목이다.
-- 이 시점의 기본값 기록.
[root@ol7a ~]# grub2-editenv list
saved_entry=Oracle Linux Server 7.9, with Unbreakable Enterprise Kernel 5.4.17-2136.338.4.2.el7uek.x86_64
-- grubenv 에는 변수가 saved_entry 하나뿐이다. 값은 번호가 아니라 메뉴 제목 문자열 전체다.
-- yum 으로 커널을 올릴 때 새 커널의 제목이 이 자리에 기록됐다.
-- 기본 항목 선택에 관여하는 설정 네 개만 골라 본다.
[root@ol7a ~]# grep -E '^GRUB_(DEFAULT|SAVEDEFAULT|TIMEOUT|DISABLE_SUBMENU)' /etc/default/grub
GRUB_TIMEOUT=5
GRUB_DEFAULT=saved
GRUB_DISABLE_SUBMENU=true
-- GRUB_DEFAULT=saved      : 기본 항목을 grubenv 의 saved_entry 로 정한다
-- GRUB_DISABLE_SUBMENU    : true 라 커널마다 최상위 항목이 하나씩 생긴다. 번호를 셀 때 하위 메뉴가 없다
-- GRUB_SAVEDEFAULT 는 없다. 있으면 메뉴에서 고른 항목이 자동으로 saved_entry 가 된다

 [1] 설치된 커널과 메뉴 항목
--------------------------------------------------------------------------------

-- rpm -q 에 패키지 이름 두 개를 주면 각각의 설치된 버전을 모두 보여 준다.
[root@ol7a ~]# rpm -q kernel-uek kernel
kernel-uek-5.4.17-2102.201.3.el7uek.x86_64
kernel-uek-5.4.17-2136.338.4.2.el7uek.x86_64
kernel-3.10.0-1160.el7.x86_64
kernel-3.10.0-1160.119.1.0.5.el7.x86_64
-- 커널 패키지는 yum update 해도 옛 버전을 지우지 않고 나란히 설치된다(installonly). 그래서 두 개씩이다.
-- UEK 는 설치 매체의 2102 와 업데이트로 받은 2136, RHCK 는 1160 과 1160.119.1.0.5.
-- 기본값을 담는 파일과 grub.cfg 로 가는 링크를 본다.
[root@ol7a ~]# ls -l /boot/grub2/grubenv; ls -l /etc/grub2.cfg
-rw-r--r--. 1 root root 1024 Nov 11 20:18 /boot/grub2/grubenv
lrwxrwxrwx. 1 root root 22 Nov 11 20:15 /etc/grub2.cfg -> ../boot/grub2/grub.cfg
-- grubenv 크기 1024 — 고정 크기 블록이다([변형] 에서 들여다본다).
-- /etc/grub2.cfg 는 BIOS 쪽 grub.cfg 로 가는 상대 경로 링크다. UEFI 서버에는 /etc/grub2-efi.cfg 가 있다.

-- grub.cfg 의 menuentry 제목을 순서대로 뽑고 0부터 번호를 붙인다.
-- -F\' 로 작은따옴표를 구분자로 삼으면 두 번째 필드가 제목이다.
[root@ol7a ~]# awk -F\' '/^menuentry /{print n++ " : " $2}' /boot/grub2/grub.cfg
0 : Oracle Linux Server 7.9, with Unbreakable Enterprise Kernel 5.4.17-2136.338.4.2.el7uek.x86_64
1 : Oracle Linux Server 7.9, with Unbreakable Enterprise Kernel 5.4.17-2102.201.3.el7uek.x86_64
2 : Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64
3 : Oracle Linux Server 7.9, with Linux 3.10.0-1160.el7.x86_64
4 : Oracle Linux Server 7.9, with Linux 0-rescue-b8658360b45f49fe8174bcd0b610ae6e
-- 다섯 항목. 맨 앞 숫자가 GRUB 이 쓰는 번호다.
-- 0·1 UEK, 2·3 RHCK, 4 rescue. 같은 계열에서는 높은 버전이 위다.
-- rescue 항목 이름의 b8658360… 은 /etc/machine-id 값이다. 설치 때 한 번 만들어진다.

-- grubby 가 보는 목록도 같은지 확인한다.
-- --info=ALL 은 항목마다 index, kernel, args, root, initrd, title 을 찍는다. 두 필드만 고른다.
[root@ol7a ~]# grubby --info=ALL | grep -E '^(index|kernel)'
index=0
kernel=/boot/vmlinuz-5.4.17-2136.338.4.2.el7uek.x86_64
index=1
kernel=/boot/vmlinuz-5.4.17-2102.201.3.el7uek.x86_64
index=2
kernel=/boot/vmlinuz-3.10.0-1160.119.1.0.5.el7.x86_64
index=3
kernel=/boot/vmlinuz-3.10.0-1160.el7.x86_64
index=4
kernel=/boot/vmlinuz-0-rescue-b8658360b45f49fe8174bcd0b610ae6e
index=5
-- grubby 의 번호도 awk 결과와 같다. kernel= 줄이 그 항목의 커널 파일이다.
-- 마지막 index=5 는 kernel 줄이 없는 빈 항목이다. grubby 가 메뉴 밖의 줄(submenu 대체용 코드)을 항목처럼 세서 생긴다.
-- 번호를 셀 때는 kernel= 줄이 있는 것만 센다.

--------------------------------------------------------------------------------
 [2] 오류 흐름 — 메뉴 번호를 1부터 셈
--------------------------------------------------------------------------------

-- 메뉴 화면에서 RHCK 최신판은 위에서 세 번째 줄이다. 그래서 3 을 준다.
[root@ol7a ~]# grub2-set-default 3
-- 아무 출력이 없다. grub2-set-default 는 값을 검사하지 않고 grubenv 에 적기만 한다.
-- 무엇이 적혔는지, grubby 가 그 값을 어떤 커널로 푸는지 본다.
[root@ol7a ~]# grub2-editenv list; grubby --default-kernel
saved_entry=3
/boot/vmlinuz-3.10.0-1160.el7.x86_64
-- saved_entry=3 이 번호로 들어갔다. grubby 가 이 번호를 풀어 보니 3번 항목은 설치 매체의 RHCK 1160 이다.
-- 재부팅 전에 이미 틀린 것을 알 수 있었던 자리다. 설정을 바꾼 뒤 grubby --default-kernel 을 꼭 친다.

-- 재부팅 (systemctl reboot 후 SSH 재접속)
[root@ol7a ~]# uname -r
3.10.0-1160.el7.x86_64
-- 증상   : RHCK 최신판(1160.119.1.0.5)을 기대했는데 uname -r 이 3.10.0-1160.el7.x86_64
-- 원인   : GRUB 메뉴 번호는 0부터 센다. "세 번째 줄"은 2번이다. 3 은 네 번째 항목인 설치 매체 커널이다
-- 조치   : 번호 대신 제목 문자열로 지정한다([3])
-- 재확인 : grubby --default-kernel 이 /boot/vmlinuz-3.10.0-1160.119.1.0.5.el7.x86_64, 재부팅 뒤 uname -r 이 같은 값
-- 옛 RHCK 로도 부팅은 정상이라 눈치채기 어렵다. 보안 패치가 빠진 커널로 몇 달을 돌 수 있다.

--------------------------------------------------------------------------------
 [3] 제목 문자열로 지정
--------------------------------------------------------------------------------

-- 번호는 커널이 하나 늘거나 줄면 가리키는 대상이 바뀐다. 제목은 그렇지 않다.
-- 이번에는 awk 결과의 2번 줄 제목을 그대로 복사해 준다.
[root@ol7a ~]# grub2-set-default 'Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64'
-- 세 값을 한 번에 본다.
[root@ol7a ~]# grub2-editenv list; grubby --default-kernel; grubby --default-index
saved_entry=Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64
/boot/vmlinuz-3.10.0-1160.119.1.0.5.el7.x86_64
2
-- saved_entry 에 제목 문자열이 들어갔고 grubby 는 그것을 2번 항목으로 풀었다.
-- 제목은 그 커널이 남아 있는 한 같은 항목을 가리킨다. 커널을 하나 더 설치해 번호가 밀려도 안전하다.

-- 재부팅 (systemctl reboot 후 SSH 재접속)
-- 재부팅 뒤 커널 확인.
[root@ol7a ~]# uname -r; uname -v
3.10.0-1160.119.1.0.5.el7.x86_64
#1 SMP Sun Sep 29 20:38:00 PDT 2024
-- uname -v 는 커널 빌드 번호와 빌드 시각이다. UEK 2136 은 #3 SMP Mon Dec 23 2024 였다.

--------------------------------------------------------------------------------
 [4] RHCK 로 부팅한 상태 확인
--------------------------------------------------------------------------------

-- 커널을 바꿔도 부트 파라미터와 장치 이름이 그대로인지 확인한다. DB 서버라면 커널을 바꾼 뒤 가장 먼저 볼 것들이다.
-- 같은 부트 파라미터가 RHCK 에도 넘어갔는지 본다.
-- 파라미터 확인.
[root@ol7a ~]# cat /proc/cmdline
BOOT_IMAGE=/vmlinuz-3.10.0-1160.119.1.0.5.el7.x86_64 root=/dev/mapper/ol-root ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync
-- BOOT_IMAGE 만 RHCK 파일로 바뀌고 나머지는 UEK 때와 같다. 모든 항목이 GRUB_CMDLINE_LINUX 를 공유한다.

-- scsi_mod.scan=sync 는 SCSI 버스를 동기식으로 훑으라는 파라미터다. 커널이 실제로 받아들였는지 모듈 파라미터로 본다.
-- 값 sync 면 반영됨.
[root@ol7a ~]# cat /sys/module/scsi_mod/parameters/scan
sync
-- RHCK 도 이 파라미터를 받아들였다. 이 값은 모든 커널 항목에 들어가 있어야 장치 이름 순서가 커널마다 같다.
-- -d 는 파티션을 빼고 디스크만, -o 로 열을 고른다.
[root@ol7a ~]# lsblk -d -o NAME,HCTL,SIZE,TYPE
NAME HCTL        SIZE TYPE
sda  0:0:0:0      40G disk
sdb  0:0:1:0      10G disk
sdc  0:0:2:0      10G disk
sdd  0:0:3:0      10G disk
sde  0:0:4:0      10G disk
sr0  2:0:0:0     4.5G rom
-- HCTL(Host:Channel:Target:LUN) 열을 BASE_FACTS 의 UEK 값과 비교한다.
-- UEK 에서 sda 는 2:0:0:0 이었다. RHCK 에서는 0:0:0:0 이다. 호스트 어댑터 번호가 커널마다 다르게 매겨진다.
-- 이름(sda~sde)과 크기는 같다. scsi_mod.scan=sync 덕분에 순서는 지켜졌다.

-- 커널 패키지와 모듈 디렉터리
-- rpm -qf 는 파일의 소유 패키지.
[root@ol7a ~]# rpm -qf /boot/vmlinuz-$(uname -r); ls /lib/modules
kernel-3.10.0-1160.119.1.0.5.el7.x86_64
3.10.0-1136.0.0.0.1.el7.x86_64
3.10.0-1160.119.1.0.5.el7.x86_64
3.10.0-1160.el7.x86_64
5.4.17-2102.201.3.el7uek.x86_64
5.4.17-2136.338.4.2.el7uek.x86_64
-- /lib/modules 에는 다섯 디렉터리가 있다. 설치된 커널 네 개 + 3.10.0-1136.0.0.0.1.
-- 마지막 것은 커널 패키지가 아니라 다른 패키지(드라이버의 weak-modules 용 디렉터리)가 남긴 것이다. rpm -qf 로 확인할 수 있다.
-- 지금 올라온 모듈 수. UEK 와 RHCK 는 모듈 구성이 다르다.
[root@ol7a ~]# lsmod | wc -l
91
-- 헤더 한 줄을 빼면 RHCK 에 올라온 모듈은 90개.

--------------------------------------------------------------------------------
 [5] 오류 흐름 — grub.cfg 를 직접 고침
--------------------------------------------------------------------------------

-- RHCK 항목의 제목에 표시를 붙이려고 grub.cfg 를 sed 로 직접 고친다.
[root@ol7a ~]# cp /boot/grub2/grub.cfg /tmp/grub.cfg.before
-- 비교용 사본을 먼저 만든다. sed -i 는 파일을 그 자리에서 고친다.
-- 제목 뒤에 [DBA TEST] 를 붙인다.
[root@ol7a ~]# sed -i "s/^menuentry 'Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64'/menuentry 'Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64 [DBA TEST]'/" /boot/grub2/grub.cfg
-- diff 형식: 줄번호c줄번호, < 옛 줄, > 새 줄.
[root@ol7a ~]# diff /tmp/grub.cfg.before /boot/grub2/grub.cfg
120c120
< menuentry 'Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64' --class oracle --class gnu-linux --class gnu --class os --unrestricted $menuentry_id_option 'gnulinux-3.10.0-1160.119.1.0.5.el7.x86_64-advanced-d527e202-9baf-46ef-9423-ac690a91a6a9' {
---
> menuentry 'Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64 [DBA TEST]' --class oracle --class gnu-linux --class gnu --class os --unrestricted $menuentry_id_option 'gnulinux-3.10.0-1160.119.1.0.5.el7.x86_64-advanced-d527e202-9baf-46ef-9423-ac690a91a6a9' {
-- 120번째 줄 하나만 바뀌었다(120c120). < 가 옛 줄, > 가 새 줄이다.
[root@ol7a ~]# grub2-editenv list; grubby --default-kernel
saved_entry=Oracle Linux Server 7.9, with Linux 3.10.0-1160.119.1.0.5.el7.x86_64
/boot/vmlinuz-5.4.17-2136.338.4.2.el7uek.x86_64
-- 여기서 두 번째 문제가 드러난다. saved_entry 는 RHCK 제목 그대로인데 grubby 는 기본 커널을 UEK 라고 답한다.
-- 제목을 바꿨기 때문에 saved_entry 와 같은 이름의 항목이 grub.cfg 에 없어졌다.
-- 이대로 재부팅하면 GRUB 도 saved_entry 를 찾지 못해 0번(UEK)으로 간다. 직접 수정이 기본 커널 선택까지 흔든다.

-- -o 를 빼면 화면에만 나오고 파일은 그대로다.

 [변형] grub2-editenv 로 직접 보고 지우기
--------------------------------------------------------------------------------

-- grubenv 는 1024바이트 고정 크기의 텍스트 블록이다. 나머지는 # 으로 채워져 있다.
-- head -c 300 은 앞 300바이트만. echo 는 줄바꿈을 하나 더 찍는다.
[root@ol7a ~]# ls -l /boot/grub2/grubenv; head -c 300 /boot/grub2/grubenv; echo
-rw-r--r--. 1 root root 1024 Nov 12 19:00 /boot/grub2/grubenv
# GRUB Environment Block
saved_entry=Oracle Linux Server 7.9, with Unbreakable Enterprise Kernel 5.4.17-2136.338.4.2.el7uek.x86_64
next_entry=
#############################################################################################################################################################
-- 첫 줄은 주석, 그다음 변수 줄, 나머지 1024바이트까지는 # 으로 채운다. head -c 300 이라 300바이트에서 잘렸다.
[root@ol7a ~]# grep -c '#' /boot/grub2/grubenv; wc -c < /boot/grub2/grubenv
2
1024
-- grep -c 는 # 이 들어간 "줄" 수다. 첫 주석 줄과 채움 줄, 두 줄이다. 파일 크기는 정확히 1024.
-- vi 로 열어 고치면 크기가 틀어질 수 있다. 반드시 grub2-editenv 로 고친다.

-- next_entry 를 손으로 넣고 지워 본다. 재부팅하지 않는다.
-- 첫 인자 - 는 기본 파일(/boot/grub2/grubenv)을 뜻한다.
[root@ol7a ~]# grub2-editenv - set next_entry=2; grub2-editenv list
saved_entry=Oracle Linux Server 7.9, with Unbreakable Enterprise Kernel 5.4.17-2136.338.4.2.el7uek.x86_64
next_entry=2
-- set 은 변수를 넣는다. next_entry 에 번호 2(RHCK 최신판)가 들어갔다. 재부팅하지 않았으니 효과는 없다.
-- unset 은 변수를 줄째 지운다.
[root@ol7a ~]# grub2-editenv - unset next_entry; grub2-editenv list
saved_entry=Oracle Linux Server 7.9, with Unbreakable Enterprise Kernel 5.4.17-2136.338.4.2.el7uek.x86_64

--  항목                    시작 전              변경 중                       되돌린 뒤
--  saved_entry 형식        제목(UEK 2136)       3 → 제목(RHCK 119)            제목(UEK 2136)
