================================================================================
 제3장 실습 06 — 서비스 유닛 작성
 대상   : ol7a (원격 제어 단계에서 ol7b)
 목표   : 셸 스크립트를 서비스 유닛으로 등록하고 Type·Restart·After 의 뜻을
          확인한다. 유닛 파일 수정 뒤 daemon-reload 가 필요한 이유, drop-in 으로
          원본을 건드리지 않고 덮어쓰는 방법, cgroup 자원 한도, 원격 systemctl -H
          를 차례로 다룬다
 배경   : 유닛 파일은 /usr/lib/systemd/system(패키지), /run/systemd/system(실행 중
          생성), /etc/systemd/system(관리자) 세 곳에 있고 뒤로 갈수록 우선한다.
          systemd 는 유닛 파일을 메모리에 읽어 두고 쓰므로 파일을 고친 뒤에는
          daemon-reload 로 다시 읽혀야 한다. 원본을 고치지 않고 일부만 바꾸려면
          <유닛>.d/*.conf drop-in 을 쓴다 (본문 3.5, 3.8)
 소요   : 약 30분
 선행   : ol7-base, ol7b-base
 계정   : root
 주의   : [9] 에서 일부러 실패를 반복하는 서비스를 만든다. 실습 끝에 반드시 지운다
 되돌림 : ol7beat 유닛·스크립트·drop-in 삭제, ol7a 의 /root/.ssh 와 ol7b 의
          authorized_keys 삭제. 확인은 systemctl status ol7beat (없음), ls /root/.ssh

 [2] 서비스로 돌릴 스크립트
--------------------------------------------------------------------------------

-- INTERVAL 이 없으면 5초.
-- 몇 초마다 한 줄씩 표준 출력에 찍는 스크립트다.
-- systemd 는 서비스의 표준 출력을 journal 로 보낸다.
[root@ol7a ~]# cat > /usr/local/bin/ol7beat.sh <<'EOS'
#!/bin/bash
INTERVAL=${INTERVAL:-5}
while true; do
    echo "ol7beat $(date +%T) interval=$INTERVAL"
    sleep "$INTERVAL"
done
EOS
-- 실행 권한을 주고 확인한다.
[root@ol7a ~]# chmod 755 /usr/local/bin/ol7beat.sh; ls -lZ /usr/local/bin/ol7beat.sh
-rwxr-xr-x. root root unconfined_u:object_r:bin_t:s0   /usr/local/bin/ol7beat.sh
-- 권한 755. -Z 는 보안 문맥 열을 더한다(bin_t).

--------------------------------------------------------------------------------
 [3] 유닛 파일 작성
--------------------------------------------------------------------------------

-- /etc 에 둔다.
-- [Unit]    Description, 순서(After)
-- [Service] Type=simple: ExecStart 의 프로세스가 곧 주 프로세스. 띄우는 즉시 active 로 본다
--           Environment: 스크립트에 넘길 환경 변수
--           Restart=on-failure: 0 이 아닌 종료 코드나 시그널로 죽었을 때만 다시 띄운다
-- [Install] WantedBy: enable 할 때 어느 타깃의 .wants 에 링크를 걸지
[root@ol7a ~]# cat > /etc/systemd/system/ol7beat.service <<'EOS'
[Unit]
Description=OL7ADMIN heartbeat demo
After=network.target

[Service]
Type=simple
Environment=INTERVAL=5
ExecStart=/usr/local/bin/ol7beat.sh
Restart=on-failure

[Install]
WantedBy=multi-user.target
EOS

-- 새 파일을 만들었으니 systemd 에게 다시 읽게 하고, 문법부터 검사한다.
[root@ol7a ~]# systemctl daemon-reload; systemd-analyze verify /etc/systemd/system/ol7beat.service; echo "verify rc=$?"
verify rc=0
-- 경고 없이 rc=0.

-- 흔한 실수다.
-- 간격을 3초로 바꾼다.
-- sed -i 로 유닛 파일을 직접 고친다.

[root@ol7a ~]# cp /etc/systemd/system/ol7beat.service /tmp/ol7beat.service.orig
[root@ol7a ~]# sed -i 's/INTERVAL=5/INTERVAL=3/' /etc/systemd/system/ol7beat.service; diff /tmp/ol7beat.service.orig /etc/systemd/system/ol7beat.service

> Environment=INTERVAL=3
-- 7번째 줄만 바뀌었다.
-- reload 없이 restart 하고 7초 뒤 로그 두 줄.
[root@ol7a ~]# systemctl restart ol7beat; sleep 7; journalctl -u ol7beat --no-pager -n 2
Warning: ol7beat.service changed on disk. Run 'systemctl daemon-reload' to reload units.
-- Logs begin at Fri 2026-11-13 08:33:05 KST, end at Fri 2026-11-13 08:34:42 KST. --

Nov 13 08:34:42 ol7a.example.com ol7beat.sh[2352]: ol7beat 08:34:42 interval=5
-- 첫 줄 경고: 파일이 바뀌었으니 daemon-reload 하라.
-- 경고만 하고 restart 는 진행됐다.
-- PID 는 2352 로 바뀌었는데 간격은 여전히 5.

[root@ol7a ~]# systemctl show -p Environment -p NeedDaemonReload ol7beat
Environment=INTERVAL=5
NeedDaemonReload=yes

-- 증상   : 유닛 파일을 INTERVAL=3 으로 고치고 restart 했는데 로그가 interval=5
-- 원인   : daemon-reload 를 하지 않아 systemd 가 옛 정의로 다시 띄웠다

-- 다시 읽힌다.
[root@ol7a ~]# systemctl daemon-reload; systemctl show -p NeedDaemonReload ol7beat

-- 다시 읽힌 뒤 no.
[root@ol7a ~]# systemctl restart ol7beat; sleep 7; journalctl -u ol7beat --no-pager -n 2

Nov 13 08:34:50 ol7a.example.com ol7beat.sh[2430]: ol7beat 08:34:50 interval=3
-- 3초 간격이 됐다.

--------------------------------------------------------------------------------
 [7] drop-in — systemctl edit
--------------------------------------------------------------------------------

-- 이 단계가 이 랩의 세 번째 오류 흐름이 됐다.
-- systemctl edit 는 편집기로 /etc/systemd/system/<유닛>.d/override.conf 를 열고,
-- 저장하면 daemon-reload 까지 해 준다.
-- 편집기를 tee 로 바꿔 표준 입력의 내용을 그대로 저장시킨다.
-- SYSTEMD_EDITOR : systemctl edit 가 쓸 편집기 (없으면 EDITOR, VISUAL, 그다음 nano·vim·vi 순서)
[root@ol7a ~]# printf '[Service]\nEnvironment=INTERVAL=2\n' | SYSTEMD_EDITOR=tee systemctl edit ol7beat
Cannot edit units if not on a tty
-- systemctl edit 가 바로 거부했다.
-- 7.9 의 systemctl edit 는 터미널(tty)에서만 동작한다.
-- 파이프로 넣는 방식은 이 버전에서 쓸 수 없다.
-- drop-in 이 생겼는지 본다.
[root@ol7a ~]# ls -l /etc/systemd/system/ol7beat.service.d/; systemctl show -p DropInPaths -p NeedDaemonReload ol7beat
ls: cannot access /etc/systemd/system/ol7beat.service.d/: No such file or directory
DropInPaths=
NeedDaemonReload=no
-- override.conf 디렉터리도 생기지 않았다.
-- DropInPaths 도 비었다.
-- 증상   : "Cannot edit units if not on a tty"
-- 원인   : 7.9 의 systemctl edit 는 표준 입력이 터미널이 아니면 거부한다
-- 조치   : 터미널에서 직접 편집하거나, drop-in 파일을 손으로 만든다([변형])
-- 재확인 : systemctl show -p DropInPaths 에 파일 경로가 나옴
-- 합쳐진 정의를 본다.
[root@ol7a ~]# systemctl cat ol7beat --no-pager

[Service]

[Install]
WantedBy=multi-user.target

-- drop-in 이 없으니 덧붙은 것도 없다.

 [8] 자원 한도 — systemctl set-property
--------------------------------------------------------------------------------

-- 실행 중인 서비스에 바로 걸린다.
-- cgroup v1 의 memory·cpu 컨트롤러 값을 유닛 속성으로 건다.
-- MemoryLimit : 메모리 상한 (7.9 의 systemd 219 이름)
-- CPUShares   : CPU 경합 시 가중치. 기본 1024
-- 실행 중인 서비스에 건다.
[root@ol7a ~]# systemctl set-property ol7beat MemoryLimit=64M CPUShares=512
-- 무엇이 남았나.
[root@ol7a ~]# ls -l /etc/systemd/system/ol7beat.service.d/; cat /etc/systemd/system/ol7beat.service.d/50-*.conf

-rw-r--r--. 1 root root 24 Nov 13 08:34 50-CPUShares.conf
-rw-r--r--. 1 root root 31 Nov 13 08:34 50-MemoryLimit.conf
[Service]
CPUShares=512
[Service]
MemoryLimit=67108864
-- set-property 는 속성마다 50-<이름>.conf 를 만들었다.
-- 64M 는 바이트로 바뀌어 기록됐다(67108864).
-- 유닛 속성.
[root@ol7a ~]# systemctl show -p MemoryLimit -p CPUShares ol7beat

MemoryLimit=67108864

-- 커널 cgroup 파일.
[root@ol7a ~]# cat /sys/fs/cgroup/memory/system.slice/ol7beat.service/memory.limit_in_bytes /sys/fs/cgroup/cpu,cpuacct/system.slice/ol7beat.service/cpu.shares

-- cgroup v1 파일에도 바로 들어갔다.
-- memory 와 cpu,cpuacct 가 따로 있는 것이 v1 구조다.

-- --runtime 을 붙이면 /run 아래에 써서 재부팅하면 사라진다.
[root@ol7a ~]# systemctl set-property --runtime ol7beat CPUShares=256; ls /run/systemd/system/ol7beat.service.d/; systemctl show -p CPUShares ol7beat
50-CPUShares.conf

-- --runtime 은 /run 에 같은 이름의 파일을 썼다.
-- /run 이 /etc 보다 늦게 읽혀 256 이 이겼다.
-- 재부팅하면 /run 이 사라져 512 로 돌아간다.

 [9] 오류 흐름 — Restart=always 와 실패 루프
--------------------------------------------------------------------------------

-- 시작 제한 값을 실측한다.
-- 시작하자마자 실패하는 명령으로 바꾸고 Restart=always 를 건다.

-- 키는 이 단계 직전에 조용히 배포했다.
-- -H 는 SSH 로 상대 노드에 붙어 그쪽 systemd 에게 명령을 보낸다.
-- 대화형 암호 입력이 안 되므로
-- 키 인증이 먼저 돼 있어야 한다.
-- (키 만들기와 배포는 15장. 여기서는 미리 배포해 두었다)
-- 먼저 그대로 쳐 본다.
[root@ol7a ~]# systemctl -H root@ol7b is-active sshd chronyd
Host key verification failed.
unknown
unknown
-- 호스트 키가 known_hosts 에 없어 SSH 가 거부했다.
-- 결과는 unknown 두 줄. 원격 상태가 아니다.

-- known_hosts 에 ol7b 를 넣고 다시 한다.
-- ssh-keyscan : 상대의 호스트 키를 받아 표준 출력으로. -t ecdsa 는 그 형식의 키만
