-- 대화형 bash 의 프롬프트 PS1 은 /etc/bashrc 가 정한다.
[root@ol7a ~]# grep -n 'PS1=' /etc/bashrc | head -3
41:  [ "$PS1" = "\\s-\\v\\\$ " ] && PS1="[\u@\h \W]\\$ "
46:  #   PS1="[\u@\h:\l \W]\\$ "
-- 41행이 실제 설정이다. PS1 이 셸 기본값("\s-\v\$ ")일
-- 때만 "[\u@\h \W]\$ " 로 바꾼다. 46행은 주석 예시
-- (\l 은 터미널 번호).
-- \u 계정, \h 호스트 이름의 첫 마디, \W 현재 디렉터리의 마지막 이름.
-- 디렉터리를 옮기면 프롬프트의 마지막 칸이 바뀐다.
[root@ol7a sysconfig]# pwd
/etc/sysconfig
-- 프롬프트 마지막 칸이 ~ 에서 sysconfig 로 바뀌었다.
[student@ol7a ~]$ pwd; whoami
/home/student
student
-- student 로 들어가면 홈 /home/student 에서 시작하고
-- 끝 기호가 $ 다. # 는 root, $ 는 일반 사용자.

--   lo     UNKNOWN 은 루프백의 정상 상태

172.16.225.0/24 dev ens32 proto kernel scope link src 172.16.225.61 metric 100 
search example.com
nameserver 172.16.225.2
-- 기본 경로는 VMware NAT 게이트웨이 .2, 그리고 대역 경로 하나.
-- proto static 은 고정 설정, metric 100 은 NetworkManager 기본값.
-- search example.com 이라 짧은 이름에 도메인을 붙여 찾는다.

[root@ol7a ~]# lsblk -o NAME,SIZE,TYPE,MOUNTPOINT,HCTL
NAME         SIZE TYPE MOUNTPOINT HCTL
sdd           10G disk            2:0:3:0
sdb           10G disk            2:0:1:0
sr0          4.5G rom             1:0:0:0
sde           10G disk            2:0:4:0
sdc           10G disk            2:0:2:0
sda           40G disk            2:0:0:0
├─sda2        39G part            
│ ├─ol-swap  3.9G lvm  [SWAP]     
│ └─ol-root 35.1G lvm  /          
└─sda1         1G part /boot      
-- 출력 순서는 sdd, sdb, sr0, sde, sdc, sda 로 뒤섞였다.

-- 장치 이름이 HCTL 의 Target 순서를 따른다. 이것이

NAME              SIZE TYPE MOUNTPOINT

│ └─ol_ol7b-root   27G lvm  /
└─sda1              1G part /boot

 [8] 오류 재현 — 호스트 키가 바뀌었다는 경고
--------------------------------------------------------------------------------

-- ol7b 를 다시 설치했거나 다른 스냅샷으로 되돌린 뒤라면, ol7a 에 기록된 ol7b 의
-- 호스트 키와 지금 ol7b 가 내미는 키가 다르다. 그 상황을 만든다:
-- ol7a 자신의 키를 ol7b 이름으로 known_hosts 에 넣는다.
[root@ol7a ~]# mkdir -m 700 /root/.ssh; ssh-keyscan -t ecdsa ol7a 2>/dev/null | sed 's/^ol7a/ol7b/' > /root/.ssh/known_hosts; cat /root/.ssh/known_hosts | cut -c1-60
ol7b ecdsa-sha2-nistp256 AAAAE2VjZHNhLXNoYTItbmlzdHAyNTYAAAA
-- known_hosts 한 줄: 호스트 이름, 키 종류, 공개 키.
-- 앞 60글자에서 잘랐다(cut -c1-60). 키는 ol7a 의 것이다.
-- BatchMode=yes 는 암호·확인 질문 없이 실패하게 한다(트랜스크립트용).
[root@ol7a ~]# ssh -o BatchMode=yes ol7b hostname; echo "rc=$?"
@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@
@       WARNING: POSSIBLE DNS SPOOFING DETECTED!          @
@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@

and the key for the corresponding IP address 172.16.225.62
is unknown. This could either mean that
DNS SPOOFING is happening or the IP address for the host

@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@
@    WARNING: REMOTE HOST IDENTIFICATION HAS CHANGED!     @
@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@

rc=255
-- 경고가 두 덩어리다.
--   POSSIBLE DNS SPOOFING DETECTED  이름 ol7b 의 키는
--     바뀌었는데 주소 .62 의 키는 기록이 없다
--   REMOTE HOST IDENTIFICATION HAS CHANGED  서버가 낸 키의
--     지문과 문제의 줄(known_hosts:1)을 알려 준다
-- "Host key verification failed." 로 끊겼고 rc=255.
-- 증상   : REMOTE HOST IDENTIFICATION HAS CHANGED!, rc=255
-- 원인   : known_hosts:1 의 ol7b 키와 서버가 낸 키가 다르다
--          (여기서는 ol7a 의 키를 넣어 일부러 만들었다)
-- 조치   : 지문 비교 → ssh-keygen -R ol7b → 키 다시 받기
-- 재확인 : 다시 접속하면 키 단계를 지나 인증 단계로 간다
-- 지문을 양쪽에서 비교한다. -l 지문 출력, -f 파일.
[root@ol7a ~]# ssh-keygen -lf /root/.ssh/known_hosts
256 SHA256:aG5yYzPFQGePobXvo1VN0T7YOWmIrh8gIYTK5k4gcCk ol7b (ECDSA)
-- 기록된 지문 aG5y… 는 경고의 MrVO… 와 다르다.
-- 열: 키 길이 256, 지문, 이름, 종류.
[root@ol7b ~]# ssh-keygen -lf /etc/ssh/ssh_host_ecdsa_key.pub
256 SHA256:MrVOo4BJEDHvlQH0owC4B+OrcIa7JonZUKBXOrS/GVo no comment (ECDSA)
-- ol7b 서버 쪽 지문 MrVO… 가 경고의 지문과 같다.
-- 서버가 진짜 키를 냈고 틀린 것은 ol7a 의 기록이다.
-- 기록을 지운다. -R 은 known_hosts 에서 그 이름의 줄을 지우고 .old 사본을 남긴다.
[root@ol7a ~]# ssh-keygen -R ol7b
# Host ol7b found: line 1
/root/.ssh/known_hosts updated.
Original contents retained as /root/.ssh/known_hosts.old
-- 1번 줄을 지우고 .old 사본을 남겼다.
-- "# Host ol7b found: line 1" 이 지운 줄 번호다. 경고의
-- "Offending ECDSA key in /root/.ssh/known_hosts:1" 과 같은 줄이다.
-- 진짜 ol7b 의 키를 받아 다시 시도한다.
[root@ol7a ~]# ssh-keyscan -t ecdsa ol7b 2>/dev/null >> /root/.ssh/known_hosts; ssh-keygen -lf /root/.ssh/known_hosts
256 SHA256:MrVOo4BJEDHvlQH0owC4B+OrcIa7JonZUKBXOrS/GVo ol7b (ECDSA)
-- 새로 받은 키의 지문이 서버 쪽 MrVO… 와 같다.
[root@ol7a ~]# ssh -o BatchMode=yes ol7b hostname; echo "rc=$?"
Warning: Permanently added the ECDSA host key for IP address '172.16.225.62' to the list of known hosts.
Permission denied (publickey,gssapi-keyex,gssapi-with-mic,password).
rc=255
-- 이번에는 주소 .62 의 키도 known_hosts 에 더해졌다.
-- 그다음 "Permission denied (publickey,…,password)."
-- 호스트 키 단계는 통과하고 인증 단계에서 멈췄다.
-- BatchMode 라 암호를 묻지 않았다. 키 인증은 15장.
-- 운영에서 같은 경고가 나오면 키가 바뀐 이유를 확인하기
-- 전에는 -R 로 지우지 않는다. 중간자 공격도 같은 경고를 낸다.

-- 프롬프트를 다시 본다. 올바른 창에서.

-- 증상   : ol7b 메모리를 확인했는데 total 3629MB 가 나왔다
-- 원인   : 프롬프트가 [root@ol7a ~]# 였다. 다른 노드 창에서 쳤다
-- 조치   : 프롬프트의 호스트 이름을 확인하고 ol7b 창에서 다시 친다
-- 재확인 : [root@ol7b ~]# 에서 total 1680, nproc 1
-- 값이 예상과 다르면 명령보다 프롬프트와 hostname 을 먼저 본다.
