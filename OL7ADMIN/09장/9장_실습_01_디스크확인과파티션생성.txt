================================================================================
 제9장 실습 01 — 디스크 확인과 파티션 생성
 대상   : ol7a
 목표   : 빈 디스크를 lsblk·blkid·/dev/disk/by-* 로 확인하고, fdisk 로 MBR,
          parted·gdisk 로 GPT 파티션을 만든 뒤 커널이 새 테이블을 읽었는지 확인
 배경   : 파티션 테이블은 디스크 앞(MBR)이나 앞뒤(GPT)에 기록된 목록이다.
          도구가 디스크에 쓴 테이블과 커널이 메모리에 들고 있는 테이블은 별개라
          쓰고 나서 커널에 알리는 단계(partprobe·partx)가 따로 있다.
          MBR 은 주 파티션 4개·2TiB 한도, GPT 는 기본 128개 (본문 9.1~9.3절)
 소요   : 약 25분
 선행   : storage-clean 스냅샷(sdb~sde 10G 가 비어 있음)
 계정   : root
 주의   : sdb~sde 의 파티션 테이블을 덮어쓴다. sda 는 절대 지정하지 않는다
 되돌림 : 끝에서 wipefs 로 네 디스크의 서명을 지워 빈 상태로 돌린다.
          확인 명령 lsblk /dev/sd[b-e], blkid -p /dev/sdb (출력 없음)
================================================================================

 [1] 디스크 장치 이름과 식별 정보
--------------------------------------------------------------------------------
-- 이름 외에 무엇으로 디스크를 구분할 수 있는지 본다.
-- -o 로 열을 고른다. HCTL = Host:Channel:Target:Lun (SCSI 주소),
-- TRAN = 전송 방식, ROTA = 회전 디스크 여부, RO = 읽기 전용 여부
[root@ol7a ~]# lsblk -d -o NAME,HCTL,SIZE,TYPE,TRAN,ROTA,RO,MODEL
NAME HCTL        SIZE TYPE TRAN ROTA RO MODEL
sdd  2:0:3:0      10G disk spi     1  0 VMware Virtual S
sdb  2:0:1:0      10G disk spi     1  0 VMware Virtual S
sr0  1:0:0:0     4.5G rom  ata     1  0 VMware IDE CDR10
sde  2:0:4:0      10G disk spi     1  0 VMware Virtual S
sdc  2:0:2:0      10G disk spi     1  0 VMware Virtual S
sda  2:0:0:0      40G disk spi     1  0 VMware Virtual S
-- HCTL 2:0:N:0 의 셋째 값(Target)이 0~4 로 이름 순서와 같다. sda=0, sdb=1 …
-- 이 일치는 랩 부트 파라미터 scsi_mod.scan=sync 덕분이다(실습 03 [10])
-- /dev/disk 아래에는 udev 가 만든 고정 이름 링크가 있다.
[root@ol7a ~]# ls -l /dev/disk/by-path/
total 0
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 pci-0000:00:07.1-ata-2.0 -> ../../sr0
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:0:0 -> ../../sda
lrwxrwxrwx. 1 root root 10 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:0:0-part1 -> ../../sda1
lrwxrwxrwx. 1 root root 10 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:0:0-part2 -> ../../sda2
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:1:0 -> ../../sdb
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:2:0 -> ../../sdc
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:3:0 -> ../../sdd
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 pci-0000:00:10.0-scsi-0:0:4:0 -> ../../sde
-- by-path 이름 = PCI 주소(00:10.0) + SCSI 주소(0:0:N:0)
-- 슬롯이 바뀌지 않으면 그대로다
-- 파티션은 -partN 이 붙는다.
-- 지금은 sda 만 -part1·-part2 가 있다
-- by-id 는 디스크가 일련번호를 내줄 때만 생긴다.
[root@ol7a ~]# ls -l /dev/disk/by-id/ | grep -v dm-
total 0
lrwxrwxrwx. 1 root root  9 Nov 13 19:47 ata-VMware_Virtual_IDE_CDROM_Drive_10000000000000000001 -> ../../sr0
lrwxrwxrwx. 1 root root 10 Nov 13 19:47 lvm-pv-uuid-ULDbRp-NtXT-dFGI-omgJ-rwPg-dl2e-CwYyeF -> ../../sda2
-- by-id 에 SCSI 디스크가 하나도 없다.
-- CD-ROM 과 LVM PV 링크뿐
-- udev 가 by-id 를 만들 때 쓰는 scsi_id 로 일련번호를 묻는다.
[root@ol7a ~]# /usr/lib/udev/scsi_id -g -u -d /dev/sdb; echo rc=$?
rc=1
-- scsi_id 가 아무것도 돌려주지 못했다(rc=1).
-- VMware 가상 디스크가 일련번호를 주지 않는다
-- by-id 가 없으니 이 랩은 by-path 와 UUID 로 디스크를 구분한다
-- VM 설정 disk.EnableUUID = "TRUE" 를 넣으면 생긴다.
-- 이 랩은 VM 설정을 바꾸지 않는다

 [2] 빈 디스크에 blkid·fdisk -l 을 걸어 보기
--------------------------------------------------------------------------------
-- blkid 는 장치의 서명(파일 시스템·테이블·스왑)을 읽는다.
-- 서명이 없으면 아무것도 찍지 않고 종료 코드 2 를 돌려준다.
[root@ol7a ~]# blkid /dev/sdb; echo rc=$?
rc=2
-- 출력 없이 rc=2.
-- 서명이 하나도 없다는 뜻의 종료 코드다(0 은 찾음)
[root@ol7a ~]# blkid -p /dev/sda; echo rc=$?
/dev/sda: PTTYPE="dos" 
-- -p 는 캐시를 거치지 않고 장치를 직접 읽는다.
-- sda 는 MBR(dos) 테이블을 가졌다
rc=0
-- fdisk -l 은 파티션 테이블을 읽어 보여 준다.
[root@ol7a ~]# fdisk -l /dev/sdb

Disk /dev/sdb: 10.7 GB, 10737418240 bytes, 20971520 sectors
Units = sectors of 1 * 512 = 512 bytes
Sector size (logical/physical): 512 bytes / 512 bytes
I/O size (minimum/optimal): 512 bytes / 512 bytes

 [3] fdisk 로 MBR 주 파티션 두 개 만들기
--------------------------------------------------------------------------------
-- fdisk 는 대화형이다. 입력을 printf 로 한 줄씩 넘긴다.
--   o        새 DOS(MBR) 테이블
--   n p 1    새 주 파티션 1번
--   (빈 줄)  첫 섹터 기본값(2048 = 1MiB 경계)
--   +2G      크기
--   n p 2    두 번째 주 파티션, 첫 섹터 기본값, +2G
--   w        디스크에 쓰고 종료

 [6] 사용 중인 디스크에서 파티션을 바꾸기
--------------------------------------------------------------------------------
-- 1번을 마운트해 "사용 중" 상태를 만든다.
[root@ol7a ~]# mkfs.xfs -q /dev/sdb1; mkdir -p /mnt/p1; mount /dev/sdb1 /mnt/p1; findmnt /mnt/p1
TARGET  SOURCE    FSTYPE OPTIONS
/mnt/p1 /dev/sdb1 xfs    rw,relatime,seclabel,attr2,inode64,logbufs=8,logbsize=32k,noquota
-- sdb1 이 /mnt/p1 에 붙었다.
-- 이제 sdb 디스크는 "사용 중" 이다

 [7] parted 로 GPT 만들기 (sdc)
--------------------------------------------------------------------------------
-- -s 는 묻지 않고 실행(스크립트 모드). 명령을 한 줄에 이어 쓸 수 있다.
--   mklabel gpt                 GPT 테이블
--   mkpart data1 ext4 1MiB 3GiB GPT 에서는 두 번째 인자가 파티션 이름
--   mkpart data2 xfs 3GiB 100%  끝까지
[root@ol7a ~]# parted -s /dev/sdc mklabel gpt mkpart data1 ext4 1MiB 3GiB mkpart data2 xfs 3GiB 100%
[root@ol7a ~]# parted -s /dev/sdc unit s print free | tail -6
Number  Start      End        Size       File system  Name   Flags
        34s        2047s      2014s      Free Space
 1      2048s      6291455s   6289408s                data1
 2      6291456s   20969471s  14678016s               data2
        20969472s  20971486s  2015s      Free Space
-- unit s(섹터)로 보니 GPT 의 구조가 보인다
--   섹터 0~33 은 보호 MBR·헤더·항목 표. 34s~2047s 는 정렬 여백
--   20971486s 가 마지막 사용 가능 섹터. 뒤 33섹터는 사본 자리
-- Name 열의 data1·data2 는 mkpart 둘째 인자.
-- File system 열은 비었다(mkfs 전)
-- GPT 디스크를 fdisk 로 읽으면 어떻게 보이나.
[root@ol7a ~]# fdisk -l /dev/sdc 2>&1 | grep -vE '^$|^Units|^Sector|^I/O'
WARNING: fdisk GPT support is currently new, and therefore in an experimental phase. Use at your own discretion.
Disk /dev/sdc: 10.7 GB, 10737418240 bytes, 20971520 sectors
Disk label type: gpt
Disk identifier: FE698F19-5DD7-4EB1-9B9A-C8CF99D8C8B9
#         Start          End    Size  Type            Name
 1         2048      6291455      3G  Microsoft basic data1
 2      6291456     20969471      7G  Microsoft basic data2
-- 첫 줄이 경고다
--   fdisk GPT support is currently new … experimental phase
-- Type 이 Microsoft basic 으로 보인다.
-- parted mkpart 의 ext4·xfs 인자가 유형 GUID 를 Linux 로
-- 정하지 않았다. 유형이 중요하면 gdisk 코드(8300)로 정한다
-- [OL8/9] fdisk 가 GPT 를 정식 지원해 이 경고가 없다

--------------------------------------------------------------------------------
 [8] gdisk 로 GPT 만들기 (sdd)
--------------------------------------------------------------------------------
-- gdisk 는 fdisk 와 비슷한 대화형 GPT 전용 도구다.
--   n 1 (빈 줄) +2G 8300   1번, 첫 섹터 기본, 2G, 유형 코드 8300(Linux filesystem)
--   n 2 (빈 줄) (빈 줄) 8200  2번, 나머지 전부, 8200(Linux swap)
--   w y                     쓰기 확인
[root@ol7a ~]# printf 'n\n1\n\n+2G\n8300\nn\n2\n\n\n8200\nw\ny\n' | gdisk /dev/sdd | tail -4
PARTITIONS!!

Do you want to proceed? (Y/N): OK; writing new GUID partition table (GPT) to /dev/sdd.
The operation has completed successfully.
-- 끝 4줄만 봤다. Do you want to proceed? 에 y 로 답해 기록됐다
[root@ol7a ~]# gdisk -l /dev/sdd | sed -n '/^Found/p;/^Number/,$p'
Found valid GPT with protective MBR; using GPT.
Number  Start (sector)    End (sector)  Size       Code  Name
   1            2048         4196351   2.0 GiB     8300  Linux filesystem
   2         4196352        20971486   8.0 GiB     8200  Linux swap
-- Found valid GPT with protective MBR: 헤더와 보호 MBR 이 모두 정상
-- Code 8300·8200 이 그대로 유형으로 들어갔다
-- Name 은 유형 이름이 기본값으로 붙는다
-- 디스크 앞 512바이트에 보호용 MBR 이 있는지 바이트로 확인한다.
-- 0x1C2 의 유형 바이트가 ee 면 보호 MBR 이다.
[root@ol7a ~]# dd if=/dev/sdd bs=512 count=1 2>/dev/null | od -A x -t x1 -j 446 -N 16
0001be 00 00 02 00 ee ff ff ff 01 00 00 00 ff ff 3f 01
0001ce
-- -j 446 -N 16 = 0x1BE 부터 16바이트, 첫 파티션 항목
--   다섯째 바이트(0x1C2) = ee → 보호 MBR 의 유형
--   마지막 4바이트 ff ff 3f 01 = 항목이 덮는 섹터 수(디스크 끝까지)
-- 0001ce 는 od 가 찍은 끝 오프셋이다

--------------------------------------------------------------------------------
 [9] 파티션이 생긴 뒤의 고정 이름
--------------------------------------------------------------------------------
-- GPT 파티션은 이름(PARTLABEL)·UUID(PARTUUID)를 따로 가진다.
[root@ol7a ~]# ls -l /dev/disk/by-partlabel/
total 0
lrwxrwxrwx. 1 root root 10 Nov 13 19:48 data1 -> ../../sdc1
lrwxrwxrwx. 1 root root 10 Nov 13 19:48 data2 -> ../../sdc2
lrwxrwxrwx. 1 root root 10 Nov 13 19:48 Linux\x20filesystem -> ../../sdd1
lrwxrwxrwx. 1 root root 10 Nov 13 19:48 Linux\x20swap -> ../../sdd2
-- by-partlabel 은 GPT 파티션 이름으로 만든 링크다.
-- sdc 의 data1·data2, sdd 의 기본 이름 두 개
-- 공백은 \x20 으로 바뀐다.
-- MBR 디스크(sdb)의 파티션은 이름이 없어 여기에 없다
[root@ol7a ~]# blkid /dev/sdb1 /dev/sdc1 /dev/sdd1
/dev/sdb1: UUID="5623939f-b00c-4b2c-b3e4-87a5339c8589" TYPE="xfs" 
/dev/sdc1: PARTLABEL="data1" PARTUUID="71e81480-b996-4021-9e21-f88721e982fb" 
/dev/sdd1: PARTLABEL="Linux filesystem" PARTUUID="8036f5b3-6943-4269-b605-2566f2c51c39" 
-- sdb1 은 [6] 에서 만든 xfs 라 UUID·TYPE 이 나온다
-- sdc1·sdd1 은 파일 시스템이 없어 PARTLABEL·PARTUUID(GPT 항목 정보)만 있다

--------------------------------------------------------------------------------
 [변형] fdisk 로 만든 MBR 배치를 parted 로 다시 만들기 (sde)
--------------------------------------------------------------------------------
-- [3] 의 배치(주 파티션 2G 둘)를 parted 로 만든다.
--   mklabel msdos                MBR
--   mkpart primary …  msdos 에서 둘째 인자는 유형(primary 등)
[root@ol7a ~]# parted -s /dev/sde mklabel msdos mkpart primary 1MiB 2049MiB mkpart primary 2049MiB 4097MiB
[root@ol7a ~]# fdisk -l /dev/sde | grep -E '^/dev|^Disk label|identifier'
Disk label type: dos
Disk identifier: 0x000730af
/dev/sde1            2048     4196351     2097152   83  Linux
/dev/sde2         4196352     8390655     2097152   83  Linux
[root@ol7a ~]# fdisk -l /dev/sdb | grep -E '^/dev/sdb[12]'
/dev/sdb1            2048     4196351     2097152   83  Linux
/dev/sdb2         4196352     8390655     2097152   82  Linux swap / Solaris
-- 시작·끝 섹터가 sdb 와 같다.
-- fdisk +2G 와 parted 1MiB~2049MiB 가 같은 배치를 만들었다
-- 다른 점: parted 는 유형을 83 으로 두었고(유형 변경은 따로),
-- fdisk 의 대화 없이 한 줄로 끝났다
-- parted 의 단위는 MB(10^6) 와 MiB(2^20) 가 다르다. 단위 없이 쓰면 MB 다.
[root@ol7a ~]# parted -s /dev/sde unit MB print | tail -3
 1      1.05MB  2149MB  2147MB  primary
 2      2149MB  4296MB  2147MB  primary
-- 같은 파티션을 unit MB(10^6)로 보면 1.05MB·2149MB 가 된다.
-- 단위 없이 숫자만 쓰면 이 MB 로 읽힌다

[root@ol7a ~]# umount /mnt/p1; rmdir /mnt/p1
-- 먼저 sdb1 을 내린다.

 [정리] 관찰 요약
--------------------------------------------------------------------------------
--  항목  시작 전  변경 후  되돌린 뒤
--  sdb 테이블  없음  MBR, 1·2·(3)·4(확장)·5·6  없음
--  sdc / sdd 테이블  없음  GPT(parted) / GPT(gdisk)  없음
--  sde 테이블  없음  MBR(parted), sdb 1·2 와 같은 섹터 없음
--  /proc/partitions sdb*  sdb 한 줄  sdb1~6 (sdb3 은 busy 로 남음)  sdb 한 줄
--  by-id 의 sd 디스크  없음  없음(VMware 일련번호 없음)  없음
--
-- 확인한 사실
--   MBR 주 파티션은 넷. 다섯째는 확장 파티션 안의 논리 파티션(5번부터)
--   사용 중인 파티션이 있으면 fdisk w 의 다시 읽기가 실패한다
--   partprobe 는 사용 중이 아닌 파티션만 골라 커널에 알린다
