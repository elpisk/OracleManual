================================================================================
 제16장 실습 02 — firewalld 영역과 서비스: runtime 과 permanent
 대상   : ol7a(방화벽), ol7b(접속 시험)
 목표   : 활성 영역과 --list-all 출력의 각 줄을 읽고, 서비스·포트를 runtime 과
          permanent 에 따로 넣었을 때 언제 적용되고 언제 사라지는지 ol7b 에서
          실제 접속으로 확인한다. XML 파일이 어디에 생기는지도 본다
 배경   : firewalld 는 두 벌의 구성을 들고 있다. runtime 은 지금 커널(netfilter)에
          들어가 있는 규칙이고, permanent 는 /etc/firewalld 의 XML 이다.
          --permanent 를 붙인 변경은 파일만 바꾸고, 붙이지 않은 변경은 커널만
          바꾼다. --reload 는 runtime 을 버리고 permanent 로 다시 만든다(본문 16.4·16.5)
 소요   : 약 25분
 선행   : ol7-base, ol7b-base 스냅샷
 계정   : root
 주의   : ssh 서비스는 건드리지 않는다(러너와 강사 PC 의 접속 경로). --panic-on 금지
 되돌림 : public 영역의 허용 목록이 [0] 과 같음(services: dhcpv6-client ssh, ports 없음).
          확인: firewall-cmd --list-all, ls /etc/firewalld/zones
================================================================================

 [1] 영역 목록과 기본 정의
--------------------------------------------------------------------------------

[root@ol7a ~]# firewall-cmd --get-zones
block dmz drop external home internal public trusted work
-- 기본 영역 아홉 개. 이름은 신뢰 수준을 뜻할 뿐 구성은 XML 에 있다
-- 사용자 영역을 만들면 이 목록에 붙는다(실습 03 [변형])

-- 영역 정의 원본. target 이 없으면 default 동작(허용 목록에 없으면 거부)
[root@ol7a ~]# cat /usr/lib/firewalld/zones/public.xml
<?xml version="1.0" encoding="utf-8"?>
<zone>
  <short>Public</short>
  <description>For use in public areas. You do not trust the other computers on networks to not harm your computer. Only selected incoming connections are accepted.</description>
  <service name="ssh"/>
  <service name="dhcpv6-client"/>
</zone>
-- 원본 정의. short·description 은 설명이고 실제 규칙은 service 두 줄이다
-- 이 파일은 firewalld 패키지 소유. yum update 때 덮어쓰일 수 있어 고치지 않는다
-- target 속성이 없으니 default

-- 영역마다 target 을 비교한다. drop·block·trusted 는 target 이 정해져 있다
[root@ol7a ~]# grep -o 'target="[A-Z%-]*"' /usr/lib/firewalld/zones/*.xml
/usr/lib/firewalld/zones/block.xml:target="%%REJECT%%"
/usr/lib/firewalld/zones/drop.xml:target="DROP"
/usr/lib/firewalld/zones/trusted.xml:target="ACCEPT"
-- target 을 정한 영역은 셋뿐이다. block 의 %%REJECT%% 는 거부 응답을 보내는 REJECT
-- 나머지 여섯 영역은 default 로 public 과 같은 방식이다
-- default 와 %%REJECT%% 는 결과가 비슷하지만 default 는 ICMP 를 허용하고 block 은 거부한다
-- 영역 정의는 이 실습에서 바꾸지 않는다

--------------------------------------------------------------------------------
 [3] 시험 대상 준비 — 80·8080 리스너
--------------------------------------------------------------------------------

-- 웹 서버 대신 python 2.7 의 SimpleHTTPServer 를 띄운다. 기본 설치에 들어 있다
[root@ol7a ~]# nohup python -m SimpleHTTPServer 80 >/tmp/http80.log 2>&1 </dev/null &
[root@ol7a ~]# nohup python -m SimpleHTTPServer 8080 >/tmp/http8080.log 2>&1 </dev/null &
[root@ol7a ~]# sleep 1; ss -ltnp | grep -E ':(80|8080) '
LISTEN 0      5            0.0.0.0:80        0.0.0.0:*    users:(("python",pid=2217,fd=3)) 
LISTEN 0      5            0.0.0.0:8080      0.0.0.0:*    users:(("python",pid=2235,fd=3)) 
-- 두 리스너 모두 0.0.0.0 에 묶였다. 모든 주소에서 받는다
-- python 2.7 은 OL7 기본 설치에 있다. 웹 서버 대신 시험용으로만 쓴다
-- 80 은 1024 아래라 root 만 열 수 있다. 러너가 root 라 가능했다
-- 열: State · Recv-Q · Send-Q(대기 큐 한도 5) · Local · Peer · 프로세스

-- ol7a 자신에서는 열린다. 자기 주소로 보낸 패킷은 lo 로 들어가고 lo 는 항상 허용된다
[root@ol7a ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a/
200
-- ol7a 자신은 200. 자기 주소로 보낸 패킷은 lo 로 들어가고 lo 는 무조건 허용이다
-- 방화벽 시험을 서버 자신에서 하면 안 되는 이유다
-- 아래 두 명령부터는 프롬프트가 [root@ol7b ~]# 다. 바깥 클라이언트 자리

-- ol7b 에서 접속. -sS : 진행 표시는 끄고 오류는 보임, -m 3 : 3초 제한, -w : 응답 코드 출력
[root@ol7b ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a/; echo "rc=$?"
curl: (7) Failed connect to ol7a:80; No route to host
000
rc=7
[root@ol7b ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a:8080/; echo "rc=$?"
curl: (7) Failed connect to ol7a:8080; No route to host
000
rc=7
-- 증상   : ol7b 에서 80·8080 모두 "curl: (7) Failed connect to ol7a:80; No route to host"
-- 원인   : 리스너는 있다(위 ss). public 에 http·8080 이 없어 기본 거부가 icmp-host-prohibited 를 돌려보냈다
-- 조치   : [4] 에서 http 를 연다
-- 재확인 : 같은 curl 이 200
-- 응답 코드 000 은 HTTP 응답을 받지 못했다는 curl 의 표시, rc=7 은 연결 실패
-- No route to host 는 라우팅 문제처럼 읽히지만 여기서는 방화벽이다. 즉시 돌아왔다
-- ping 은 된다. public 은 ICMP 를 허용한다([10] 마지막 줄)
-- 같은 증상에서 ping 부터 치는 습관이 있으면 "네트워크는 정상" 까지 바로 좁혀진다

--------------------------------------------------------------------------------
 [4] runtime 에 서비스 추가
--------------------------------------------------------------------------------

-- --permanent 없이 : 커널 규칙만 바뀐다. 즉시 적용
[root@ol7a ~]# firewall-cmd --add-service=http
success
[root@ol7b ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a/; echo "rc=$?"
200
-- 옵션 없는 --add-service 는 즉시 적용됐다. 재시작·reload 없음
-- 방금 No route to host 였던 같은 명령이 1초도 안 돼 200 이다
rc=0

-- runtime 과 permanent 를 나란히 본다
[root@ol7a ~]# echo "runtime  : $(firewall-cmd --list-services)"; echo "permanent: $(firewall-cmd --permanent --list-services)"
runtime  : dhcpv6-client http ssh
permanent: dhcpv6-client ssh
-- runtime 에는 http 가 있고 permanent 에는 없다. 두 벌이 갈라졌다
-- 이 상태에서 재부팅·firewalld 재시작·reload 중 하나만 일어나도 http 는 닫힌다

-- 파일은 그대로다
[root@ol7a ~]# ls -l /etc/firewalld/zones/
total 8
-rw-r--r--. 1 root root 315 Nov 11 19:49 public.xml
-rw-r--r--. 1 root root 315 Nov 11 19:49 public.xml.old
-- 파일 시각이 그대로 11-11 19:49. runtime 변경은 디스크에 아무것도 쓰지 않는다
-- 이 시점에 firewalld 를 재시작하면 http 는 사라진다. reload 와 같은 효과다

--------------------------------------------------------------------------------
 [5] 오류 재현 ② — runtime 에만 넣고 reload
--------------------------------------------------------------------------------

-- 다른 관리자가 다른 변경을 반영하려고 reload 를 쳤다고 하자
[root@ol7a ~]# firewall-cmd --reload
success
[root@ol7a ~]# firewall-cmd --list-services
dhcpv6-client ssh
-- reload 뒤 http 가 사라졌다. runtime 을 버리고 permanent 로 다시 만들었기 때문이다
[root@ol7b ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a/; echo "rc=$?"
curl: (7) Failed connect to ol7a:80; No route to host
000
rc=7
-- 증상   : 방금 200 이던 ol7b 의 curl 이 다시 No route to host
-- 원인   : http 는 runtime 에만 있었다. reload 가 permanent(http 없음)로 다시 만들었다
-- 조치   : 처음부터 두 벌에 넣는다([7])
-- 재확인 : [7] 뒤 --permanent --list-services 에 http
-- 운영에서는 다른 사람의 reload 로 일어난다. 연 사람은 열려 있다고 믿고 있다
-- 반대로 "reload 하면 runtime 이 permanent 에 저장된다"고 오해하는 경우도 많다. 방향이 반대다

 [6] 오류 재현 ① — --permanent 만 쓰고 reload 안 함
--------------------------------------------------------------------------------

[root@ol7a ~]# firewall-cmd --permanent --add-port=8080/tcp
success
[root@ol7b ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a:8080/; echo "rc=$?"
curl: (7) Failed connect to ol7a:8080; No route to host
000
rc=7
-- 이번에는 permanent 에만 넣었다. success 가 나왔지만 접속은 여전히 거부
-- success 는 "파일에 썼다" 는 뜻이지 "열렸다" 는 뜻이 아니다

[root@ol7a ~]# echo "runtime  : [$(firewall-cmd --list-ports)]"; echo "permanent: [$(firewall-cmd --permanent --list-ports)]"
runtime  : []
permanent: [8080/tcp]
-- runtime 빈칸, permanent 8080/tcp. [4] 와 정반대로 갈라졌다
-- 대괄호는 빈 값을 눈에 보이게 하려고 echo 에 붙인 것이다
-- 포트는 --list-ports, 서비스는 --list-services 로 따로 본다

-- 이번에는 파일이 생겼다
[root@ol7a ~]# ls -l /etc/firewalld/zones/; cat /etc/firewalld/zones/public.xml
total 8
-rw-r--r--. 1 root root 352 Nov 16 10:38 public.xml
-rw-r--r--. 1 root root 315 Nov 11 19:49 public.xml.old
<?xml version="1.0" encoding="utf-8"?>
<zone>
  <short>Public</short>
  <description>For use in public areas. You do not trust the other computers on networks to not harm your computer. Only selected incoming connections are accepted.</description>
  <service name="ssh"/>
  <service name="dhcpv6-client"/>
  <port protocol="tcp" port="8080"/>
</zone>
-- public.xml 이 352바이트로 바뀌고 시각이 10:38. port 줄 하나가 늘었다
-- .old 는 315바이트 그대로 남아 있다. firewalld 가 쓰기 직전 판을 .old 로 돌린다
-- 파일 안 순서: service 들 다음에 port. firewalld 가 다시 쓸 때 형식을 정리한다
-- 주석·들여쓰기를 손으로 넣어도 firewall-cmd 가 다음에 쓰면 사라진다

-- 조치 : reload 로 permanent 를 runtime 에 올린다
[root@ol7a ~]# firewall-cmd --reload; firewall-cmd --list-ports
success
8080/tcp
[root@ol7b ~]# curl -sS -m 3 -o /dev/null -w '%{http_code}\n' http://ol7a:8080/; echo "rc=$?"
200
rc=0
-- 증상   : --permanent --add-port 가 success 인데 ol7b 는 No route to host
-- 원인   : 파일에만 있고 runtime(--list-ports)에는 없다
-- 조치   : firewall-cmd --reload
-- 재확인 : --list-ports 에 8080/tcp, ol7b 의 curl 이 200
-- reload 는 ol7a 의 기존 SSH 세션을 끊지 않았다. 연결 추적 정보를 유지한다
-- [OL8/9] 같은 명령 순서가 그대로 통한다. 차이는 백엔드뿐이다

--------------------------------------------------------------------------------
 [7] 두 벌에 함께 넣기 — 두 번 치기와 --runtime-to-permanent
--------------------------------------------------------------------------------

-- 방법 1 : 같은 명령을 --permanent 유무로 두 번
[root@ol7a ~]# firewall-cmd --add-service=http; firewall-cmd --permanent --add-service=http
success
success
-- 방법 1. 두 번 친다. 각각 success 하나씩

-- 방법 2 : runtime 에서 시험해 보고 맞으면 통째로 저장
[root@ol7a ~]# firewall-cmd --add-service=https; firewall-cmd --runtime-to-permanent
success
success
-- 방법 2. https 는 runtime 에만 넣고 runtime-to-permanent 로 통째로 저장했다
-- runtime-to-permanent 는 runtime 의 모든 영역을 저장한다. 시험 중인 잘못된 규칙까지 함께 들어간다
-- 저장 직전에 --list-all 로 runtime 을 한 번 더 본다
[root@ol7a ~]# firewall-cmd --permanent --list-services; grep -E 'service|port' /etc/firewalld/zones/public.xml
dhcpv6-client http https ssh
  <service name="ssh"/>
  <service name="dhcpv6-client"/>
  <service name="http"/>
  <service name="https"/>
  <port protocol="tcp" port="8080"/>
-- permanent 에 http·https 가 모두 들어갔다. 서비스 줄은 넣은 순서대로 붙는다

-- 백업 파일 public.xml.old 는 firewalld 가 permanent 를 쓸 때마다 직전 판을 남긴 것이다
[root@ol7a ~]# ls /etc/firewalld/zones/; diff /etc/firewalld/zones/public.xml.old /etc/firewalld/zones/public.xml
public.xml
public.xml.old
7a8
>   <service name="https"/>
-- .old 와의 차이는 https 한 줄이다. .old 는 "바로 직전 한 번" 만 담는다
-- http 를 넣은 쓰기가 먼저 있었고, 그 결과가 지금 .old 다
-- 두 단계 전으로 돌아가야 하면 .old 로는 부족하다. 바꾸기 전에 직접 사본을 둔다([변형])

 [8] 시간 제한 규칙 --timeout
--------------------------------------------------------------------------------

-- 점검 작업처럼 잠깐만 열 때. runtime 전용이고 시간이 지나면 저절로 빠진다
[root@ol7a ~]# firewall-cmd --add-port=9090/tcp --timeout=15s; firewall-cmd --list-ports
success
8080/tcp 9090/tcp
-- 넣은 직후 9090/tcp 가 보인다
-- timeout 값은 s·m·h 단위를 붙인다. 숫자만 쓰면 초
[root@ol7a ~]# firewall-cmd --list-ports
8080/tcp
-- 18초 뒤. 9090 이 저절로 빠졌다. reload 하지 않았다
-- 점검 창구를 잠깐 열 때 닫는 것을 잊어도 남지 않는다
-- 시간이 지나기 전에 reload 하면 그때 바로 사라진다(runtime 이므로)

-- --permanent 와 함께 쓰면 거부된다
[root@ol7a ~]# firewall-cmd --permanent --add-port=9090/tcp --timeout=15s; echo "rc=$?"

Can't specify timeout for permanent action.

-- --timeout 은 runtime 전용이다. 문구 원문: Can't specify timeout for permanent action.
-- rc=2 는 인자 오류. 아무것도 바뀌지 않았다
-- usage 줄은 표준 오류로, 원인 문장은 그다음 줄로 나왔다

--------------------------------------------------------------------------------
 [10] netfilter 에 실제로 들어간 규칙
--------------------------------------------------------------------------------

-- OL7.9 의 firewalld 0.6.3 은 iptables 백엔드를 쓴다. 영역별 체인 IN_public_allow 를 본다
[root@ol7a ~]# iptables -S IN_public_allow
-N IN_public_allow
-A IN_public_allow -p tcp -m tcp --dport 22 -m conntrack --ctstate NEW,UNTRACKED -j ACCEPT
-A IN_public_allow -p tcp -m tcp --dport 8080 -m conntrack --ctstate NEW,UNTRACKED -j ACCEPT
-A IN_public_allow -p tcp -m tcp --dport 80 -m conntrack --ctstate NEW,UNTRACKED -j ACCEPT
-A IN_public_allow -p tcp -m tcp --dport 443 -m conntrack --ctstate NEW,UNTRACKED -j ACCEPT
-- firewalld 가 만든 iptables 규칙. ssh(22)·8080·http(80)·https(443) 이 한 줄씩
-- ctstate NEW,UNTRACKED : 새 연결만 이 체인에서 판정한다. 이미 맺어진 연결은 INPUT 앞쪽에서 통과
-- 줄 순서는 넣은 순서다. 22(기본) → 8080([6]) → 80([7]) → 443([7])
-- --list-all 은 이름순으로 정렬해 보이지만 커널 규칙은 이 순서다
-- [OL8/9] firewalld 0.8 이후는 nftables 를 써서 이 자리를 nft list ruleset 으로 본다
[root@ol7a ~]# iptables -S INPUT_ZONES; iptables -S IN_public
-N INPUT_ZONES
-A INPUT_ZONES -i ens32 -g IN_public
-A INPUT_ZONES -g IN_public
-N IN_public
-A IN_public -j IN_public_log
-A IN_public -j IN_public_deny
-A IN_public -j IN_public_allow
-A IN_public -p icmp -j ACCEPT
-- INPUT_ZONES 는 ens32 로 들어온 패킷을 IN_public 으로 보낸다(-g : 돌아오지 않는 점프)
-- 둘째 줄은 인터페이스가 지정되지 않은 패킷의 기본 영역 처리다
-- IN_public 안에서 log → deny → allow 순서로 하위 체인을 지난다(실습 03 에서 다시 본다)
-- 마지막 줄 : ICMP 는 따로 막지 않는 한 허용
-- 영역 체인의 이름 규칙은 IN_영역 / IN_영역_log·_deny·_allow. 영역마다 같은 모양이다

-- 거부 방식. 허용 목록에 없으면 icmp-host-prohibited 로 REJECT
[root@ol7a ~]# iptables -S INPUT | tail -3
-A INPUT -j INPUT_ZONES
-A INPUT -m conntrack --ctstate INVALID -j DROP
-A INPUT -j REJECT --reject-with icmp-host-prohibited
-- 영역 체인에서 결정되지 않은 패킷은 INVALID 를 버린 뒤 이 REJECT 에 닿는다
-- 클라이언트가 본 No route to host 가 바로 이 줄의 icmp-host-prohibited 다
-- 이 세 줄 앞에는 ESTABLISHED 허용, lo 허용, INPUT_direct, INPUT_ZONES_SOURCE 가 있다
-- iptables -S INPUT 전체는 실습 03 [4] 에서 본다

 [변형] XML 을 직접 고쳐 같은 결과 만들기
--------------------------------------------------------------------------------

-- firewall-cmd 대신 파일을 고친다. 8081 포트 한 줄을 넣는다
[root@ol7a ~]# cp /etc/firewalld/zones/public.xml /root/public.xml.before
[root@ol7a ~]# sed -i 's|^</zone>|  <port protocol="tcp" port="8081"/>\n</zone>|' /etc/firewalld/zones/public.xml
[root@ol7a ~]# diff /root/public.xml.before /etc/firewalld/zones/public.xml
9a10
>   <port protocol="tcp" port="8081"/>
-- </zone> 앞에 port 한 줄을 넣었다. firewall-cmd 를 쓰지 않았다

-- 파일만 바꿨으니 runtime 에는 아직 없다. 검사 후 reload
[root@ol7a ~]# firewall-cmd --list-ports; firewall-cmd --check-config; echo "rc=$?"
8080/tcp
success
rc=0
-- 파일만 바꿔서 runtime 에는 아직 8080 뿐. --check-config 는 success, rc=0
-- --check-config 는 permanent 파일 전체(영역·서비스·direct)를 읽어 검사한다
[root@ol7a ~]# firewall-cmd --reload; firewall-cmd --list-ports
success
8080/tcp 8081/tcp
-- reload 뒤 8081 이 runtime 에 올라왔다. --permanent 명령과 결과가 같다
-- 여러 서버에 같은 영역을 배포할 때 이 방법을 쓴다. 파일을 복사하고 reload

-- 일부러 XML 을 깨뜨려 본다. 닫는 태그를 지운다
[root@ol7a ~]# sed -i 's|^</zone>||' /etc/firewalld/zones/public.xml; firewall-cmd --check-config; echo "rc=$?"
Error: INVALID_ZONE: 'public.xml': not a valid zone file: no element found: line 12, column 0
rc=112
-- 닫는 태그를 지우자 --check-config 가 오류를 냈다
-- 12번째 줄 0열 = 파일 끝. 닫는 태그가 있어야 할 자리를 정확히 가리킨다
-- 원문: INVALID_ZONE: 'public.xml': not a valid zone file: no element found: line 12, column 0
-- 깨진 파일로 reload 하지 않는다. 파일 이름과 줄 번호가 함께 나오니 바로 찾는다
-- rc 112 는 firewall-cmd 의 INVALID_ZONE 오류 코드다

[root@ol7a ~]# cp /root/public.xml.before /etc/firewalld/zones/public.xml; firewall-cmd --reload; firewall-cmd --list-ports
success
8080/tcp
-- 사본으로 복구하고 reload. 8081 이 빠지고 8080 만 남았다
-- 사본(public.xml.before)은 8081 을 넣기 전에 떠 두었다. 그래서 8081 도 함께 없어졌다
