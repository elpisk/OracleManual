================================================================================
 제7장 실습 04 — 부팅 시 적재와 블랙리스트
 대상   : ol7a
 목표   : /etc/modules-load.d 로 부팅 때 모듈을 올리고, blacklist 가 무엇을 막고
          무엇을 못 막는지(이름 직접 적재, 의존성, 별칭) 구분한다. install ... /bin/false
          로 완전히 막는 방법과, initramfs 안에 든 모듈은 dracut -f 로 다시 만들기
          전까지 블랙리스트가 듣지 않는 것을 재부팅으로 확인한다
 배경   : 부팅 중 모듈은 세 경로로 올라온다. initramfs 안의 udev, 루트 전환 뒤의
          udev(장치 별칭 modalias), systemd-modules-load(/etc/modules-load.d).
          blacklist 는 "별칭으로 찾을 때 이 모듈을 후보에서 뺀다" 는 뜻일 뿐이라
          이름을 직접 부르거나 의존성으로 끌려오는 적재는 막지 못한다. initramfs 는
          만들 때의 /etc/modprobe.d 사본을 들고 있다(본문 7.5)
 소요   : 약 40분 (재부팅 4회, dracut 2회)
 선행   : ol7-base 스냅샷
 계정   : root
 주의   : initramfs 를 다시 만든다. 잘못 만들면 부팅이 멈출 수 있으므로 스냅샷이
          있는 상태에서만 한다. 실습 끝에서 원래 구성으로 다시 만든다
 되돌림 : 만든 conf 네 개 삭제, initramfs 재생성, 재부팅. lsmod 의 dummy 없음,
          pcspkr 적재(기준 상태와 같음), lsinitrd 에 pcspkr 없음

 [3] blacklist 는 이름 직접 적재를 막지 못한다
--------------------------------------------------------------------------------
-- 블랙리스트 파일은 이름을 아무렇게나 지어도 되지만 .conf 로 끝나야 한다.
[root@ol7a ~]# echo 'blacklist vfat' > /etc/modprobe.d/m07-blacklist.conf; modprobe -c | grep -c '^blacklist'; modprobe -c | grep -E '^blacklist (vfat|fat)$'
32
blacklist vfat
-- modprobe -c 의 blacklist 줄 32 개. 패키지가 넣은 31 개(dccp, 프레임버퍼, snd_pcsp 등)에 방금 넣은 vfat 하나

-- 이름으로 부르면 그냥 올라간다.
[root@ol7a ~]# modprobe -v vfat; lsmod | grep -w -E 'vfat|fat'
insmod /lib/modules/5.4.17-2136.338.4.2.el7uek.x86_64/kernel/fs/fat/fat.ko.xz 
insmod /lib/modules/5.4.17-2136.338.4.2.el7uek.x86_64/kernel/fs/fat/vfat.ko.xz 
vfat                   20480  0 
fat                    73728  1 vfat
-- blacklist vfat 이 있는데도 modprobe -v vfat 이 insmod 두 줄로 성공했다
-- 이름을 직접 부르는 적재는 블랙리스트를 보지 않는다

-- -b(--use-blacklist) 를 붙이면 그때서야 블랙리스트가 적용된다.
[root@ol7a ~]# modprobe -r vfat; modprobe -v -b vfat; echo "rc=$?"; lsmod | grep -w vfat; echo "rc=$?"

rc=1
-- modprobe -v -b vfat: 출력 없이 rc=0, lsmod 에 vfat 없음(rc=1)
-- -b 를 주면 블랙리스트를 적용한다. 적재하지 않았는데 rc=0 이라 실패로 보이지 않는다는 점에 주의

 [4] blacklist 가 막는 것 — 별칭으로 찾는 적재
--------------------------------------------------------------------------------
-- 커널이 모르는 파일 시스템을 마운트하려 하면 "fs-이름" 별칭으로 modprobe 를 부른다.
[root@ol7a ~]# modinfo -F alias vfat
fs-vfat
-- vfat 의 alias 는 fs-vfat 하나. 커널이 vfat 마운트를 처리하려고 부르는 이름이다

[root@ol7a ~]# mkdir -p /root/m07 && dd if=/dev/zero of=/root/m07/fat.img bs=1M count=32 2>/dev/null && mkfs.vfat /root/m07/fat.img >/dev/null && ls -l /root/m07/fat.img
-rw-r--r--. 1 root root 33554432 Nov 13 20:10 /root/m07/fat.img
-- 32MiB 이미지를 vfat 으로 포맷했다. 출력은 ls 한 줄만 남기고 버렸다(2>/dev/null, >/dev/null)

[root@ol7a ~]# modprobe -v fs-vfat; echo "rc=$?"; lsmod | grep -w vfat; echo "rc=$?"
rc=0
rc=1
-- modprobe -v fs-vfat: 출력 없이 rc=0, vfat 미적재(rc=1)
-- 별칭으로 찾을 때는 블랙리스트가 적용됐다. modprobe 가 조용히 아무것도 고르지 않았다

[root@ol7a ~]# mount -o loop /root/m07/fat.img /mnt; echo "rc=$?"; dmesg | tail -3
mount: unknown filesystem type 'vfat'
rc=32
[    3.979491] IPv6: ADDRCONF(NETDEV_CHANGE): ens33: link becomes ready
[   15.184967] loop: module loaded
[   15.226717] request_module fs-vfat succeeded, but still no fs?
-- mount 원문: mount: unknown filesystem type 'vfat', rc=32(mount 의 실패 코드)
-- dmesg 마지막 줄: request_module fs-vfat succeeded, but still no fs?
-- 커널의 modprobe fs-vfat 요청은 성공(0)했지만 파일 시스템이 등록되지 않았다

--------------------------------------------------------------------------------
 [5] 오류 ① — fat 을 막았는데 fat 이 올라온다
--------------------------------------------------------------------------------
-- 블랙리스트를 vfat 에서 fat 으로 바꾼다. vfat 은 fat 에 의존한다.
[root@ol7a ~]# echo 'blacklist fat' > /etc/modprobe.d/m07-blacklist.conf; modprobe -c | grep -c '^blacklist'; modprobe -c | grep -E '^blacklist (vfat|fat)$'
32
blacklist fat
-- 이번에는 vfat 대신 fat 을 막았다. 개수 32 는 같고(vfat 이 빠지고 fat 이 들어감) 걸린 이름은 fat

[root@ol7a ~]# mount -o loop /root/m07/fat.img /mnt; echo "rc=$?"; findmnt /mnt; lsmod | grep -w -E 'vfat|fat'
rc=0
TARGET SOURCE     FSTYPE OPTIONS
/mnt   /dev/loop0 vfat   rw,relatime,fmask=0022,dmask=0022,codepage=437,iocharset=ascii,shortname=mixed,errors=remount-ro
vfat                   20480  1 
fat                    73728  1 vfat
-- 마운트 성공(rc=0), /dev/loop0 vfat. lsmod 에 vfat 과 fat 이 모두 있다
-- 커널의 fs-vfat 요청 → vfat 선택(블랙리스트 아님) → vfat 의 의존 모듈 fat 을 함께 적재
-- 블랙리스트는 별칭 후보를 고를 때만 적용된다. 의존성으로 끌려오는 fat 은 막지 않았다

[root@ol7a ~]# umount /mnt; modprobe -r vfat; lsmod | grep -w -E 'vfat|fat'; echo "rc=$?"
rc=1
-- umount 후 vfat 을 내리자 fat 도 함께 내려갔다(rc=1)
--
-- 증상   : /etc/modprobe.d/m07-blacklist.conf 에 blacklist fat 을 넣었는데 vfat 이미지 마운트가 rc=0 으로
--          성공하고 lsmod 에 'fat 73728 1 vfat' 이 보임
-- 원인   : blacklist 는 별칭으로 모듈을 찾을 때만 적용된다. fat 은 별칭이 아니라 vfat 의 의존성으로
--          적재됐다(modules.dep 의 vfat → fat)
-- 조치   : 적재 자체를 막으려면 install fat /bin/false 를 같은 파일에 넣는다([6])
-- 재확인 : 같은 mount 가 unknown filesystem type 'vfat' 로 실패, lsmod 에 fat 없음

--------------------------------------------------------------------------------
 [6] install ... /bin/false — 정말로 막기
--------------------------------------------------------------------------------
-- install 은 "적재 대신 이 명령을 실행". /bin/false 는 아무것도 안 하고 실패를 돌려준다.
[root@ol7a ~]# echo 'install fat /bin/false' >> /etc/modprobe.d/m07-blacklist.conf; cat /etc/modprobe.d/m07-blacklist.conf
blacklist fat
install fat /bin/false
-- 파일에 두 줄. blacklist 는 별칭 경로, install 은 나머지 모든 경로를 맡는다

[root@ol7a ~]# modprobe -v fat; echo "rc=$?"
modprobe: ERROR: Error running install command for fat
modprobe: ERROR: could not insert 'fat': Operation not permitted
install /bin/false 
rc=1
-- modprobe -v fat 원문 두 줄
--   modprobe: ERROR: Error running install command for fat
--   modprobe: ERROR: could not insert 'fat': Operation not permitted
-- 셋째 줄 install /bin/false 는 -v 가 보여 준 실제 실행 명령이다. /bin/false 는 1 을 돌려준다

[root@ol7a ~]# modprobe -v vfat; echo "rc=$?"
modprobe: ERROR: Error running install command for fat
modprobe: ERROR: could not insert 'vfat': Operation not permitted
install /bin/false 
rc=1
-- vfat 을 부르자 의존 모듈 fat 단계에서 같은 install 이 실행돼 vfat 까지 실패했다
-- 오류 문구의 'vfat' 부분만 다르다. 의존성 경로도 막혔다

[root@ol7a ~]# mount -o loop /root/m07/fat.img /mnt; echo "rc=$?"; lsmod | grep -w -E 'vfat|fat'; echo "rc=$?"
mount: unknown filesystem type 'vfat'
rc=32
rc=1
-- 마운트도 unknown filesystem type 'vfat'(rc=32), lsmod 에 둘 다 없음(rc=1)

-- 비상구: --ignore-install 은 install 줄을 무시하고 원래대로 적재한다.
[root@ol7a ~]# modprobe -v --ignore-install fat; lsmod | grep -w fat; modprobe -r fat
insmod /lib/modules/5.4.17-2136.338.4.2.el7uek.x86_64/kernel/fs/fat/fat.ko.xz 
fat                    73728  0 
-- --ignore-install 은 install 줄을 건너뛰고 원래 insmod 를 실행했다. 장애 대응 때만 쓰는 비상구다

--------------------------------------------------------------------------------
 [7] 재부팅 — modules-load.d 와 udev 경로에 blacklist 걸기
--------------------------------------------------------------------------------
-- pcspkr 를 블랙리스트에 넣는다. udev 가 별칭(platform:pcspkr)으로 부르므로 막혀야 한다.
-- dummy 도 블랙리스트에 넣어 systemd-modules-load 가 블랙리스트를 따르는지 본다.
[root@ol7a ~]# printf 'blacklist pcspkr\nblacklist dummy\n' > /etc/modprobe.d/m07-boot.conf; cat /etc/modprobe.d/m07-boot.conf
blacklist pcspkr
blacklist dummy
-- pcspkr 는 udev 경로(장치 별칭), dummy 는 systemd-modules-load 경로. 두 경로를 한 번의 재부팅으로 본다

-- 재부팅

[root@ol7a ~]# uptime; lsmod | grep -w -E 'dummy|pcspkr'; echo "rc=$?"

rc=1
-- up 0 min. 재부팅 직후 dummy 도 pcspkr 도 없다(rc=1)
-- 기준 상태에서 늘 올라와 있던 pcspkr 가 빠졌다. udev 의 별칭 적재에 블랙리스트가 적용됐다
