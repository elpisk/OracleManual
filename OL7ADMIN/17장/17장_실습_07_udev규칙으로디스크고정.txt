================================================================================
 제17장 실습 07 — udev 규칙으로 ASM 후보 디스크 이름·소유자·권한 고정
 대상   : ol7a
 목표   : scsi_id 로 디스크 식별자를 읽고, VMware 에서 그 값이 비어 있을 때 쓸 수
          있는 다른 식별자(GPT 파티션 이름, ID_PATH)를 고른다. 규칙 파일로
          /dev/oracleasm/ 아래 고정 이름과 grid:asmadmin 0660 을 주고, udevadm
          test·trigger 로 적용한 뒤 재부팅해도 유지되는지 확인한다
 배경   : /dev/sdX 이름은 장치 발견 순서로 정해져 부팅마다 바뀔 수 있고, 장치 노드의
          소유자는 부팅 때마다 root:disk 로 다시 만들어진다. udev 규칙은 장치가
          나타날 때마다 같은 조건으로 이름(SYMLINK)·OWNER·GROUP·MODE 를 붙인다.
          조건(==)과 대입(=, +=)을 헷갈리면 규칙이 조용히 무시된다(본문 17.10)
 소요   : 약 30분
 선행   : ol7-base 스냅샷(sdb~sde 비어 있음)
 계정   : root
 주의   : sdb·sdc 에 GPT 파티션을 만든다. 대상이 빈 디스크인지 HCTL 로 먼저 확인
 되돌림 : 규칙 삭제, 파티션 지움, 시험 사용자·그룹 삭제. 확인은 lsblk, ls /dev/oracleasm

 [1] scsi_id — 디스크 고유 번호
--------------------------------------------------------------------------------

-- -g : 화이트리스트로 취급, -u : 공백을 _ 로, -d : 대상 장치
[root@ol7a ~]# /usr/lib/udev/scsi_id -g -u -d /dev/sdb; echo "exit=$?"
exit=1
-- scsi_id 는 /usr/lib/udev 아래에 있어
--   PATH 에 없다. 전체 경로로 부른다
-- RAC 설치 가이드 예제 규칙의 PROGRAM=="scsi_id
--   …", RESULT=="…" 형식이 이 출력을 쓴다
-- 증상   : scsi_id 가 아무것도 출력하지 않고 종료 코드 1
-- 원인   : VMware 가상 디스크는 vmx 에
--          disk.EnableUUID = "TRUE" 가 없으면
--          SCSI VPD 페이지(0x80·0x83)의 일련번호를 내주지 않는다
--          scsi_id 는 그 값을 읽어 WWID 를 만드는 도구라 읽을 것이 없다
[root@ol7a ~]# /usr/lib/udev/scsi_id -g -u -d /dev/sda; echo "exit=$?"

-- OS 디스크 sda 도 같다. 디스크 하나의 문제가 아니라 이
--   가상 머신 전체의 설정이다
-- 실무에서는 VM 을 끄고 vmx 에
--   disk.EnableUUID 를 넣은 뒤 다시 켜면
--   scsi_id 가 값을 낸다
-- 이 랩은 VM 설정을 바꾸지 않는 규칙이라 다른 식별자로 간다

 [3] GPT 파티션과 파티션 이름
--------------------------------------------------------------------------------

-- 대상이 HCTL 2:0:1:0(sdb)·2:0:2:0(sdc) 이고 비어 있을 때만 parted 가 돈다
[root@ol7a ~]# for d in sdb:2:0:1:0 sdc:2:0:2:0; do n=${d%%:*}; h=${d#*:}; [ "$(lsblk -dno HCTL /dev/$n | xargs)" = "$h" ] && [ -z "$(lsblk -no FSTYPE /dev/$n | xargs)" ] && parted -s /dev/$n mklabel gpt mkpart asm_${n/sd/} 1MiB 100% && echo "$n ok"; done
sdb ok
sdc ok
-- 조건 세 가지를 &&로 이었다: 이름과 HCTL 이 맞는가, 파일
--   시스템 서명이 없는가, 그 다음에 parted
-- lsblk -dno HCTL : -d 디스크만, -n 머리줄
--   없이, -o HCTL 한 열. xargs 로 앞뒤 공백을 지운다
-- mklabel gpt : GPT 파티션 테이블, mkpart
--   asm_b 1MiB 100% : 파티션 이름 asm_b,
--   1MiB 정렬부터 끝까지
-- ${n/sd/} 는 sdb 에서 sd 를 지워 b 만 남긴다.
--   그래서 이름이 asm_b, asm_c
[root@ol7a ~]# udevadm settle; lsblk -o NAME,SIZE,TYPE,PARTLABEL,PARTUUID /dev/sdb /dev/sdc
NAME   SIZE TYPE PARTLABEL PARTUUID
sdb     10G disk           
└─sdb1  10G part asm_b     388bc711-f78a-4eb5-beca-1a4f15747c88
sdc     10G disk           
└─sdc1  10G part asm_c     2827309f-8666-484c-90ca-095eba5b1b68
-- PARTLABEL(파티션 이름)과 PARTUUID 가 생겼다.
--   둘 다 GPT 에 기록돼 디스크를 따라다닌다
-- MBR(msdos) 파티션에는 이름이 없다. 파티션 이름을
--   쓰려면 GPT 여야 한다
-- 파티션 이름은 사람이 정하니 규칙이 읽기 쉽다.
--   PARTUUID 는 parted 가 무작위로 만든다
[root@ol7a ~]# udevadm info --query=property --name=/dev/sdb1 | grep -E '^ID_PART_ENTRY_(NAME|UUID|SCHEME)'
ID_PART_ENTRY_NAME=asm_b
ID_PART_ENTRY_SCHEME=gpt
ID_PART_ENTRY_UUID=388bc711-f78a-4eb5-beca-1a4f15747c88
-- udev 는 이 값을 ID_PART_ENTRY_NAME·UUID
--   속성으로 갖는다. 규칙의 ENV{...} 로 비교할 수 있다
-- 처음 실측에서는 settle 없이 lsblk 를 쳐서
--   PARTLABEL 이 빈칸으로 나왔다. udev
--   처리가 끝나기 전이었다

--------------------------------------------------------------------------------
 [4] 규칙 작성 — OWNER 를 == 로 쓰면
--------------------------------------------------------------------------------

-- 한 줄이 규칙 하나다. 쉼표로 이은 키 중 == 는
--   조건, = 와 += 는 대입
-- 한 줄이 길어지면 끝에 \ 를 두고 다음 줄로 이을 수 있다
--   KERNEL=="sd?1"                  : 커널 이름이 sd 한 글자 1 인 장치(sdb1, sdc1 …)
--   SUBSYSTEM=="block"              : 블록 장치만
--   ENV{ID_PART_ENTRY_NAME}=="asm_b" : 파티션 이름이 asm_b 인 것
--   SYMLINK+="oracleasm/asm_b"       : /dev/oracleasm/asm_b 링크를 더한다
--   OWNER·GROUP·MODE                 : 장치 노드의 소유자·그룹·권한
-- 설계: OWNER 를 일부러 == 로 써서 "대입 키를 조건으로
--   쓰면 규칙이 무시된다" 를 보이려 했다
[root@ol7a ~]# cat > /etc/udev/rules.d/99-oracle-asmdevices.rules <<'EOF'
KERNEL=="sd?1", SUBSYSTEM=="block", ENV{ID_PART_ENTRY_NAME}=="asm_b", SYMLINK+="oracleasm/asm_b", OWNER=="grid", GROUP="asmadmin", MODE="0660"
EOF
[root@ol7a ~]# ls -l /dev/sdb1; udevadm test /sys/block/sdb/sdb1 2>&1 | grep -E '99-oracle' | head -6
brw-rw----. 1 root disk 8, 17 Nov 16 11:11 /dev/sdb1
Reading rules file: /etc/udev/rules.d/99-oracle-asmdevices.rules
OWNER 54331 /etc/udev/rules.d/99-oracle-asmdevices.rules:1
GROUP 54329 /etc/udev/rules.d/99-oracle-asmdevices.rules:1
MODE 0660 /etc/udev/rules.d/99-oracle-asmdevices.rules:1
LINK 'oracleasm/asm_b' /etc/udev/rules.d/99-oracle-asmdevices.rules:1
-- udevadm test 는 장치에 이벤트가 왔다고 가정하고 규칙을
--   돌려 무엇이 붙는지 보여 준다(실제로 바꾸지는 않음)
-- grep 99-oracle 로 이 파일과 관련된 줄만 남겼다.
--   전체 출력은 수십 줄이다
-- 각 줄의 :1 은 규칙 파일의 줄 번호다. 어느 줄이 값을
--   붙였는지 바로 보인다
-- OWNER 54331 이 1 번 줄에서 붙었다. udev
--   219 는 OWNER 에 쓴 == 를 대입으로 받아들였다
-- 설계한 오류는 7.9 에서 재현되지 않았다. 다른 버전의 udev
--   에서는 오류로 다룰 수 있으니 = 로 쓰는 것이 맞다
[root@ol7a ~]# udevadm control --reload; udevadm trigger --action=change --sysname-match=sdb1; udevadm settle; ls -l /dev/oracleasm/ /dev/sdb1
brw-rw----. 1 grid asmadmin 8, 17 Nov 16 11:11 /dev/sdb1

/dev/oracleasm/:
total 0
lrwxrwxrwx. 1 root root 7 Nov 16 11:11 asm_b -> ../sdb1
-- control --reload : 규칙 파일을 다시 읽게 한다.
--   trigger --action=change
--   --sysname-match=sdb1 : sdb1 에 change 이벤트
-- udev 219 는 규칙 파일이 바뀌면 스스로 다시 읽기도 하지만
--   reload 를 해 두면 확실하다
-- trigger 를 빼면 규칙은 다음 이벤트(재부팅, 디스크
--   재연결)까지 적용되지 않는다
-- settle : 이벤트 처리가 끝날 때까지 기다린다. 결과:
--   sdb1 이 grid asmadmin, 링크
--   asm_b → ../sdb1
[root@ol7a ~]# sed -i 's/OWNER=="grid"/OWNER="grid"/' /etc/udev/rules.d/99-oracle-asmdevices.rules; cat /etc/udev/rules.d/99-oracle-asmdevices.rules
KERNEL=="sd?1", SUBSYSTEM=="block", ENV{ID_PART_ENTRY_NAME}=="asm_b", SYMLINK+="oracleasm/asm_b", OWNER="grid", GROUP="asmadmin", MODE="0660"
-- 결과와 상관없이 OWNER="grid" 로 바로잡아 둔다

--------------------------------------------------------------------------------
 [5] 두 번째 디스크 — SYMLINK 를 == 로 쓰면
--------------------------------------------------------------------------------

-- 둘째 디스크 규칙에서 이번에는 SYMLINK 를 == 로 썼다.
--   SYMLINK 는 조건으로도 쓸 수 있는 키다
-- echo '…' >> 파일 : 홑따옴표라 안의
--   큰따옴표와 == 가 그대로 들어간다
[root@ol7a ~]# echo 'KERNEL=="sd?1", SUBSYSTEM=="block", ENV{ID_PART_ENTRY_NAME}=="asm_c", SYMLINK=="oracleasm/asm_c", OWNER="grid", GROUP="asmadmin", MODE="0660"' >> /etc/udev/rules.d/99-oracle-asmdevices.rules; tail -1 /etc/udev/rules.d/99-oracle-asmdevices.rules
KERNEL=="sd?1", SUBSYSTEM=="block", ENV{ID_PART_ENTRY_NAME}=="asm_c", SYMLINK=="oracleasm/asm_c", OWNER="grid", GROUP="asmadmin", MODE="0660"
[root@ol7a ~]# udevadm control --reload; udevadm trigger --action=change --sysname-match=sdc1; udevadm settle; ls -l /dev/oracleasm/; ls -l /dev/sdc1
total 0
lrwxrwxrwx. 1 root root 7 Nov 16 11:11 asm_b -> ../sdb1
brw-rw----. 1 root disk 8, 33 Nov 16 11:11 /dev/sdc1
-- 증상   : 오류 메시지 없이 asm_c 링크가 생기지 않았고,
--          sdc1 은 root disk 그대로다
-- 원인   : SYMLINK=="oracleasm/asm_c" 는
--          "지금 그 링크가 있는가" 를 묻는
--          조건이다. 아직 없으니 거짓,
--          같은 줄의 OWNER·GROUP·MODE 대입까지 통째로 건너뛰었다
[root@ol7a ~]# udevadm test /sys/block/sdc/sdc1 2>&1 | grep -E '99-oracle' | head -6; echo "lines=$(udevadm test /sys/block/sdc/sdc1 2>&1 | grep -c 99-oracle-asmdevices.rules:2)"
Reading rules file: /etc/udev/rules.d/99-oracle-asmdevices.rules
lines=0
-- udevadm test 에도 2 번 줄이 한 번도 나오지
--   않는다(lines=0). 규칙 파일은 읽었지만
--   그 줄이 맞지 않았다
-- grep -c 로 2 번 줄 기록을 셌다. 0 이면 그 규칙이
--   한 번도 맞지 않았다
-- 1 번 줄과 달리 OWNER·GROUP·LINK 기록이
--   없다는 것으로 알아본다
[root@ol7a ~]# sed -i 's|SYMLINK=="oracleasm/asm_c"|SYMLINK+="oracleasm/asm_c"|' /etc/udev/rules.d/99-oracle-asmdevices.rules; tail -1 /etc/udev/rules.d/99-oracle-asmdevices.rules
KERNEL=="sd?1", SUBSYSTEM=="block", ENV{ID_PART_ENTRY_NAME}=="asm_c", SYMLINK+="oracleasm/asm_c", OWNER="grid", GROUP="asmadmin", MODE="0660"
-- 조치   : SYMLINK== 를 SYMLINK+= 로 고쳤다.
--          sed 의 구분자를 | 로 바꿔 / 가
--          든 경로를 그대로 썼다
-- SYMLINK= 로 쓰면 다른 규칙이 만든 링크(by-path
--   등)를 지우고 이것 하나만 남긴다. += 가 안전하다
[root@ol7a ~]# udevadm test /sys/block/sdc/sdc1 2>&1 | grep -E '99-oracle' | head -6
Reading rules file: /etc/udev/rules.d/99-oracle-asmdevices.rules

GROUP 54329 /etc/udev/rules.d/99-oracle-asmdevices.rules:2
MODE 0660 /etc/udev/rules.d/99-oracle-asmdevices.rules:2
LINK 'oracleasm/asm_c' /etc/udev/rules.d/99-oracle-asmdevices.rules:2
-- 이제 2 번 줄에서 OWNER·GROUP·MODE·LINK 네
--   줄이 붙는다. 적용하기 전에 test 로 먼저 본 것이다
-- 고친 뒤에도 trigger 전까지는 장치가 바뀌지 않는다.
--   test 는 미리 보기일 뿐이다
[root@ol7a ~]# udevadm control --reload; udevadm trigger --action=change --sysname-match='sd?1'; udevadm settle; ls -l /dev/oracleasm/; ls -l /dev/sdb1 /dev/sdc1
total 0

brw-rw----. 1 grid asmadmin 8, 33 Nov 16 11:11 /dev/sdc1
-- 재확인 : sd?1 전체에 change 이벤트를 보내자 asm_c
--         → ../sdc1 링크가 생기고 sdc1 이
--         grid asmadmin 이 됐다
-- 링크 자체는 root 소유 777 이다. 권한은 링크가 아니라

 [7] 손으로 준 chown 은 어디까지 가나
--------------------------------------------------------------------------------

[root@ol7a ~]# chown grid:asmadmin /dev/sde; ls -l /dev/sde
brw-rw----. 1 grid asmadmin 8, 64 Nov 16 11:11 /dev/sde
-- 규칙이 없는 sde 를 chown 으로 grid 에게 줬다.
--   급할 때 흔히 하는 방법이다
-- chown 은 /dev 의 장치 노드(tmpfs 위의 파일)만
--   바꾼다. 디스크 내용과는 관계없다
[root@ol7a ~]# udevadm trigger --action=change --sysname-match=sde; udevadm settle; ls -l /dev/sde
brw-rw----. 1 grid asmadmin 8, 64 Nov 16 11:11 /dev/sde
-- change 이벤트를 보내도 grid asmadmin 그대로다.
--   udev 는 규칙이 정한 값만 다시 쓰고, 정한 것이
--   없으면 건드리지 않았다
-- 그래서 손으로 한 chown 이 "잘 되는 것처럼" 보인다.
--   남는지는 [8] 재부팅에서 본다

 [변형] 파티션 없이 디스크 전체를 ID_PATH 로

[root@ol7a ~]# udevadm info --query=property --name=/dev/sdd | grep -E '^ID_PATH='
ID_PATH=pci-0000:00:10.0-scsi-0:0:3:0

--   표준)를 위해 연결 위치로 식별한다
-- sdd 는 아직 아무 규칙도 없고 파티션도 없다
[root@ol7a ~]# P=$(udevadm info --query=property --name=/dev/sdd | sed -n 's/^ID_PATH=//p'); echo "KERNEL==\"sd?\", SUBSYSTEM==\"block\", ENV{DEVTYPE}==\"disk\", ENV{ID_PATH}==\"$P\", SYMLINK+=\"oracleasm/asm_d\", OWNER=\"grid\", GROUP=\"asmadmin\", MODE=\"0660\"" >> /etc/udev/rules.d/99-oracle-asmdevices.rules; tail -1 /etc/udev/rules.d/99-oracle-asmdevices.rules
KERNEL=="sd?", SUBSYSTEM=="block", ENV{DEVTYPE}=="disk", ENV{ID_PATH}=="pci-0000:00:10.0-scsi-0:0:3:0", SYMLINK+="oracleasm/asm_d", OWNER="grid", GROUP="asmadmin", MODE="0660"
-- 셸 변수 P 에 ID_PATH 값을 받아 규칙 줄을 만들었다.
--   큰따옴표를 \" 로 넣었다
-- sed -n 's/^ID_PATH=//p' : ID_PATH= 줄에서 값만 찍는다
-- KERNEL=="sd?" 와
--   DEVTYPE=="disk" : 파티션이 아니라
--   디스크 자체에만 맞는다

--   → ../sdd. 파티션 이름 방식과 결과가 같다
-- 차이는 식별 기준이다. 파티션 이름은 디스크를 다른

--   sde 와 같은 동작이다

--   직접 chown 하거나 재부팅해야 한다

[root@ol7a ~]# wipefs -a /dev/sdb1 /dev/sdc1 >/dev/null; wipefs -a /dev/sdb /dev/sdc >/dev/null; partprobe /dev/sdb /dev/sdc; udevadm settle; chown root:disk /dev/sdd; ls -l /dev/sdd

--   sdd 는 chown root:disk 로 직접 돌렸다
-- wipefs -a 를 디스크에 하면 GPT 앞뒤 사본과 보호
--   MBR 서명을 모두 지운다
-- partprobe 로 커널이 파티션 표를 다시 읽게 했다
[root@ol7a ~]# userdel -r grid; groupdel asmadmin; groupdel asmdba; groupdel oinstall
