================================================================================
 제17장 실습 05 — THP 비활성화: sysfs, 부트 파라미터, 모든 커널에 적용
 대상   : ol7a
 목표   : Transparent HugePages 의 현재 모드를 sysfs 에서 읽고, never 로 바꾼 값이
          재부팅하면 사라지는 것을 본다. grubby 로 부트 파라미터에 넣어 영구화하되
          한 커널에만 넣으면 다른 커널로 부팅했을 때 빠진다는 것을 RHCK 부팅으로
          확인하고, 모든 커널과 앞으로 설치될 커널에 적용한다
 배경   : THP 는 커널이 일반 메모리를 2MB 페이지로 자동으로 묶어 주는 기능이다.
          Oracle 은 THP 가 메모리 할당 지연을 만들 수 있어 끄라고 권고한다.
          HugePages(풀)와 이름이 비슷하지만 별개 기능이다. sysfs 값은 실행 중인
          커널에만 있고, 부트 파라미터 transparent_hugepage=never 가 영구 설정이다(본문 17.8)
 소요   : 약 30분 (재부팅 세 번)
 선행   : ol7-base 스냅샷
 계정   : root
 주의   : 부팅할 커널을 한 번 RHCK 로 바꾼다(grub2-reboot, 한 번만 적용)
 되돌림 : 부트 파라미터 제거, sysfs 값 [0] 의 값. 확인은 grubby --info=ALL | grep args
================================================================================

 [0] 시작 전 상태 기록
--------------------------------------------------------------------------------

-- 바꿀 대상: THP 의 sysfs 두 파일, 커널 항목별 부트
--   파라미터, /etc/default/grub.
--   시작 값을 모두 남긴다
-- 재부팅이 네 번 있다. 각 재부팅 뒤 uname
--   -r 과 /proc/cmdline 으로 어느
--   커널로 떴는지 먼저 본다
[root@ol7a ~]# uname -r; cat /proc/cmdline
5.4.17-2136.338.4.2.el7uek.x86_64
BOOT_IMAGE=/vmlinuz-5.4.17-2136.338.4.2.el7uek.x86_64 root=/dev/mapper/ol-root ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync
-- UEK R6 5.4.17-2136.338.4.2 로 부팅.
--   명령행에 transparent_hugepage= 가 없다
-- BOOT_IMAGE 는 GRUB 이 읽은 커널 파일,
--   root= 는 루트 파일 시스템 LV
-- 끝의 console=·scsi_mod.scan=sync 는 랩
--   콘솔 캡처와 장치 순서 고정용 설정이다
[root@ol7a ~]# cat /sys/kernel/mm/transparent_hugepage/enabled /sys/kernel/mm/transparent_hugepage/defrag
[always] madvise never
always defer defer+madvise [madvise] never
-- enabled [always] : 모든 익명 메모리에 THP 를
--   쓴다(대괄호가 현재 값)
--   madvise : 프로그램이 madvise(MADV_HUGEPAGE)로 요청한 영역에만, never : 쓰지 않음
-- defrag [madvise] : 2MB 페이지를 바로 못
--   구할 때 압축(defrag)을 기다릴지.
--   madvise 영역에서만 기다린다
-- defer : 기다리지 않고 kswapd·kcompactd 에
--   압축을 맡긴다. defer+madvise 는 두 방식의 혼합
--   always 면 모든 할당이 압축을 기다릴 수 있다. Oracle 이 지적하는 지연이 이 부분이다
[root@ol7a ~]# grubby --info=ALL | grep -E '^(index|kernel|args)'
index=0
kernel=/boot/vmlinuz-5.4.17-2136.338.4.2.el7uek.x86_64
args="ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync "
index=1
kernel=/boot/vmlinuz-5.4.17-2102.201.3.el7uek.x86_64
args="ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync "
index=2
kernel=/boot/vmlinuz-3.10.0-1160.119.1.0.5.el7.x86_64
args="ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync "
index=3
kernel=/boot/vmlinuz-3.10.0-1160.el7.x86_64
args="ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync "
index=4
kernel=/boot/vmlinuz-0-rescue-b8658360b45f49fe8174bcd0b610ae6e
args="ro crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync "
index=5
-- grubby --info=ALL : 커널 항목마다
--   index·kernel·args. 다섯 항목(UEK 둘,
--   RHCK 둘, rescue)
-- args 는 다섯 항목이 모두 같다. 이 줄에 무엇을
--   더하느냐가 이 실습의 내용이다
-- 항목 순서 index 0~4 가 GRUB 메뉴 순서와 같다. 0
--   번이 기본(saved_entry)이다
-- 마지막 index=5 는 kernel 줄이 없는 빈
--   항목이다(grubby 7.9 판의 출력 습관)
[root@ol7a ~]# grep -E '^GRUB_(DEFAULT|CMDLINE_LINUX)' /etc/default/grub; cp -p /etc/default/grub /root/grub.default.orig
GRUB_DEFAULT=saved
GRUB_CMDLINE_LINUX="crashkernel=auto rd.lvm.lv=ol/root rd.lvm.lv=ol/swap rhgb quiet console=tty0 console=ttyS0,115200 scsi_mod.scan=sync"
-- GRUB_DEFAULT=saved : 기본 항목을
--   grubenv 의 saved_entry 로 정한다. [5] 의
--   grub2-reboot 가 이것을 쓴다
-- grubby 는 grub.cfg 를 직접
--   고치고, /etc/default/grub 은
--   grub2-mkconfig 때만 읽힌다
-- GRUB_CMDLINE_LINUX : grub2-mkconfig 가 grub.cfg 를 새로 만들 때 모든 항목에 넣는 값
[root@ol7a ~]# for f in enabled defrag; do sed 's/.*\[\(.*\)\].*/\1/' /sys/kernel/mm/transparent_hugepage/$f > /root/thp.$f.orig; done; head /root/thp.*.orig
==> /root/thp.defrag.orig <==
madvise

==> /root/thp.enabled.orig <==
always
-- 대괄호 안의 값만 뽑아 파일로 남겼다. 되돌릴 때 이 값을 그대로
--   다시 쓴다(enabled always,
--   defrag madvise)
-- sed 's/.*\[\(.*\)\].*/\1/' : 대괄호 사이만
--   남긴다. head 는 두 파일 이름과 내용을 함께 찍는다

--------------------------------------------------------------------------------
 [1] THP 가 실제로 쓰이고 있나
--------------------------------------------------------------------------------

[root@ol7a ~]# grep AnonHugePages /proc/meminfo
AnonHugePages:      4096 kB
-- AnonHugePages : THP 로 묶여 있는 익명 메모리.
--   부팅 직후라 4096 kB(2MB 페이지 둘)뿐이다
-- HugePages 풀(실습 04)은 HugePages_Total
--   줄로 따로 나온다. 둘을 헷갈리지 않는다
-- 이 값이 수 GB 로 커지는 서버에서 THP 의 영향이 크다
[root@ol7a ~]# ls /sys/kernel/mm/transparent_hugepage/
defrag
enabled
hpage_pmd_size
khugepaged
shmem_enabled
use_zero_page
-- hpage_pmd_size : THP 페이지 크기(2MB),
--   khugepaged : 뒤에서 페이지를 묶는 커널 스레드 설정
-- 이 디렉터리 전체가 커널이 만든 가상 파일이다. 값을 쓰면
--   즉시 커널 변수가 바뀐다
-- shmem_enabled 는 tmpfs·공유 메모리용 THP.
--   SGA(System V 공유 메모리)와 관련된 값이다
[root@ol7a ~]# cat /sys/kernel/mm/transparent_hugepage/khugepaged/pages_collapsed /sys/kernel/mm/transparent_hugepage/khugepaged/full_scans
0
1
-- pages_collapsed 0 : khugepaged 가 아직
--   묶은 페이지가 없다. full_scans
--   1 : 전체 훑기를 한 번 했다
-- khugepaged 는 주기적으로 프로세스 메모리를 훑어 4KB
--   페이지 512 개를 2MB 하나로 바꾼다
-- 이 작업이 메모리 잠금을 잡아 순간 지연을 만들 수 있다
[root@ol7a ~]# grep -E 'thp_(fault_alloc|collapse_alloc|split_page)' /proc/vmstat
thp_fault_alloc 25
thp_collapse_alloc 0
thp_collapse_alloc_failed 0
thp_split_page 0
thp_split_page_failed 0
-- /proc/vmstat 의 thp_ 계수기. fault_alloc
--   25 는 페이지 폴트 때 바로 2MB 로 받은 횟수
-- 부팅 뒤 누적값이다. 같은 명령을 시간 차를 두고 두 번 쳐서
--   증가량을 보는 식으로 쓴다
-- collapse_alloc 은 khugepaged 가 묶은 횟수,
--   split_page 는 2MB 를 다시 쪼갠 횟수
[root@ol7a ~]# ps -eo pid,comm | grep -w khugepaged
   30 khugepaged
-- khugepaged 는 커널 스레드(PID 30). THP 가
--   never 여도 스레드는 남아 있고 일만 하지 않는다

 [2] sysfs 로 끄기
--------------------------------------------------------------------------------

[root@ol7a ~]# echo never > /sys/kernel/mm/transparent_hugepage/enabled; echo never > /sys/kernel/mm/transparent_hugepage/defrag
[root@ol7a ~]# cat /sys/kernel/mm/transparent_hugepage/enabled /sys/kernel/mm/transparent_hugepage/defrag
always madvise [never]
always defer defer+madvise madvise [never]
-- 두 파일 모두 [never]. sysfs 에 쓰는 순간 바로
--   바뀐다. 재시작할 서비스도 없다
-- echo never > 파일 : 리다이렉션이라 root 셸에서
--   해야 한다. sudo echo 는 리다이렉션에
--   권한이 없어 실패한다
[root@ol7a ~]# grep AnonHugePages /proc/meminfo
AnonHugePages:      4096 kB
-- 이미 묶인 4096 kB 는 그대로다. never 는 새 할당부터
--   적용되고 기존 페이지를 풀지 않는다
-- 한 번 묶인 2MB 페이지는 그 메모리가 해제될 때까지 남는다
-- 그래서 DB 서버는 인스턴스를 띄우기 전에,
--   가능하면 부팅할 때부터 끈다

-- 재부팅(@reboot a). next_entry 가 비어 있어
--   기본 UEK 로 뜬다
[root@ol7a ~]# uname -r; grep -o 'transparent_hugepage=[a-z]*' /proc/cmdline
5.4.17-2136.338.4.2.el7uek.x86_64
transparent_hugepage=never
-- 재확인 : 기본 UEK 에서 /proc/cmdline 에
--         transparent_hugepage=never
-- grep -o : 맞은 부분만 찍는다. 값이 never 하나다
[root@ol7a ~]# cat /sys/kernel/mm/transparent_hugepage/enabled /sys/kernel/mm/transparent_hugepage/defrag
always madvise [never]
always defer defer+madvise [madvise] never
-- enabled [never]. defrag 는 [madvise]
--   그대로다. 부트 파라미터는 enabled 만 정한다
-- enabled 가 never 면 defrag 값과 상관없이
--   THP 를 만들지 않는다
[root@ol7a ~]# grep AnonHugePages /proc/meminfo; grep -E 'thp_fault_alloc' /proc/vmstat
AnonHugePages:         0 kB
thp_fault_alloc 0
-- AnonHugePages 0 kB, thp_fault_alloc
--   0. 부팅 때부터 꺼져 THP 페이지가 하나도 없다
-- 운영 서버는 이 상태에서 인스턴스를 띄우고 THP 경고가
--   없는지 경고 로그를 본다
-- [2] 의 sysfs 방식은 이미 묶인 4096
--   kB 가 남았던 것과 다르다

 [8] 부트 파라미터가 있을 때의 tuned
--------------------------------------------------------------------------------

[root@ol7a ~]# cat /root/thp.enabled.orig > /sys/kernel/mm/transparent_hugepage/enabled; cat /sys/kernel/mm/transparent_hugepage/enabled; rpm -q tuned; tuned-adm active
[always] madvise never
tuned-2.11.0-12.0.3.el7_9.noarch
Current active profile: virtual-guest
-- 실행 중인 값을 일부러 always 로 돌려 두고
--   tuned 를 다시 시험한다
-- 부트 파라미터가 never 여도 sysfs 에는 always 를
--   쓸 수 있다. 실행 중 값과 부팅 값은 별개다
[root@ol7a ~]# tuned-adm profile oracle-nothp; tuned-adm active; cat /sys/kernel/mm/transparent_hugepage/enabled
Current active profile: oracle-nothp
[always] madvise never
-- 이번에는 oracle-nothp 를 켜도 enabled 가
--   [always] 그대로다
[root@ol7a ~]# grep -i transparent /var/log/tuned/tuned.log | tail -2 | cut -c1-150
2026-11-16 11:14:29,048 INFO     tuned.plugins.plugin_vm: transparent_hugepage is already set in kernel boot cmdline, ingoring value from profile
-- tuned.log : "transparent_hugepage is
--   already set in kernel boot
--   cmdline, ingoring value
--   from profile"
-- 명령행에 transparent_hugepage= 가 있으면
--   tuned 의 vm 플러그인이 값을 건드리지
--   않는다(ingoring 은 원문 철자)
-- 부트 파라미터가 tuned 보다 우선한다. 두 방식을
--   섞지 말고 하나만 쓴다
-- 사내 표준이 tuned 프로파일이라면 부트 파라미터 대신
--   프로파일 쪽을 고친다
[root@ol7a ~]# tuned-adm profile virtual-guest; tuned-adm active; cat /sys/kernel/mm/transparent_hugepage/enabled
Current active profile: virtual-guest
[always] madvise never
-- virtual-guest 로 돌려도 그대로. tuned 가
--   이번에는 바꾼 것이 없어 되살릴 것도 없다
[root@ol7a ~]# rm -rf /etc/tuned/oracle-nothp
