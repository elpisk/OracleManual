================================================================================
 제10장 실습 02 — 온라인 확장: 마운트된 채로 LV 와 파일 시스템 늘리기
 대상   : ol7a
 목표   : 마운트된 XFS·Ext4 LV 를 늘린다. lvextend 만 하면 파일 시스템은
          그대로라는 것을 보고, -r(--resizefs) 로 두 단계를 한 번에 한다.
          VG 가 모자라면 vgextend 로 PV 를 더하고, LV 가 두 디스크에 걸치는
          모습을 세그먼트로 확인한다
 배경   : LV 크기와 그 위 파일 시스템 크기는 따로 관리된다. lvextend 는 LV 에
          PE 를 붙일 뿐이고, 파일 시스템은 xfs_growfs·resize2fs 가 늘린다.
          -r 은 fsadm 을 불러 이 둘째 단계를 대신한다. XFS 와 Ext4 모두
          마운트된 상태에서 늘릴 수 있다 (본문 10.2, 10.3)
 소요   : 약 25분
 선행   : storage-clean 스냅샷
 계정   : root
 주의   : sdb·sdc 의 내용을 지운다
 되돌림 : 마운트 해제, LV·VG·PV 삭제. 확인은 pvs·lsblk (sdb·sdc 비어 있음)
================================================================================

--------------------------------------------------------------------------------
 [0] 시작 전 상태 기록
--------------------------------------------------------------------------------

-- 이 실습은 sdb·sdc 두 디스크만 쓴다.
-- 인자로 두 장치만 주면 다른 디스크는 나오지 않는다
[root@ol7a ~]# lsblk -o NAME,SIZE,TYPE,FSTYPE,MOUNTPOINT /dev/sdb /dev/sdc
NAME SIZE TYPE FSTYPE MOUNTPOINT
sdb   10G disk        
sdc   10G disk        
-- sdb·sdc 모두 FSTYPE 빈칸, 아래에 아무것도 없다
[root@ol7a ~]# vgs
  VG #PV #LV #SN Attr   VSize   VFree
  ol   1   2   0 wz--n- <39.00g 4.00m
-- VG 는 OS 의 ol 하나. 끝에서 이 두 출력과 비교한다

--------------------------------------------------------------------------------
 [1] 준비 — 디스크 하나로 VG, LV 두 개
--------------------------------------------------------------------------------

-- 처음에는 sdb 하나만 VG 에 넣는다. 공간 부족 상황을 [5] 에서 만들려는 것이다.
[root@ol7a ~]# vgcreate vgdata /dev/sdb
  Physical volume "/dev/sdb" successfully created.
  Volume group "vgdata" successfully created
-- sdb 는 PV 가 아니었지만 vgcreate 가 pvcreate 를 먼저 했다
-- 2G 두 개를 차례로 만든다. 두 LV 가 sdb 에서 번갈아 PE 를 받게 되어
-- [6] 에서 조각난 세그먼트를 볼 수 있다
[root@ol7a ~]# lvcreate -L 2G -n lvxfs vgdata; lvcreate -L 2G -n lvext vgdata
  Logical volume "lvxfs" created.
  Logical volume "lvext" created.
-- lvcreate 두 번이 sdb 의 PE 0~511,
-- 512~1023 을 차례로 가져갔다([6] 에서 확인)
-- -q : 만들 때의 긴 출력을 생략. 형식 세부는 9장·11장에서 본다
[root@ol7a ~]# mkfs.xfs -q /dev/vgdata/lvxfs; mkfs.ext4 -q /dev/vgdata/lvext
[root@ol7a ~]# mkdir -p /data/xfs /data/ext; mount /dev/vgdata/lvxfs /data/xfs; mount /dev/vgdata/lvext /data/ext
-- 같은 크기(2G) LV 두 개에 XFS 와 Ext4 를 따로 올렸다.
-- 두 형식의 확장 방식을 비교한다

-- 확장 도중에도 데이터가 그대로인지 보려고 파일을 하나씩 두고 체크섬을 남긴다.
-- dd 옵션
--   if=/dev/urandom : 압축되지 않는 난수
--   bs=1M count=200 : 1MiB 씩 200번 = 200MiB
--   status=none     : 진행 통계를 찍지 않음
[root@ol7a ~]# dd if=/dev/urandom of=/data/xfs/f1 bs=1M count=200 status=none; cp /data/xfs/f1 /data/ext/f1
[root@ol7a ~]# md5sum /data/xfs/f1 /data/ext/f1
bc2f833734c9b01f9b7358fd33f6f76c  /data/xfs/f1
bc2f833734c9b01f9b7358fd33f6f76c  /data/ext/f1
-- 두 파일의 체크섬이 같다(복사본). [7] 에서 다시 비교한다

--------------------------------------------------------------------------------
 [2] 확장 전 크기 — LV 와 파일 시스템
--------------------------------------------------------------------------------

-- df -hT 의 일곱 열: 장치, 형식, 크기, 사용, 남음, 사용률, 마운트 지점
[root@ol7a ~]# df -hT /data/xfs /data/ext
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvxfs xfs   2.0G  234M  1.8G  12% /data/xfs
/dev/mapper/vgdata-lvext ext4  2.0G  201M  1.6G  11% /data/ext
-- 이 실습에서 볼 열은 Size 다
--   xfs  2.0G : LV 2G 와 거의 같다
--   ext4 2.0G : Avail 1.6G 는 Size-Used 보다 작다. 예약 블록 5% 몫
-- Used 는 xfs 234M, ext4 201M.
-- 같은 200M 파일인데 xfs 는 로그 영역이 더해진다
[root@ol7a ~]# lvs vgdata; vgs vgdata
  LV    VG     Attr       LSize Pool Origin Data%  Meta%  Move Log Cpy%Sync Convert
  lvext vgdata -wi-ao---- 2.00g                                                    
  lvxfs vgdata -wi-ao---- 2.00g                                                    
  VG     #PV #LV #SN Attr   VSize   VFree 
  vgdata   1   2   0 wz--n- <10.00g <6.00g
-- LV 둘 다 2.00g, Attr 6번째 'o' → 마운트되어 열려 있음
-- VG 는 PV 하나(#PV 1), 남은 공간 <6.00g

--------------------------------------------------------------------------------
 [3] lvextend 만 한 경우 — XFS
--------------------------------------------------------------------------------

-- -L +1G : 지금보다 1GiB 더. (+ 가 없으면 "그 크기로" 다)
[root@ol7a ~]# lvextend -L +1G vgdata/lvxfs
  Size of logical volume vgdata/lvxfs changed from 2.00 GiB (512 extents) to 3.00 GiB (768 extents).
  Logical volume vgdata/lvxfs successfully resized.
-- 출력 첫 줄이 PE 수까지 알려 준다. 512 extents → 768 extents
-- 파일 시스템 얘기는 한 줄도 없다
-- LV 쪽과 파일 시스템 쪽을 나란히 놓고 본다
[root@ol7a ~]# lvs vgdata/lvxfs; df -hT /data/xfs
  LV    VG     Attr       LSize Pool Origin Data%  Meta%  Move Log Cpy%Sync Convert
  lvxfs vgdata -wi-ao---- 3.00g                                                    
-- LSize 3.00g. Attr 은 그대로 -wi-ao----
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvxfs xfs   2.0G  234M  1.8G  12% /data/xfs
-- 증상   : lvs 는 3.00g 인데 df 의 Size 는 2.0G 그대로. 오류 메시지는 없다
-- 원인   : lvextend 는 LV 에 PE 를 붙일 뿐이다. 그 위 XFS 는 자기 크기를
--          슈퍼블록에 들고 있어 LV 가 커져도 모른다
-- 경보가 울리는 디스크에서 이 상태로 두면 공간을 늘렸다고 착각한다

-- 파일 시스템을 늘린다. XFS 는 마운트 지점을 인자로 준다. 마운트된 상태에서만 동작한다.
-- xfs_growfs 옵션
--   (없음)  : 장치 끝까지 늘림
--   -D 블록 : 지정한 데이터 블록 수까지 (줄이는 값은 거부, 실습 03)
--   -n      : 바꾸지 않고 현재 구성만 출력 (xfs_info 와 같음)
[root@ol7a ~]# xfs_growfs /data/xfs
meta-data=/dev/mapper/vgdata-lvxfs isize=256    agcount=4, agsize=131072 blks
         =                       sectsz=512   attr=2, projid32bit=1
         =                       crc=0        finobt=0, sparse=0, rmapbt=0
         =                       reflink=0
data     =                       bsize=4096   blocks=524288, imaxpct=25
         =                       sunit=0      swidth=0 blks
naming   =version 2              bsize=4096   ascii-ci=0, ftype=1
log      =internal log           bsize=4096   blocks=2560, version=2
         =                       sectsz=512   sunit=0 blks, lazy-count=1
realtime =none                   extsz=4096   blocks=0, rtextents=0
data blocks changed from 524288 to 786432
-- 조치   : xfs_growfs /data/xfs
-- xfs_growfs 는 먼저 "늘리기 전" 구성을 찍고 마지막 줄에 바뀐 값을 쓴다
--   meta-data : agcount=4 (할당 그룹 4개), agsize=131072 블록(512MiB)
--   data      : bsize=4096, blocks=524288 (= 2GiB)
--   log       : internal log. 로그가 데이터 영역 안에 있다
--   마지막 줄 : 524288 → 786432 블록 (2GiB → 3GiB)
-- 인자는 장치가 아니라 마운트 지점이다. XFS 는 마운트된 상태에서만 늘린다
-- 이 VM 의 mkfs.xfs 결과는 crc=0,
-- isize=256 이다. 구성 값 세부는 11장
[root@ol7a ~]# df -hT /data/xfs
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvxfs xfs   3.0G  234M  2.8G   8% /data/xfs
-- 재확인 : df Size 3.0G. 마운트를 내리지 않고 늘었다

--------------------------------------------------------------------------------
 [4] lvextend -r — Ext4 를 한 번에
--------------------------------------------------------------------------------

-- -r (--resizefs) : LV 를 늘린 뒤 fsadm 으로 파일 시스템도 늘린다
-- Ext4 는 resize2fs 로 늘린다.
-- 마운트된 채 늘리는 것은 커널의 온라인 확장 기능이다
-- 줄이기는 온라인으로 되지 않는다(실습 03)
[root@ol7a ~]# lvextend -r -L +1G vgdata/lvext
  Size of logical volume vgdata/lvext changed from 2.00 GiB (512 extents) to 3.00 GiB (768 extents).
  Logical volume vgdata/lvext successfully resized.
resize2fs 1.45.4 (23-Sep-2019)
Filesystem at /dev/mapper/vgdata-lvext is mounted on /data/ext; on-line resizing required
old_desc_blocks = 1, new_desc_blocks = 1
The filesystem on /dev/mapper/vgdata-lvext is now 786432 (4k) blocks long.
-- 위 두 줄은 lvextend,
-- 아래 네 줄은 fsadm 이 부른 resize2fs 의 출력이다
--   on-line resizing required : 마운트된 상태라 온라인 확장으로 진행
--   old_desc_blocks / new_desc_blocks : 그룹 디스크립터 블록 수. 그대로 1
--   786432 (4k) blocks : 4KiB × 786432 = 3GiB
[root@ol7a ~]# df -hT /data/ext
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvext ext4  2.9G  201M  2.6G   8% /data/ext
-- 한 명령으로 LV 와 파일 시스템이 함께 늘었다
-- 3G 인데 2.9G 로 보이는 것은 Ext4 메타데이터(아이노드 표 등) 몫

-- 같은 방법으로 XFS 도 한 번 더 늘려 본다. fsadm 이 형식을 보고 xfs_growfs 를 부른다.
[root@ol7a ~]# lvextend -r -L +512M vgdata/lvxfs
  Size of logical volume vgdata/lvxfs changed from 3.00 GiB (768 extents) to 3.50 GiB (896 extents).
  Logical volume vgdata/lvxfs successfully resized.
meta-data=/dev/mapper/vgdata-lvxfs isize=256    agcount=6, agsize=131072 blks
         =                       sectsz=512   attr=2, projid32bit=1
         =                       crc=0        finobt=0, sparse=0, rmapbt=0
         =                       reflink=0
data     =                       bsize=4096   blocks=786432, imaxpct=25
         =                       sunit=0      swidth=0 blks
naming   =version 2              bsize=4096   ascii-ci=0, ftype=1
log      =internal log           bsize=4096   blocks=2560, version=2
         =                       sectsz=512   sunit=0 blks, lazy-count=1
realtime =none                   extsz=4096   blocks=0, rtextents=0
data blocks changed from 786432 to 917504
-- fsadm 이 xfs 를 알아보고 xfs_growfs 를 불렀다.
-- 출력 형식이 [3] 과 같다
-- agcount=6 → 앞의 xfs_growfs 로 이미 4 → 6 으로 늘어 있었다
[root@ol7a ~]# df -hT /data/xfs
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvxfs xfs   3.5G  234M  3.3G   7% /data/xfs
-- 3.5G. 형식과 상관없이 -r 하나로 끝난다. 실무에서는 -r 을 기본으로 붙인다

--------------------------------------------------------------------------------
 [5] VG 에 공간이 모자라면 — vgextend
--------------------------------------------------------------------------------

[root@ol7a ~]# vgs vgdata
  VG     #PV #LV #SN Attr   VSize   VFree 
  vgdata   1   2   0 wz--n- <10.00g <3.50g
-- 남은 공간 <3.50g. 여기에 +5G 를 요청한다
-- 남은 3.5G 보다 큰 5G 를 일부러 요청한다
[root@ol7a ~]# lvextend -r -L +5G vgdata/lvxfs
  Insufficient free space: 1280 extents needed, but only 895 available
-- 증상   : "Insufficient free space: 1280 extents needed, but only 895 available"
-- 원인   : 5GiB = 1280 PE 가 필요한데 VG 에 남은 PE 가 895 개뿐이다
--          LV 는 VG 를 넘어 자랄 수 없다. 일부만 늘리지도 않는다

-- 새 디스크를 VG 에 넣는다. sdc 는 아직 PV 가 아니지만 vgextend 가 초기화까지 한다.
[root@ol7a ~]# vgextend vgdata /dev/sdc
  Physical volume "/dev/sdc" successfully created.
-- vgextend 도 vgcreate 처럼 PV 가 아닌 디스크면 pvcreate 를 먼저 한다
-- 다른 서명이 있는 디스크라면 실습 01 처럼 거부된다
  Volume group "vgdata" successfully extended
-- 조치   : vgextend vgdata /dev/sdc. 새 디스크를 PV 로 만들며 VG 에 넣었다
-- PV 쪽과 VG 쪽에서 각각 늘어난 것을 확인한다
[root@ol7a ~]# pvs; vgs vgdata
  PV         VG     Fmt  Attr PSize   PFree  
  /dev/sda2  ol     lvm2 a--  <39.00g   4.00m
  /dev/sdb   vgdata lvm2 a--  <10.00g  <3.50g
  /dev/sdc   vgdata lvm2 a--  <10.00g <10.00g
  VG     #PV #LV #SN Attr   VSize  VFree 
  vgdata   2   2   0 wz--n- 19.99g 13.49g
-- 바뀐 열
--   pvs : sdc 가 VG vgdata 에 들어감, PFree <10.00g
--   vgs : #PV 1 → 2, VSize <10.00g → 19.99g, VFree <3.50g → 13.49g
[root@ol7a ~]# lvextend -r -L +5G vgdata/lvxfs
  Size of logical volume vgdata/lvxfs changed from 3.50 GiB (896 extents) to 8.50 GiB (2176 extents).
-- 재확인 : 같은 lvextend 가 896 → 2176 extents 로 성공
  Logical volume vgdata/lvxfs successfully resized.
meta-data=/dev/mapper/vgdata-lvxfs isize=256    agcount=7, agsize=131072 blks
         =                       sectsz=512   attr=2, projid32bit=1
         =                       crc=0        finobt=0, sparse=0, rmapbt=0
         =                       reflink=0
data     =                       bsize=4096   blocks=917504, imaxpct=25
         =                       sunit=0      swidth=0 blks
naming   =version 2              bsize=4096   ascii-ci=0, ftype=1
log      =internal log           bsize=4096   blocks=2560, version=2
         =                       sectsz=512   sunit=0 blks, lazy-count=1
realtime =none                   extsz=4096   blocks=0, rtextents=0
data blocks changed from 917504 to 2228224
-- 917504 → 2228224 블록 = 3.5GiB → 8.5GiB
[root@ol7a ~]# df -hT /data/xfs
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvxfs xfs   8.5G  234M  8.3G   3% /data/xfs
-- df 도 8.5G. Used 234M 은 처음부터 그대로다

--------------------------------------------------------------------------------
 [6] 두 디스크에 걸친 LV — 세그먼트
--------------------------------------------------------------------------------

-- 늘린 몫은 남은 PE 에서 붙는다. sdb 에 남은 것을 먼저 쓰고 모자라는 몫을 sdc 에서 가져온다.
-- -o 로 열을 고른다. --segments 를 주면 LV 한 줄이 세그먼트 수만큼 나뉜다
[root@ol7a ~]# lvs -o lv_name,lv_size,seg_start_pe,seg_size_pe,devices --segments vgdata
  LV    LSize Start SSize Devices       
  lvext 3.00g     0   512 /dev/sdb(512) 
  lvext 3.00g   512   256 /dev/sdb(1280)
  lvxfs 8.50g     0   512 /dev/sdb(0)   
  lvxfs 8.50g   512   256 /dev/sdb(1024)
  lvxfs 8.50g   768  1023 /dev/sdb(1536)
  lvxfs 8.50g  1791   385 /dev/sdc(0)   
-- 열 해설
--   LSize   : LV 전체 크기
--   Start   : 이 세그먼트가 LV 안에서 시작하는 LE 번호
--   SSize   : 세그먼트 길이(PE 수)
--   Devices : 놓인 PV(시작 PE)
-- lvxfs 는 네 조각이다
--   0~511    : sdb 0~511    (처음 만든 2G)
--   512~767  : sdb 1024~    ([3] 의 +1G. 그 사이 sdb 512~1023 은 lvext 가 차지)
--   768~1790 : sdb 1536~    ([4]·[5] 에서 sdb 에 남은 것 전부)
--   1791~    : sdc 0~384    (모자란 몫 385 PE)
-- lvext 도 sdb 512~ 와 1280~ 두 조각
-- LVM 은 같은 PV 의 남은 공간을 먼저 쓰고, 모자랄 때 다음 PV 로 넘어간다
-- LV 가 PE 의 목록이라 조각이 몇 개든 파일 시스템은 하나로 본다
-- PV 마다 쓴 양과 남은 양
[root@ol7a ~]# pvs -o pv_name,pv_size,pv_used,pv_free
  PV         PSize   Used    PFree
  /dev/sda2  <39.00g  38.99g 4.00m
  /dev/sdb   <10.00g <10.00g    0 
  /dev/sdc   <10.00g   1.50g 8.49g
-- sdb 는 PFree 0 → 꽉 참. sdc 는 1.50g 사용(385 PE)
-- 이제 sdb 나 sdc 하나가 고장 나면 lvxfs 전체를 잃는다.
-- LVM 의 linear 는 이중화가 아니다

--------------------------------------------------------------------------------
 [7] 데이터 확인 — 확장 동안 내려간 적이 없다
--------------------------------------------------------------------------------

-- 확장 세 번 동안 마운트를 내린 적이 없다. 그 사이 데이터가 바뀌지 않았는지 본다
[root@ol7a ~]# md5sum /data/xfs/f1 /data/ext/f1
bc2f833734c9b01f9b7358fd33f6f76c  /data/xfs/f1
bc2f833734c9b01f9b7358fd33f6f76c  /data/ext/f1
-- 체크섬이 [1] 과 같다. 세 번 늘리는 동안 데이터는 그대로다
-- 마운트 옵션이 바뀌거나 ro 로 다시 마운트된 흔적이 없는지 본다
[root@ol7a ~]# findmnt -o TARGET,SOURCE,FSTYPE,OPTIONS /data/xfs; findmnt -o TARGET,SOURCE,FSTYPE,OPTIONS /data/ext
TARGET    SOURCE                   FSTYPE OPTIONS
/data/xfs /dev/mapper/vgdata-lvxfs xfs    rw,relatime,seclabel,attr2,inode64,logbufs=8,logbsize=32k,noquota
TARGET    SOURCE                   FSTYPE OPTIONS
/data/ext /dev/mapper/vgdata-lvext ext4   rw,relatime,seclabel
-- findmnt 열 해설: TARGET(마운트 지점) SOURCE(장치) FSTYPE OPTIONS
-- 두 파일 시스템 모두 rw 로 처음 마운트 그대로다. 다시 마운트한 적이 없다

-- XFS 의 할당 그룹(agcount)은 늘린 만큼 늘어난다. 2G 로 만들 때와 비교한다.
-- xfs_info 는 마운트된 XFS 의 구성을 보여 준다.
-- 첫 두 줄에 agcount 가 있다
[root@ol7a ~]# xfs_info /data/xfs | head -2
meta-data=/dev/mapper/vgdata-lvxfs isize=256    agcount=17, agsize=131072 blks
         =                       sectsz=512   attr=2, projid32bit=1
-- agcount 4 → 17. agsize 는 그대로 131072 블록(512MiB)이고 그룹 수만 늘었다
-- 처음부터 크게 만든 XFS 보다 그룹이 잘다.
-- 작게 만들어 여러 번 크게 늘리는 것을 피하는 이유(11장)

--------------------------------------------------------------------------------
 [8] Ext4 를 남은 공간 비율로 늘리기 — -l +PERCENT
--------------------------------------------------------------------------------

-- -l +50%FREE : VG 에 남은 PE 의 절반을 붙인다
-- 비율로 늘리기 전에 남은 PE 를 숫자로 본다
[root@ol7a ~]# vgs -o vg_name,vg_free_count,vg_free vgdata
  VG     Free VFree
  vgdata 2174 8.49g
-- 남은 PE 2174 개. 절반은 1087 개
-- -l 의 비율 단위
--   %FREE : VG 에 남은 공간 기준
--   %VG   : VG 전체 기준
--   %PVS  : 명령에 적은 PV 들 기준
--   %ORIGIN : 스냅샷의 원본 크기 기준
[root@ol7a ~]# lvextend -r -l +50%FREE vgdata/lvext
  Size of logical volume vgdata/lvext changed from 3.00 GiB (768 extents) to <7.25 GiB (1855 extents).
  Logical volume vgdata/lvext successfully resized.
resize2fs 1.45.4 (23-Sep-2019)
Filesystem at /dev/mapper/vgdata-lvext is mounted on /data/ext; on-line resizing required
old_desc_blocks = 1, new_desc_blocks = 1
The filesystem on /dev/mapper/vgdata-lvext is now 1899520 (4k) blocks long.
-- 768 + 1087 = 1855 extents. 퍼센트도 PE 개수로 바뀌어 붙는다
-- -l +50%FREE 의 + 를 빼면 "LV 를 남은 공간의 50% 크기로" 가 되어 줄이라는 뜻이 된다
-- Ext4 크기와 VG 남은 공간을 함께 본다
[root@ol7a ~]# df -hT /data/ext; vgs vgdata
Filesystem               Type  Size  Used Avail Use% Mounted on
/dev/mapper/vgdata-lvext ext4  7.1G  201M  6.6G   3% /data/ext
-- 7.25GiB 인데 df 는 7.1G.
-- 크기가 커질수록 Ext4 메타데이터 몫도 함께 커진다
-- VFree <4.25g 는 남은 1087 PE. 다음 단계에서 256M 를 더 쓴다
  VG     #PV #LV #SN Attr   VSize  VFree 
  vgdata   2   2   0 wz--n- 19.99g <4.25g
-- ext4 7.1G, VFree <4.25g(1087 PE)

--------------------------------------------------------------------------------
 [변형] lvresize 와 fsadm 을 따로
--------------------------------------------------------------------------------

-- lvresize 는 늘리기·줄이기를 겸하는 명령이다. 여기서는 -r 없이 늘리고
-- fsadm 을 직접 불러 파일 시스템을 맞춘다. lvextend -r 이 속에서 하는 일과 같다.
-- [변형] 은 -r 이 속에서 하는 두 단계를 따로 쳐서 확인한다
-- 256MiB = 64 PE
[root@ol7a ~]# lvresize -L +256M vgdata/lvext
  Size of logical volume vgdata/lvext changed from <7.25 GiB (1855 extents) to <7.50 GiB (1919 extents).
  Logical volume vgdata/lvext successfully resized.
-- lvresize 는 + 면 늘리고 - 면 줄인다.
-- lvextend 와 달리 줄이는 쪽도 거부하지 않는다
[root@ol7a ~]# df -h /data/ext | tail -1
/dev/mapper/vgdata-lvext  7.1G  201M  6.6G   3% /data/ext
-- LV 는 <7.50 GiB 인데 df 는 7.1G 그대로. [3] 과 같은 상태다
[root@ol7a ~]# fsadm -v resize /dev/vgdata/lvext
fsadm: "ext4" filesystem found on "/dev/mapper/vgdata-lvext".
fsadm: Device "/dev/mapper/vgdata-lvext" size is 8048869376 bytes
fsadm: Parsing tune2fs -l "/dev/mapper/vgdata-lvext"
fsadm: Resizing filesystem on device "/dev/mapper/vgdata-lvext" to 8048869376 bytes (1899520 -> 1965056 blocks of 4096 bytes)
fsadm: Executing resize2fs /dev/mapper/vgdata-lvext 1965056
resize2fs 1.45.4 (23-Sep-2019)
Filesystem at /dev/mapper/vgdata-lvext is mounted on /data/ext; on-line resizing required
old_desc_blocks = 1, new_desc_blocks = 1
The filesystem on /dev/mapper/vgdata-lvext is now 1965056 (4k) blocks long.
-- fsadm -v 가 하는 일
--   1) blkid 로 형식 확인 (ext4)
--   2) 장치 크기 확인 (8048869376 바이트)
--   3) tune2fs -l 로 현재 블록 수 확인
--   4) resize2fs 장치 블록수 실행 (1899520 → 1965056)
-- lvextend -r 이 속에서 부르는 것이 바로 이 fsadm resize 다
[root@ol7a ~]# df -h /data/ext | tail -1
/dev/mapper/vgdata-lvext  7.4G  201M  6.8G   3% /data/ext
-- 7.1G → 7.4G

-- 파일 시스템 확장은 이미 맞춰진 상태에서 다시 해도 해가 없다.
-- XFS 쪽에도 같은 명령을 다시 쳐 본다. 이미 맞는 크기면 무엇이 나오는가
[root@ol7a ~]# xfs_growfs /data/xfs | tail -1
realtime =none                   extsz=4096   blocks=0, rtextents=0
-- 마지막 줄에 "data blocks changed" 가 없다.
-- 이미 LV 크기와 같아 바꿀 것이 없었다

 [9] 되돌림

-- 마운트를 내리고 마운트 지점 디렉터리도 지운다
[root@ol7a ~]# umount /data/xfs /data/ext; rm -rf /data
[root@ol7a ~]# vgremove -y vgdata
  Logical volume "lvxfs" successfully removed
  Logical volume "lvext" successfully removed
  Volume group "vgdata" successfully removed
-- vgremove -y 는 VG 안의 LV 를 먼저 지운다.
-- 마운트를 내려 두지 않았으면 거부한다
-- VG 를 지운 뒤 두 디스크의 LVM 레이블을 지운다
[root@ol7a ~]# pvremove /dev/sdb /dev/sdc
  Labels on physical volume "/dev/sdb" successfully wiped.
  Labels on physical volume "/dev/sdc" successfully wiped.
-- 순서는 [1] 의 역순. 파일 시스템 → LV → VG → PV

--  항목              시작 전      작업 중                              되돌린 뒤

--  vgdata VSize      -            <10.00g → 19.99g                     -
