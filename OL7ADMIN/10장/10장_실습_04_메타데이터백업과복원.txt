================================================================================
 제10장 실습 04 — 메타데이터 백업과 복원: 지운 LV 되살리기
 대상   : ol7a
 목표   : /etc/lvm/backup 과 /etc/lvm/archive 가 언제 어떻게 쌓이는지 보고,
          파일 내용을 필드 단위로 읽는다. 실수로 지운 LV 를 vgcfgrestore 로
          되살리고 데이터가 그대로인지 확인한다. 복원 지점을 잘못 고르는
          경우와 복원 뒤 LV 가 비활성인 경우를 겪는다
 배경   : LVM 명령은 메타데이터를 바꾸기 직전 상태를 archive 에, 바꾼 뒤
          상태를 backup 에 텍스트로 남긴다. lvremove 는 메타데이터에서 LV 를
          지울 뿐 PE 의 데이터를 지우지 않는다. 그 PE 를 다른 LV 가 덮어쓰기
          전이면 메타데이터를 되돌려 LV 를 살릴 수 있다 (본문 10.4)
 소요   : 약 25분
 선행   : storage-clean 스냅샷
 계정   : root
 주의   : sdb·sdc 의 내용을 지운다
 되돌림 : LV·VG·PV 삭제, /root 에 만든 파일 삭제. 확인은 pvs·lsblk
================================================================================

 [0] 시작 전 상태 기록
--------------------------------------------------------------------------------
-- 두 디렉터리에 무엇이 있는지 먼저 남긴다
-- grep -v total : ls 의 합계 줄을 뺀다
[root@ol7a ~]# ls -l /etc/lvm/backup/ /etc/lvm/archive/ | grep -v total
/etc/lvm/archive/:
-rw-------. 1 root root 1746 Nov 11 19:58 ol_00000-215091343.vg

/etc/lvm/backup/:
-rw-------. 1 root root 1745 Nov 11 19:58 ol
-- 설치 때 만든 ol 의 사본만 있다
--   archive : ol_00000 하나
--   backup  : ol 하나
-- VG 와 쓸 디스크 둘의 상태
[root@ol7a ~]# vgs; lsblk -o NAME,SIZE,TYPE /dev/sdb /dev/sdc
  VG #PV #LV #SN Attr   VSize   VFree
  ol   1   2   0 wz--n- <39.00g 4.00m
NAME SIZE TYPE
sdb   10G disk
sdc   10G disk
-- VG 는 ol 하나, 두 디스크는 비어 있다
-- 백업·아카이브 동작은 lvm.conf 의 backup 절이 정한다. 주석을 뺀 값만 본다.
[root@ol7a ~]# lvmconfig backup
backup {
        backup=1
        backup_dir="/etc/lvm/backup"
        archive=1
        archive_dir="/etc/lvm/archive"
        retain_min=10
        retain_days=30
}
-- 필드
--   backup=1, archive=1 : 둘 다 켜짐(기본)
--   backup_dir·archive_dir : 저장 위치
--   retain_min=10  : 최소 10개는 남긴다
--   retain_days=30 : 30일 안의 것은 남긴다
-- 두 조건을 모두 넘어야 지운다

--------------------------------------------------------------------------------
 [1] VG 와 LV 만들기 — 명령마다 아카이브가 하나씩
--------------------------------------------------------------------------------
-- 명령 셋을 차례로 친다. 각 명령이 무엇을 남기는지 뒤에서 센다
[root@ol7a ~]# vgcreate vgdata /dev/sdb /dev/sdc
  Physical volume "/dev/sdb" successfully created.
  Physical volume "/dev/sdc" successfully created.
  Volume group "vgdata" successfully created
-- vgcreate 가 pvcreate 를 먼저 했다(sdb·sdc 두 줄)
[root@ol7a ~]# lvcreate -L 2G -n lvdata vgdata
  Logical volume "lvdata" created.
-- lvtmp 는 [8] 에서 "덮어쓴 뒤" 를 보여 줄 LV 다
[root@ol7a ~]# lvcreate -L 1G -n lvtmp vgdata
  Logical volume "lvtmp" created.
-- lvdata 에만 파일 시스템을 만들고 데이터를 넣는다
[root@ol7a ~]# mkfs.ext4 -q /dev/vgdata/lvdata; mkdir -p /data; mount /dev/vgdata/lvdata /data
-- /etc 사본(작은 파일 많이)과 100M 난수 파일
[root@ol7a ~]# cp -a /etc /data/; dd if=/dev/urandom of=/data/f1 bs=1M count=100 status=none
[root@ol7a ~]# md5sum /data/f1 | tee /root/f1.md5; du -sh /data
3c4ddf6d45141188be54d32a51469c41  /data/f1
-- f1 체크섬. [6] 에서 비교한다
152M    /data
-- 체크섬은 /root 에 저장. LV 가 사라져도 비교 기준이 남는다
-- [0] 과 같은 명령. 무엇이 늘었나
[root@ol7a ~]# ls -l /etc/lvm/backup/ /etc/lvm/archive/ | grep -v total
/etc/lvm/archive/:
-rw-------. 1 root root 1746 Nov 11 19:58 ol_00000-215091343.vg
-rw-------. 1 root root 1145 Nov 13 18:18 vgdata_00000-611916045.vg
-rw-------. 1 root root 1143 Nov 13 18:18 vgdata_00001-427146574.vg
-rw-------. 1 root root 1568 Nov 13 18:18 vgdata_00002-1030314331.vg

/etc/lvm/backup/:
-rw-------. 1 root root 1745 Nov 11 19:58 ol
-rw-------. 1 root root 1975 Nov 13 18:18 vgdata
-- 명령 셋에 아카이브 셋(00000~00002)
--   00000 : vgcreate 직전
--   00001 : lvcreate lvdata 직전
--   00002 : lvcreate lvtmp 직전
-- backup/vgdata 는 마지막 명령 뒤 상태 하나
-- mkfs·mount 는 LVM 메타데이터를 바꾸지 않아 기록이 없다

[root@ol7a ~]# sed -n '1,12p' /etc/lvm/backup/vgdata

        id = "bkdA4t-y2gB-hBmB-80sT-n6Ud-3L0A-4WOR8P"

--   # Generated by : 만든 lvm2 판과 시각
--   description    : 어떤 명령 *after* 의 상태인지
--   creation_host  : 만든 호스트와 커널
--   vgdata { id    : VG UUID
-- logical_volumes 절에서 첫 LV 정의가 끝나는 곳(탭 두 개 + })까지
[root@ol7a ~]# sed -n '/^\tlogical_volumes/,/^\t\t}/p' /etc/lvm/backup/vgdata
        logical_volumes {

                lvdata {
-- LV 이름 아래에 그 LV 의 모든 정보가 들어 있다
                        id = "8yxNU1-SQsh-7G3e-KMo1-yYMA-DJYI-mKHrDE"
                        status = ["READ", "WRITE", "VISIBLE"]
                        flags = []
                        creation_time = 1794561521      # 2026-11-13 18:18:41 +0900
                        creation_host = "ol7a.example.com"
                        segment_count = 1

                        segment1 {
                                start_extent = 0
                                extent_count = 512      # 2 Gigabytes

                                type = "striped"
                                stripe_count = 1        # linear

                                stripes = [
                                        "pv0", 0
                                ]
                        }
                }
-- lvdata 정의
--   status READ, WRITE, VISIBLE : 쓰기 가능, 숨지 않음
--   segment_count 1 : 한 조각

--   stripes "pv0", 0 : pv0 의 0 번 PE 부터
-- 이 숫자가 그대로 dmsetup table 의 매핑이 된다
-- PV 정의에서 pe_start·pe_count 를 본다. 장치 이름(device)은 참고용이고 실제 대조는 id 로 한다.

                        device = "/dev/sdb"     # Hint only

                        pe_count = 2559 # 9.99609 Gigabytes

--   dev_size 20971520 : 섹터 수(10GiB)

--   pe_count 2559     : PE 개수
                }

 [5] 복원 — 활성 LV, 그리고 잘못 고른 지점
--------------------------------------------------------------------------------
-- description 에 lvremove 가 적힌 아카이브를 찾아 바로 복원해 본다.
-- -f 파일 : 그 파일로 복원
-- grep -l : 일치하는 줄이 아니라 파일 이름만
[root@ol7a ~]# grep -l "lvremove -y vgdata/lvdata" /etc/lvm/archive/vgdata_*
/etc/lvm/archive/vgdata_00003-1486134953.vg
-- description 문자열로 찾았다. 파일 하나만 걸린다
-- $( ) 안의 결과(파일 이름)가 -f 의 인자가 된다
[root@ol7a ~]# vgcfgrestore -f $(grep -l "lvremove -y vgdata/lvdata" /etc/lvm/archive/vgdata_*) vgdata
  Volume group vgdata has active volume: lvtmp.
  WARNING: Found 1 active volume(s) in volume group "vgdata".
  Restoring VG with active LVs, may cause mismatch with its metadata.
Do you really want to proceed with restore of volume group "vgdata", while 1 volume(s) are active? [y/n]: [n]
-- 러너가 표준 입력을 주지 않아 기본 답 [n] 으로 처리됐다
  Restore aborted.
-- 증상   : "Volume group vgdata has active volume: lvtmp." → Restore aborted.
-- 원인   : VG 안에 활성 LV(lvtmp)가 있다. 활성 LV 의 매핑과 메타데이터가
--          어긋날 수 있어 확인을 묻고, 답이 없으니 [n] 으로 멈췄다
-- VG 의 LV 를 모두 비활성화한 뒤 다시 한다. -an : 활성 해제(available no)
[root@ol7a ~]# vgchange -an vgdata
  0 logical volume(s) in volume group "vgdata" now active
-- 조치   : vgchange -an vgdata. 활성 LV 가 0 개
-- lvtmp 를 쓰는 응용이 있다면 이 순간 멈춘다. 복원은 정지 작업이다
-- 이번에는 "lvdata 를 만들 때 상태로 돌리면 되겠지" 하고 lvcreate 쪽 아카이브를 고른다.
[root@ol7a ~]# F=$(grep -l "executing 'lvcreate -L 2G -n lvdata" /etc/lvm/archive/vgdata_*); echo $F; grep description $F
/etc/lvm/archive/vgdata_00001-427146574.vg
description = "Created *before* executing 'lvcreate -L 2G -n lvdata vgdata'"
-- 00001 은 lvdata 를 "만들기 직전" 이다. 이 시점에는 lvdata 가 없다
[root@ol7a ~]# vgcfgrestore -f $(grep -l "executing 'lvcreate -L 2G -n lvdata" /etc/lvm/archive/vgdata_*) vgdata
  Restored volume group vgdata
  Scan of VG vgdata from /dev/sdb found metadata seqno 2 vs previous 4.
  Scan of VG vgdata from /dev/sdc found metadata seqno 2 vs previous 4.
-- 재확인 : 이번에는 Restored volume group vgdata 로 진행됐다
-- 아래 두 줄은 디스크의 seqno 가 4 에서 2 로 내려갔다는 알림
-- 복원 결과를 본다
[root@ol7a ~]# lvs vgdata
-- 증상   : 복원은 성공했는데 lvs 가 빈칸. lvdata 도 lvtmp 도 없다
-- 원인   : before 는 "그 명령 직전" 이다. lvcreate 직전 상태엔 LV 가 하나도 없었다
-- 조치   : lvremove 직전(00003)으로 다시 복원
-- lvremove 직전 아카이브로 다시 복원한다. 복원은 몇 번이든 덮어쓸 수 있다.
[root@ol7a ~]# vgcfgrestore -f $(grep -l "lvremove -y vgdata/lvdata" /etc/lvm/archive/vgdata_*) vgdata
  Restored volume group vgdata
  Scan of VG vgdata from /dev/sdb found metadata seqno 4 vs previous 2.
  Scan of VG vgdata from /dev/sdc found metadata seqno 4 vs previous 2.
-- seqno 2 → 4. 복원은 앞으로도 뒤로도 할 수 있다
[root@ol7a ~]# lvs vgdata
  LV     VG     Attr       LSize Pool Origin Data%  Meta%  Move Log Cpy%Sync Convert
  lvdata vgdata -wi------- 2.00g                                                    
-- lvdata 의 Attr -wi------- : 일반 LV, 쓰기 가능, 비활성
  lvtmp  vgdata -wi------- 1.00g                                                    
-- 재확인 : lvdata 2.00g 와 lvtmp 1.00g 가 돌아왔다. seqno 2 → 4
-- Attr 다섯째가 '-' → 둘 다 비활성

 [6] 복원했는데 마운트가 안 된다
--------------------------------------------------------------------------------
-- 돌아온 lvdata 를 바로 마운트해 본다
[root@ol7a ~]# mount /dev/vgdata/lvdata /data
mount: special device /dev/vgdata/lvdata does not exist
-- 증상   : "special device /dev/vgdata/lvdata does not exist"
-- lv_active 열과 /dev/vgdata 링크를 함께 본다
[root@ol7a ~]# lvs -o lv_name,lv_attr,lv_active vgdata; ls /dev/vgdata/
  LV     Attr       Active
  lvdata -wi-------       
-- 두 LV 모두 Attr 다섯째 '-'
  lvtmp  -wi-------       
ls: cannot access /dev/vgdata/: No such file or directory
-- 원인   : 비활성이라 매퍼 장치도 /dev/vgdata 디렉터리도 없다
--   Active 열이 빈칸
-- -ay : 활성화(available yes)
[root@ol7a ~]# lvchange -ay vgdata/lvdata
-- 조치   : lvchange -ay vgdata/lvdata. 출력 없음
[root@ol7a ~]# lvs -o lv_name,lv_attr,lv_active vgdata
  LV     Attr       Active
  lvdata -wi-a----- active
-- lvdata 만 활성. 필요한 LV 만 골라 켤 수 있다
  lvtmp  -wi-------       
-- lvdata 만 a·active. lvtmp 는 아직 비활성
-- 마운트 전에 읽기 전용 검사로 파일 시스템이 온전한지 본다
[root@ol7a ~]# e2fsck -fn /dev/vgdata/lvdata | tail -2
e2fsck 1.45.4 (23-Sep-2019)
Pass 5: Checking group summary information
-- tail -2 로 마지막 두 줄만. 앞의 Pass 1~4 도 오류 없었다는 뜻이다
/dev/vgdata/lvdata: 2629/131072 files (0.2% non-contiguous), 64279/524288 blocks
-- -n : 고치지 않고 검사만. 오류 없이 Pass 5 까지
[root@ol7a ~]# mount /dev/vgdata/lvdata /data; md5sum -c /root/f1.md5; du -sh /data
/data/f1: OK
-- f1 체크섬 일치. /etc 사본까지 152M 그대로
152M    /data
-- 재확인 : 마운트 성공, f1 체크섬 OK, 152M 그대로
-- lvremove 는 PE 의 내용을 지우지 않았다. 메타데이터만 돌아오면 된다

-- lvtmp 에 파일 시스템과 파일을 만든다.
[root@ol7a ~]# lvchange -ay vgdata/lvtmp; mkfs.ext4 -q /dev/vgdata/lvtmp; mkdir -p /mnt/t; mount /dev/vgdata/lvtmp /mnt/t; echo important > /mnt/t/a.txt; umount /mnt/t
-- 출력 없음. lvtmp 에 ext4 와 파일 하나가 생겼다
-- lvtmp 를 지우고 바로 새 LV 를 만든다. 비어 있는 PE 중 앞쪽, 곧 lvtmp 자리가 쓰인다.
-- 이번에는 lvtmp 를 지운다
[root@ol7a ~]# lvremove -y vgdata/lvtmp
  Logical volume "lvtmp" successfully removed
-- lvtmp 의 PE 512~767 이 빈 공간이 됐다
[root@ol7a ~]# lvcreate -y -L 1G -n lvnew vgdata
  Wiping ext4 signature on /dev/vgdata/lvnew.
  Logical volume "lvnew" created.
-- 새 LV 가 어느 PE 를 받았는지 본다
[root@ol7a ~]# lvs -o lv_name,lv_size,devices vgdata
  LV     LSize Devices      
  lvdata 2.00g /dev/sdb(0)  
  lvnew  1.00g /dev/sdb(512)
-- lvnew 가 sdb(512), 곧 lvtmp 가 있던 자리를 받았다
-- 첫 줄 "Wiping ext4 signature" 는 lvcreate 가 옛 ext4 서명을 지운 기록
-- 새 LV 를 만드는 것만으로 옛 내용의 첫 부분이 사라진다
-- 새 LV 의 앞 10MB 를 0 으로 채우고 지운다. 다른 서비스가 잠깐 쓴 상황이다

-- lvnew 는 사라졌지만 lvtmp 자리의 앞 10MB 는 이미 0 이다
-- 옛 lvtmp 를 되살린다. 복원 전에 VG 를 비활성화한다.
[root@ol7a ~]# umount /data; vgchange -an vgdata

[root@ol7a ~]# vgcfgrestore -f $(grep -l "lvremove -y vgdata/lvtmp" /etc/lvm/archive/vgdata_*) vgdata
  Restored volume group vgdata

  Scan of VG vgdata from /dev/sdc found metadata seqno 5 vs previous 7.
-- seqno 7 → 5. lvtmp 를 지우기 직전 판으로 돌아갔다
-- 메타데이터 복원은 성공했다
-- vgchange -ay : VG 의 LV 를 모두 활성화
[root@ol7a ~]# vgchange -ay vgdata; lvs -o lv_name,lv_size,devices vgdata

  LV     LSize Devices

[root@ol7a ~]# mount /dev/vgdata/lvtmp /mnt/t

       missing codepage or helper program, or other error

       In some cases useful info is found in syslog - try
       dmesg | tail or so.
-- 증상   : mount 가 "wrong fs type, bad option, bad superblock" 으로 실패
-- 파일 시스템 서명이 남았는지 blkid 로 본다.

blkid 종료 코드 2

-- 조치   : LVM 으로는 되살릴 수 없다. 백업에서 복구한다
-- 재확인 : 같은 VG 의 lvdata 는 마운트·체크섬 OK. 덮어쓴 LV 만 잃었다
-- 다른 LV 는 무사한가
