================================================================================
 제5장 실습 05 — 로컬 ISO 저장소
 대상   : ol7a
 목표   : 설치 ISO 를 마운트해 인터넷 없이 쓰는 저장소를 만들고, 다른 저장소를
          모두 끈 채 그 저장소만으로 설치해 본다. createrepo 로 RPM 몇 개를 모은
          저장소를 직접 만들고, 메타데이터를 다시 만들지 않거나 캐시가 남아
          새 패키지가 안 보이는 상황을 확인한다
 배경   : yum 저장소는 RPM 파일과 repodata/ 디렉터리(repomd.xml 과 메타데이터)
          로 이루어진다. 설치 ISO 의 루트에 이미 repodata 가 있어 마운트만 하면
          저장소가 된다. baseurl 은 file:// 뒤에 절대 경로를 붙이므로 슬래시가
          세 개다 (본문 5.6)
 소요   : 약 30분
 선행   : ol7-base 스냅샷, ol7a 의 sr0 에 설치 ISO 연결
 계정   : root
 주의   : /root 에 ISO 사본(약 4.5GB)을 만들었다 지운다
 되돌림 : 마운트 해제, 추가한 .repo·디렉터리 삭제. yum repolist 가 [0] 과
          같으면 끝
================================================================================

-- VMware 설정에서 CD/DVD 가 연결돼 있어야 sr0 이 보인다.
-- 연결이 끊긴 VM 이면 lsblk 에 sr0 이 없거나 SIZE 가 0 이다.
-- 저장소로 쓸 ISO 가 sr0 에 붙어 있는지부터 본다.
[root@ol7a ~]# lsblk /dev/sr0; blkid /dev/sr0
NAME MAJ:MIN RM  SIZE RO TYPE MOUNTPOINT
sr0   11:0    1  4.5G  0 rom  
/dev/sr0: UUID="2021-05-28-10-00-48-00" LABEL="OL-7.9 Server.x86_64" TYPE="iso9660" PTTYPE="dos" 
-- ISO 는 읽기 전용 매체다.
-- lsblk 열: NAME / MAJ:MIN / RM(이동식 1) / SIZE / RO(읽기 전용 0) / TYPE(rom) / MOUNTPOINT(비어 있음)
-- blkid 의 UUID 는 ISO 를 만든 시각(2021-05-28 10:00:48)이다.
-- LABEL 은 "OL-7.9 Server.x86_64", TYPE 은 iso9660 이다.
-- 레이블만 봐도 7.9 Server 매체인 것을 알 수 있다.
[root@ol7a ~]# ls /etc/yum.repos.d/; findmnt /mnt; echo "findmnt rc=$?"

-- 마운트 지점은 비어 있어야 한다.
-- /mnt 는 임시 마운트용으로 비워 두는 관례적 자리다.
-- 형식(iso9660)은 mount 가 장치를 읽고 스스로 알아낸다.
-- 옵션 없이 붙인다. 광학 매체라 커널이 알아서 ro 로 바꾼다.
[root@ol7a ~]# mount /dev/sr0 /mnt; findmnt /mnt

--------------------------------------------------------------------------------
 [2] 저장소 정의 — 슬래시 두 개
--------------------------------------------------------------------------------

-- 구역 이름 [ol79_iso] 가 저장소 ID 가 된다.
-- 공백을 쓰지 않는다.
-- name 은 사람이 읽는 설명, gpgcheck=1 이면 ISO 의 서명도 확인한다.
-- 일부러 baseurl 을 file://mnt 로 쓴다(슬래시 둘).
[root@ol7a ~]# cat > /etc/yum.repos.d/ol79-iso.repo <<'EOF'
[ol79_iso]
name=Oracle Linux 7.9 ISO
baseurl=file://mnt
gpgkey=file:///etc/pki/rpm-gpg/RPM-GPG-KEY-oracle
gpgcheck=1
enabled=1
EOF
-- 파일은 다섯 줄짜리다.
-- gpgkey 는 슬래시 셋으로 바르게 썼다.
-- 둘을 비교해 보는 셈이다.
[root@ol7a ~]# yum --disablerepo='*' --enablerepo=ol79_iso repolist 2>&1 | tail -8
Loaded plugins: langpacks, ulninfo
file://mnt/repodata/repomd.xml: [Errno 14] curl#37 - "Couldn't open file /repodata/repomd.xml"
Trying other mirror.
file://mnt/repodata/repomd.xml: [Errno 14] curl#37 - "Couldn't open file /repodata/repomd.xml"
Trying other mirror.
repo id                        repo name                                  status
ol79_iso                       Oracle Linux 7.9 ISO                       0
repolist: 0
-- 오류 흐름 ① — file:// 뒤 슬래시가 둘
-- 증상   : "file://mnt/repodata/repomd.xml: [Errno 14] curl#37 - "Couldn't open file /repodata/repomd.xml"", status 0
-- 원인   : URL 문법에서 // 뒤는 호스트 자리다.
--          mnt 가 호스트로 읽히고 경로는 /repodata/... 만 남았다.
--          메시지의 경로에 /mnt 가 빠져 있는 것이 단서다.
-- 조치   : baseurl=file:///mnt 로 고친다(슬래시 셋)
-- 재확인 : 아래 repolist 에서 ol79_iso status 5,210 저장소 하나가 실패해도 yum 은 오류 코드로 멈추지 않고 0 으로 보인다.
-- "Trying other mirror." 는 다른 주소가 없어 같은 실패를 반복한 것.

[root@ol7a ~]# yum list available tree createrepo 2>&1 | tail -3

 [6] createrepo 로 저장소 만들기
--------------------------------------------------------------------------------

-- /srv 는 서버가 내보낼 데이터를 두는 FHS 표준 자리다.
-- 사내 RPM 이나 벤더 RPM 을 이렇게 모아 배포한다.
-- RPM 두 개를 모아 저장소를 직접 만든다.
[root@ol7a ~]# mkdir -p /srv/myrepo; cp /mnt/Packages/tree-*.rpm /mnt/Packages/nmap-ncat-*.rpm /srv/myrepo/; ls /srv/myrepo

tree-1.6.0-10.el7.x86_64.rpm
-- createrepo 는 RPM 머리를 읽어 repodata/ 를 만든다.
[root@ol7a ~]# createrepo /srv/myrepo; ls /srv/myrepo/repodata

-- worker 두 개가 RPM 하나씩 읽었다.

-- primary(이름·의존성), filelists(파일 목록), other(changelog).
-- 서명 없는 사내 RPM 도 넣을 수 있게 gpgcheck=0, 평소엔 꺼 둠(enabled=0).
[root@ol7a ~]# cat > /etc/yum.repos.d/myrepo.repo <<'EOF'

name=My local repo

 [변형] ISO 파일 루프 마운트와 /etc/fstab
--------------------------------------------------------------------------------

-- 운영에서는 ISO 파일을 USB·공유 디스크로 옮겨 오는 것이 보통이다.
-- 여기서는 sr0 을 그대로 떠서 그 과정을 대신한다.
-- 문서(YUM 7장) 방식: ISO 파일을 디스크에 두고 루프 마운트.
-- dd 로 sr0 을 파일로 뜬다.
-- bs=4M 은 한 번에 읽을 크기.
[root@ol7a ~]# umount /mnt; dd if=/dev/sr0 of=/root/OL79.iso bs=4M 2>&1 | tail -1; ls -lh /root/OL79.iso
4857004032 bytes (4.9 GB) copied, 35.0572 s, 139 MB/s
-rw-r--r--. 1 root root 4.6G Nov 13 20:15 /root/OL79.iso
-- 원본 sr0 은 그대로 둔다.
-- 4,857,004,032 바이트를 35 초에 복사(139 MB/s). ls -lh 로 4.6G.
[root@ol7a ~]# mkdir -p /var/OSimage/OL7.9_x86_64; echo '/root/OL79.iso /var/OSimage/OL7.9_x86_64 iso9660 loop,ro 0 0' >> /etc/fstab; tail -1 /etc/fstab
/root/OL79.iso /var/OSimage/OL7.9_x86_64 iso9660 loop,ro 0 0
-- >> 로 덧붙였다.
-- > 로 쓰면 fstab 전체가 지워져 부팅이 깨진다.
-- fstab 여섯 칸: 장치(파일) / 마운트 지점 / 형식 / 옵션 / dump / fsck 순서.
-- loop 는 파일을 블록 장치처럼 붙이는 옵션, ro 는 읽기 전용.
[root@ol7a ~]# mount -a; findmnt /var/OSimage/OL7.9_x86_64; losetup -a
TARGET                    SOURCE     FSTYPE  OPTIONS
/var/OSimage/OL7.9_x86_64 /dev/loop0 iso9660 ro,relatime,nojoliet,check=s,map=n,blocksize=2048,iocharset=utf8
/dev/loop0: [64512]:135056715 (/root/OL79.iso)
-- 파일이 블록 장치처럼 붙었다.
-- [64512]:135056715 는 ISO 파일이 있는 장치 번호와 inode 다.
-- mount -a 가 fstab 을 읽어 붙였다.
-- 재부팅해도 같은 방식으로 붙는다.
-- SOURCE 가 /dev/loop0. losetup 은 loop0 이 어느 파일인지 보여 준다.
[root@ol7a ~]# sed -i 's#^baseurl=file:///mnt#baseurl=file:///var/OSimage/OL7.9_x86_64#' /etc/yum.repos.d/ol79-iso.repo; yum clean all >/dev/null 2>&1; yum --disablerepo='*' --enablerepo=ol79_iso repolist 2>&1 | tail -3
repo id                        repo name                                  status
ol79_iso                       Oracle Linux 7.9 ISO                       5,210
repolist: 5,210
-- baseurl 만 새 경로로 바꾸고 clean all 뒤 확인. 같은 5,210.

-- 같은 tree 가 두 저장소에서 온다.

-- 오류 흐름 ② — ISO 와 인터넷 저장소가 함께 켜짐
-- 증상   : repolist 합계 34,040. [5] 에서 tree 가 ol79_iso 와 ol7_latest 두 곳에서 나왔다
-- 원인   : ol79-iso.repo 를 enabled=1 로 만들었고 기본 저장소도 그대로 켜져 있다.
--          같은 패키지가 두 출처에서 온다.
--          인터넷이 끊긴 서버라면 ol7_latest 접속 실패가 매번 난다.
-- 조치   : 문서 5단계대로 yum-config-manager --disable \* 로 모두 끄고 --enable ol79_iso 로 하나만 켠다
-- 재확인 : 아래 repolist 에 ol79_iso 하나, 5,210
[root@ol7a ~]# yum-config-manager --disable \* >/dev/null; yum-config-manager --enable ol79_iso >/dev/null; yum repolist 2>&1 | tail -4
