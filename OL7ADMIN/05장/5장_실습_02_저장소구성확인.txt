================================================================================
 제5장 실습 02 — 저장소 구성 확인
 대상   : ol7a
 목표   : yum.conf 와 /etc/yum.repos.d 의 .repo 파일을 읽고, 저장소를 켜고 끄는
          세 가지 방법(파일 편집, yum-config-manager, 명령줄 --enablerepo)을 비교한다.
          oraclelinux-release-el7 이 .repo 파일을 관리한다는 점과 $releasever·$basearch
          가 무엇으로 바뀌는지 확인한다. GPG 키와 패키지 서명도 같이 본다
 배경   : yum 은 [main] 구역과 저장소 구역을 합쳐 하나의 구성을 만든다.
          OL7 의 저장소 정의는 2019 년 이후 모듈형 release 패키지로 배포되고,
          기본 활성 저장소는 ol7_latest 와 ol7_UEKR6 둘이다. 꺼져 있는 저장소의
          패키지는 검색에도 나오지 않는다 (본문 5.3)
 소요   : 약 25분
 선행   : ol7-base 스냅샷, 인터넷(yum.oracle.com) 접속
 계정   : root
 주의   : ol7_addons 를 켰다 끈다. 임시 .repo 파일 하나를 만들었다 지운다
 되돌림 : yum repolist 결과가 [0] 과 같은 두 저장소로 돌아오면 끝

--------------------------------------------------------------------------------
 [1] /etc/yum.conf — [main] 구역
--------------------------------------------------------------------------------

-- 주석 줄도 그대로 보인다.
-- [main] 구역만 있고 저장소 구역은 없다.
[root@ol7a ~]# cat /etc/yum.conf
[main]
cachedir=/var/cache/yum/$basearch/$releasever
keepcache=0
debuglevel=2
logfile=/var/log/yum.log
exactarch=1
obsoletes=1
gpgcheck=1
plugins=1
installonly_limit=3

#  This is the default, if you make this bigger yum won't see if the metadata
# is newer on the remote and so you'll "gain" the bandwidth of not having to
# download the new metadata and "pay" for it by yum not having correct
# information.
#  It is esp. important, to have correct metadata, for distributions like
# Fedora which don't keep old packages around. If you don't like this checking
# interupting your command line usage, it's much better to have something
# manually check the metadata once an hour (yum-updatesd will do this).
# metadata_expire=90m

# PUT YOUR REPOS HERE OR IN separate files named file.repo
# in /etc/yum.repos.d
-- 이 랩의 yum.conf 는 설치 기본값 그대로다.
-- 지시어 해설
--   cachedir         : $basearch/$releasever 가 풀려 x86_64/7Server
--   keepcache=0      : 설치 뒤 RPM 을 지움
--   debuglevel=2     : 출력 양(0~10)
--   logfile          : /var/log/yum.log
--   exactarch=1      : 같은 아키텍처로만 업데이트
--   obsoletes=1      : 대체 관계 반영
--   gpgcheck=1       : 서명 확인. 저장소 구역이 덮어쓸 수 있다
--   plugins=1        : ulninfo, langpacks 플러그인 사용
--   installonly_limit=3 : 커널 보존 개수(실습 06)
-- metadata_expire 는 주석이다. 기본값이 적용된다.
-- proxy 줄이 없다. 이 랩은 프록시 없이 직접 나간다.

 [2] .repo 파일은 누가 설치했나
--------------------------------------------------------------------------------

-- 셋 다 같은 패키지 소속인지 본다.
-- .repo 도 파일이니 주인 패키지를 물어본다.
[root@ol7a ~]# rpm -qf /etc/yum.repos.d/*.repo
oraclelinux-release-el7-1.0-17.el7.x86_64
-- 1.0-17 은 release 패키지 자신의 판이다.
oraclelinux-release-el7-1.0-17.el7.x86_64
oraclelinux-release-el7-1.0-17.el7.x86_64
-- 세 파일 모두 oraclelinux-release-el7 소속.
-- 2019 년 1월부터 저장소 정의를 패키지로 배포한다(YUM 2장).
-- yum update 를 하면 저장소 정의도 함께 갱신된다.
[root@ol7a ~]# rpm -qi oraclelinux-release-el7 | sed -n '1,4p;/^Summary/p'
Name        : oraclelinux-release-el7
Version     : 1.0
Release     : 17.el7
Architecture: x86_64
Summary     : Oracle Linux yum repository configuration
-- sed 로 머리 네 줄과 Summary 만 골랐다.
-- 배포판 판(7.9)과 따로 1.0-17 이라는 자기 판을 갖는다.
[root@ol7a ~]# rpm -qc oraclelinux-release-el7
/etc/yum.repos.d/oracle-linux-ol7.repo
/etc/yum.repos.d/uek-ol7.repo
/etc/yum.repos.d/virt-ol7.repo
-- .rpmnew 가 생기면 diff 로 비교해 합친다.
-- -qc 에 세 파일이 모두 있다.
-- %config 라 관리자가 고쳐도 업데이트가 덮어쓰지 않고 .rpmnew 로 새 판을 옆에 둔다.

--------------------------------------------------------------------------------
 [3] 저장소 구역 읽기
--------------------------------------------------------------------------------

-- 정규식은 줄 맨 앞의 [ 를 찾는다.
-- 구역 머리 [ ... ] 의 개수 = 저장소 정의 수.
[root@ol7a ~]# grep -c '^\[' /etc/yum.repos.d/oracle-linux-ol7.repo /etc/yum.repos.d/uek-ol7.repo
/etc/yum.repos.d/oracle-linux-ol7.repo:19
/etc/yum.repos.d/uek-ol7.repo:10
-- 둘을 합치면 29 개다.
-- oracle-linux-ol7.repo 19 개, uek-ol7.repo 10 개.
-- 대부분 꺼져 있고 켜진 것은 둘뿐이다([5]).
-- sed 의 /시작/,/끝/p 는 범위를 찍는다.
-- 두 구역만 잘라 본다. 빈 줄까지가 한 구역이다.
[root@ol7a ~]# sed -n '/^\[ol7_latest\]/,/^$/p;/^\[ol7_addons\]/,/^$/p' /etc/yum.repos.d/oracle-linux-ol7.repo
[ol7_latest]
name=Oracle Linux $releasever Latest ($basearch)
baseurl=https://yum$ociregion.$ocidomain/repo/OracleLinux/OL7/latest/$basearch/
gpgkey=file:///etc/pki/rpm-gpg/RPM-GPG-KEY-oracle
gpgcheck=1
enabled=1

[ol7_addons]
name=Oracle Linux $releasever Add ons ($basearch)
baseurl=https://yum$ociregion.$ocidomain/repo/OracleLinux/OL7/addons/$basearch/
gpgkey=file:///etc/pki/rpm-gpg/RPM-GPG-KEY-oracle
gpgcheck=1
enabled=0
-- 경로의 latest·addons 가 저장소 디렉터리다.
-- 두 구역의 차이는 baseurl 의 경로(latest/addons)와 enabled 뿐.
-- baseurl 의 변수 셋
--   $ociregion  : /etc/yum/vars/ociregion 내용
--   $ocidomain  : /etc/yum/vars/ocidomain 내용
--   $basearch   : x86_64
-- gpgkey 는 file:/// 로 로컬 키 파일을 가리킨다.

 [4] yum 변수 — $releasever, $basearch, $uekr6
--------------------------------------------------------------------------------

-- 파일 이름이 곧 변수 이름이다.
-- yum 변수 파일은 /etc/yum/vars 에 이름=변수명으로 둔다.
[root@ol7a ~]# ls /etc/yum/vars/; head /etc/yum/vars/*
ocidomain
ociregion
==> /etc/yum/vars/ocidomain <==
oracle.com

==> /etc/yum/vars/ociregion <==
-- 빈 파일도 변수로 읽힌다(빈 문자열).
-- ocidomain 은 oracle.com, ociregion 은 비어 있다.
-- 그래서 baseurl 이 https://yum.oracle.com/... 로 풀린다.
-- OCI 인스턴스라면 ociregion 에 -phx 같은 값이 들어가 리전 yum 서버로 간다.
[root@ol7a ~]# python -c 'import yum; yb=yum.YumBase(); yb.preconf.debuglevel=0; print yb.conf.yumvar["releasever"], yb.conf.yumvar["basearch"]' 2>&1 | tail -1
7Server x86_64
-- 7Server 는 경로에 쓰이는 이름이다.
-- yum 이 내부에서 계산한 값을 파이썬으로 직접 꺼냈다.
-- releasever 는 7.9 가 아니라 7Server 다.
[root@ol7a ~]# rpm -q --provides oraclelinux-release | grep -i release
config(oraclelinux-release) = 7:7.9-1.0.13.el7
oraclelinux-release = 7:7.9-1.0.13.el7
oraclelinux-release(x86-64) = 7:7.9-1.0.13.el7
system-release = 7.9-1.0.13.el7
system-release(releasever) = 7Server
-- 7Server 는 Server 변형이라는 뜻이다.
-- releasever 는 system-release(releasever) 를 제공하는 패키지에서 온다.
-- oraclelinux-release 7.9-1.0.13 이 7Server 를 제공한다.
-- 에포크 7: 이 붙어 있다. 판 비교에서 가장 먼저 본다.
-- 그래서 $releasever 는 7.x 업데이트를 해도 바뀌지 않는다.

 [6] 꺼진 저장소의 패키지는 보이지 않는다
--------------------------------------------------------------------------------

-- 기본 설정 그대로 찾는다.
-- ol7_addons 에만 있는 패키지를 기본 설정으로 찾아본다.
[root@ol7a ~]# yum info WALinuxAgent 2>&1 | tail -3
Loaded plugins: langpacks, ulninfo
Error: No matching Packages to list
-- 이름을 틀린 것과 메시지가 같다.

--------------------------------------------------------------------------------
 [7] yum-config-manager 로 켜기
--------------------------------------------------------------------------------

-- 이 도구가 어느 패키지에 있는지부터 본다.
-- which 가 경로를 찾고 rpm -qf 가 그 경로의 주인을 답한다.
-- yum-config-manager 는 yum-utils 에 들어 있다.
[root@ol7a ~]# rpm -qf $(which yum-config-manager)
yum-utils-1.1.31-54.0.1.el7_8.noarch
-- -p 는 시각과 권한을 지킨다.
-- 바꾸기 전 사본을 떠 두고 diff 로 비교한다.
[root@ol7a ~]# cp -p /etc/yum.repos.d/oracle-linux-ol7.repo /root/ol7.repo.before
[root@ol7a ~]# yum-config-manager --enable ol7_addons 2>&1 | head -4
Loaded plugins: langpacks
=============================== repo: ol7_addons ===============================
[ol7_addons]
async = True
-- 고친 결과를 다시 읽어 보여 준다.
-- --enable 은 설정을 바꾼 뒤 결과 구역을 다시 찍어 보여 준다.
-- 잘린 뒤쪽에 baseurl, enabled = 1 등이 이어진다.
-- 출력은 그 저장소의 전체 설정을 길게 찍는다. head 로 4줄만.
[root@ol7a ~]# diff /root/ol7.repo.before /etc/yum.repos.d/oracle-linux-ol7.repo
90c90
< enabled=0
---
> enabled=1
-- 파일 한 줄이 바뀐 것이 전부다.
-- 파일에서 바뀐 것은 90번째 줄 하나다. enabled=0 → 1.
-- 다른 줄은 손대지 않았다. 주석도 그대로 남는다.
[root@ol7a ~]# yum repolist 2>&1 | tail -6
Loaded plugins: langpacks, ulninfo
repo id            repo name                                              status
!ol7_UEKR6/x86_64  Latest Unbreakable Enterprise Kernel Release 6 for Ora  1,401
ol7_addons/x86_64  Oracle Linux 7Server Add ons (x86_64)                     981
!ol7_latest/x86_64 Oracle Linux 7Server Latest (x86_64)                   27,429
repolist: 29,811
-- ol7_addons 줄에는 ! 가 없다.
-- 방금 메타데이터를 새로 받았다.
-- ol7_addons 981 개가 더해져 29,811.
-- 이 상태는 재부팅해도 유지된다(파일을 고쳤으므로).

--------------------------------------------------------------------------------
 [8] 끄고 비교 — 영구 변경과 일회성 지정
--------------------------------------------------------------------------------

[root@ol7a ~]# yum-config-manager --disable ol7_addons >/dev/null; diff /root/ol7.repo.before /etc/yum.repos.d/oracle-linux-ol7.repo; echo "diff rc=$?"
diff rc=0
-- 파일이 원래대로 돌아왔다.
-- --disable 로 되돌리니 diff 가 비어 있다.
-- 원본과 같은 파일이다.
[root@ol7a ~]# yum --enablerepo=ol7_addons repolist 2>&1 | tail -6
Loaded plugins: langpacks, ulninfo
repo id            repo name                                              status
!ol7_UEKR6/x86_64  Latest Unbreakable Enterprise Kernel Release 6 for Ora  1,401
ol7_addons/x86_64  Oracle Linux 7Server Add ons (x86_64)                     981
!ol7_latest/x86_64 Oracle Linux 7Server Latest (x86_64)                   27,429
repolist: 29,811
-- --enablerepo 로 이 명령 한 번만 켰다. 29,811.
[root@ol7a ~]# yum repolist 2>&1 | tail -3
!ol7_UEKR6/x86_64  Latest Unbreakable Enterprise Kernel Release 6 for Ora  1,401
!ol7_latest/x86_64 Oracle Linux 7Server Latest (x86_64)                   27,429
repolist: 28,830

 [9] GPG 키와 패키지 서명
--------------------------------------------------------------------------------

-- --qf 로 이름과 요약만 뽑았다.
-- 가져온 공개 키는 gpg-pubkey 라는 가짜 패키지로 보인다.
[root@ol7a ~]# rpm -q gpg-pubkey --qf '%{NAME}-%{VERSION}-%{RELEASE}  %{SUMMARY}\n'
gpg-pubkey-ec551f03-53619141  gpg(Oracle OSS group (Open Source Software group) <build@oss.oracle.com>)
-- 이 키 하나로 ol7_latest·UEKR6 패키지를 모두 확인한다.
-- VERSION ec551f03 = 키 ID 끝 8자리, RELEASE 53619141 = 키 생성 시각(16진).

[root@ol7a ~]# ls -l /etc/pki/rpm-gpg/

-rw-r--r--. 1 root root 1011 Jun 26  2024 RPM-GPG-KEY-oracle
-- 키 파일은 텍스트(ASCII armor)다.
-- RPM-GPG-KEY 와 RPM-GPG-KEY-oracle 은 크기 1011 로 같다.
-- fedora 키 두 개는 업스트림 호환용으로 들어 있을 뿐 쓰이지 않는다.
-- RPM-GPG-KEY-oracle-ol7 이라는 파일은 없다.
-- [10] 에서 쓴다.
[root@ol7a ~]# rpm -q --qf '%{NAME}  %{SIGPGP:pgpsig}\n' chrony yum

yum  RSA/SHA256, Sat 17 Jun 2023 07:04:24 AM KST, Key ID 72f97b74ec551f03
-- 설치된 패키지의 서명 정보다.
-- 두 패키지 모두 Key ID 72f97b74ec551f03 으로 서명됐다.
[root@ol7a ~]# yumdownloader --destdir=/tmp tree 2>&1 | tail -2; rpm -K /tmp/tree-*.rpm; rpm -Kv /tmp/tree-*.rpm
Loaded plugins: langpacks
/tmp/tree-1.6.0-10.el7.x86_64.rpm: rsa sha1 (md5) pgp md5 OK
/tmp/tree-1.6.0-10.el7.x86_64.rpm:
    Header V3 RSA/SHA256 Signature, key ID ec551f03: OK
    Header SHA1 digest: OK (3cb264e42a866a097afe1bb60e2dc0343d3421c3)
    V3 RSA/SHA256 Signature, key ID ec551f03: OK
    MD5 digest: OK (30e4a24937cf5ed4788e0da917db0b48)
-- 설치 전 파일도 같은 키로 확인한다.
-- rpm -K 의 요약 줄: rsa sha1 (md5) pgp md5 OK
--   rsa·pgp 는 서명, sha1·md5 는 다이제스트 검사. 모두 통과.
-- -Kv 는 항목별로: 머리 서명 / 머리 SHA1 / 전체 서명 / MD5.
-- 실습이 끝나면 키를 다시 들여온다.

[root@ol7a ~]# rpm -e gpg-pubkey-ec551f03-53619141; rpm -q gpg-pubkey; rpm -K /tmp/tree-*.rpm
package gpg-pubkey is not installed

-- rpm -K 의 종료 코드는 1 이 된다.
-- 대문자와 괄호는 검사하지 못한 항목. MISSING KEYS: PGP#ec551f03.
-- 파일은 그대로인데 판정이 바뀌었다.
-- 키가 없으면 서명을 믿을 수 없다.
