================================================================================
 제19장 실습 04 — 템플릿으로 호스트별 파일
 대상   : ol7a (전송), ol7b (수신)
 목표   : 수신 서버 ol7b 에 template() 으로 파일 경로를 만들어 원격 메시지를
          /var/log/remote/<호스트>/<프로그램>.log 로 나눠 담는다. ruleset 으로 원격
          메시지를 ol7b 자신의 messages 와 분리하고, 줄 형식 템플릿을 하나 더 만든다
 배경   : 템플릿은 메시지 속성(%HOSTNAME%, %programname%, %timegenerated% 등)을 끼워
          넣은 문자열이다. omfile 의 dynaFile 에 템플릿 이름을 주면 메시지마다 경로가
          달라진다. 템플릿 이름이 틀리면 그 액션이 꺼진다. 없는 디렉터리는 omfile 의
          createDirs(기본 on)가 만든다 (본문 19.3·19.4)
 소요   : 약 30분
 선행   : ol7-base, ol7b-base 스냅샷
 계정   : root
 주의   : ol7b 의 firewalld 영구 설정을 바꿨다가 되돌린다. 끝에 ol7b 를 끈다
 되돌림 : 두 노드 rsyslog.d 비움, /var/log/remote 없음, 514/tcp 닫힘 → ls /etc/rsyslog.d, ss -tlnp

 [2] 경로 템플릿과 ruleset — 이름을 틀리게
--------------------------------------------------------------------------------

-- 오류 흐름 ①. 템플릿은 PerHost 로 정의하고 dynaFile 에는 PerHosts 로 적는다.
--   template(name= type="string" string=)   %속성% 을 끼운 문자열
--   ruleset(name=){ ... }                   이 input 으로 들어온 메시지만 처리하는 규칙 묶음
--   input(... ruleset="remote")             TCP 514 로 들어온 메시지는 기본 규칙(RULES)을 타지 않는다
-- 수신 쪽 설정 한 파일
[root@ol7b ~]# cat > /etc/rsyslog.d/10-remote.conf <<'EOF'
template(name="PerHost" type="string"
         string="/var/log/remote/%HOSTNAME%/%programname%.log")
ruleset(name="remote") {
    action(type="omfile" dynaFile="PerHosts")
    stop
}
module(load="imtcp")
input(type="imtcp" port="514" ruleset="remote")
EOF

-- 이 파일 하나에 템플릿·규칙 묶음·입력 모듈이 다 있다. 순서는 상관없다
-- %HOSTNAME% 은 메시지의 호스트 이름, %programname% 은 태그에서 콜론·PID 를 뗀 이름
-- 재시작 전에 검사부터
[root@ol7b ~]# rsyslogd -N1; echo "exit=$?"

-- createDirs 가 만든 디렉터리의 권한·소유자. 파일 권한은 실습 02 와 같은 규칙을 따른다.

 [6] createDirs 를 끄면
--------------------------------------------------------------------------------

-- 오류 흐름 ②. 경로 템플릿을 날짜별 디렉터리로 바꾸고 createDirs="off" 를 준다.
-- 경로에 날짜를 넣고 디렉터리 자동 생성을 끈다
[root@ol7b ~]# sed -i 's|string="/var/log/remote/%HOSTNAME%/%programname%.log"|string="/var/log/remote/%HOSTNAME%/%$YEAR%%$MONTH%%$DAY%/%programname%.log"|; s|dynaFile="PerHost")|dynaFile="PerHost" createDirs="off")|' /etc/rsyslog.d/10-remote.conf
-- sed 두 군데(경로, createDirs) 결과
[root@ol7b ~]# cat /etc/rsyslog.d/10-remote.conf
template(name="PerHost" type="string"
         string="/var/log/remote/%HOSTNAME%/%$YEAR%%$MONTH%%$DAY%/%programname%.log")
ruleset(name="remote") {
    action(type="omfile" dynaFile="PerHost" createDirs="off")
    stop
}
module(load="imtcp")
input(type="imtcp" port="514" ruleset="remote")
-- 경로에 %$YEAR%%$MONTH%%$DAY% 를 넣었다. 날짜마다 새 디렉터리가 필요하다
-- createDirs="off" 로 디렉터리 자동 생성을 끈다
[root@ol7b ~]# rsyslogd -N1; systemctl restart rsyslog

 [7] 줄 형식 템플릿
--------------------------------------------------------------------------------

-- 경로만이 아니라 한 줄의 모양도 템플릿으로 정한다. 기본 형식에는 연도와 초 아래 자릿수가 없다.
--   %timegenerated:::date-rfc3339%   수신한 시각을 RFC 3339(연도·밀리초·시간대) 로
--   %fromhost-ip%                    보낸 쪽 IP (HOSTNAME 은 메시지 안의 이름)
--   %syslogfacility-text%.%syslogseverity-text%   facility.priority 이름
-- 09- 로 시작해 10-remote.conf 보다 먼저 정의된다
[root@ol7b ~]# cat > /etc/rsyslog.d/09-fmt.conf <<'EOF'
template(name="LabFmt" type="string"
         string="%timegenerated:::date-rfc3339% %fromhost-ip% %HOSTNAME% %syslogfacility-text%.%syslogseverity-text% %syslogtag%%msg%\n")
EOF
-- 줄 형식 템플릿. 끝의 \n 이 줄바꿈이다. 문자열 템플릿은 줄바꿈을 직접 넣어야 한다
-- omfile 에 template= 로 줄 형식을 건다
[root@ol7b ~]# sed -i 's|dynaFile="PerHost")|dynaFile="PerHost" template="LabFmt")|' /etc/rsyslog.d/10-remote.conf; grep omfile /etc/rsyslog.d/10-remote.conf
    action(type="omfile" dynaFile="PerHost" template="LabFmt")
[root@ol7b ~]# rsyslogd -N1; systemctl restart rsyslog
rsyslogd: version 8.24.0-57.0.3.el7_9.3, config validation run (level 1), master config /etc/rsyslog.conf
rsyslogd: End of config validation run. Bye.
[root@ol7a ~]# systemctl restart rsyslog
-- facility·priority 가 줄에 찍히는지 보려고 local2.warning 으로 보낸다
[root@ol7a ~]# logger -t app1 -p local2.warning "LAB19-07 formatted"; sleep 3
-- 같은 파일에 기본 형식 줄과 새 형식 줄이 이어진다
[root@ol7b ~]# tail -2 /var/log/remote/ol7a/2*/app1.log
Nov 16 16:05:46 ol7a app1: LAB19-06 withdir
2026-11-16T16:05:53.380286+09:00 172.16.225.61 ol7a local2.warning app1: LAB19-07 formatted

 [8] 같은 템플릿을 옛 형식으로
--------------------------------------------------------------------------------

-- $template 지시어 형식. 기존 설정 파일에서 자주 보인다. 검사만 하고 지운다.
-- 옛 형식 두 줄: $template 정의, 규칙 끝 ;이름
[root@ol7b ~]# printf '%s\n' '$template LabFmtOld,"%timegenerated:::date-rfc3339% %HOSTNAME% %syslogtag%%msg%\n"' 'local1.*  /var/log/lab-old.log;LabFmtOld' > /etc/rsyslog.d/08-old.conf; cat /etc/rsyslog.d/08-old.conf
$template LabFmtOld,"%timegenerated:::date-rfc3339% %HOSTNAME% %syslogtag%%msg%\n"
local1.*  /var/log/lab-old.log;LabFmtOld
[root@ol7b ~]# rsyslogd -N1; systemctl restart rsyslog; logger -t oldfmt -p local1.info "LAB19-08 legacy template"; sleep 2; cat /var/log/lab-old.log
rsyslogd: version 8.24.0-57.0.3.el7_9.3, config validation run (level 1), master config /etc/rsyslog.conf
rsyslogd: End of config validation run. Bye.
2026-11-16T16:06:00.488281+09:00 ol7b oldfmt:LAB19-08 legacy template
-- $template 이름,"문자열" 과 규칙 끝의 ;이름 형식도 똑같이 동작한다
-- 태그 뒤에 공백이 없다. 옛 템플릿 문자열에 %syslogtag% 와 %msg% 를 붙여 썼기 때문이다
-- 비교만 하고 지운다
[root@ol7b ~]# rm -f /etc/rsyslog.d/08-old.conf /var/log/lab-old.log

 [9] 보내는 쪽 형식 — RSYSLOG_ForwardFormat
--------------------------------------------------------------------------------

-- 전송 기본 형식(RSYSLOG_TraditionalForwardFormat)은 초 단위다. 정밀 시각 형식으로 바꾼다.
-- 규칙 끝 ;템플릿이름 은 전송 형식을 정한다
[root@ol7a ~]# echo '*.* @@ol7b.example.com:514;RSYSLOG_ForwardFormat' > /etc/rsyslog.d/50-fwd.conf; rsyslogd -N1; systemctl restart rsyslog
rsyslogd: version 8.24.0-57.0.3.el7_9.3, config validation run (level 1), master config /etc/rsyslog.conf
rsyslogd: End of config validation run. Bye.
-- 줄 형식에 timereported 를 더한다
[root@ol7b ~]# sed -i 's|%timegenerated:::date-rfc3339% %fromhost-ip%|%timegenerated:::date-rfc3339% %timereported:::date-rfc3339% %fromhost-ip%|' /etc/rsyslog.d/09-fmt.conf; rsyslogd -N1; systemctl restart rsyslog
rsyslogd: version 8.24.0-57.0.3.el7_9.3, config validation run (level 1), master config /etc/rsyslog.conf
rsyslogd: End of config validation run. Bye.
[root@ol7a ~]# systemctl restart rsyslog
-- 정밀 시각 형식으로 한 건
[root@ol7a ~]# logger -t app1 "LAB19-09 forwardformat"; sleep 3
-- 시각 두 개를 나란히 본다
[root@ol7b ~]# tail -1 /var/log/remote/ol7a/2*/app1.log
2026-11-16T16:06:07.552080+09:00 2026-11-16T16:06:07.551847+09:00 172.16.225.61 ol7a user.notice app1: LAB19-09 forwardformat
-- 둘째 시각(timereported)은 ol7a 가 메시지에 적은 시각, 첫째(timegenerated)는 ol7b 가 받은 시각
-- RSYSLOG_ForwardFormat 은 보낼 때 마이크로초 시각을 넣는다. 둘의 차이로 전송 지연을 잴 수 있다
-- 기본 전송 형식(Traditional)은 초 단위라 timereported 의 자릿수가 줄어든다
