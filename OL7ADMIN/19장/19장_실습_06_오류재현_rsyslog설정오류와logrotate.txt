================================================================================
 제19장 실습 06 — 오류 재현: rsyslog 설정 오류와 logrotate
 대상   : ol7a
 목표   : 규칙 파일의 오타·괄호 누락을 rsyslogd -N1 로 찾아 고친다. 새 로그 파일에
          logrotate 규칙을 붙여 -d(시험)·-f(강제)로 돌려 보고, postrotate 를 빠뜨렸을 때
          rsyslog 가 회전된 옛 파일에 계속 쓰는 것을 확인한다. 괄호 누락이 /dev/log 를 지우는 장애도 재현한다
 배경   : rsyslog 는 설정 오류가 있어도 대개 데몬을 띄운다. 틀린 줄만 버리고 나머지로
          동작하므로 서비스 상태만 보면 오류를 놓친다. logrotate 는 파일 이름만 바꾸고,
          파일을 열어 둔 rsyslog 에 알리는 일은 postrotate 의 HUP 이 한다. 7.9 에서는
          cron.daily 가 logrotate·logwatch 를 하루 한 번 돌린다 (본문 19.2·19.7·19.8)
 소요   : 약 40분
 선행   : ol7-base 스냅샷
 계정   : root
 주의   : logwatch 패키지를 설치했다가 지운다(인터넷 저장소 사용). 만든 파일은 끝에서 지운다
 되돌림 : rsyslog.d·logrotate.d 원래 목록, logwatch 미설치 → ls, rpm -q logwatch

 [6] 새 파일에 규칙 붙이기 — postrotate 없이
--------------------------------------------------------------------------------

-- [2] 의 labapp.log 를 회전 대상으로 만든다. postrotate 를 일부러 넣지 않는다(오류 흐름 ②).
--   size 1k      크기가 1KB 를 넘으면 회전 (daily 같은 주기 대신)
--   rotate 3     옛 파일 3개까지 보관
--   missingok    파일이 없어도 오류로 치지 않음
--   notifempty   빈 파일은 회전하지 않음
--   create 0600 root root   회전 직후 같은 이름의 빈 파일을 이 권한으로 만듦
-- 규칙 파일 작성
[root@ol7a ~]# cat > /etc/logrotate.d/labapp <<'EOF'
/var/log/labapp.log {
    size 1k
    rotate 3
    missingok
    notifempty
    create 0600 root root
}
EOF
-- size 1k 와 rotate 3 만 준 최소 규칙. postrotate 가 없다

-- 파일을 1KB 넘게 채운다.
-- 30줄을 쓴다

l-wx------. 1 root root 64 Nov 16 18:06 8 -> /var/log/labapp-all.log
-- fd 7 이 labapp.log.1 을 가리킨다
--
-- 증상   : logrotate -f 뒤 새 labapp.log 는 0 바이트, 새 메시지는 labapp.log.1 에 기록
-- 원인   : 규칙에 postrotate 가 없어 rsyslog 가 회전 전 파일(같은 inode)을 계속 열고 있음.
--          /proc/PID/fd 의 7 -> /var/log/labapp.log.1 로 확인
-- 조치   : 규칙에 sharedscripts·postrotate kill -HUP 추가, 지금 상태는 kill -HUP 한 번
-- 재확인 : 다시 회전 뒤 "with HUP" 이 labapp.log 에, fd 가 labapp.log 를 가리킴

--------------------------------------------------------------------------------
 [8] postrotate 를 넣는다
--------------------------------------------------------------------------------

-- syslog 규칙과 같은 HUP 을 붙인다.
--   sharedscripts   패턴이 여러 파일에 맞아도 postrotate 를 한 번만 실행
-- 원본을 남긴다
[root@ol7a ~]# cp /etc/logrotate.d/labapp /root/labapp.v1
-- sed 의 a\ 는 맞는 줄 뒤에 붙인다
[root@ol7a ~]# sed -i '/create 0600 root root/a\    sharedscripts\n    postrotate\n        /bin/kill -HUP $(cat /var/run/syslogd.pid 2> /dev/null) 2> /dev/null || true\n    endscript' /etc/logrotate.d/labapp
-- 바뀐 줄 확인
[root@ol7a ~]# diff /root/labapp.v1 /etc/logrotate.d/labapp
6a7,10
>     sharedscripts
>     postrotate
>         /bin/kill -HUP $(cat /var/run/syslogd.pid 2> /dev/null) 2> /dev/null || true
>     endscript
-- create 다음에 네 줄. 백틱 대신 $( ) 를 썼다

-- 지금 상태를 바로잡으려고 한 번 HUP 을 보낸 뒤 다시 채우고 강제 회전.
[root@ol7a ~]# kill -HUP $(cat /var/run/syslogd.pid); for i in $(seq 1 30); do logger -t labapp -p local4.info "LAB19-08 fill $i"; done; sleep 2
-- 다시 강제 회전
[root@ol7a ~]# logrotate -v -f /etc/logrotate.d/labapp 2>&1 | grep -E '^renaming /var/log/labapp.log to|^creating|postrotate|does not need'
renaming /var/log/labapp.log to /var/log/labapp.log.1
creating new /var/log/labapp.log mode = 0600 uid = 0 gid = 0
running postrotate script
-- 이번에는 running postrotate script 가 나온다
-- 회전 뒤 한 건
[root@ol7a ~]# logger -t labapp -p local4.info "LAB19-08 after rotate with HUP"; sleep 2
-- 이번에는 어디에
[root@ol7a ~]# ls -l /var/log/labapp.log*; grep -H "with HUP" /var/log/labapp.log*
-rw-------. 1 root root   60 Nov 16 18:07 /var/log/labapp.log
-rw-------. 1 root root 1371 Nov 16 18:07 /var/log/labapp.log.1
-rw-------. 1 root root 1759 Nov 16 18:07 /var/log/labapp.log.2
/var/log/labapp.log:Nov 16 18:07:07 ol7a labapp: LAB19-08 after rotate with HUP
-- 새 labapp.log 에 기록됐다. .2 는 [7] 의 .1 이 밀린 것
-- 열린 파일 다시 확인
[root@ol7a ~]# ls -l /proc/$(pidof rsyslogd)/fd | grep labapp
l-wx------. 1 root root 64 Nov 16 18:06 6 -> /var/log/labapp.log
l-wx------. 1 root root 64 Nov 16 18:06 7 -> /var/log/labapp-all.log
-- fd 6 이 labapp.log 를 가리킨다

 [9] 상태 파일과 compress
--------------------------------------------------------------------------------

-- logrotate 는 마지막으로 회전한 날짜를 상태 파일에 적는다. 주기(daily·weekly) 판단의 근거다.
-- 상태 파일의 labapp 줄
[root@ol7a ~]# grep labapp /var/lib/logrotate/logrotate.status
"/var/log/labapp.log" 2026-11-16-18:7:7
-- 마지막 회전 시각. weekly 면 7일 뒤

-- compress·delaycompress 를 더한다. 가장 최근 옛 파일(.1)은 다음 회전 때 압축된다.
-- 압축 두 줄 추가
[root@ol7a ~]# sed -i 's/^    sharedscripts/    compress\n    delaycompress\n    sharedscripts/' /etc/logrotate.d/labapp; grep -n compress /etc/logrotate.d/labapp
7:    compress
8:    delaycompress
-- 채우고 회전을 두 번
[root@ol7a ~]# for n in 1 2; do for i in $(seq 1 30); do logger -t labapp -p local4.info "LAB19-09 round $n line $i"; done; sleep 2; logrotate -f /etc/logrotate.d/labapp; done; ls -l /var/log/labapp.log*
-rw-------. 1 root root    0 Nov 16 18:07 /var/log/labapp.log
-rw-------. 1 root root 1611 Nov 16 18:07 /var/log/labapp.log.1
-rw-------. 1 root root 1759 Nov 16 18:07 /var/log/labapp.log.2
-rw-------. 1 root root  182 Nov 16 18:07 /var/log/labapp.log.2.gz
-rw-------. 1 root root  146 Nov 16 18:07 /var/log/labapp.log.3.gz
-- .1 은 평문(delaycompress), 그 뒤는 .gz
-- 평문 .2 는 압축 전 규칙의 잔재다. compress 뒤에는 .N.gz 만 민다

-- Logwatch 는 로그 파일을 읽어 서비스별 요약 보고서를 만드는 Perl 스크립트다. 기본 설치에는 없다.

[root@ol7a ~]# ls /etc/cron.daily/ | xargs; grep -Ev '^#|^$' /etc/cron.daily/0logwatch

 [변형] copytruncate — 신호 없이 회전
--------------------------------------------------------------------------------

-- HUP 을 받을 수 없는 프로그램(로그 파일을 계속 열어 두는 셸 스크립트 등)에 쓰는 방법.
-- 파일을 복사한 뒤 원본을 0 바이트로 자른다. inode 가 그대로라 쓰던 프로그램이 모른다.
-- 시험 규칙 파일
[root@ol7a ~]# printf '/var/log/labcopy.log {\n    copytruncate\n    rotate 2\n    missingok\n}\n' > /etc/logrotate.d/labcopy; cat /etc/logrotate.d/labcopy
/var/log/labcopy.log {
    copytruncate
    rotate 2
    missingok
}
-- 500줄 파일을 회전
[root@ol7a ~]# seq 1 500 > /var/log/labcopy.log; ls -li /var/log/labcopy.log; logrotate -f /etc/logrotate.d/labcopy; ls -li /var/log/labcopy.log*

67602768 -rw-r--r--. 1 root root 1892 Nov 16 18:07 /var/log/labcopy.log.1
-- 원본 inode 가 그대로, 크기만 0. 사본 .1 은 새 inode
-- 복사와 자르기 사이에 쓰인 줄은 잃을 수 있다
