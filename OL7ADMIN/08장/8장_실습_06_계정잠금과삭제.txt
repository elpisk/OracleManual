 목표   : passwd -l, usermod -L, chage -E 0, nologin 셸이 각각 무엇을 막고 무엇을

--------------------------------------------------------------------------------

[root@ol7a ~]# getent shadow user1 | cut -c1-24 > /root/u1.sh0; passwd -l user1; echo "exit=$?"
Locking password for user user1.
passwd: Success
exit=0
-- 'Locking password for user user1.' 뒤 'passwd: Success'

-- 2번 필드 앞에 무엇이 붙었는지 본다.
[root@ol7a ~]# cat /root/u1.sh0; getent shadow user1 | cut -c1-26; passwd -S user1
user1:$6$TsPdI/BikNVFQF$
user1:!!$6$TsPdI/BikNVFQF$
user1 LK 2026-11-16 0 99999 7 -1 (Password locked.)
-- 전: user1:$6$TsPdI/... 후: user1:!!$6$TsPdI/...

-- passwd -S 는 PS 에서 LK 로 바뀌었다
[student@ol7a ~]$ printf 'Ol7#Pass2026\n' | su - user1 -c 'echo pw-ok'; echo "exit=$?"

-- 'su: Authentication failure',
-- exit=1. 맞는 암호인데 거부됐다
-- !! 가 붙은 해시는 어떤 입력과도 맞지 않는다

 [3] 잠근 계정이 SSH 키로는 들어와진다 (오류 ①)
--------------------------------------------------------------------------------

[user1@ol7a ~]$ ssh -o BatchMode=yes user1@localhost 'echo key-ok; id -un'; echo "exit=$?"
key-ok
user1
exit=0
-- 증상   : passwd -l(LK) 로 잠근 user1 이 ssh 키로 key-ok·user1 을 출력, exit=0
-- 원인   : passwd -l 은 shadow 의 해시만 바꾼다. 키 인증은 해시를 보지 않고,
--          UsePAM yes 의 sshd 는 PAM 의 auth 를 건너뛰고 account·session 만 부른다
-- 조치   : 계정 만료(chage -E 0 또는 usermod -e 1)를 함께 건다([5]·[변형])
--          authorized_keys 를 치우는 것도 같이 한다
-- 재확인 : chage -E 0 뒤 같은 ssh 가 'Your account has expired', exit=255
[root@ol7a ~]# grep 'sshd.*user1' /var/log/secure | tail -3 | cut -c17-
ol7a sshd[2394]: Accepted publickey for user1 from ::1 port 32872 ssh2: RSA SHA256:73Pvuh9FXtB+Knf10uK+/HLrmb4VETsyGCWteud1gSQ
ol7a sshd[2394]: pam_unix(sshd:session): session opened for user user1 by (uid=0)
ol7a sshd[2394]: pam_unix(sshd:session): session closed for user user1
-- secure 의 'Accepted publickey for user1 from ::1' 이 들어온 기록이다
-- 잠긴 계정인데도 session opened·closed 가 정상으로 남았다
-- 감사 시 잠근 계정의 Accepted 줄을 찾는 것이 이 구멍을 찾는 방법이다

-- 잠금이 암호 칸에만 있다는 것을 확인한 뒤 푼다.
[root@ol7a ~]# passwd -u user1; passwd -S user1
Unlocking password for user user1.
passwd: Success
user1 PS 2026-11-16 0 99999 7 -1 (Password set, SHA512 crypt.)
-- 'Unlocking password' 뒤 PS. 해시 앞의 !! 가 빠졌다

 [4] usermod -L 과 passwd -l 비교
--------------------------------------------------------------------------------

-- usermod -L 도 해시 앞에 ! 를 붙인다. passwd -l 과 붙는 개수가 다른지 본다.
[root@ol7a ~]# usermod -L user1; getent shadow user1 | cut -c1-26; passwd -S user1
user1:!$6$TsPdI/BikNVFQF$u
user1 LK 2026-11-16 0 99999 7 -1 (Password locked.)
-- usermod -L 은 ! 한 글자. passwd -l 은 !! 두 글자였다
-- passwd -S 는 둘 다 LK 로 보인다. 구분하려면 shadow 를 직접 본다
[root@ol7a ~]# usermod -U user1; getent shadow user1 | cut -c1-26
user1:$6$TsPdI/BikNVFQF$uy
-- ! 가 빠졌다. 해시(TsPdI/...)는 처음과 같다

-- 암호가 없는 계정(!! 만 있는 새 계정)을 usermod -U 로 풀면 어떻게 되는가.
[root@ol7a ~]# useradd user8; getent shadow user8 | cut -d: -f1,2; usermod -U user8; echo "exit=$?"; passwd -u user8; echo "exit=$?"; userdel -r user8
user8:!!
usermod: unlocking the user's password would result in a passwordless account.
You should set a password with usermod -p to unlock this user's password.
exit=1
passwd: Warning: unlocked password would be empty.
Unlocking password for user user8.
passwd: Unsafe operation (use -f to force)
exit=254
-- 암호가 없는 계정(!!)은 둘 다 풀기를 거부했다
--   usermod -U  'unlocking ... would result in a passwordless account', exit=1
--   passwd -u   'unlocked password would be empty', 'Unsafe operation', exit=254
-- 풀면 암호 없이 들어오는 계정이 되기 때문이다. 새 계정은 암호를 줘서 연다

 [5] chage -E 0 — 계정 만료는 키 로그인도 막는다
--------------------------------------------------------------------------------

-- -E 0 은 만료일을 1970-01-01 로 둔다. 이미 지난 날짜라 즉시 만료된다.
[root@ol7a ~]# chage -E 0 user1; chage -l user1 | sed -n 4p
Account expires                                         : Jan 01, 1970
-- 8번 필드 0 = 1970-01-01. 오늘보다 이전이라 즉시 만료
[user1@ol7a ~]$ ssh -o BatchMode=yes user1@localhost 'echo key-ok'; echo "exit=$?"
Your account has expired; please contact your system administrator
Authentication failed.
exit=255
-- 'Your account has expired; please contact your system administrator',
-- 'Authentication failed.',
-- exit=255(ssh 의 연결 실패 코드)
-- 키 인증은 통과했지만 account 단계의 pam_unix 가 막았다
[student@ol7a ~]$ printf 'Ol7#Pass2026\n' | su - user1 -c 'echo pw-ok'; echo "exit=$?"
Password: Your account has expired; please contact your system administrator
su: User account has expired
exit=1
-- su 도 같은 문구 뒤 'su: User account has expired'
[root@ol7a ~]# grep 'user1' /var/log/secure | tail -3 | cut -c17-
ol7a sshd[2587]: fatal: Access denied for user user1 by PAM account configuration [preauth]
ol7a su: pam_unix(su-l:session): session closed for user user1
ol7a su: pam_unix(su-l:account): account user1 has expired (account expired)
-- sshd 줄: 'Access denied for user user1 by PAM account configuration [preauth]'
-- su 줄: pam_unix(su-l:account) — 막은 곳이 account 단계라는 것이 이름에 보인다

-- -E -1 은 만료일을 지운다(필드 8 을 빈 칸으로).
[root@ol7a ~]# chage -E -1 user1; chage -l user1 | sed -n 4p
Account expires                                         : never
-- 8번 필드가 비었다

 [7] pam_faillock 켜기 — authconfig
--------------------------------------------------------------------------------

-- 7.9 의 PAM 파일은 authconfig 가 만든다(파일 머리의 경고 주석). 손으로 고치면
-- 다음 authconfig --update 때 사라진다.
[root@ol7a ~]# head -3 /etc/pam.d/system-auth-ac
#%PAM-1.0
# This file is auto-generated.
# User changes will be destroyed the next time authconfig is run.
-- 둘째·셋째 줄의 경고: authconfig 가 만든 파일이며 손으로 고친 것은 다음 실행 때 사라진다

-- --enablefaillock 이 faillock 줄을 넣는다. deny=3 세 번 실패면 잠금, unlock_time=120 초 뒤 자동 해제
[root@ol7a ~]# cp -a /etc/pam.d/system-auth-ac /root/sa.before; authconfig --enablefaillock --faillockargs="deny=3 unlock_time=120" --update; echo "exit=$?"
exit=0
-- authconfig 는 성공하면 아무것도 출력하지 않는다
[root@ol7a ~]# diff /root/sa.before /etc/pam.d/system-auth-ac
5a6
> auth        required      pam_faillock.so preauth silent deny=3 unlock_time=120
8a10
> auth        required      pam_faillock.so authfail deny=3 unlock_time=120
10a13
> account     required      pam_faillock.so
-- 세 줄이 들어갔다
--   auth preauth silent  인증 전에 잠김 여부 확인. silent 는 메시지 없음
--   auth authfail        pam_unix 가 실패하면 실패 횟수 +1
--   account required     성공 시 기록 정리
-- 인자는 deny=3 unlock_time=120 이 그대로 붙었다

 [8] 세 번 틀리고 네 번째에 맞는 암호
--------------------------------------------------------------------------------

-- 틀린 암호로 세 번 시도한다.
[student@ol7a ~]$ for i in 1 2 3; do printf 'wrong%s\n' $i | su - user1 -c id; done
Password: su: Authentication failure
Password: su: Authentication failure
Password: su: Authentication failure
-- 세 번 모두 같은 문구. 틀린 암호와 잠긴 계정이 화면에서는 구분되지 않는다

-- faillock 은 실패 기록을 사용자별 파일에 둔다. --user 로 한 사람 것만 본다.
[root@ol7a ~]# faillock --user user1; ls -l /var/run/faillock
user1:
When                Type  Source                                           Valid
2026-11-16 10:50:53 SVC   su-l                                                 V
2026-11-16 10:50:56 SVC   su-l                                                 V
2026-11-16 10:51:00 SVC   su-l                                                 V
total 4
-rw-------. 1 user1 student 192 Nov 16 10:51 user1
-- faillock 의 열
--   When    실패 시각. 10:50:53·56, 10:51:00 (pam_faildelay 로 약 3~4초 간격)
--   Type    SVC 는 서비스(PAM 서비스 이름)로 기록된 실패
--   Source  su-l. su - 가 쓰는 PAM 서비스 이름
--   Valid   V = 아직 unlock_time 안이라 셈에 들어가는 기록
-- /var/run/faillock/user1 파일.
-- 그룹이 student 인 것은 su 를 친 사용자의 그룹이다

-- 이제 맞는 암호를 준다.
[student@ol7a ~]$ printf 'Ol7#Pass2026\n' | su - user1 -c 'echo pw-ok'; echo "exit=$?"
Password: su: Authentication failure
exit=1
-- 맞는 암호인데 'su: Authentication failure'.
-- 이미 세 번 실패해 잠겼다
[root@ol7a ~]# grep -E 'faillock|user1' /var/log/secure | tail -4 | cut -c17-
ol7a su: pam_unix(su-l:auth): authentication failure; logname= uid=1000 euid=0 tty= ruser=student rhost=  user=user1
ol7a unix_chkpwd[2873]: password check failed for user (user1)
ol7a su: pam_unix(su-l:auth): authentication failure; logname= uid=1000 euid=0 tty= ruser=student rhost=  user=user1
ol7a su: pam_faillock(su-l:auth): Consecutive login failures for user user1 account temporarily locked
-- unix_chkpwd 의 'password check failed' 가 틀린 암호의 기록
-- 마지막 줄 pam_faillock 의 'account temporarily locked' 이 잠금의 기록이다
-- 화면에는 아무것도 나오지 않았다(preauth 의 silent)
-- 증상   : 맞는 암호 Ol7#Pass2026 에도 'su: Authentication failure', exit=1
-- 원인   : 앞의 세 번 실패로 deny=3 에 닿았다. secure 에 'Consecutive login
--          failures for user user1 account temporarily locked'
-- 조치   : faillock --user user1 --reset (또는 unlock_time 120 초 대기)
-- 재확인 : faillock --user user1 의 목록이 비고 su 가 pw-ok

-- 키 로그인은 어떤가. sshd 는 password-auth 를 쓰고, faillock 은 auth 단계에 걸려 있다.
[user1@ol7a ~]$ ssh -o BatchMode=yes user1@localhost 'echo key-ok'; echo "exit=$?"
key-ok
exit=0
-- 잠긴 상태에서도 키 로그인은 key-ok. faillock 은 auth 단계에만 있다

When                Type  Source                                           Valid

[student@ol7a ~]$ printf 'Ol7#Pass2026\n' | su - user1 -c 'echo pw-ok'; echo "exit=$?"
Password: pw-ok
exit=0

--------------------------------------------------------------------------------
 [10] userdel -r — 쓰는 중인 계정
--------------------------------------------------------------------------------

-- user1 의 프로세스를 하나 띄워 둔 채로 지워 본다.
[root@ol7a ~]# nohup su - user1 -c 'sleep 600' >/dev/null 2>&1 </dev/null &
-- 출력 없이 프롬프트가 돌아왔다. sleep 이 뒤에서 돈다
[root@ol7a ~]# sleep 1; pgrep -u user1 -l; userdel -r user1; echo "exit=$?"
3094 sleep
userdel: user user1 is currently used by process 3094
exit=8
-- 'userdel: user user1 is currently used by process 3094', exit=8
-- 프로세스가 남아 있으면 지우지 않는다. 지웠다면 3094 는 주인 없는 프로세스가 된다

-- 프로세스를 정리하고 다시 지운다. 지우기 전에 홈 밖에 있는 user1 의 파일을 찾아 둔다.
[root@ol7a ~]# pkill -u user1; sleep 1; find / -xdev -user user1 2>/dev/null | grep -v '^/home/user1'
/var/tmp/u1_work
/var/tmp/u1_work/data
/var/spool/mail/user1
/tmp/u1_report.txt
-- 홈 밖에 남은 user1 의 파일 넷. 메일 스풀은 userdel -r 이 지운다
-- 나머지 셋(/var/tmp 둘, /tmp 하나)은 -r 이 모른다
[root@ol7a ~]# userdel -r user1; echo "exit=$?"; ls -ld /home/user1 /var/spool/mail/user1
exit=0
ls: cannot access /home/user1: No such file or directory
ls: cannot access /var/spool/mail/user1: No such file or directory
-- 이번에는 exit=0. 홈과 메일 스풀은 지워졌다

--------------------------------------------------------------------------------
 [11] 남은 파일과 UID 재사용 (오류 ②)
--------------------------------------------------------------------------------

-- 주인 없는 파일이 생겼다. 소유자 칸에 이름 대신 숫자가 보인다.
[root@ol7a ~]# find / -xdev -nouser 2>/dev/null; ls -l /tmp/u1_report.txt
/var/tmp/u1_work
/var/tmp/u1_work/data
/tmp/u1_report.txt
-rw-rw-r--. 1 1001 1001 7 Nov 16 10:50 /tmp/u1_report.txt
-- 주인 없는 파일 셋. ls -l 이 이름 대신 1001 을 보인다(-n 없이도)

-- 이 상태에서 새 사용자를 만든다. 비어 있는 다음 UID 는 몇 번인가.
[root@ol7a ~]# useradd user9; id user9
uid=1001(user9) gid=1001(user9) groups=1001(user9)
-- user9 가 1001 을 받았다. 지운 user1 이 가장 큰 UID 였기 때문이다
[root@ol7a ~]# ls -l /tmp/u1_report.txt; ls -la /var/tmp/u1_work; find / -xdev -nouser 2>/dev/null | wc -l
-rw-rw-r--. 1 user9 user9 7 Nov 16 10:50 /tmp/u1_report.txt
total 8
drwxrwxr-x. 2 user9 user9   18 Nov 16 10:50 .
drwxrwxrwt. 6 root  root  4096 Nov 16 10:51 ..
-rw-rw-r--. 1 user9 user9    2 Nov 16 10:50 data
0

-- 증상   : 새로 만든 user9 가 /tmp/u1_report.txt, /var/tmp/u1_work 의 소유자로 보임.
--          find -nouser 결과가 3 에서 0 으로
-- 원인   : 파일에는 UID 1001 만 적혀 있다. userdel -r 이 홈 밖의 파일을 남겼고,
--          가장 큰 UID 를 지웠으니 같은 번호가 바로 다음 계정에 갔다
-- 조치   : 삭제 전에 find / -xdev -user 로 찾아 정리한다. 이미 늦었으면 UID 로
--          찾아 지운다(아래 find -uid 1001)
-- 재확인 : ls /tmp /var/tmp | grep u1_ 가 결과 없음
-- 남은 파일을 정리한다. 이번에는 UID 로 찾는다(이름은 이미 user9 를 가리킨다).
-- -maxdepth 1 은 /tmp, /var/tmp 바로 아래만 본다. 디렉터리는 rm -rf 가 통째로 지운다.
[root@ol7a ~]# find /tmp /var/tmp -maxdepth 1 -uid 1001 -print -exec rm -rf {} +; ls /tmp /var/tmp | grep u1_ || echo "u1_ 파일 없음"
-- -print 로 지운 두 항목을 먼저 찍었다.
-- u1_work 안의 data 는 디렉터리와 함께 지워졌다
/tmp/u1_report.txt
/var/tmp/u1_work
u1_ 파일 없음

--------------------------------------------------------------------------------
 [변형] usermod -e 와 chage -E 비교
--------------------------------------------------------------------------------

-- usermod 의 -e(expiredate)도 shadow 8번 필드를 고친다. 날짜 형식은 같다.
[root@ol7a ~]# usermod -e 1 user9; getent shadow user9 | cut -d: -f1,8; chage -l user9 | sed -n 4p
user9:1
Account expires                                         : Jan 02, 1970
-- usermod -e 1 은 8번 필드에 1(1970-01-02)을 넣는다.
-- 날짜 대신 일수를 줘도 된다
[root@ol7a ~]# chage -E 0 user9; getent shadow user9 | cut -d: -f1,8; usermod -e '' user9; getent shadow user9 | cut -d: -f1,8
user9:0
user9:

-- chage -E 0 은 0, usermod -e '' 는 빈 칸(만료 없음)
-- 두 명령이 같은 칸을 고친다. 빈 값을 주는 방법만 다르다(chage 는 -1)
-- 잠금을 한 번에 단단히 하려면 암호 잠금과 계정 만료를 함께 건다.
[root@ol7a ~]# usermod -L -e 1 user9; passwd -S user9; chage -l user9 | sed -n 4p
user9 LK 2026-11-16 0 99999 7 -1 (Password locked.)
Account expires                                         : Jan 02, 1970
-- 한 줄로 두 가지가 걸렸다. LK(암호 잠금)와 Jan 02, 1970(계정 만료)

 [정리] 관찰 요약
--------------------------------------------------------------------------------
--  방법               shadow/passwd 의 변화     su(암호)       ssh 키        해제
--  passwd -l          해시 앞 !!                거부           들어와짐      passwd -u
--  usermod -L         해시 앞 !                 (LK)           -             usermod -U
--  chage -E 0         8번 = 0 (Jan 01, 1970)    expired        expired       chage -E -1
--  nologin 셸         7번 /sbin/nologin         -              not available 셸 되돌림
--  faillock deny=3    /var/run/faillock/user1   3회 뒤 거부    들어와짐      --reset
--  usermod -L -e 1    !, 8번 = 1                -              -             -
--  항목                 시작 전        변경 후                         되돌린 뒤
--  PAM -ac 두 파일 md5  af8a…/ee5e…    faillock 3줄 추가               af8a…/ee5e…
--  find -nouser         0              3 → user9 생성 뒤 0             0
--  userdel(프로세스)     -             used by process 3094, exit=8    -
-- 확인한 사실
--   암호 잠금은 auth 만 막고, 계정 만료는 account 에서 모든 경로를 막는다
--   faillock 은 auth 단계에서 세므로 키 로그인에는 효과가 없다
--   userdel -r 뒤 남은 파일은 같은 UID 를 받은 새 계정의 것이 된다
