================================================================================
 제12장 실습 05 — 오류 재현: RHCK 에서의 Btrfs
 대상   : ol7a (추가 디스크 sdb 사용, 재부팅 두 번)
 목표   : UEK 에서 quota group 으로 서브볼륨 한도를 걸고 초과 쓰기를 본다. 같은 Btrfs 를
          RHCK 로 부팅해 마운트해 보고 모듈·커널 메시지로 지원 범위를 확인한다.
          끝에 UEK 로 돌아온 것을 보인다
 배경   : Oracle Linux 7 에서 Btrfs 는 UEK R4 이후 UEK 에서만 정식 지원된다. RHCK 의
          Btrfs 는 7.4 부터 deprecated 다(RN79 4장). 같은 디스크라도 어느 커널로
          부팅했느냐에 따라 지원 여부가 갈린다. grub2-reboot 는 다음 한 번만 다른 항목으로
          부팅하게 하므로 기본 커널을 바꾸지 않고 시험할 수 있다(본문 12.1·12.8절)
 소요   : 약 25분(재부팅 포함)
 선행   : storage-clean 스냅샷, UEK R6 로 부팅, RHCK 3.10.0-1160.119.1.0.5 설치돼 있음
 계정   : root
 주의   : 기본 부팅 항목(saved_entry)을 바꾸지 않는다. grub2-reboot 만 쓴다
 되돌림 : UEK 로 부팅된 상태, Btrfs 해제와 wipefs.
          확인 uname -r, grub2-editenv list, lsblk -f /dev/sdb

 [1] UEK 에서 Btrfs 와 quota 준비
--------------------------------------------------------------------------------

[root@ol7a ~]# mkfs.btrfs -L qg /dev/sdb | grep -E 'Label|UUID|Incompat'; mkdir -p /qg; mount /dev/sdb /qg
Label:              qg
UUID:               5ef469cc-29a9-4fab-ae72-d12b1f44f4df
Incompat features:  extref, skinny-metadata
-- grep 으로 세 줄만. RHCK 3.10 이 이 두 기능을 아는지가 [7] 마운트의 관건이다
-- 레이블 qg 는 재부팅 뒤 장치를 찾을 때 쓴다.
-- UUID 5ef469cc-... 는 이 파일 시스템 하나의 값
-- 문서 권고대로 서브볼륨을 만들기 전에 quota 를 켠다.
[root@ol7a ~]# btrfs quota enable /qg
[root@ol7a ~]# btrfs subvolume create /qg/proj1; btrfs subvolume create /qg/proj2
Create subvolume '/qg/proj1'
Create subvolume '/qg/proj2'
-- quota enable 은 출력이 없다. 서브볼륨 둘을 그 뒤에 만들었다
-- 문서는 서브볼륨을 만들기 전에 quota 를 켜라고 한다.
-- 나중에 켜면 기존 데이터를 다시 세는
--   rescan 이 돌아 그동안 값이 맞지 않는다(btrfs quota rescan -s 로 진행 확인)
-- proj1 은 ID 257, proj2 는 ID 258
[root@ol7a ~]# btrfs qgroup show /qg
qgroupid         rfer         excl 
--------         ----         ---- 
0/5          16.00KiB     16.00KiB 
0/257        16.00KiB     16.00KiB 
0/258        16.00KiB     16.00KiB 
-- qgroup show 의 열
--   qgroupid : 0/ID. 0 은 레벨(서브볼륨 자신), ID 는 서브볼륨 ID. 0/5 는 최상위
--   rfer     : referenced. 이 서브볼륨이 가리키는 전체 양
--   excl     : exclusive. 다른 서브볼륨·스냅샷과 공유하지 않는 양
-- 빈 서브볼륨이라 셋 다 16.00KiB(메타데이터 노드 하나)
-- quota 를 켠 뒤 만든 서브볼륨은 qgroup 이 자동으로 생겼다

--------------------------------------------------------------------------------
 [2] qgroup 한도 걸기
--------------------------------------------------------------------------------

-- 한도는 서브볼륨 경로로 지정한다. 해당 서브볼륨의 0/ID qgroup 에 걸린다.
[root@ol7a ~]# btrfs qgroup limit 100m /qg/proj1
-- -r 은 참조(referenced) 한도, -e 는 배타(exclusive) 한도 열을 보탠다.
[root@ol7a ~]# btrfs qgroup show -re /qg
qgroupid         rfer         excl     max_rfer     max_excl 
--------         ----         ----     --------     -------- 
0/5          16.00KiB     16.00KiB         none         none 
0/257        16.00KiB     16.00KiB    100.00MiB         none 
0/258        16.00KiB     16.00KiB         none         none 
-- -r·-e 로 max_rfer·max_excl 두 열이 붙었다
-- proj1(0/257)의 max_rfer 가 100.00MiB.
-- limit 은 기본으로 referenced 에 건다(-e 를 주면 excl)
-- proj2·최상위는 none
-- 한도는 서브볼륨 경로로 지정했지만 실제로는 qgroup 0/257 에 걸린다
--   btrfs qgroup limit 100m 0/257 /qg 처럼 qgroup ID 로 지정해도 같다

--------------------------------------------------------------------------------
 [3] 한도를 넘겨 쓰기
--------------------------------------------------------------------------------

[root@ol7a ~]# dd if=/dev/zero of=/qg/proj1/fill bs=1M count=150; echo "rc=$?"
dd: error writing ‘/qg/proj1/fill’: Disk quota exceeded
100+0 records in
99+0 records out
104792064 bytes (105 MB) copied, 0.310302 s, 338 MB/s
rc=1
-- 증상   : 150MiB 를 쓰려던 dd 가 'dd: error writing ‘/qg/proj1/fill’: Disk quota exceeded', rc=1
-- 원인   : proj1 의 qgroup 0/257 에 max_rfer 100.00MiB 를 걸었다. 99MiB 째에서 한도에 닿았다
-- 조치   : 필요 없는 파일을 지우거나 btrfs qgroup limit 300m /qg/proj1 로 한도를 늘린다([4])
-- 재확인 : 같은 dd 가 rc=0,
-- qgroup show 의 max_rfer 300.00MiB
-- dd 의 records in 100+0,
-- out 99+0 : 100번째 1MiB 블록을 쓰다 실패했다
-- 104792064 바이트(약 99.94MiB)가 기록된 채 남았다
-- 일반 Ext·XFS quota 의 'Disk quota exceeded' 와 같은 errno(EDQUOT)라 응용 프로그램은 구별하지 못한다
[root@ol7a ~]# ls -l /qg/proj1/fill; sync; btrfs qgroup show -re /qg
-rw-r--r--. 1 root root 104792064 Nov 16 12:04 /qg/proj1/fill
qgroupid         rfer         excl     max_rfer     max_excl 
--------         ----         ----     --------     -------- 
0/5          16.00KiB     16.00KiB         none         none 
0/257        99.95MiB     99.95MiB    100.00MiB         none 
0/258        16.00KiB     16.00KiB         none         none 
-- fill 파일은 지워지지 않았다. 크기 104792064.
-- 응용 프로그램은 잘린 파일을 처리해야 한다
-- rfer·excl 99.95MiB 로 한도 100MiB 바로 아래.
-- 일반 quota 처럼 유예(grace) 기간이 없다
-- sync 뒤에 qgroup show 를 봤다.
-- qgroup 값은 트랜잭션 커밋 때 갱신된다
--   sync 없이 바로 보면 직전 값이 보일 수 있다
-- 한도가 없는 proj2 에는 같은 크기가 그대로 들어간다.
[root@ol7a ~]# dd if=/dev/zero of=/qg/proj2/fill bs=1M count=150 status=none; echo "rc=$?"; sync; btrfs qgroup show -re /qg | tail -2
rc=0
0/257        99.95MiB     99.95MiB    100.00MiB         none 
0/258       150.02MiB    150.02MiB         none         none 
-- 한도가 없는 proj2 에는 150MiB 가 그대로 들어갔다(rc=0)
-- tail -2 로 두 서브볼륨 줄만 보였다. 머리 줄과 0/5 는 뺐다
-- 서브볼륨마다 한도가 따로다. 파일 시스템 전체에는 여유가 있다
-- status=none 으로 dd 의 통계 줄을 껐다. 실패하면 오류 줄은 그대로 나온다

-- 뒤의 grub2-editenv 에 next_entry=2 가 생겼다

-- 이어서 systemctl reboot 으로 재부팅했다

 [7] RHCK 에서 같은 Btrfs 마운트

[root@ol7a ~]# cat /proc/sys/kernel/tainted

-- 마운트 뒤에도 tainted 0
