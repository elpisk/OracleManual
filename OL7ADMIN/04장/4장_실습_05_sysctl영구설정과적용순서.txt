================================================================================
 제4장 실습 05 — sysctl 영구 설정과 적용 순서
 대상   : ol7a
 목표   : sysctl -w 로 런타임 값을 바꾸고, /etc/sysctl.d 에 파일을 두어 영구화한다.
          sysctl --system 이 파일을 읽는 순서와, 같은 키가 여러 파일에 있을 때 이기는
          쪽을 확인한다. 부팅 때 값을 넣는 systemd-sysctl 이 sysctl --system 과 같은
          결과를 내는지 재부팅과 디버그 실행으로 확인한다
 배경   : 커널 파라미터는 /proc/sys 아래 파일이고 sysctl 은 그 파일을 읽고 쓰는 도구다.
          재부팅 뒤에도 남기려면 설정 파일에 적는다. OL 7 은 /usr/lib/sysctl.d(패키지
          기본값), /run/sysctl.d, /etc/sysctl.d(관리자) 세 곳과 예전 방식의
          /etc/sysctl.conf 를 함께 쓴다. /etc/sysctl.d/99-sysctl.conf 는 /etc/sysctl.conf
          를 가리키는 링크다 (본문 4.6 절)
 소요   : 약 20분 (재부팅 1회 포함)
 선행   : ol7a 는 ol7-base
 계정   : root
 주의   : kernel.panic, kernel.sysrq 를 바꾼다. 끝에서 파일을 지우고 sysctl --system 으로 되돌린다
 되돌림 : kernel.panic = 0, kernel.sysrq = 16, /etc/sysctl.d 에 99-sysctl.conf 링크만 남음
================================================================================

--------------------------------------------------------------------------------
 [0] 시작 전 상태 기록
--------------------------------------------------------------------------------

-- 이 실습에서 다룰 키 세 개의 현재 값. sysctl 에 키를 여러 개 줄 수 있다.
[root@ol7a ~]# sysctl kernel.panic kernel.sysrq vm.swappiness
kernel.panic = 0
kernel.sysrq = 16
vm.swappiness = 30

-- 출력 형식은 "키 = 값". sysctl 은 키를 /proc/sys 경로로 바꿔 파일을 읽은
--   것뿐이다.
--   kernel.panic = 0   : 패닉이 나면 자동 재부팅하지 않고 멈춘 채 둔다(커널
--     기본값)
--   kernel.sysrq = 16  : Magic SysRq 기능 가운데 sync(16)만
--     허용하는 비트 마스크
--   vm.swappiness = 30 : 커널 기본값 60 이 아니다. 누가 30 을
--     넣었는지는 [9] 에서 찾는다
-- 이 세 값이 [10] 되돌림 뒤에 그대로 돌아와야 한다.

-- 설정 파일이 놓이는 세 곳과 /etc/sysctl.conf. /run/sysctl.d 는 없을
--   수 있다.
[root@ol7a ~]# ls -l /etc/sysctl.d /usr/lib/sysctl.d; ls -ld /run/sysctl.d
/etc/sysctl.d:
total 0
lrwxrwxrwx. 1 root root 14 Nov 11 20:15 99-sysctl.conf -> ../sysctl.conf

-- /etc/sysctl.d 에는 링크 하나뿐이다. 99-sysctl.conf ->
--   ../sysctl.conf
--   이름이 99 로 시작해 다른 파일 대부분보다 뒤에 읽히게 만들어 둔 장치다.
-- /usr/lib/sysctl.d 의 네 파일은 패키지가 깐 기본값이다. 날짜 열을 보면
--   2020~2024 년
--   패키지 빌드 날짜가 찍혀 있다. 관리자가 고치는 파일이 아니다.
--   00-system.conf            : 브리지 넷필터 관련 기본값
--   01-unprivileged-bpf.conf  : 일반 사용자의 BPF 사용 금지
--   10-default-yama-scope.conf: ptrace 범위
--   50-default.conf           : systemd 가 깐 일반 기본값 ([1] 에서 내용을 본다)
-- /run/sysctl.d 는 없다. 실행 중에 프로그램이 임시 설정을 둘 때만 생기는 곳이다.

/usr/lib/sysctl.d:
total 16
-rw-r--r--. 1 root root  293 Sep 12  2024 00-system.conf
-rw-r--r--. 1 root root  280 Oct 10  2024 01-unprivileged-bpf.conf
-rw-r--r--. 1 root root 1810 May 24  2020 10-default-yama-scope.conf
-rw-r--r--. 1 root root 1285 Oct 10  2024 50-default.conf
ls: cannot access /run/sysctl.d: No such file or directory
[root@ol7a ~]# grep -v '^#' /etc/sysctl.conf | grep -c .
0

-- 주석이 아닌 줄이 0 개. 기준 상태의 /etc/sysctl.conf 는 주석뿐인 빈 파일이다.

--------------------------------------------------------------------------------
 [1] sysctl --system 이 읽는 순서
--------------------------------------------------------------------------------

-- --system 은 모든 설정 파일을 읽어 적용하고, 어떤 파일을 읽는지 "*
--   Applying" 줄로 보인다.
-- 실제 값 줄은 빼고 순서만 본다.
[root@ol7a ~]# sysctl --system | grep Applying
* Applying /usr/lib/sysctl.d/00-system.conf ...
* Applying /usr/lib/sysctl.d/01-unprivileged-bpf.conf ...
* Applying /usr/lib/sysctl.d/10-default-yama-scope.conf ...
* Applying /usr/lib/sysctl.d/50-default.conf ...
* Applying /etc/sysctl.d/99-sysctl.conf ...
* Applying /etc/sysctl.conf ...

-- 읽는 순서를 정리하면 다음과 같다.
--   1~4  /usr/lib/sysctl.d 의 네 파일 (파일 이름순)
--   5    /etc/sysctl.d/99-sysctl.conf  — 링크라서 실제로는 /etc/sysctl.conf 의 내용
--   6    /etc/sysctl.conf              — 같은 파일을 마지막에 한 번 더
-- 세 디렉터리를 합쳐 이름순으로 정렬한 뒤, 맨 끝에 /etc/sysctl.conf 를 따로
--   붙인다.
-- 6번 줄이 이 실습의 핵심 관찰 대상이다. [5]·[6] 에서 이 줄 때문에 결과가 정해진다.
-- [OL8/9] sysctl.d 세 곳을 이름순으로 읽고 /etc/sysctl.conf 를 끝에
--   다시 읽는 구조는 같다(문서 기준).

-- 50-default.conf 의 내용. 패키지(systemd)가 깐 기본값이다.
[root@ol7a ~]# grep -v '^#' /usr/lib/sysctl.d/50-default.conf | grep -v '^$'
kernel.sysrq = 16
kernel.core_uses_pid = 1
kernel.kptr_restrict = 1
net.ipv4.conf.default.rp_filter = 1
net.ipv4.conf.all.rp_filter = 1
net.ipv4.conf.default.accept_source_route = 0
net.ipv4.conf.all.accept_source_route = 0
net.ipv4.conf.default.promote_secondaries = 1
net.ipv4.conf.all.promote_secondaries = 1
fs.protected_hardlinks = 1
fs.protected_symlinks = 1

-- kernel.sysrq = 16 이 여기서 온다. [0] 의 값과 같다.
-- rp_filter = 1 은 들어온 패킷의 출발지 주소가 그 NIC 로 돌아갈 수 있는
--   주소인지 검사한다.
--   RAC 인터커넥트처럼 NIC 가 여럿인 서버에서 2(느슨한 검사)로 바꾸는 일이
--     있다(17장).
-- protected_hardlinks·symlinks = 1 은 /tmp 같은 공유 디렉터리에서
--   링크를 이용한 공격을 막는다.

--------------------------------------------------------------------------------
 [2] 런타임 변경 — sysctl -w
--------------------------------------------------------------------------------

-- -w 는 쓰기. 키=값 사이에 공백이 없어야 한다.
[root@ol7a ~]# sysctl -w kernel.panic=10
kernel.panic = 10

-- 성공하면 쓴 값을 "키 = 값" 으로 다시 찍는다. 이 줄이 실제 값이라는 보장은 없다(실습
--   06 [3]).

[root@ol7a ~]# cat /proc/sys/kernel/panic
10

-- sysctl -w 가 바꾼 것은 이 파일이다. 같은 값 10 이 보인다.
-- 이 값은 메모리에만 있다. 설정 파일 어디에도 적히지 않았다.

--------------------------------------------------------------------------------
 [3] 영구 설정 — /etc/sysctl.d 에 파일 두기
--------------------------------------------------------------------------------

-- 파일 이름은 숫자로 시작하고 .conf 로 끝나야 읽힌다. 80 은 기본값(50)보다 뒤라는
--   뜻이다.
[root@ol7a ~]# echo 'kernel.panic = 20' > /etc/sysctl.d/80-lab.conf; cat /etc/sysctl.d/80-lab.conf
kernel.panic = 20

-- 파일을 만든 것만으로는 바뀌지 않는다.
[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 10

-- 파일을 만들었는데 값은 여전히 10 이다. 파일을 두는 것과 적용하는 것은 별개다.
-- 부팅 때가 아니면 누군가 그 파일을 읽어 줘야 한다.

-- 파일 하나만 적용하려면 sysctl -p 파일. 모든 파일을 다시 읽으려면 sysctl
--   --system.
[root@ol7a ~]# sysctl -p /etc/sysctl.d/80-lab.conf
kernel.panic = 20

-- -p 뒤에 파일을 주면 그 파일 하나만 읽어 적용하고, 적용한 줄을 찍는다.

[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 20

-- 20 이 들어갔다. 즉시 반영되고 재시작할 서비스는 없다.

--------------------------------------------------------------------------------
 [4] 오류 ② — 인자 없는 sysctl -p 는 sysctl.d 를 읽지 않음
--------------------------------------------------------------------------------

-- 값을 런타임에 0 으로 돌린 뒤, 파일을 다시 적용하려고 습관대로 sysctl -p 만 친다.
[root@ol7a ~]# sysctl -w kernel.panic=0
kernel.panic = 0
[root@ol7a ~]# sysctl -p; echo "rc=$?"
rc=0

-- 아무 줄도 찍히지 않고 종료 코드 0. 오류처럼 보이지 않는다.

[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 0

-- 80-lab.conf 에 20 이 있는데도 0 그대로다.
-- 증상   : sysctl.d 에 파일을 두고 "sysctl -p" 를 실행했는데 값이 바뀌지
--          않음. 출력 없음, rc=0
-- 원인   : 인자 없는 -p 는 /etc/sysctl.conf 하나만 읽는다. 그 파일은
--          주석뿐이라 아무것도
--          적용하지 않았다. /etc/sysctl.d 의 파일은 읽는 대상이 아니다
-- 조치   : sysctl -p /etc/sysctl.d/80-lab.conf 처럼 파일을
--          지정하거나 sysctl --system
-- 재확인 : 아래 sysctl --system 출력에 "* Applying
--          /etc/sysctl.d/80-lab.conf" 와
--          kernel.panic = 20 이 찍히고, sysctl
--            kernel.panic 이 20

-- man 페이지의 -p 설명.
[root@ol7a ~]# man sysctl 2>/dev/null | grep -A3 -E '^ +-p\['
       -p[FILE], --load[=FILE]
              Load   in   sysctl   settings   from   the   file  specified  or
              /etc/sysctl.conf if none given.  Specifying - as filename  means
              reading  data  from standard input.  Using this option will mean

-- man 페이지가 직접 말한다. "/etc/sysctl.conf if none given".

[root@ol7a ~]# sysctl --system | grep -E 'Applying|kernel.panic'
* Applying /usr/lib/sysctl.d/00-system.conf ...
* Applying /usr/lib/sysctl.d/01-unprivileged-bpf.conf ...
* Applying /usr/lib/sysctl.d/10-default-yama-scope.conf ...
* Applying /usr/lib/sysctl.d/50-default.conf ...
* Applying /etc/sysctl.d/80-lab.conf ...
kernel.panic = 20
* Applying /etc/sysctl.d/99-sysctl.conf ...
* Applying /etc/sysctl.conf ...

-- [1] 의 순서에 80-lab.conf 가 끼어들었다. 50-default.conf 다음,
--   99-sysctl.conf 앞이다.
-- 파일 이름의 숫자가 읽는 순서를 정한다. 그 바로 아래에 적용된 값 kernel.panic =
--   20.

[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 20

-- 20. --system 은 sysctl.d 를 포함한 모든 곳을 읽는다.

--------------------------------------------------------------------------------
 [5] 같은 키가 sysctl.d 와 sysctl.conf 양쪽에 있으면 — sysctl --system
--------------------------------------------------------------------------------

-- /etc/sysctl.conf 에 30 을 넣는다. 80-lab.conf 에는 20 이 있다.
[root@ol7a ~]# cp -p /etc/sysctl.conf /root/sysctl.conf.orig; echo 'kernel.panic = 30' >> /etc/sysctl.conf
[root@ol7a ~]# grep -r kernel.panic /etc/sysctl.conf /etc/sysctl.d/
/etc/sysctl.conf:kernel.panic = 30
/etc/sysctl.d/80-lab.conf:kernel.panic = 20

-- grep -r 이 /etc/sysctl.d/99-sysctl.conf 는 찾지 않았다.
--   grep -r 은 디렉터리 안의 심볼릭
--   링크를 따라가지 않기 때문이다. 링크까지 보려면 grep -R(대문자)을 쓴다.
-- 지금 kernel.panic 이 적힌 곳은 두 파일, 값은 20 과 30.

[root@ol7a ~]# sysctl --system | grep -E 'Applying|kernel.panic'
* Applying /usr/lib/sysctl.d/00-system.conf ...
* Applying /usr/lib/sysctl.d/01-unprivileged-bpf.conf ...
* Applying /usr/lib/sysctl.d/10-default-yama-scope.conf ...
* Applying /usr/lib/sysctl.d/50-default.conf ...
* Applying /etc/sysctl.d/80-lab.conf ...
kernel.panic = 20
* Applying /etc/sysctl.d/99-sysctl.conf ...
kernel.panic = 30
* Applying /etc/sysctl.conf ...
kernel.panic = 30

-- 같은 키가 세 번 적용됐다. 80-lab.conf 에서 20,
--   99-sysctl.conf(링크)에서 30, sysctl.conf 에서 30.
-- 마지막에 쓴 값이 남는다. 30.

[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 30

-- 30. /etc/sysctl.conf 의 값이 sysctl.d 의 값을 이겼다. 여기까지는
--   예상대로다.

--------------------------------------------------------------------------------
 [6] 오류 ① — 99-sysctl.conf 보다 뒤에 오는 파일
--------------------------------------------------------------------------------

-- 튜닝 값을 모아 둔다며 99-tuning.conf 를 만든 상황이다. 이름순으로
--   99-sysctl.conf 뒤에 온다.
[root@ol7a ~]# echo 'kernel.panic = 40' > /etc/sysctl.d/99-tuning.conf; ls /etc/sysctl.d
80-lab.conf
99-sysctl.conf
99-tuning.conf

-- 관리자가 "99 보다 뒤에 오면 이긴다" 고 생각하고 99-tuning.conf 를 만들었다고
--   하자.
-- 이름순으로 99-sysctl.conf < 99-tuning.conf 다(s 가 t 보다 앞).

[root@ol7a ~]# sysctl --system | grep -E 'Applying|kernel.panic'
* Applying /usr/lib/sysctl.d/00-system.conf ...
* Applying /usr/lib/sysctl.d/01-unprivileged-bpf.conf ...
* Applying /usr/lib/sysctl.d/10-default-yama-scope.conf ...
* Applying /usr/lib/sysctl.d/50-default.conf ...
* Applying /etc/sysctl.d/80-lab.conf ...
kernel.panic = 20
* Applying /etc/sysctl.d/99-sysctl.conf ...
kernel.panic = 30
* Applying /etc/sysctl.d/99-tuning.conf ...
kernel.panic = 40
* Applying /etc/sysctl.conf ...
kernel.panic = 30

-- 99-tuning.conf 는 기대한 자리에서 40 을 넣었다. 그런데 그 뒤에
--   /etc/sysctl.conf 가 한 번
--   더 읽혀 30 으로 덮었다. 이 마지막 줄 때문에 sysctl.d 의 어떤 파일도
--     sysctl.conf 를 이길 수 없다.

[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 30

-- 증상   : /etc/sysctl.d/99-tuning.conf 에 kernel.panic =
--          40 을 넣고 sysctl --system 을
--          실행했는데 "kernel.panic = 30" 이 남음
-- 원인   : OL 7 의 sysctl --system 은 sysctl.d 를 다 읽은 뒤 "*
--          Applying /etc/sysctl.conf"
--          를 마지막에 한 번 더 한다. 같은 키가 /etc/sysctl.conf 에
--            30 으로 남아 있었다.
--          파일 이름을 아무리 뒤로 미뤄도 소용없다
-- 조치   : 키를 한 곳에만 둔다. 이 실습에서는 [10] 에서 sysctl.conf 를
--          원본(주석뿐)으로 돌린다
-- 재확인 : grep -R kernel.panic /etc/sysctl.conf
--          /etc/sysctl.d 가 한 줄만 보이고,
--          sysctl --system 출력에서 그 키가 마지막으로 찍힌 값이 기대값

-- 부팅 때는 sysctl --system 이 아니라 systemd-sysctl.service 가
--   값을 넣는다. 그 유닛이 실행하는 명령.
[root@ol7a ~]# systemctl cat systemd-sysctl | grep -E 'ExecStart|Before|After|Conditions|Description'
Description=Apply Kernel Variables
After=systemd-readahead-collect.service systemd-readahead-replay.service
After=systemd-modules-load.service
Before=sysinit.target shutdown.target
RemainAfterExit=yes
ExecStart=/usr/lib/systemd/systemd-sysctl

-- Before=sysinit.target — 부팅 초기, 일반 서비스보다 먼저 돈다.
-- After=systemd-modules-load.service — 모듈을 올린 뒤에 돈다.
--   모듈이 만드는 키(예: 브리지
--   넷필터)는 그 모듈이 올라온 뒤에야 /proc/sys 에 생기기 때문이다.
-- ExecStart 가 sysctl --system 이 아니라 systemd 자체 도구
--   /usr/lib/systemd/systemd-sysctl 이다.
-- RemainAfterExit=yes — 한 번 실행하고 끝나지만 상태는 active 로 남는
--   oneshot 유닛.

-- 재부팅해서 확인한다 (러너가 재부팅 후 SSH 가 열릴 때까지 기다린다)
[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 30

-- 재부팅 뒤에도 30. 부팅 때 값을 넣은 systemd-sysctl 도
--   /etc/sysctl.conf 의 값을 남겼다. 이유는 [8] 에서 본다.

[root@ol7a ~]# journalctl -b -u systemd-sysctl --no-pager | tail -n 3
Nov 12 20:30:22 ol7a.example.com systemd[1]: Stopped Apply Kernel Variables.
Nov 12 20:30:23 ol7a.example.com systemd[1]: Starting Apply Kernel Variables...
Nov 12 20:30:23 ol7a.example.com systemd[1]: Started Apply Kernel Variables.

-- Stopped 와 Starting 이 같은 초에 찍혔다. 이번 부팅의 initramfs
--   단계에서 한 번 돌고,
--   실제 루트로 넘어온 뒤 다시 돈 흔적이다. 실패 메시지는 없다.

--------------------------------------------------------------------------------
 [7] /etc 의 같은 이름 파일이 /usr/lib 파일을 가림
--------------------------------------------------------------------------------

-- 패키지 기본값 50-default.conf 의 kernel.sysrq = 16 을 바꾸고 싶다.
-- /usr/lib 의 파일을 고치면 패키지 업데이트 때 덮인다. /etc 에 같은 이름으로 둔다.
[root@ol7a ~]# echo 'kernel.sysrq = 1' > /etc/sysctl.d/50-default.conf

-- 출력 없음. /etc/sysctl.d/50-default.conf 와
--   /usr/lib/sysctl.d/50-default.conf 가 이름이 같다.

[root@ol7a ~]# sysctl --system | grep -E 'Applying|kernel.sysrq'
* Applying /usr/lib/sysctl.d/00-system.conf ...
* Applying /usr/lib/sysctl.d/01-unprivileged-bpf.conf ...
* Applying /usr/lib/sysctl.d/10-default-yama-scope.conf ...
* Applying /etc/sysctl.d/50-default.conf ...
kernel.sysrq = 1
* Applying /etc/sysctl.d/80-lab.conf ...
* Applying /etc/sysctl.d/99-sysctl.conf ...
* Applying /etc/sysctl.d/99-tuning.conf ...
* Applying /etc/sysctl.conf ...

-- "* Applying /usr/lib/sysctl.d/50-default.conf" 가
--   목록에서 사라지고 /etc 쪽이 그 자리에 들어갔다.
-- 같은 이름이면 /etc 가 /usr/lib 를 통째로 가린다. 50-default.conf 의
--   다른 줄(rp_filter 등)도
--   이제 적용되지 않는다는 점에 주의한다. 패키지 파일을 가릴 때는 원본을 복사한 뒤 고친다.

[root@ol7a ~]# sysctl kernel.sysrq
kernel.sysrq = 1

-- 1. /etc 쪽 값이 들어갔다.

--------------------------------------------------------------------------------
 [8] systemd-sysctl 을 직접 실행 — 부팅 때 순서를 재부팅 없이 보기
--------------------------------------------------------------------------------

-- 바이너리를 직접 부르면 부팅 때와 같은 규칙으로 적용한다. 출력은 없다.
[root@ol7a ~]# sysctl -w kernel.panic=0 kernel.sysrq=0
kernel.panic = 0
kernel.sysrq = 0
[root@ol7a ~]# /usr/lib/systemd/systemd-sysctl; echo "rc=$?"; sysctl kernel.panic kernel.sysrq
rc=0
kernel.panic = 30
kernel.sysrq = 1

-- 일부러 둘 다 0 으로 내린 뒤 부팅 때 쓰는 도구를 직접 실행했다. 출력은 없고 rc=0.
-- kernel.panic = 30 (sysctl.conf 의 값), kernel.sysrq =
--   1 (/etc/sysctl.d/50-default.conf).
-- 재부팅하지 않고도 부팅 때 들어갈 값을 미리 볼 수 있다.

-- 파일 하나만 넘기면 그 파일만 적용한다.
[root@ol7a ~]# sysctl -w kernel.panic=0; /usr/lib/systemd/systemd-sysctl 80-lab.conf; sysctl kernel.panic
kernel.panic = 0
kernel.panic = 20

-- 파일 이름만 주면 sysctl.d 디렉터리들에서 찾아 그 파일만 적용한다. 20.

-- 99-tuning.conf 가 99-sysctl.conf 보다 뒤인데도 30 이 남았다.
--   systemd-sysctl 이 무엇을 읽는지 직접 본다.
-- SYSTEMD_LOG_LEVEL=debug 는 systemd 도구의 디버그 메시지를 표준
--   오류로 찍게 한다.
[root@ol7a ~]# sysctl -w kernel.panic=0; SYSTEMD_LOG_LEVEL=debug /usr/lib/systemd/systemd-sysctl 2>&1 | grep -E 'panic|Parsing|Overwriting'; sysctl kernel.panic
kernel.panic = 0
kernel.panic = 30

-- 디버그 줄은 하나도 찍히지 않았다. 이 도구는 로그를 표준
--   오류가 아니라 journal 쪽으로 보낸 것으로 보인다.
-- 결과만 남는다. 0 으로 내린 kernel.panic 이 30 이 됐다.
--   99-tuning.conf 의 40 이 아니다.

-- 바이너리 안에 박힌 경로 문자열을 뽑는다. grep -a 는 바이너리를 글자로 취급, -o 는
--   맞은 부분만.
[root@ol7a ~]# grep -a -o -E '/[a-z/]*sysctl\.(conf|d)' /usr/lib/systemd/systemd-sysctl | sort -u
/etc/sysctl.conf
/etc/sysctl.d
/run/sysctl.d
/usr/lib/sysctl.d
/usr/local/lib/sysctl.d

-- 바이너리 안에 /etc/sysctl.conf 라는 문자열이 따로 있다.
-- 디렉터리 넷(/etc, /run, /usr/local/lib, /usr/lib 의
--   sysctl.d)과 별도로 /etc/sysctl.conf 를 직접 가리킨다.
-- OL 7 의 systemd-sysctl 도 sysctl --system 처럼
--   /etc/sysctl.conf 를 따로 읽는다고 보는 근거다. 재부팅
--   뒤와 직접 실행 모두 30 이 남은 결과와 맞는다.
-- /usr/local/lib/sysctl.d 도 읽는 곳이다. sysctl --system
--   의 Applying 목록에는 나오지 않았다(파일이 없어서).


--------------------------------------------------------------------------------
 [9] tuned 가 넣는 값 — vm.swappiness
--------------------------------------------------------------------------------

-- vm.swappiness 는 어느 sysctl 설정 파일에도 없다. 값은 tuned 프로필이
--   넣는다.
[root@ol7a ~]# grep -r swappiness /etc/sysctl.conf /etc/sysctl.d /usr/lib/sysctl.d; echo "rc=$?"
rc=1

-- rc=1 은 grep 이 하나도 찾지 못했다는 뜻이다. swappiness 30 은
--   sysctl 설정 파일 출신이 아니다.

[root@ol7a ~]# tuned-adm active; grep -E '^\[|swappiness|dirty|include' /usr/lib/tuned/virtual-guest/tuned.conf
Current active profile: virtual-guest
[main]
include=throughput-performance
[sysctl]
# The generator of dirty data starts writeback at this percentage (system default
vm.dirty_ratio = 30
vm.swappiness = 30

-- tuned 는 용도별 프로필로 커널 값과 장치 설정을 한꺼번에 넣는 데몬이다.
-- VM 에 설치하면 virtual-guest 프로필이 자동으로 골라진다.
--   [main] include=throughput-performance : 그 프로필을 물려받고
--   [sysctl] vm.dirty_ratio = 30, vm.swappiness = 30 :
--     두 값을 덮어쓴다
-- [0] 의 30 이 여기서 왔다.

[root@ol7a ~]# grep -E '^reapply_sysctl|^#.*reapply' /etc/tuned/tuned-main.conf
# Whether to reapply sysctl from /run/sysctl.d/, /etc/sysctl.d/ and
reapply_sysctl = 1

-- reapply_sysctl = 1 : tuned 가 자기 프로필 값을 넣은 뒤 sysctl
--   설정 파일들을 다시 적용한다.
--   그래서 관리자가 sysctl.d 에 적은 값이 tuned 프로필 값보다 우선한다.

-- sysctl.d 에 vm.swappiness 를 넣고 tuned 를 재시작하면 어느 값이
--   남는가.
[root@ol7a ~]# echo 'vm.swappiness = 5' > /etc/sysctl.d/80-swap.conf; systemctl restart tuned; sleep 3; sysctl vm.swappiness
vm.swappiness = 5

-- tuned 를 재시작했는데 프로필 값 30 이 아니라 sysctl.d 의 5 가 남았다.
--   reapply_sysctl 의 효과다.
-- 재부팅 뒤 값이 예상과 다르면 sysctl 파일 세 곳과 함께 tuned 프로필을 본다.

--------------------------------------------------------------------------------
 [변형] /proc/sys 에 직접 쓰기와 sysctl -w 비교
--------------------------------------------------------------------------------

-- 두 방법은 같은 파일에 쓴다. 결과도 같다.
[root@ol7a ~]# echo 15 > /proc/sys/kernel/panic; sysctl kernel.panic
kernel.panic = 15

-- echo 로 파일에 직접 쓴 값을 sysctl 이 그대로 읽는다.

[root@ol7a ~]# sysctl -w kernel.panic=25; cat /proc/sys/kernel/panic
kernel.panic = 25
25

-- 반대로 sysctl -w 가 쓴 값을 cat 이 읽는다. 두 방법은 같은 파일을 다룬다.
-- 차이는 편의뿐이다. sysctl 은 키를 여러 개 한 번에 쓰고, 결과 줄을 찍고, 파일을
--   읽어 일괄
--   적용한다. echo 는 셸 리다이렉션이라 sudo echo 1 > 파일 처럼 쓰면 권한
--     오류가 난다.

-- -q 는 결과 줄을 찍지 않는다. 스크립트에서 쓴다.
[root@ol7a ~]# sysctl -q -w kernel.panic=0; echo "rc=$?"; cat /proc/sys/kernel/panic
rc=0
0

-- -q 로 결과 줄이 사라졌다. 값은 0 으로 들어갔다.

[root@ol7a ~]# cp -p /root/sysctl.conf.orig /etc/sysctl.conf; ls -l /etc/sysctl.d
total 0

-- 그런데 kernel.panic 은 런타임 값이라 파일을 지웠다고 0 으로 돌아가지 않는다.
--   [변형] 에서 0 으로 맞춰 두었기 때문에 지금 0 이다. 순서가 바뀌면 직접 sysctl

[root@ol7a ~]# systemctl restart tuned; sleep 3; sysctl kernel.panic kernel.sysrq vm.swappiness
kernel.panic = 0
kernel.sysrq = 16
vm.swappiness = 30

-- 80-swap.conf 를 지우고 tuned 를 재시작하자 swappiness 가 프로필 값
--   30 으로 돌아왔다.
-- 세 값이 [0] 과 같다.
