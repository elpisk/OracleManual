================================================================================
 제4장 실습 04 — /proc 로 시스템 읽기
 대상   : ol7a
 목표   : /proc 의 시스템 정보 파일(cpuinfo·meminfo·loadavg·mounts 등)과 프로세스별
          디렉터리(/proc/PID 의 status·fd·cmdline·limits)를 읽고, /proc/sys 와 sysctl
          이름의 대응을 확인한다. /proc/sys 에 echo 로 쓴 값이 재부팅 뒤 어떻게 되는지 본다
 배경   : /proc 는 디스크에 없는 가상 파일 시스템이다. 파일을 읽는 순간 커널이 내용을
          만들어 준다. 그래서 ls -l 로 보면 크기가 0 인데 cat 하면 내용이 나온다.
          /proc/sys 아래 파일은 쓸 수 있고, 쓰면 커널 동작이 바로 바뀐다. 다만 메모리에만
          있으므로 재부팅하면 원래 값으로 돌아간다 (본문 4.5 절)
 소요   : 약 20분 (재부팅 1회 포함)
 선행   : ol7a 는 ol7-base
 계정   : root
 주의   : /proc/sys/vm/swappiness 를 런타임에 바꾼다. 재부팅으로 되돌아온다
 되돌림 : vm.swappiness 가 [0] 의 값과 같음 (cat /proc/sys/vm/swappiness)
================================================================================

--------------------------------------------------------------------------------
 [0] 시작 전 상태 기록
--------------------------------------------------------------------------------

-- 이 실습에서 바꾸는 값은 vm.swappiness 하나다.
[root@ol7a ~]# cat /proc/sys/vm/swappiness; uname -r; uptime
30
5.4.17-2136.338.4.2.el7uek.x86_64
 20:29:13 up 0 min,  0 users,  load average: 0.07, 0.02, 0.00

-- 첫 줄 30 이 이 실습에서 바꿨다가 되돌릴 값이다. 둘째 줄은 기준 커널 UEK
--   R6.
-- uptime 의 "up 0 min" 은 실습 시작 때 스냅샷으로 되돌리고 막
--   부팅했기 때문이다.

-- /proc 가 어떤 파일 시스템으로 붙어 있는지. 디스크 장치가 아니라 proc
--   라는 이름이다.
[root@ol7a ~]# findmnt /proc
TARGET SOURCE FSTYPE OPTIONS
/proc  proc   proc   rw,nosuid,nodev,noexec,relatime

-- SOURCE 가 장치가 아니라 그냥 proc 다. 커널이 만든 것이다.
-- nosuid·nodev·noexec — 실행·장치 파일이 의미 없는 곳이다.

 [1] 크기 0 인 파일
--------------------------------------------------------------------------------

-- ls -l 의 크기 열과 실제 내용 길이(wc -c)를 비교한다.
[root@ol7a ~]# ls -l /proc/cpuinfo /proc/meminfo /proc/loadavg /proc/version
-r--r--r--. 1 root root 0 Nov 12 20:29 /proc/cpuinfo
-r--r--r--. 1 root root 0 Nov 12 20:29 /proc/loadavg
-r--r--r--. 1 root root 0 Nov 12 20:29 /proc/meminfo
-r--r--r--. 1 root root 0 Nov 12 20:29 /proc/version

-- 네 파일 모두 크기 열이 0 이다. 날짜·시각 열은 지금 시각이다. 파일이 디스크에
--   저장된 적이 없어
--   ls 가 보여 줄 수정 시각도 크기도 없다.
-- 권한은 모두 -r--r--r-- (0444). 누구나 읽고 아무도 쓰지 못한다.

[root@ol7a ~]# wc -c /proc/cpuinfo /proc/meminfo /proc/loadavg /proc/version
2388 /proc/cpuinfo
1475 /proc/meminfo
26 /proc/loadavg
182 /proc/version
4071 total

-- 크기 0 인 파일에서 수천 바이트가 나왔다.

--     가상화는 꺼져 있다

 [3] /proc/meminfo — 메모리
--------------------------------------------------------------------------------

-- 앞 16줄과 HugePages 관련 줄. free 가 이 파일을 읽어 계산한다.
[root@ol7a ~]# head -16 /proc/meminfo
MemTotal:        3716120 kB
MemFree:         3365448 kB
MemAvailable:    3326088 kB
Buffers:            2116 kB
Cached:           136104 kB
SwapCached:            0 kB
Active:           138356 kB
Inactive:          80732 kB
Active(anon):      81288 kB
Inactive(anon):     8496 kB
Active(file):      57068 kB
Inactive(file):    72236 kB
Unevictable:           0 kB
Mlocked:               0 kB
SwapTotal:       4063228 kB
SwapFree:        4063228 kB

-- 단위는 모두 kB.
-- MemTotal : 커널이 쓸 수 있는 전체. 4GB 보다 조금 작다
-- MemFree : 아무 데도 안 쓴 메모리
-- MemAvailable : 캐시를 비우면 쓸 수 있는 추정치.
--   여유 메모리는 이 값으로 판단한다
-- Buffers, Cached : 블록·파일 캐시
-- (anon) 은 프로그램 메모리, (file) 은 파일 캐시
-- SwapTotal 4063228, SwapFree 같음 — 스왑을 안 씀

[root@ol7a ~]# grep -E '^(HugePages_|Hugepagesize|CommitLimit|Committed_AS)' /proc/meminfo
CommitLimit:     5921288 kB
Committed_AS:     361200 kB
HugePages_Total:       0
HugePages_Free:        0
HugePages_Rsvd:        0
HugePages_Surp:        0
Hugepagesize:       2048 kB

-- HugePages_Total 0 — 큰 페이지를 하나도 잡지 않았다. 17장에서
--   SGA 용으로 잡은 뒤 이 줄이 바뀐다.
-- Hugepagesize 2048 kB — x86_64 의 기본 큰 페이지 크기
--   2MB.
-- CommitLimit 은 스왑 + RAM×overcommit_ratio(기본
--   50%), Committed_AS 는 지금까지 약속한 메모리 합이다.

[root@ol7a ~]# free -k
              total        used        free      shared  buff/cache   available
Mem:        3716120      181844     3365364        8880      168912     3326112
Swap:       4063228           0     4063228

-- free 는 meminfo 를 읽어 계산한다. total 과 free 가 위
--   MemTotal·MemFree 와 같은 크기다
--   (두 명령 사이 몇 kB 차이는 그사이 메모리가 조금 움직였기 때문).
-- buff/cache 는 Buffers + Cached + 회수 가능한 슬랩.
--   available 은 MemAvailable 과 같다.

 [5] /proc/mounts — 커널이 아는 마운트
--------------------------------------------------------------------------------

-- /etc/mtab 은 OL 7 에서 /proc/self/mounts 로 가는 링크다.
[root@ol7a ~]# ls -l /etc/mtab
lrwxrwxrwx. 1 root root 17 Nov 11 19:45 /etc/mtab -> /proc/self/mounts

-- 링크다. 예전에는 mount 명령이 이 파일에 기록했지만 지금은 커널의 목록을
--   그대로 보인다.
--   그래서 /etc/mtab 과 실제 마운트 상태가 어긋날 일이 없다.

[root@ol7a ~]# grep -E ' (/|/boot) ' /proc/mounts
/dev/mapper/ol-root / xfs rw,seclabel,relatime,attr2,inode64,logbufs=8,logbsize=32k,noquota 0 0
/dev/sda1 /boot xfs rw,seclabel,relatime,attr2,inode64,logbufs=8,logbsize=32k,noquota 0 0

-- 여섯 필드 — 장치 | 마운트 지점 | 종류 | 옵션 | dump | fsck
--   순서(/proc/mounts 에서는 늘 0 0)
-- 옵션의 seclabel 은 SELinux 레이블을 지원한다는 표시,
--   attr2·inode64·logbufs 는 XFS 옵션이다(11장).

[root@ol7a ~]# wc -l /proc/mounts; grep -c -v -E '^#|^$' /etc/fstab
32 /proc/mounts
3

-- 커널이 아는 마운트는 32 개인데 fstab 에 적은 것은 3 줄뿐이다.
-- 나머지는 systemd 가 부팅 때 붙인 가상 파일 시스템(proc, sysfs,
--   cgroup, tmpfs 등)이다.

--------------------------------------------------------------------------------
 [6] /proc/PID/status — 프로세스 한 개
--------------------------------------------------------------------------------

-- sshd 의 부모 프로세스(데몬 본체)를 본다. pidof 는 여러 PID 를 줄
--   수 있어 systemctl 로 주 PID 를 얻는다.
[root@ol7a ~]# systemctl show -p MainPID sshd
MainPID=1411

-- 이 PID 가 sshd 데몬 본체다. 실습 러너가 SSH 로 접속해 있어 sshd
--   자식 프로세스도 여럿 있다.

[root@ol7a ~]# P=$(systemctl show -p MainPID sshd | cut -d= -f2); grep -E '^(Name|State|Tgid|PPid|Uid|Gid|VmPeak|VmSize|VmRSS|Threads|voluntary_ctxt_switches)' /proc/$P/status
Name:   sshd
State:  S (sleeping)
Tgid:   1411
PPid:   1
Uid:    0       0       0       0
Gid:    0       0       0       0
VmPeak:   113048 kB
VmSize:   113016 kB
VmRSS:      7800 kB
Threads:        1
voluntary_ctxt_switches:        21

-- 필드별
--   Name     : 실행 파일 이름(15자까지)
--   State    : S 잠듦(대기), R 실행, D 디스크 대기(끊을 수 없음), Z 좀비, T 정지
--   Tgid     : 스레드 그룹 ID. 단일 스레드 프로세스라 PID 와 같다
--   PPid 1   : 부모가 systemd. 데몬은 모두 이렇다
--   Uid, Gid : 네 칸은 실제·유효·저장·파일시스템 ID. 모두 0(root)
--   VmPeak, VmSize : 가상 메모리 최대치·현재치. 실제 쓰는 양이 아니다
--   VmRSS    : 실제로 RAM 에 올라와 있는 양
--   Threads  : 1
--   voluntary_ctxt_switches : 스스로 CPU 를 내놓은
--     횟수(대기하러 간 횟수)

--------------------------------------------------------------------------------
 [7] /proc/PID 의 cmdline·exe·fd·limits
--------------------------------------------------------------------------------

-- cmdline 은 인자 사이가 NUL 문자라 tr 로 공백으로 바꿔 본다.
[root@ol7a ~]# P=$(systemctl show -p MainPID sshd | cut -d= -f2); tr '\0' ' ' < /proc/$P/cmdline; echo; ls -l /proc/$P/exe /proc/$P/cwd
/usr/sbin/sshd -D 
lrwxrwxrwx. 1 root root 0 Nov 12 20:29 /proc/1411/cwd -> /
lrwxrwxrwx. 1 root root 0 Nov 12 20:29 /proc/1411/exe -> /usr/sbin/sshd

-- cmdline 은 /usr/sbin/sshd -D. -D 는 "데몬으로 분리하지
--   말고 앞에서 돌라" 는 뜻으로, systemd 가
--   프로세스를 직접 지켜보려고 붙인 옵션이다(유닛의 ExecStart 와 같다).
-- exe 와 cwd 는 링크다. cwd 가 / 인 것은 데몬이 시작할 때 루트로 옮겨
--   가기 때문이다.

[root@ol7a ~]# P=$(systemctl show -p MainPID sshd | cut -d= -f2); ls -l /proc/$P/fd
total 0
lr-x------. 1 root root 64 Nov 12 20:29 0 -> /dev/null
lrwx------. 1 root root 64 Nov 12 20:29 1 -> socket:[26102]
lrwx------. 1 root root 64 Nov 12 20:29 2 -> socket:[26102]
lrwx------. 1 root root 64 Nov 12 20:29 3 -> socket:[26127]
lrwx------. 1 root root 64 Nov 12 20:29 4 -> socket:[26129]

-- 0 은 /dev/null, 1·2 는 소켓(journald 로 가는 표준
--   출력·오류), 3·4 는 22번 포트를 듣는 소켓(IPv4·IPv6).
-- socket:[번호] 의 번호는 inode 다. ss -lptne 출력의 ino
--   값과 맞춰 볼 수 있다.

[root@ol7a ~]# P=$(systemctl show -p MainPID sshd | cut -d= -f2); grep -E 'Limit|open files|processes|locked' /proc/$P/limits
Limit                     Soft Limit           Hard Limit           Units     
Max processes             14400                14400                processes 
Max open files            1024                 4096                 files     
Max locked memory         65536                65536                bytes     

-- 세 열 — 소프트 한도 | 하드 한도 | 단위
--   Max processes      : 사용자당 프로세스 수
--   Max open files     : 1024 / 4096. 데몬은 로그인 셸의 ulimit 이 아니라 systemd 가 준 한도로 돈다
--   Max locked memory  : 64KB. HugePages 를 쓰는
--     오라클은 이 한도를 크게 올려야 한다(17장)

--------------------------------------------------------------------------------
 [9] 오류 ② — 끝난 프로세스의 /proc/PID
--------------------------------------------------------------------------------

-- 백그라운드로 sleep 을 띄워 PID 를 파일에 적고, 끝낸 뒤 다시 읽는다.
[root@ol7a ~]# nohup sleep 300 >/tmp/sleep.log 2>&1 </dev/null &
[root@ol7a ~]# pgrep -x sleep > /tmp/sleep.pid; cat /tmp/sleep.pid; grep -E '^(Name|State|PPid)' /proc/$(cat /tmp/sleep.pid)/status
2414
Name:   sleep
State:  S (sleeping)
PPid:   1

-- pgrep -x 는 이름이 정확히 sleep 인 프로세스만 찾는다.
-- PPid 가 1 이다. nohup 으로 띄운 뒤 러너의 셸이 끝나 고아가 됐고
--   systemd 가 부모를 이어받았다.

[root@ol7a ~]# kill $(cat /tmp/sleep.pid); sleep 1; cat /proc/$(cat /tmp/sleep.pid)/status
cat: /proc/2414/status: No such file or directory

-- 증상   : 끝난 프로세스의 /proc/PID/status 를 읽자 "No such
--          file or directory"
-- 원인   : /proc/PID 는 살아 있는 프로세스에만 있다. kill 로 끝나자
--          디렉터리가 즉시 사라졌다.
--          부모가 바로 거둬 가서 좀비(Z)로 남지도 않았다
-- 조치   : 모니터링 스크립트는 PID 를 저장해 두지 말고 매번
--          pgrep·systemctl show -p MainPID 로 다시
--          구한다.
--          파일을 읽기 전에 [ -d /proc/$PID ] 로 존재를 확인한다
-- 재확인 : 아래 ls -d 와 ps -p 가 모두 없음을 보인다

[root@ol7a ~]# ls -d /proc/$(cat /tmp/sleep.pid); ps -p $(cat /tmp/sleep.pid); echo "rc=$?"
ls: cannot access /proc/2414: No such file or directory
  PID TTY          TIME CMD
rc=1

-- ps -p 는 머리줄만 찍고 rc=1. 스크립트에서 프로세스 생존을 확인할 때 이
--   종료 코드를 쓴다.
-- PID 는 언젠가 다른 프로세스에게 다시 쓰인다(kernel.pid_max
--   32768 에 이르면 처음부터).
--   오래된 PID 파일만 믿고 kill 하면 엉뚱한 프로세스를 죽일 수 있다.

[root@ol7a ~]# cat /proc/sys/vm/swappiness; sysctl vm.swappiness
30
vm.swappiness = 30

-- 여기도 같다. sysctl 은 이 파일을 읽어 "키 = 값" 형식으로 바꿔 찍을
--   뿐이다.

[root@ol7a ~]# sysctl -a 2>/dev/null | wc -l; find /proc/sys -type f | wc -l
1120
1123

-- 파일 수가 sysctl -a 줄 수보다 많다. sysctl -a 는 읽다가 오류가
--   난 키를 표준 오류로 보내고
--   건너뛰는데, 여기서는 2>/dev/null 로 그 오류를 버렸다.

-- 파일 권한이 쓰기 가능 여부를 말해 준다. 읽기 전용(0444)과 쓰기
--   가능(0644)을 비교한다.
[root@ol7a ~]# ls -l /proc/sys/kernel/osrelease /proc/sys/kernel/hostname /proc/sys/vm/swappiness /proc/sys/fs/file-nr
-r--r--r--. 1 root root 0 Nov 12 20:29 /proc/sys/fs/file-nr
-rw-r--r--. 1 root root 0 Nov 12 20:29 /proc/sys/kernel/hostname
-r--r--r--. 1 root root 0 Nov 12 20:29 /proc/sys/kernel/osrelease
-rw-r--r--. 1 root root 0 Nov 12 20:29 /proc/sys/vm/swappiness

-- hostname·swappiness 는 -rw-r--r-- (root 가 쓸 수
--   있음), osrelease·file-nr 은 -r--r--r-- (정보용).
-- 쓰기 가능한 키만 sysctl -w 나 echo 로 바꿀 수 있다. 실습 06 에서
--   읽기 전용 키에 써 본다.

 [11] 오류 ① — echo 로 쓴 값은 재부팅 뒤 사라짐
--------------------------------------------------------------------------------

[root@ol7a ~]# echo 10 > /proc/sys/vm/swappiness; cat /proc/sys/vm/swappiness; sysctl vm.swappiness
10
vm.swappiness = 10

-- 파일에 쓰자 즉시 10. sysctl 로 읽어도 10. 런타임 변경은 이 한 줄로
--   끝난다.

-- 재부팅 (러너가 재부팅 후 SSH 가 열릴 때까지 기다린다)
[root@ol7a ~]# cat /proc/sys/vm/swappiness; uptime
30
 20:29:41 up 0 min,  0 users,  load average: 0.07, 0.02, 0.00

-- 증상   : echo 10 > /proc/sys/vm/swappiness 로 바꾼
--          값이 재부팅 뒤 30 으로 돌아감
-- 원인   : /proc/sys 는 커널 메모리의 값을 보여 줄 뿐 디스크에 저장하지
--          않는다. 부팅 때
--          sysctl 설정 파일과 tuned 프로필이 다시 값을 넣는다
-- 조치   : 영구화하려면 /etc/sysctl.d/숫자-이름.conf 에
--          vm.swappiness = 10 을 적는다(실습 05)
-- 재확인 : 재부팅 뒤 sysctl vm.swappiness. 이 실습은 되돌리는
--          것이 목적이라 파일을 만들지 않았다
-- uptime 이 다시 0 min 이다. 재부팅이 실제로 일어났다는 근거다.

-- 값을 정한 곳을 찾는다. sysctl 설정 파일 세 곳과 tuned 프로필을
--   grep 한다.
[root@ol7a ~]# grep -r swappiness /etc/sysctl.conf /etc/sysctl.d /usr/lib/sysctl.d; echo "rc=$?"
rc=1

-- rc=1 — sysctl 설정 파일 어디에도 swappiness 가 없다. 30 의
--   출처가 따로 있다.

[root@ol7a ~]# tuned-adm active; grep -r swappiness /usr/lib/tuned/virtual-guest /usr/lib/tuned/throughput-performance
Current active profile: virtual-guest
/usr/lib/tuned/virtual-guest/tuned.conf:vm.swappiness = 30
/usr/lib/tuned/throughput-performance/tuned.conf:# The swappiness parameter controls the tendency of the kernel to move
/usr/lib/tuned/throughput-performance/tuned.conf:vm.swappiness=10

-- 지금 프로필은 virtual-guest, 그 안에 vm.swappiness =
--   30. 출처를 찾았다.
-- virtual-guest 는 throughput-performance 를
--   물려받는데, 부모의 10 을 자식이 30 으로 덮는다.

 [13] 지운 파일을 잡고 있는 프로세스 — fd 의 (deleted)
--------------------------------------------------------------------------------

-- 200MB 짜리 파일을 만들고 tail -f 로 열어 둔다. 로그 파일을 열어 둔
--   데몬을 흉내 낸다.
-- dd 의 bs=1M count=200 은 1MB 블록 200개. 마지막 요약 줄만
--   본다.
[root@ol7a ~]# dd if=/dev/zero of=/tmp/big.log bs=1M count=200 2>&1 | tail -1
209715200 bytes (210 MB) copied, 0.329569 s, 636 MB/s

-- 200 MiB 를 썼다. dd 의 마지막 줄은 바이트 수와 걸린 시간·속도다.

[root@ol7a ~]# nohup tail -f /tmp/big.log >/dev/null 2>&1 </dev/null &
[root@ol7a ~]# df -m / | tail -1
/dev/mapper/ol-root     35943  4561     31383  13% /

-- 열 — 장치 | 전체(MB) | 사용 | 남음 | 사용률 | 마운트 지점. /tmp
--   는 루트 파일 시스템 안에 있다.

-- 파일을 지운다. 이름은 사라졌지만 tail 이 아직 열고 있다.
[root@ol7a ~]# rm -f /tmp/big.log; ls /tmp/big.log; df -m / | tail -1
ls: cannot access /tmp/big.log: No such file or directory
/dev/mapper/ol-root     35943  4561     31383  13% /

-- ls 가 파일이 없다고 한다. 그런데 df 의 Used 는 바로 위와 같은 4561
--   이다.

[root@ol7a ~]# ls -l /proc/$(pgrep -x tail)/fd | grep deleted
lr-x------. 1 root root 64 Nov 12 20:29 3 -> /tmp/big.log (deleted)

-- tail 의 fd 3 이 "/tmp/big.log (deleted)" 를 가리킨다.
--   이름은 지워졌지만 열린 파일은 살아 있다.
-- 증상   : rm 으로 200MB 파일을 지웠는데 df 의 Used 가 줄지 않음
-- 원인   : 그 파일을 tail 이 열고 있다. 마지막 참조가 닫힐 때까지 블록이
--          반환되지 않는다.
--          /proc/PID/fd 의 링크 끝 "(deleted)" 가 근거다
-- 조치   : 프로세스를 재시작할 수 없으면 truncate -s 0
--          /proc/PID/fd/3 으로 내용만 비운다
-- 재확인 : 아래 df 의 Used 가 약 200MB 줄어듦

-- 공간을 돌려받는 방법 두 가지 — 프로세스를 끝내거나, fd 를 통해 파일을 0
--   바이트로 자른다.
[root@ol7a ~]# truncate -s 0 /proc/$(pgrep -x tail)/fd/3; df -m / | tail -1
/dev/mapper/ol-root     35943  4360     31584  13% /

-- Used 가 4561 에서 4360 으로, 약 200MB 줄었다. tail 은 살아
--   있지만 파일은 0 바이트가 됐다.

[root@ol7a ~]# kill $(pgrep -x tail); sleep 1; pgrep -x tail; echo "rc=$?"; df -m / | tail -1
rc=1
/dev/mapper/ol-root     35943  4360     31584  13% /

-- tail 을 끝냈다. pgrep 이 아무것도 못 찾아 rc=1. df 는 바로 위와
--   같다(이미 0 바이트였다).
