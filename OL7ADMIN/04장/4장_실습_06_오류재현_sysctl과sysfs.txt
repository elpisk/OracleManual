================================================================================
 제4장 실습 06 — 오류 재현: sysctl 과 sysfs 에 잘못된 값 넣기
 대상   : ol7a
 목표   : 없는 키(오타), 범위 밖 값, 숫자 자리에 문자, 읽기 전용 키, 일반 사용자 권한으로
          sysctl 을 써 보고 각각 어떤 메시지가 나오는지 모은다. 설정 파일 안의 오타가
          부팅 때 어디에 기록되는지 본다. /sys/block/sdX/queue/scheduler 를 런타임에
          바꾸고, 재부팅 뒤에도 남기려면 udev 규칙을 쓴다
 배경   : sysctl 은 /proc/sys 아래 파일에 값을 써 넣을 뿐이고, 값이 맞는지는 커널이
          판단한다. 그래서 오류 메시지는 대부분 파일 쓰기 오류(No such file, Invalid
          argument, Permission denied)의 모양이다. /sys(sysfs)는 장치와 드라이버의
          속성을 파일로 보이는 곳인데, 여기 쓴 값도 재부팅하면 사라진다. sysfs 값은
          sysctl.d 로 영구화할 수 없고 udev 규칙이나 tuned 로 넣는다 (본문 4.6·4.7 절)
 소요   : 약 20분 (재부팅 1회 포함)
 선행   : ol7a 는 ol7-base. 추가 디스크 sdb·sdc 는 비어 있음(파티션 없음)
 계정   : root, student
 주의   : 디스크 데이터는 건드리지 않는다. 스케줄러 속성만 바꾼다
 되돌림 : 오타 파일과 udev 규칙 삭제, sdb·sdc 스케줄러 [mq-deadline], vm.swappiness 30

 [1] 오류 ① — 없는 키 이름 (오타)
--------------------------------------------------------------------------------

-- swappiness 의 p 를 하나 빠뜨렸다.
[root@ol7a ~]# sysctl -w vm.swapiness=10; echo "rc=$?"
sysctl: cannot stat /proc/sys/vm/swapiness: No such file or directory
rc=255

-- sysctl 은 키 이름을 /proc/sys/vm/swapiness 경로로 바꿔 파일을
--   찾았고, 없어서 실패했다.
-- 메시지가 "키가 없다" 가 아니라 파일 시스템 오류(cannot stat ... No such
--   file or directory)
--   모양인 것이 sysctl 의 정체를 보여 준다. 종료 코드는 255.

[root@ol7a ~]# sysctl vm.swapiness; echo "rc=$?"
sysctl: cannot stat /proc/sys/vm/swapiness: No such file or directory
rc=255

-- 읽기도 같은 메시지, 같은 rc=255.

-- 올바른 이름은 sysctl -a 에서 찾는다. 출력이 많아 grep 으로 좁힌다.
[root@ol7a ~]# sysctl -a 2>/dev/null | grep swap
vm.kswapd_threads = 1
vm.swappiness = 30

-- p 가 두 개인 vm.swappiness 가 보인다. vm.kswapd_threads 는
--   이름에 swap 이 들어 있어 함께 걸렸다.

[root@ol7a ~]# sysctl -w vm.swappiness=10; sysctl vm.swappiness
vm.swappiness = 10
vm.swappiness = 10

-- 증상   : sysctl -w vm.swapiness=10 이 "sysctl: cannot
--          stat /proc/sys/vm/swapiness:
--          No such file or directory" 와 rc=255 로 실패
-- 원인   : 키 이름 오타. sysctl 은 키를 /proc/sys 아래 경로로 바꿔 열기
--          때문에 없는 키는
--          파일이 없다는 오류로 나타난다
-- 조치   : sysctl -a | grep 으로 정확한 이름을 찾아 sysctl -w
--          vm.swappiness=10
-- 재확인 : sysctl vm.swappiness 가 "vm.swappiness = 10" (위
--          두 줄 중 아래 줄)

--------------------------------------------------------------------------------
 [2] 오타가 설정 파일 안에 있으면
--------------------------------------------------------------------------------

-- 앞줄에 오타, 뒷줄에 정상 키를 둔 파일. 오타 한 줄 때문에 나머지도 안 들어가는지 본다.
[root@ol7a ~]# printf 'vm.swapiness = 20\nkernel.panic = 20\n' > /etc/sysctl.d/80-typo.conf; cat /etc/sysctl.d/80-typo.conf
vm.swapiness = 20
kernel.panic = 20

-- printf 의 \n 이 줄바꿈이 돼 두 줄짜리 파일이 생겼다.

[root@ol7a ~]# sysctl -p /etc/sysctl.d/80-typo.conf; echo "rc=$?"
sysctl: cannot stat /proc/sys/vm/swapiness: No such file or directory
kernel.panic = 20
rc=255

-- 첫 줄은 실패, 둘째 줄 kernel.panic = 20 은 적용됐다. 한 줄의 오류가
--   나머지를 막지 않는다.
-- 대신 전체 종료 코드는 255. 스크립트는 일부만 적용된 상태를 실패로 받게 된다.

[root@ol7a ~]# sysctl kernel.panic
kernel.panic = 20

-- 20 이 들어가 있다. 오류가 난 파일에서도 정상 줄은 살아 있다.

-- -e 는 없는 키를 무시하고 오류를 내지 않는다.
[root@ol7a ~]# sysctl -e -p /etc/sysctl.d/80-typo.conf; echo "rc=$?"
kernel.panic = 20
rc=0

-- -e 를 붙이자 오타 줄의 메시지가 사라지고 rc=0. 없는 키를 조용히 건너뛴다.
-- 여러 커널(UEK·RHCK)이 공유하는 설정 파일에서 한쪽 커널에만 있는 키를 넣을 때 쓴다.
-- 오타까지 숨기므로 평소에는 쓰지 않는다.

--------------------------------------------------------------------------------
 [3] 범위 밖 값과 형식이 틀린 값
--------------------------------------------------------------------------------

-- vm.swappiness 는 0~100.
[root@ol7a ~]# sysctl -w vm.swappiness=200; echo "rc=$?"; sysctl vm.swappiness
sysctl: setting key "vm.swappiness": Invalid argument
vm.swappiness = 200
rc=0
vm.swappiness = 10

-- 출력 네 줄을 차례로 읽는다.
--   1 sysctl: setting key "vm.swappiness": Invalid
--     argument  — 커널이 200 을 거부했다
--   2 vm.swappiness = 200                                     — 그런데 sysctl 은 쓴 값을 그대로 찍었다
--   3 rc=0                                                    — 종료 코드도 성공
--   4 vm.swappiness = 10                                      — 실제 값은 바뀌지 않았다
-- 7.9 의 sysctl(procps-ng)은 쓰기 오류가 EINVAL 일 때 메시지를 내고도
--   0 으로 끝난다.
--   종료 코드나 둘째 줄만 보는 스크립트는 성공으로 오판한다. 반드시 다시 읽어 확인한다.

-- 숫자 자리에 문자.
[root@ol7a ~]# sysctl -w vm.swappiness=abc; echo "rc=$?"; sysctl vm.swappiness
sysctl: setting key "vm.swappiness": Invalid argument
vm.swappiness = abc
rc=0
vm.swappiness = 10

-- 숫자 자리에 문자도 같은 모양이다. Invalid argument, 그러나 rc=0, 실제
--   값은 10 그대로.

-- 값이 두 개인 키에 하나만, 또는 앞뒤가 뒤집힌 범위.
[root@ol7a ~]# sysctl -w net.ipv4.ip_local_port_range="60000 30000"; echo "rc=$?"; sysctl net.ipv4.ip_local_port_range
sysctl: setting key "net.ipv4.ip_local_port_range": Invalid argument
net.ipv4.ip_local_port_range = 60000 30000
rc=0
net.ipv4.ip_local_port_range = 32768    60999

-- 시작 포트가 끝 포트보다 크다. 커널이 거부했고 값은 32768 60999 그대로다.
-- 값이 두 개인 키는 따옴표로 묶어 한 인자로 넘긴다. 따옴표가 없으면 셸이 둘로 나눠
--   두 번째 숫자를 다른 키 이름으로 해석한다.

--------------------------------------------------------------------------------
 [4] = 앞뒤의 공백 — 명령줄에서
--------------------------------------------------------------------------------

-- 설정 파일에서는 = 앞뒤 공백이 허용되지만 명령줄에서는 인자가 셋으로 나뉜다.
[root@ol7a ~]# sysctl -w vm.swappiness = 10; echo "rc=$?"
sysctl: "vm.swappiness" must be of the form name=value
sysctl: malformed setting "="
sysctl: "10" must be of the form name=value
rc=252

-- 셸이 인자를 vm.swappiness, =, 10 세 개로 나눴다. 셋 다 키=값 형식이
--   아니라 각각 거부됐다.
-- 종료 코드 252 는 사용법 오류. [1] 의 255(키 없음)와 구분된다.
-- 설정 파일 안에서는 "vm.swappiness = 10" 처럼 공백이 있어도 된다.
--   sysctl 이 줄을 직접 나누기 때문이다.

[root@ol7a ~]# sysctl vm.swappiness
vm.swappiness = 10

-- 10 그대로. 실패한 명령은 아무것도 바꾸지 않았다.

 [5] 읽기 전용 키
--------------------------------------------------------------------------------

[root@ol7a ~]# ls -l /proc/sys/kernel/osrelease /proc/sys/fs/file-nr
-r--r--r--. 1 root root 0 Nov 12 20:31 /proc/sys/fs/file-nr
-r--r--r--. 1 root root 0 Nov 12 20:31 /proc/sys/kernel/osrelease

-- 두 파일 모두 -r--r--r-- (0444). 소유자 root 에게도 쓰기 권한이 없다.
-- osrelease 는 실행 중인 커널 릴리스 문자열, file-nr 는 열린 파일 핸들 수
--   통계다. 둘 다 커널이 보여 주기만 하는 값이다.

[root@ol7a ~]# sysctl -w kernel.osrelease=1.0; echo "rc=$?"
sysctl: setting key "kernel.osrelease": No such file or directory
rc=0

-- 메시지가 Permission denied 가 아니라 "No such file or
--   directory" 다. rc 도 0.
-- sysctl 은 파일에 쓰기 권한 비트가 없으면 쓰기를 시도하지 않고 경고만 낸다. 그 경고에
--   붙은
--   "No such file or directory" 는 실제 원인과 상관없는 문장이다.
--   문구만 보고 "키가 없다" 고 판단하면 틀린다. ls -l 로 권한을 확인한다.

[root@ol7a ~]# echo 1 > /proc/sys/fs/file-nr; echo "rc=$?"
/tmp/_ol7lab_root.sh: line 1: /proc/sys/fs/file-nr: Permission denied
rc=1

-- 셸 리다이렉션으로 쓰면 메시지가 정직하다. Permission denied, rc=1.
-- 줄 앞의 /tmp/_ol7lab_root.sh: line 1: 은 실습 러너가 명령을 스크립트
--   파일로 실행해서 붙은 것이다.
--   터미널에서 직접 치면 같은 자리에 -bash: 가 나온다.
-- 증상   : 읽기 전용 키에 쓰기 — sysctl 은 "setting key
--          "kernel.osrelease": No such file or
--          directory"(rc=0), echo 는 "Permission
--            denied"(rc=1)
-- 원인   : 파일 권한 0444. 커널이 쓰기 처리 함수를 두지 않은 정보용 키다. root
--          도 바꿀 수 없다
-- 조치   : 바꿀 수 있는 키인지 먼저 ls -l /proc/sys/경로 로 확인한다.
--          osrelease 는 다른 커널로
--          부팅해야 바뀌는 값이고, file-nr 의 한도는 쓰기 가능한
--            fs.file-max 로 정한다
-- 재확인 : ls -l /proc/sys/fs/file-max 가 -rw-r--r-- 인지
--          확인하고 그 키를 쓴다

--------------------------------------------------------------------------------
 [7] sysfs — 디스크 I/O 스케줄러를 런타임에 바꿈
--------------------------------------------------------------------------------

-- 대괄호 안이 현재 선택된 스케줄러다. UEK R6 는 다중 큐(blk-mq) 스케줄러만 보인다.
[root@ol7a ~]# cat /sys/block/sdb/queue/scheduler
[mq-deadline] kyber bfq none

-- 목록은 UEK R6 의 다중 큐(blk-mq) 스케줄러 네 개. 대괄호 안이 현재 값이다.
--   mq-deadline : 요청마다 마감 시간을 두어 읽기가 오래 굶지 않게 함. 기본값
--   kyber       : 지연 목표 기반, 빠른 SSD 용
--   bfq         : 프로세스별 공정 배분, 데스크톱 용
--   none        : 스케줄링 없이 그대로 내려보냄. 가상 디스크·NVMe 에서 흔히 고름

[root@ol7a ~]# echo none > /sys/block/sdb/queue/scheduler; cat /sys/block/sdb/queue/scheduler
[none] mq-deadline kyber bfq 

-- 즉시 바뀌었다. 대괄호가 none 으로 옮겨 가고 목록 순서도 바뀌었다(현재 값이 앞).
-- 재시작할 서비스도, 적용 명령도 없다. 다음 I/O 부터 새 스케줄러를 쓴다.

-- 목록에 없는 이름. RHCK 3.10 의 예전 이름(deadline, cfq)을 쓰는 실수다.
[root@ol7a ~]# echo cfq > /sys/block/sdb/queue/scheduler; echo "rc=$?"; cat /sys/block/sdb/queue/scheduler
/tmp/_ol7lab_root.sh: line 1: echo: write error: Invalid argument
rc=1
[none] mq-deadline kyber bfq 

-- cfq 는 RHCK 3.10 의 단일 큐 스케줄러다. UEK R6 에는 없다.
-- 증상   : "echo: write error: Invalid argument", rc=1,
--          스케줄러는 [none] 그대로
-- 원인   : 목록에 없는 이름. UEK R6(5.4) 는 blk-mq 스케줄러만 지원한다.
--          예전 문서의
--          cfq·deadline·noop 은 RHCK 로 부팅했을 때의 이름이다
-- 조치   : 먼저 cat 으로 목록을 읽고 그 안의 이름만 쓴다. deadline 대신
--          mq-deadline, noop 대신 none
-- 재확인 : cat /sys/block/sdb/queue/scheduler 의 대괄호 위치

--------------------------------------------------------------------------------
 [8] sysfs 의 읽기 전용 속성
--------------------------------------------------------------------------------

[root@ol7a ~]# ls -l /sys/block/sdb/size /sys/block/sdb/queue/scheduler /sys/block/sdb/queue/rotational
-rw-r--r--. 1 root root 4096 Nov 12 20:31 /sys/block/sdb/queue/rotational
-rw-r--r--. 1 root root 4096 Nov 12 20:31 /sys/block/sdb/queue/scheduler
-r--r--r--. 1 root root 4096 Nov 12 20:31 /sys/block/sdb/size

-- sysfs 파일은 크기가 4096 으로 찍힌다(/proc 는 0). 둘 다 실제 크기와는
--   상관없는 값이다.
-- size 는 0444 읽기 전용, scheduler·rotational 은 0644 쓰기 가능.

[root@ol7a ~]# cat /sys/block/sdb/size; echo 1 > /sys/block/sdb/size; echo "rc=$?"
20971520
/tmp/_ol7lab_root.sh: line 1: /sys/block/sdb/size: Permission denied
rc=1

-- size 는 512 바이트 섹터 수다. 20971520 × 512 = 10 GiB, sdb 의
--   크기와 맞는다.
-- 읽기 전용이라 Permission denied. 디스크 크기는 장치 쪽이 정하는 값이다.

-- sysctl 은 /proc/sys 만 다룬다. sysfs 경로를 키처럼 넣으면?
[root@ol7a ~]# sysctl block.sdb.queue.scheduler; echo "rc=$?"
sysctl: cannot stat /proc/sys/block/sdb/queue/scheduler: No such file or directory
rc=255

-- sysctl 은 /proc/sys 아래만 본다. /proc/sys/block/... 을 찾다가
--   실패했다.
-- sysfs 값은 sysctl 로 읽을 수도, sysctl.d 로 영구화할 수도 없다.

--------------------------------------------------------------------------------
 [9] udev 규칙으로 영구화 — sdc 만
--------------------------------------------------------------------------------

-- sdb 는 런타임 변경만(none), sdc 는 udev 규칙으로 none 을 넣는다.
-- ACTION 은 장치가 추가·변경될 때, KERNEL 은 커널 장치 이름, ATTR{} 는
--   sysfs 속성 쓰기.
[root@ol7a ~]# echo 'ACTION=="add|change", KERNEL=="sdc", ATTR{queue/scheduler}="none"' > /etc/udev/rules.d/60-lab-scheduler.rules
[root@ol7a ~]# cat /etc/udev/rules.d/60-lab-scheduler.rules
ACTION=="add|change", KERNEL=="sdc", ATTR{queue/scheduler}="none"

-- 규칙 한 줄의 구성
--   ACTION=="add|change"            : 장치가 생기거나 바뀌는 이벤트에서
--   KERNEL=="sdc"                   : 커널 장치 이름이 sdc 이면
--   ATTR{queue/scheduler}="none"    : /sys/block/sdc/queue/scheduler 에 none 을 쓴다
-- == 는 비교, = 는 대입이다. 반대로 쓰면 규칙이 엉뚱하게 동작한다.

-- 규칙은 다음 이벤트 때 적용된다. change 이벤트를 sdc 에 강제로 보낸다.
[root@ol7a ~]# udevadm control --reload; udevadm trigger --action=change --sysname-match=sdc; sleep 2

-- 규칙 파일만 두면 다음 부팅이나 장치 이벤트 때까지 적용되지 않는다.
--   control --reload : udevd 가 규칙 파일을 다시 읽음
--   trigger --action=change --sysname-match=sdc : sdc
--     에 change 이벤트를 보내 규칙을 지금 돌림

[root@ol7a ~]# cat /sys/block/sdb/queue/scheduler /sys/block/sdc/queue/scheduler
[none] mq-deadline kyber bfq 
[none] mq-deadline kyber bfq 

-- 지금은 둘 다 [none]. sdb 는 [7] 에서 echo 로, sdc 는 방금 udev
--   규칙으로 바뀌었다.
-- 겉으로는 같다. 차이는 재부팅하면 드러난다.

--------------------------------------------------------------------------------
 [10] 오류 ② — 재부팅 뒤 런타임 값이 사라짐 (오타 파일도 함께 확인)
--------------------------------------------------------------------------------

-- 80-typo.conf 는 그대로 두고 재부팅한다. sdb 는 런타임 값만, sdc 는
--   udev 규칙.
-- 재부팅 (러너가 재부팅 후 SSH 가 열릴 때까지 기다린다)
[root@ol7a ~]# cat /sys/block/sdb/queue/scheduler /sys/block/sdc/queue/scheduler
[mq-deadline] kyber bfq none
[none] mq-deadline kyber bfq 

-- 재부팅 뒤 sdb 는 [mq-deadline], sdc 는 [none].
-- 증상   : 재부팅 뒤 sdb 의 스케줄러가 [none] 에서 [mq-deadline] 으로
--          돌아감
-- 원인   : /sys 에 쓴 값은 커널 메모리에만 있다. 부팅 때 커널이 장치를 다시 만들며
--          기본값을 넣는다
-- 조치   : udev 규칙(sdc 처럼) 또는 tuned 프로필([변형])로 부팅 때마다 다시
--          넣게 한다
-- 재확인 : 재부팅 뒤 cat /sys/block/sdc/queue/scheduler 가
--          [none] — 규칙을 둔 sdc 만 남았다

[root@ol7a ~]# sysctl vm.swappiness kernel.panic
vm.swappiness = 30
kernel.panic = 20

-- vm.swappiness 는 30 으로 돌아왔다. [1] 에서 sysctl -w 로 넣은 10
--   은 런타임 값이었다.
-- kernel.panic 은 20. 80-typo.conf 의 정상 줄이 부팅 때 적용됐다.
--   오타 줄은 아무 일도 하지 않았다.

-- 부팅 때 sysctl 값을 넣는 systemd-sysctl 의 기록. 오타 줄이 어떻게
--   남았는지 본다.
[root@ol7a ~]# journalctl -b -u systemd-sysctl --no-pager
-- Logs begin at Thu 2026-11-12 20:31:22 KST, end at Thu 2026-11-12 20:31:39 KST. --

Nov 12 20:31:24 ol7a.example.com systemd[1]: Starting Apply Kernel Variables...
Nov 12 20:31:24 ol7a.example.com systemd[1]: Started Apply Kernel Variables.

-- 오타 줄에 대한 경고가 없다. Started 만 남았다.
-- 첫 줄 "-- Logs begin at ..." 은 journalctl 이 찍는

           man:sysctl.d(5)

 [변형] udev 대신 tuned 프로필로 sysfs 값 넣기 — sdd
--------------------------------------------------------------------------------

-- tuned 의 disk 플러그인에는 elevator 항목이 있다. 기본 프로필들이 [disk]
--   를 어떻게 쓰는지 먼저 본다.
[root@ol7a ~]# grep -r -E 'elevator|^\[sysfs\]|^\[disk\]' /usr/lib/tuned/*/tuned.conf | head -8
/usr/lib/tuned/balanced/tuned.conf:[disk]
/usr/lib/tuned/hpc-compute/tuned.conf:[disk]
/usr/lib/tuned/powersave/tuned.conf:[disk]
/usr/lib/tuned/throughput-performance/tuned.conf:[disk]
[root@ol7a ~]# grep -A3 '^\[disk\]' /usr/lib/tuned/throughput-performance/tuned.conf
[disk]
# The default unit for readahead is KiB.  This can be adjusted to sectors
# by specifying the relevant suffix, eg. (readahead => 8192 s). There must
# be at least one space between the number and suffix (if suffix is specified).

-- 기본 프로필의 [disk] 는 readahead 를 다룬다. elevator 를
--   쓰는 기본 프로필은 없다. 위 grep 결과에도 elevator 가 없다.

[root@ol7a ~]# udevadm info -q property -n sdc | grep -E 'DEVNAME|ID_SERIAL|ID_PATH='
DEVNAME=/dev/sdc
ID_PATH=pci-0000:00:10.0-scsi-0:0:2:0

-- 지금 프로필(virtual-guest)을 물려받고 sdd 의 스케줄러만 바꾸는 사용자 프로필을
--   만든다.
-- /etc/tuned/이름/tuned.conf 에 두면 tuned-adm list 에 나타난다.
--   include 는 부모 프로필.
[root@ol7a ~]# mkdir /etc/tuned/lab-sched
[root@ol7a ~]# printf '[main]\ninclude=virtual-guest\n\n[disk]\ndevices=sdd\nelevator=none\n' > /etc/tuned/lab-sched/tuned.conf; cat /etc/tuned/lab-sched/tuned.conf
[main]
include=virtual-guest

-- 사용자 프로필 파일 내용
--   [main] include=virtual-guest : 지금 프로필을 물려받는다
--   [disk] devices=sdd           : 이 절은 sdd 에만 적용
--          elevator=none         : 스케줄러를 none 으로

[disk]
devices=sdd
elevator=none
[root@ol7a ~]# cat /sys/block/sdd/queue/scheduler; tuned-adm profile lab-sched; tuned-adm active; cat /sys/block/sdd/queue/scheduler
[mq-deadline] kyber bfq none
Current active profile: lab-sched
[none] mq-deadline kyber bfq 

-- 프로필을 바꾸기 전 [mq-deadline], 바꾼 뒤 [none].
-- tuned-adm profile 은 프로필을 켜고 그 설정을 즉시 넣는다.

[root@ol7a ~]# sysctl vm.swappiness; cat /etc/tuned/active_profile
vm.swappiness = 30
lab-sched

-- include 로 물려받아 swappiness 는 그대로 30.
-- 지금 프로필 이름은 /etc/tuned/active_profile 에 저장된다.
--   tuned 가 재시작·재부팅 뒤 이 파일을 보고 같은 프로필을 켠다.
--   udev 규칙과 같은 영구화 효과다.

-- 프로필을 원래대로 돌리면 tuned 가 바꾼 값을 되돌리는지 본다.
[root@ol7a ~]# tuned-adm profile virtual-guest; tuned-adm active; cat /sys/block/sdd/queue/scheduler
Current active profile: virtual-guest
[mq-deadline] kyber bfq none

-- 프로필을 되돌리자 sdd 도 [mq-deadline] 으로 돌아왔다.
-- tuned 는 자기가 바꾼 값을 기억했다가 프로필을 끌 때 원래대로
--   돌린다. udev 규칙에는 없는 성질이다.
