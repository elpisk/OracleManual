--------------------------------------------------------------------------------
 [1] /etc/crontab — 비어 있는 시스템 crontab
--------------------------------------------------------------------------------

[root@ol7a ~]# cat /etc/crontab
SHELL=/bin/bash
PATH=/sbin:/bin:/usr/sbin:/usr/bin
MAILTO=root

# For details see man 4 crontabs

# Example of job definition:
# .---------------- minute (0 - 59)
# |  .------------- hour (0 - 23)
# |  |  .---------- day of month (1 - 31)
# |  |  |  .------- month (1 - 12) OR jan,feb,mar,apr ...
# |  |  |  |  .---- day of week (0 - 6) (Sunday=0 or 7) OR sun,mon,tue,wed,thu,fri,sat
# |  |  |  |  |
# *  *  *  *  * user-name  command to be executed

-- 맨 위 세 줄은 이 파일 안의 작업에만 쓰이는 변수
--   SHELL=/bin/bash  : 작업 명령을 bash 로 해석 (지정이 없으면 /bin/sh)
--   PATH=/sbin:/bin:/usr/sbin:/usr/bin : sbin 이 들어 있어 관리 명령을 찾음
--   MAILTO=root      : 작업 출력이 있으면 root 에게 메일
-- 그림의 마지막 줄에 user-name 이 있다. 사용자 crontab 에는 없는 필드
-- 작업 줄은 하나도 없다. 7.9 의 /etc/crontab 은 사실상 설명서다
-- [OL8/9] 같은 crontabs 패키지 구성이라 내용이 거의 같다

--------------------------------------------------------------------------------
 [2] /etc/cron.d/0hourly — cron.hourly 의 실행 주체
--------------------------------------------------------------------------------

[root@ol7a ~]# cat /etc/cron.d/0hourly
# Run the hourly jobs
SHELL=/bin/bash
PATH=/sbin:/bin:/usr/sbin:/usr/bin
MAILTO=root
01 * * * * root run-parts /etc/cron.hourly

-- 0hourly 는 /etc/crontab 과 같은 형식의 시스템 crontab 조각
--   01 * * * *  : 매시 01분
--   root        : 사용자 필드. 이 작업은 root 로 돈다
--   run-parts /etc/cron.hourly : 디렉터리 안 스크립트를 차례로 실행
-- cron.d 의 파일은 자기 변수(SHELL·PATH·MAILTO)를 따로 가진다.
-- /etc/crontab 의 변수가 cron.d 로 넘어오지 않기 때문이다

-- run-parts 는 셸 스크립트다. 실행 권한 검사와 제외 규칙이 있는 부분을 본다

[root@ol7a ~]# file /usr/bin/run-parts; rpm -qf /usr/bin/run-parts; wc -l /usr/bin/run-parts
/usr/bin/run-parts: Bourne-Again shell script, ASCII text executable
crontabs-1.11-6.20121102git.el7.noarch
101 /usr/bin/run-parts

-- run-parts 는 101줄짜리 bash 스크립트. 실행 파일이 아니라 읽을 수 있다

[root@ol7a ~]# grep -n -E 'for i in|-x|~|rpmsave|rpmorig|rpmnew|swp|cfsaved|logger' /usr/bin/run-parts
45:# Ignore *~ and *, scripts
46:for i in $(LC_ALL=C; echo ${1%/}/*[^~,]) ; do
48:     # Don't run *.{rpmsave,rpmorig,rpmnew,swp,cfsaved} scripts
49:     [ "${i%.cfsaved}" != "${i}" ] && continue
50:     [ "${i%.rpmsave}" != "${i}" ] && continue
51:        [ "${i%.rpmorig}" != "${i}" ] && continue
52:        [ "${i%.rpmnew}" != "${i}" ] && continue
53:        [ "${i%.swp}" != "${i}" ] && continue
73:             elif [ -x $i ]; then
89:                     logger -p cron.notice -t "run-parts($1)[$$]" "starting $(basename $i)"
96:                     logger -i -p cron.notice -t "run-parts($1)" "finished $(basename $i)"

-- 줄 번호별로 읽는다
--   46행 : 글롭 *[^~,] → 이름이 ~ 나 , 로 끝나는 파일은 목록에서 빠짐
--          LC_ALL=C 로 정렬해 이름 순(숫자·대문자·소문자 순)으로 돈다
--   49~53: .cfsaved .rpmsave .rpmorig .rpmnew .swp 로 끝나면 continue
--   73행 : [ -x $i ] 일 때만 실행. 아닌 파일은 else 없이 넘어감
--   89·96: 실행 앞뒤로 logger 가 cron.notice 로 starting·finished 를 남김
-- 즉 실행 권한이 없는 파일은 로그에 이름조차 남지 않는다. [7] 에서 확인
-- 생략된 줄(54~72행)에는 디렉터리 건너뛰기와 기타 제외 규칙이 있다.
-- 전체는 sed -n '40,101p' /usr/bin/run-parts 로 볼 수 있다

 [4] /etc/cron.d 에 작업 파일 넣기
--------------------------------------------------------------------------------

-- 작업이 쓸 스크립트. 실행한 사용자와 시각을 /tmp/lab-sys.log 에 남긴다

[root@ol7a ~]# cat > /usr/local/bin/lab-sys.sh <<'EOF'
#!/bin/bash
echo "$(date +%T) $1 ran as $(id -un)" >> /tmp/lab-sys.log
EOF

-- id -un : 실행 중인 실효 사용자 이름. 어느 계정으로 돌았는지 기록하려고 넣음

[root@ol7a ~]# chmod 755 /usr/local/bin/lab-sys.sh; touch /tmp/lab-sys.log; chmod 666 /tmp/lab-sys.log

-- /tmp/lab-sys.log 를 666 으로 만든 이유: student 로 도는 작업도 써야 하므로

-- 시스템 crontab 형식: 분 시 일 월 요일 사용자 명령.
-- 같은 스크립트를 root 와 student 로 한 줄씩

[root@ol7a ~]# cat > /etc/cron.d/lab-sys <<'EOF'
# 실습 02 — 시스템 cron 작업
* * * * * root    /usr/local/bin/lab-sys.sh A
* * * * * student /usr/local/bin/lab-sys.sh B
EOF
[root@ol7a ~]# ls -l /etc/cron.d
total 16
-rw-r--r--. 1 root root 128 Apr 26  2023 0hourly
-rw-r--r--. 1 root root 130 Nov 12 18:07 lab-sys

-- lab-sys 가 18:07 에 생겼다. 권한 644, root 소유
-- 파일 이름에는 점(.)을 넣지 않는다. cron.d 는 이름에 점이 든 파일을 무시하는
-- 배포판이 있어 관례상 영문·숫자·- 만 쓴다
-- cron.d 파일은 root 소유여야 하고 그룹·기타 쓰기 권한이 없어야 crond 가 읽는다
-rw-r--r--. 1 root root 108 Feb 23  2022 raid-check
-rw-------. 1 root root 235 Aug 24  2023 sysstat

-- crond 를 재시작하지 않는다. 다음 분에 읽히는지 본다

[root@ol7a ~]# sleep $((62 - 10#$(date +%S))); date '+%T'; cat /tmp/lab-sys.log
18:08:02

-- 18:08:02 에 확인했는데 로그가 비어 있다. grep 도 아무것도 찾지 못했다

[root@ol7a ~]# grep 'lab-sys' /var/log/cron | tail -3

-- 실행되지 않은 것이 아니다. [5] 의 grep 결과 첫 줄을 보면
-- 'Nov 12 18:08:02 ... CROND[2296]: (student) CMD (... B)' 가 있다.
-- 새 파일을 읽어 들인 첫 분에는 작업이 :01 이 아니라 :02 에 시작했고,
-- 확인 명령이 그보다 아주 조금 빨랐다. [5] 의 tail 에도 '18:08:02 A ran as root' 가 보인다
-- 교훈: cron 결과는 분 경계 직후가 아니라 몇 초 여유를 두고 본다

 [5] 오류 흐름 ① — 사용자 필드 누락
--------------------------------------------------------------------------------

-- 사용자 crontab 습관대로 사용자 필드 없이 한 줄을 더 넣는다

[root@ol7a ~]# echo '* * * * * /usr/local/bin/lab-sys.sh C' >> /etc/cron.d/lab-sys; cat /etc/cron.d/lab-sys

-- 세 번째 줄은 시간 다섯 필드 뒤에 바로 명령이 온다
-- 사용자 crontab(crontab -e)에서는 맞는 형식이라 손에 익은 사람일수록 실수한다
-- crond 는 이 줄을 '분 시 일 월 요일 사용자=/usr/local/bin/lab-sys.sh 명령=C' 로 나눈다
# 실습 02 — 시스템 cron 작업
* * * * * root    /usr/local/bin/lab-sys.sh A
* * * * * student /usr/local/bin/lab-sys.sh B
* * * * * /usr/local/bin/lab-sys.sh C
[root@ol7a ~]# sleep $((62 - 10#$(date +%S))); date '+%T'; tail -3 /tmp/lab-sys.log
18:09:02
18:08:02 A ran as root
18:09:01 B ran as student
18:09:01 A ran as root

-- 18:09 에 A·B 는 돌았는데 C 가 없다. 오류 메시지도 화면에는 없다

[root@ol7a ~]# grep -E 'lab-sys|ERROR|getpwnam' /var/log/cron | tail -4
Nov 12 18:08:02 ol7a CROND[2296]: (student) CMD (/usr/local/bin/lab-sys.sh B)
Nov 12 18:09:01 ol7a crond[1039]: (/usr/local/bin/lab-sys.sh) ERROR (getpwnam() failed)
Nov 12 18:09:01 ol7a CROND[2308]: (student) CMD (/usr/local/bin/lab-sys.sh B)
Nov 12 18:09:01 ol7a CROND[2310]: (root) CMD (   /usr/local/bin/lab-sys.sh A)

-- 18:08:02 의 B 줄은 [4] 에서 늦게 본 첫 실행이다
-- 둘째 줄이 crond 가 남긴 오류
--   crond[1039]   : 상주 데몬 자신(PID 1039)이 남긴 줄
--   (/usr/local/bin/lab-sys.sh) : 괄호 안은 원래 사용자 이름 자리
--   ERROR (getpwnam() failed)   : 그런 이름의 사용자를 /etc/passwd 에서 못 찾음
-- 반면 CROND[2308]·CROND[2310] 은 작업마다 갈라진 자식 프로세스가 남긴 줄
-- 마지막 줄 CMD 의 명령 앞 공백 셋은 파일에 'root    ' 로 맞춘 공백이 남은 것
--
-- 증상   : /etc/cron.d/lab-sys 의 C 작업이 실행되지 않음. /var/log/cron 에
--          'crond[1039]: (/usr/local/bin/lab-sys.sh) ERROR (getpwnam() failed)'
-- 원인   : 시스템 crontab 은 여섯째 필드가 사용자다. 사용자 필드를 빠뜨려
--          명령 경로가 사용자 이름으로 읽혔다
-- 조치   : 시간 필드 뒤에 root 를 넣는다 (아래 sed)
-- 재확인 : [9] 에서 '18:11:02 C ran as root' 가 기록됨

-- 조치: 사용자 필드 root 를 넣는다

[root@ol7a ~]# sed -i 's#^\* \* \* \* \* /usr/local#* * * * * root    /usr/local#' /etc/cron.d/lab-sys; cat /etc/cron.d/lab-sys
# 실습 02 — 시스템 cron 작업
* * * * * root    /usr/local/bin/lab-sys.sh A
* * * * * student /usr/local/bin/lab-sys.sh B
* * * * * root    /usr/local/bin/lab-sys.sh C

-- sed 의 구분자로 # 를 썼다. 패턴에 / 가 많아 \/ 로 막는 수고를 줄인다
-- 패턴 앞의 \* 는 * 를 글자로 찾게 한다. 고친 줄이 원하는 모양인지 cat 으로 바로 확인

 [6] 오류 흐름 ② — 명령 칸의 %

-- 날짜를 파일명에 넣는 흔한 작업. 시스템 crontab 이든 사용자 crontab 이든

* * * * * root date +%Y%m%d > /tmp/lab-date.txt

-- cron.hourly 에 스크립트 두 개를 넣는다. 하나는 실행 권한이 없다.
-- 매시 01분을 기다리는 대신 0hourly 가 치는 명령을 그대로 직접 친다

[root@ol7a ~]# printf '#!/bin/bash\necho "$(date +%%T) hourly-x" >> /tmp/lab-sys.log\n' > /etc/cron.hourly/lab-x

[root@ol7a ~]# chmod 755 /etc/cron.hourly/lab-noexec; run-parts /etc/cron.hourly; grep hourly /tmp/lab-sys.log

-- 같은 작업을 /etc/cron.d 대신 /etc/crontab 에 넣어도 형식은 같다.
-- 다른 점은 파일 하나를 여러 작업이 나눠 쓴다는 것. 패키지 업데이트로

[root@ol7a ~]# rm -f /etc/cron.d/lab-sys; cp -p /etc/crontab /root/crontab.orig
