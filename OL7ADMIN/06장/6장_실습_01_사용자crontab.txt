--------------------------------------------------------------------------------
 [2] crontab 파일로 등록
--------------------------------------------------------------------------------

-- 먼저 작업 정의 파일을 만들고 crontab 파일명 으로 통째로 올린다.
-- crontab 파일명 은 기존 crontab 을 덮어쓴다. 합치지 않는다.
-- 명령 칸의 % 는 cron 에서 줄바꿈으로 바뀌는 특수 문자라 \% 로 막는다(실습 02 에서 재현)

[student@ol7a ~]$ cat > ~/mycron <<'EOF'
# 매 분 시각 기록
* * * * * date '+\%T' >> /home/student/cron-date.log
EOF
[student@ol7a ~]$ crontab ~/mycron; crontab -l
# 매 분 시각 기록
* * * * * date '+\%T' >> /home/student/cron-date.log

-- 파일의 두 줄이 그대로 설치됐다. 주석 줄도 같이 들어간다
-- * 다섯 개 = 매 분. 명령 칸의 \% 는 crontab 이 % 로 바꿔 date 에 넘긴다
-- crontab 명령은 성공하면 아무 말도 하지 않는다. -l 로 다시 읽어 확인하는 습관을 둔다

 [5] 터미널에서 되는 스크립트 — 준비

-- 열린 TCP 포트 수를 세어 보고서 디렉터리에 저장하는 스크립트.
-- 보고서 위치는 REPORT_DIR 변수로 받는다. 변수는 ~/.bash_profile 에 둔다

[student@ol7a ~]$ mkdir -p ~/bin ~/reports; echo 'export REPORT_DIR=$HOME/reports' >> ~/.bash_profile
[student@ol7a ~]$ cat > ~/bin/listen-report.sh <<'EOF'
#!/bin/bash
echo "== $(date +%T) REPORT_DIR=[$REPORT_DIR]"
ss -tln | tail -n +2 | wc -l
ss -tln > "$REPORT_DIR/listen.txt" && echo saved

-- 터미널(로그인 셸)에서는 세 줄 모두 정상
--   REPORT_DIR=[/home/student/reports] : 방금 .bash_profile 에 넣은 값
--   6 : 열린 TCP 리슨 소켓 수(ss -tln 의 머리 줄을 뺀 줄 수)
--   saved : 보고서 파일 저장 성공
-- 이 실습의 student 명령은 su - student 로 들어간 로그인 셸에서 돈다. 그래서 방금 덧붙인
-- .bash_profile 줄이 바로 반영됐다
EOF
[student@ol7a ~]$ chmod +x ~/bin/listen-report.sh; ~/bin/listen-report.sh
== 18:33:05 REPORT_DIR=[/home/student/reports]
6
saved

-- 이 PATH 는 /etc/profile 과 ~/.bash_profile 이 만든 것이다

 [6] 같은 스크립트를 cron 에 — 오류 흐름 ①, ②

-- crontab 을 세 줄로 바꾼다.
--   1행 : 스크립트 실행, 출력과 오류를 파일로 (>> 파일 2>&1)
--   2행 : cron 이 작업에 주는 환경 변수를 그대로 덤프
--   3행 : 출력을 리다이렉트하지 않은 작업 — 출력은 어디로 가는가

-- crontab -l 이 세 줄. 앞의 date 작업은 사라졌다. crontab 파일명 은 덮어쓰기다

[student@ol7a ~]$ cat > ~/mycron <<'EOF'
* * * * * /home/student/bin/listen-report.sh >> /home/student/cron-report.log 2>&1
* * * * * env > /home/student/cron-env.txt
* * * * * echo "hello from cron on $(hostname)"
EOF
[student@ol7a ~]$ crontab ~/mycron; crontab -l | wc -l
3

[root@ol7a ~]# sleep $((62 - 10#$(date +%S))); date '+%T'; cat /home/student/cron-report.log
18:34:02
== 18:34:01 REPORT_DIR=[]
/home/student/bin/listen-report.sh: line 3: ss: command not found
0
/home/student/bin/listen-report.sh: line 4: /listen.txt: Permission denied

-- cron 에서 돈 결과(18:34:01). 터미널과 같은 스크립트가 두 곳에서 실패했다
--   REPORT_DIR=[] : 변수가 비었다
--   line 3: ss: command not found : ss 를 찾지 못했다
--   0 : ss 출력이 없으니 wc -l 이 0
--   line 4: /listen.txt: Permission denied : "$REPORT_DIR/listen.txt" 가 /listen.txt 가
--     되어 student 가 / 에 쓰려다 거부됨
-- 1행에 2>&1 을 붙였기 때문에 오류 메시지도 로그 파일에 들어왔다

-- cron 의 환경과 로그인 셸의 환경을 나란히 놓는다

[root@ol7a ~]# cat /home/student/cron-env.txt
XDG_SESSION_ID=4
SHELL=/bin/sh
USER=student
PATH=/usr/bin:/bin
PWD=/home/student
LANG=en_US.UTF-8
SHLVL=1
HOME=/home/student
LOGNAME=student
XDG_RUNTIME_DIR=/run/user/1000
_=/usr/bin/env

-- cron 이 작업에 준 환경 전체. 열한 개뿐이다
--   SHELL=/bin/sh, PATH=/usr/bin:/bin : crond 의 기본값
--   HOME·LOGNAME·USER : /etc/passwd 의 student 항목에서 채움
--   XDG_SESSION_ID·XDG_RUNTIME_DIR : pam_systemd 가 세션을 만들며 넣은 값
--   PWD=/home/student : 작업은 홈 디렉터리에서 시작한다
-- REPORT_DIR 도, MAIL 도 없다. ~/.bash_profile 을 읽지 않았다는 증거다

[student@ol7a ~]$ env | sort > ~/login-env.txt; diff <(sort ~/cron-env.txt) ~/login-env.txt | grep -E '^[<>] (PATH|SHELL|REPORT_DIR|HOME|LOGNAME|USER|MAIL)='
< PATH=/usr/bin:/bin
> MAIL=/var/spool/mail/student
> PATH=/usr/local/bin:/bin:/usr/bin:/usr/local/sbin:/usr/sbin:/home/student/.local/bin:/home/student/bin
< SHELL=/bin/sh
> REPORT_DIR=/home/student/reports
> SHELL=/bin/bash

-- diff 표기: < 는 cron 쪽, > 는 로그인 셸 쪽. 두 파일 다 sort 해서 비교
--   PATH  : /usr/bin:/bin  대  /usr/local/bin:…:/usr/sbin:…  → ss 를 못 찾은 원인
--   SHELL : /bin/sh  대  /bin/bash  → crontab 명령 줄을 sh 가 해석
--   REPORT_DIR : cron 쪽에 없음 → /listen.txt 로 쓰려던 원인
--   MAIL  : 로그인 셸에만 있음
-- HOME·LOGNAME·USER 는 양쪽이 같아 diff 에 나오지 않았다
--
-- 증상   : 'line 3: ss: command not found', 'line 4: /listen.txt: Permission denied'
-- 원인   : cron 은 로그인 셸을 거치지 않아 PATH=/usr/bin:/bin 이고
--          ~/.bash_profile 의 REPORT_DIR 이 없다(위 diff)
-- 조치   : crontab 맨 위에 PATH·REPORT_DIR 을 이름=값 으로 정의([7])
-- 재확인 : 다음 분 cron-report.log 에 'REPORT_DIR=[/home/student/reports]', 6, saved

-- 증상   : 3행 작업의 출력이 어디에도 보이지 않음(터미널에도 파일에도 없음)
-- 원인   : 리다이렉트하지 않은 cron 출력은 MAILTO(없으면 crontab 주인)에게 메일로 감
-- 조치   : 출력이 필요 없으면 MAILTO="" 를 두거나 >> 파일 2>&1 로 받는다([7])
-- 재확인 : [7] 이후 메일함 줄 수가 25 에서 늘지 않음

--------------------------------------------------------------------------------
 [7] 고치기 — crontab 안의 변수 정의
--------------------------------------------------------------------------------

-- crontab 맨 위에 '이름=값' 줄을 두면 그 아래 작업의 환경이 된다.
-- 값 안의 $HOME 같은 변수는 펼쳐지지 않으므로 절대 경로로 쓴다.
--   PATH=...     ss 가 있는 /usr/sbin 을 넣음
--   REPORT_DIR=  스크립트가 기대하는 변수
--   MAILTO=""    출력 메일을 보내지 않음

[student@ol7a ~]$ cat > ~/mycron <<'EOF'
PATH=/usr/local/bin:/usr/bin:/usr/sbin
REPORT_DIR=/home/student/reports
MAILTO=""
* * * * * /home/student/bin/listen-report.sh >> /home/student/cron-report.log 2>&1
* * * * * echo "hello from cron on $(hostname)"
EOF
[student@ol7a ~]$ crontab ~/mycron; crontab -l | head -3; : > ~/cron-report.log; wc -l < /var/spool/mail/student
PATH=/usr/local/bin:/usr/bin:/usr/sbin
REPORT_DIR=/home/student/reports
MAILTO=""
25

-- crontab -l | head -3 으로 변수 세 줄이 들어갔는지 보고, 보고서 로그를 비운 뒤
-- 메일함 줄 수 25 를 기록했다. 다음 분과 비교할 기준이다

[root@ol7a ~]# sleep $((62 - 10#$(date +%S))); date '+%T'; cat /home/student/cron-report.log; wc -l < /var/spool/mail/student
18:35:02
== 18:35:01 REPORT_DIR=[/home/student/reports]
6
saved
25

-- 18:35:01 의 결과는 터미널과 같다. REPORT_DIR 이 차고 ss 가 6 을 셌고 saved
-- 메일함은 25 줄 그대로. hello 작업은 여전히 출력을 내지만 MAILTO="" 라 메일이 없다
-- crond 재시작 없이 다음 분에 바로 반영됐다
-- PATH 에 /bin 을 넣지 않았는데도 동작한 것은 7.9 에서 /bin 이 /usr/bin 의 링크이기 때문
-- [OL8/9] 같은 구조(/bin → usr/bin)다. 이 부분은 차이가 없다

--------------------------------------------------------------------------------
 [8] 문법 검사 — 잘못된 시간 값
--------------------------------------------------------------------------------

-- crontab 명령은 올리기 전에 문법을 본다. 분 필드에 61 을 넣는다.
-- crontab - : 표준 입력에서 읽음

[student@ol7a ~]$ echo '61 * * * * date' | crontab -; crontab -l | tail -1
"-":1: bad minute
errors in crontab file, can't install.

-- 설치 거부 메시지 두 줄
--   "-":1: bad minute : 표준 입력(-)의 1행, 분 필드가 범위(0-59) 밖
--   errors in crontab file, can't install. : 아무것도 바꾸지 않음
* * * * * echo "hello from cron on $(hostname)"

-- crontab -l 의 마지막 줄이 [7] 의 hello 줄이다. 기존 crontab 이 그대로 남았다
-- 문법이 틀린 crontab 을 넣어도 운영 중인 작업이 사라지지 않는다는 뜻

-- 요일 필드 7 과 이름 표기는 받아들여지는가

[student@ol7a ~]$ printf '0 9 * * 7 date\n0 9 * jan sun date\n*/15 8-18 * * mon-fri date\n' | crontab -; crontab -l

-- 세 줄 모두 설치됐다
--   0 9 * * 7          : 요일 7 도 일요일로 받아들임
--   0 9 * jan sun      : 월·요일 이름 표기
--   */15 8-18 * * mon-fri : 평일 8~18시대에 15분마다(마지막은 18:45)
-- crontab - 로 넣었으니 [7] 의 변수 세 줄과 작업 두 줄은 사라졌다
0 9 * * 7 date
0 9 * jan sun date
*/15 8-18 * * mon-fri date

--------------------------------------------------------------------------------
 [변형] crontab -e 로 한 줄 추가
--------------------------------------------------------------------------------

-- crontab -e 는 crontab 을 임시 파일로 꺼내 편집기($VISUAL, $EDITOR, 없으면 vi)로 열고
-- 저장하면 문법 검사 후 올린다. 화면 편집기 대신 sed 를 편집기로 지정해 같은 경로를 탄다.
--   sed -i '$a 줄' : 파일 끝에 한 줄 추가

[student@ol7a ~]$ EDITOR="sed -i '\$a 30 2 * * 1-5 /home/student/bin/listen-report.sh'" crontab -e; crontab -l

-- 'crontab: installing new crontab' 은 -e 경로에서만 나오는 문구다
-- 편집기(sed)가 임시 파일 끝에 한 줄을 붙였고, crontab 이 검사 뒤 설치했다
-- 기존 세 줄 뒤에 네 번째 줄이 붙었다. -e 는 덮어쓰기가 아니라 편집이다
-- 운영에서 crontab -e 를 쓸 때도 저장 뒤 이 문구가 나오는지 본다
crontab: installing new crontab
0 9 * * 7 date
0 9 * jan sun date
*/15 8-18 * * mon-fri date
30 2 * * 1-5 /home/student/bin/listen-report.sh

-- root 는 -u 로 다른 사용자의 crontab 을 다룬다

[root@ol7a ~]# crontab -u student -l | wc -l; ls -l --time-style=+%T /var/spool/cron/student
4
-rw-------. 1 student student 109 18:35:02 /var/spool/cron/student

-- root 가 -u student 로 본 줄 수 4. 파일은 109 바이트로 커졌다
-- 변경 시각 18:35:02 는 crontab -e 가 설치한 순간

--------------------------------------------------------------------------------
 [9] crontab -r 과 되돌림
--------------------------------------------------------------------------------

-- -r : 확인 없이 crontab 삭제. -i 를 붙이면 묻는다(터미널에서만)

[student@ol7a ~]$ crontab -r; crontab -l
no crontab for student

-- -r 은 묻지 않고 지운다. 바로 no crontab for student 로 돌아갔다

[root@ol7a ~]# ls -la /var/spool/cron; cp -p /root/bash_profile.student /home/student/.bash_profile
total 4
drwx------.  2 root root    6 Nov 12 18:35 .
drwxr-xr-x. 13 root root 4096 Nov 11 19:46 ..

-- /var/spool/cron 이 다시 비었다(크기 6)

[student@ol7a ~]$ rm -rf ~/bin ~/reports ~/mycron ~/cron-*.log ~/cron-env.txt ~/login-env.txt; ls ~
[root@ol7a ~]# diff /root/bash_profile.student /home/student/.bash_profile && echo same; : > /var/spool/mail/student; rm -f /root/bash_profile.student

-- ls ~ 의 출력이 없다. 실습 중 만든 파일이 모두 지워졌다
same

-- .bash_profile 이 [0] 의 사본과 같다(same). 메일함은 비우고 사본도 지웠다

--------------------------------------------------------------------------------
 [정리] 관찰 요약
--------------------------------------------------------------------------------

--  항목                     시작 전             변경 후                         되돌린 뒤
--  student crontab          없음                1~4 줄로 여러 번 교체           없음
--  /var/spool/cron          비어 있음           student (600, 77→109B)          비어 있음
--  첫 실행 시각             -                   설치 다음 분 :03                -
--  cron PATH                -                   /usr/bin:/bin                   -
--  REPORT_DIR (cron)        -                   없음 → crontab 변수로 지정      -
--  3행 출력                 -                   student 메일함(896B)            메일함 비움
--  MAILTO=""                -                   메일 25 줄에서 늘지 않음        -
--  61 * * * *               -                   bad minute, 설치 거부           -
--  ~/.bash_profile          원본                export 한 줄 추가               원본과 같음
--
-- 확인한 사실
--   cron 작업은 로그인 셸을 거치지 않는다. PATH 와 프로파일 변수를 스크립트나 crontab 에서 정한다
--   리다이렉트하지 않은 출력은 메일로 간다. X-Cron-Env 머리로 작업 환경을 확인할 수 있다
--   crontab 명령은 설치 전에 문법을 검사하고, 실패하면 기존 crontab 을 그대로 둔다
